\section{Introduction}

\subsection{Statistical Inference}

\begin{definition}[The general statistical problem]
    We observe the data values $X_1, \dotsc, X_n$ from a distribution $F$.
    We then want to estimate
    (or learn)
    $F$ itself,
    or some characteristics of $F$
    (e.g. its mean).
\end{definition}

\subsubsection{Models and parameters}
A \textbf{statistical model}
(also known as a family of distributions)
is a set of distributions
(or densities)
$\mathcal{M}$.

A \textbf{parametric model} is any model $\mathcal{M}$ that can be parameterised by a finite number of parameters.

In general,
a parametric model for the data would take the form:
\[
\mathcal{M} = \{f(x; \theta)\,:\,\theta \in \Omega\}
\]
where $\theta$ is an unknown parameter
(or vector of parameters)
that can take values in the parameter space $\Omega$.
If $\theta$ is a vector,
we are only interested in one component of $\theta$,
we call the remaining parameters nuisance parameters.

A nonparametric model is a set $\mathcal{M}$ that cannot be parameterised by a finite number of parameters.

\subsubsection{Fundamental Inference Problems}

\textbf{Point Estimation}:

Point estimation refers to providing a single "best guess" for some quantity of interest.
This is often a parameter $\theta$ in a parametric model,
but can also be the entire distribution $F$,
a prediction for a future value $Y$,
or any combination or function of the above.

\textbf{Uncertainty Estimation}:

We need to understand the accuracy of the estimates we make,
we can do this by estimating attributes of our potential error
(such as means and variances),
or producing intervals of plausible values consistent with our estimate.

\textbf{Hypothesis Tests}:

In hypothesis testing,
we start with some default theory about the model - the null hypothesis,
and we ask if the data provides sufficient evidence to reject the theory.

\section{Distribution Theory}

\subsection{Multivariate probability}

\begin{definition}[Random vector]
    If $X_1, \dotsc, X_p$ is a collection of real-valued random variables,
    then
    \[
    \mathbf{X} = \begin{bmatrix}
        X_1 \\ \vdots \\ X_p
    \end{bmatrix} = (X_1, \dotsc, X_p) ^ T,
    \]
    is a $p$-dimensional random vector in $\R ^ p$,
    and $\mathbf{x} = (x_1, \dotsc, x_p) ^ T$ is a particular realisation of $\mathbf{X}$.
\end{definition}

\begin{definition}[PDF of a random vector]
    The pdf of a random vector $\mathbf{X} = (X_1, \dotsc, X_p) ^ T$ is the joint probability density function $f(x_1, \dotsc, x_p)$ from $\R ^ p$ to $\R_{0}^{+} = [0, \infty)$ such that:
    
    \begin{enumerate}[label = (\roman*)]
        \item $f(\mathbf{x}) \geq 0$ for all $\mbf{x} \in \R ^ p$

        \item $\displaystyle\int_{\R ^ p}f(\mbf{x})\,\mbf{dx} = \int_{-\infty}^{\infty} \dotsi\int_{-\infty}^{\infty}f(x_1, \dotsc, x_p)\,\mbf{d}x_1, \dotsc, \mbf{d}x_p = 1$

        \item For any subset $\mathcal{A}$ of $\R ^ p\,:\,\P[\mbf{X} \in \mathcal{A}] = \int_{\mathcal{A}}f(\mbf{x})\,\mbf{dx}$.
    \end{enumerate}
\end{definition}

\begin{definition}[CDF of a random vector]
    Similarly,
    the joint cdf is defined as
    \[
    F(\mbf{x}) = F(x_1', \dotsc, x_p') = \P[X_1 \leq x_1', \dotsc, X_p \leq x_p'] = \int_{-\infty}^{x_1'}\dotsi\int_{-\infty}^{x_p'}f(x_1, \dotsc, x_p)\,\mbf{d}x_p, \dotsc, \mbf{d}x_1,
    \]
    and we can recover the pdf by
    (partial)
    differentiation
    \[
    f(\mbf{x}) = \frac{\partial ^ p}{\partial x_1\dotsi\partial x_p}F(\mbf{x}).
    \]
\end{definition}

\subsubsection{Key operations and relations}

\begin{definition}[Marginal pdf]
    A subcollection $(X_1, \dotsc, X_q)$ of $(X_1, \dotsc, X_p)$,
    where $q < p$,
    has a marginal pdf defined as
    \[
    f(x_1, \dotsc, x_q) = \int_{-\infty}^{\infty}\dotsi\int_{-\infty}^{\infty}f(x_1, \dotsc, x_p)\,\mbf{d}x_{q + 1}\dotsc\mbf{d}x_p.
    \]
\end{definition}

\begin{definition}[Conditional pdf]
    If we know the values $X_{q + 1} = x_{q + 1}, \dotsc, X_p = x_p$,
    but not the values of $X_1, \dotsc, X_q$,
    then the $X_1, \dotsc, X_q$ are still random variables and have a conditional pdf:
    \[
    f(x_1, \dotsc, x_q\mid x_{q + 1}, \dotsc, x_p) = \frac{f(x_1, \dotsc, x_q, x_{q + 1}, \dotsc, x_p)}{f(x_{q + 1}, \dotsc, x_p)}.
    \]
\end{definition}

\begin{definition}[Independence]
    A collection of random variables $(X_1, \dotsc, X_p)$ is said to be
    (mutually)
    independent if and only if their joint pdf factorises as
    \[
    f(x_1, x_2, \dotsc, x_p) = f_1(x_1)f_2(x_2)\dotsc f_p(x_p),
    \]
    for all $(x_1, x_2, \dotsc, x_p) \in \R ^ p$.
    We sometimes write this as $X_1 \independent X_2 \dotsi \independent X_p$,
    where $\independent$ means 'independent of'.

    An equivalent statement,
    is that the conditional densities are equal to the marginal densities:
    \[
    f(x_i\mid x_1, \dotsc, x_{i - 1}, x_{i + 1}, \dotsc, x_p) = f(x_i),
    \]
    for all $i = 1, \dotsc, p$.
\end{definition}

\begin{definition}[Random sample,
IID]
    The random variables $X_1, \dotsc, X_p$ are called independent and identically distributed
    (iid)
    random variables with pdf or pmf $f(x)$ if they are all independent of each other,
    and the marginal pdf or pmf of each $X_i$ is the same function $f(x)$.
    This means that the joint pdf simplifies to
    \[
    f(x_1, x_2, \dotsc, x_p) = f(x_1)f(x_2)\dotsc f(x_p) = \prod_{i = 1}^{p}f(x_i).
    \]
    Alternatively,
    $X_1, \dotsc, X_p$ are called a random sample of size $n$ from the distribution
    (or population)
    $f(x)$.
\end{definition}

\subsection{Multivariate expectation and variance}

\begin{definition}[Expectation of a functional]
    Let $g(\mbf{X})\,:\, \R ^ p \to \R$ be a function of the random vector $\mbf{X}$.
    Then $g(\mbf{X})$ is a random variable and its expectation is
    \[
    \E\left(g(\mbf{X})\right) = \int_{-\infty}^{\infty}\dotsi\int_{-\infty}^{\infty}g(x_1, \dotsc, x_p)f(x_1, \dotsc, x_p)\,\mbf{d}x_1\dotsc\mbf{d}x_p = \int_{\R ^ p}g(\mbf{x})f(\mbf{x})\,\mbf{dx}.
    \]
\end{definition}

\begin{definition}[Mean vector]
    The expectation $\E(\mbf{X})$ of a random vector $\mbf{X} = (X_1, \dotsc, X_p)$ is defined as:
    \[
    \E(\mbf{X}) = \int\mbf{x}f(\mbf{x})\,\mbf{dx} = \begin{bmatrix}
        \int_{-\infty}^{\infty}x_1f(\mbf{x})\,\mbf{dx} \\
        \vdots \\
        \int_{-\infty}^{\infty}x_pf(\mbf{x})\,\mbf{dx}
    \end{bmatrix}
    =
    \begin{bmatrix}
        \E(\mbf{X}_1) \\
        \vdots \\
        \E(\mbf{X}_p)
    \end{bmatrix}
    = \mbf{\mu},
    \]
    which is simply the $p$-dimensional vector whose components are the expectations of the individual components of $\mbf{X}$ and is known as the mean vector,
    $\mbf{\mu}$,
    of $\mbf{X}$.
\end{definition}

\begin{definition}[Variance matrix]
    The variance matrix of the random vector $\mbf{X}$ is defined to be the $p \times p$ matrix such that,
    for $i, j = 1, \dotsc p$,
    the $(i, j)$ element of the matrix is $\Cov(X_i, X_j)$:
    \[
    \Var(\mbf{X}) = \E\left((\mbf{X} - \mbf{\mu})(\mbf{X} - \mbf{\mu}) ^ T\right) = \E(\mbf{X}\mbf{X} ^ T) - \mbf{\mu}\mbf{\mu} ^ T = \begin{bmatrix}
        \sum_{11} & \dotsi & \sum_{1p} \\
        \vdots & \ddots & \vdots \\
        \sum_{p1} & \dotsi & \sum_{pp}
    \end{bmatrix} = \mbf{\Sigma}
    \]
    where the components of the matrix comprise the individual variances and covariances of the random variables
    \begin{align*}
        \Sigma_{jj} &= \E(X_j ^ 2) - \E(X_j) ^ 2 = \Var(X_j) = \sigma_j ^ 2 \\
        \Sigma_{ij} &= \E(X_iX_j) - \E(X_i)\E(X_j) = \Cov(X_i, X_j) = \Cov(X_j, X_i) = \sigma_{ji}.
    \end{align*}
\end{definition}

\subsubsection{Properties of expectation and variance}
For shorthand let $\mbf{X} \sim [\mbf{\mu}, \mbf{\Sigma}]$ denote the random vector $\mbf{X}$ which has $\E(\mbf{X}) = \mbf{\mu}$ and variance $\Var(\mbf{X}) = \mbf{\Sigma}$.

\begin{proposition}
    Let $\mbf{X}$ be a random $p$-vector $\mbf{X} \sim [\mbf{\mu}, \mbf{\Sigma}]$.
    Then for constant $p \times p$ matrices $\mbf{A}, \mbf{C}$ and constant $p$-vectors $\mbf{b}, \mbf{d}$:

    \begin{enumerate}[label = (\roman*)]
        \item $\E(\mbf{A}\mbf{X} + \mbf{b}) = \mbf{A}\E(\mbf{X}) + \mbf{b}$,
        
        \item $\Var(\mbf{A}\mbf{X} + \mbf{b}) = \mbf{A}\Var(\mbf{X})\mbf{A} ^ T$,
        
        \item $\Cov(\mbf{A}\mbf{X} + \mbf{b}, \mbf{C}\mbf{X} + \mbf{d}) = \mbf{A}\Var(\mbf{X})\mbf{C} ^ T$.
    \end{enumerate}

    \begin{proof}
        \begin{enumerate}[label = (\roman*)]
            \item
            \begin{align*}
                \E(\mbf{A}\mbf{X} + \mbf{b}) &= \int(\mbf{Ax} + \mbf{b})f(\mbf{x})\,\mbf{dx} + \mbf{b} = \int\mbf{Ax}f(\mbf{x}) + \mbf{b}f(\mbf{x})\,\mbf{dx} = \int\mbf{Ax}f(\mbf{x})\,\mbf{dx} + \int\mbf{b}f(\mbf{x})\,\mbf{dx} \\
                &= \mbf{A}\int\mbf{x}f(\mbf{x})\,\mbf{dx} + \mbf{b} \\
                &= \mbf{A}\E(\mbf{X}) + \mbf{b}.
            \end{align*}

            \item
            Define $\mbf{Y} = \mbf{AX} + \mbf{b}$,
            then by (i) $\mbf{\mu}_Y = \mbf{A}\mbf{\mu} + \mbf{b}$.

            \begin{align*}
                \Var(\mbf{AX} + \mbf{b}) &= \E\left((\mbf{Y} - \mbf{\mu}_Y)(\mbf{Y} - \mbf{\mu}_Y) ^ T\right) = \E\left((\mbf{AX} + \mbf{b} - \mbf{A\mu} - \mbf{b})(\mbf{AX} + \mbf{b} - \mbf{A\mu} - \mbf{b}) ^ T\right) \\
                &= \E\left(\mbf{A}(\mbf{X} - \mbf{\mu})(\mbf{A}(\mbf{X} - \mbf{\mu})) ^ T\right)
                \intertext{Now,
                note that $(\mbf{AB}) ^ T = \mbf{B} ^ T\mbf{A} ^ T$ so:}
                &= \E\left(\mbf{A}(\mbf{X} - \mbf{\mu})(\mbf{X} - \mbf{\mu}) ^ T\mbf{A} ^ T\right)
                \intertext{Then by property (i) we can pull out the constant $\mbf{A}$ on either side to get:}
                &= \mbf{A}\E\left((\mbf{X} - \mbf{\mu})(\mbf{X} - \mbf{\mu}) ^ T\right)\mbf{A} ^ T \\
                &= \mbf{A}\Var(\mbf{X})\mbf{A} ^ T.
            \end{align*}
        \end{enumerate}
    \end{proof}
\end{proposition}

\begin{proposition}
    The variance matrix $\mbf{\Sigma}$ is symmetric and positive semi-definite.
\end{proposition}

\subsection{A change of variables}

\subsubsection{Univariate Transforms}
To look at the distribution and density of $Y$ we can use the following methods:
\begin{enumerate}[label = (\Alph*)]
    \item The cumulative distribution function method:
    find and differentiate the cdf of the transformed random variable to recover the pdf.

    \item The change of variables theorem:
    a general result that is easier to apply when the transformation function $y(\cdot)$ obeys certain properties.
\end{enumerate}

\textbf{The cumulative distribution function method}

This method has two simple steps and works for all general problems where we can find $\P(Y \leq y)$:
\begin{enumerate}[label = \arabic*.]
    \item Find $F_Y(y) = \P(Y \leq y)$ by direct probability calculations using $f_X(x)$ and $y(\cdot)$.

    \item Differentiate $F_Y(y)$ with respect to $y$:
    $f_Y(y) = \frac{d}{dy}F_Y(y)$.
\end{enumerate}

\textbf{The $\chi ^ 2$ distribution}
\begin{example}
    If $Z \sim \mathcal{N}(0, 1)$,
    then $Y = Z ^ 2$ is said to have a chi-squared distribution with one degree of freedom,
    $Y \sim \chi_1 ^ 2$.
    Find the pdf of $Y = Z ^ 2$.

    \begin{solution}
        We start by finding the cdf of $Y$:
        \begin{align*}
            F_Y(y) &= \P(Y \leq y) = \P(Z ^ 2< y) \\
            &= \P(-\sqrt{y} \leq Z \leq \sqrt{y}) \\
            &= \Phi(\sqrt{y}) - \Phi(-\sqrt{y}),
        \end{align*}
        where $\Phi(\cdot)$ is the cdf of the standard normal distribution.
        Note $Z$ can be any real number,
        the transformation $Y = Z ^ 2$ is not one-to-one.
        Thus the probability in the second line is more complicated than the single equality that we might naively expect.
        We then find the pdf by differentiation:
        \begin{align*}
            f_Y(y) &= \frac{\mbf{d}}{\mbf{d}y}F_Y(y) \\
            &= \frac{1}{2}y ^ {-\frac{1}{2}}\phi(\sqrt{y}) + \frac{1}{2}y ^ {-\frac{1}{2}\phi(-\sqrt{y})} \\
            &= y ^ {-\frac{1}{2}\phi(\sqrt{y})} \\
            &= \frac{y ^ {-\frac{1}{2}}}{\sqrt{2\pi}}e ^ {-\frac{1}{2}y},
        \end{align*}
        where $\phi(\cdot)$ is the pdf of the standard normal distribution and we have used the facts that $\phi(z) = \frac{\mbf{d}}{\mbf{d}z}\Phi(z)$,
        and $\phi(z)$ is symmetric about $z = 0$.
        Thus $f_Y(y)$ is the pdf of the $\chi_1 ^ 2$ distribution.
    \end{solution}
\end{example}

\textbf{Properties:}

\begin{enumerate}[label = \arabic*.]
    \item If $U \sim \chi_1 ^ 2$ then $\E(U) = 1$ and $\Var(U) = 2$.
    
    \item If $U_1, \dotsc, U_k$ are all iid $\chi_1 ^ 2$ and $V = \sum_{i = 1}^{k}U_i$,
    then $V \sim \chi_k ^ 2$.

    \item If $V \sim \chi_k ^ 2$ then $\E(V) = k$ and $\Var(V) = 2k$.
\end{enumerate}

\textbf{Univariate transformation theorem}

If the function $y(X)$ is monotonic,
invertible and differentiable over the support,
$\chi$,
of $X$,
then we can find a general method to determine the pdf of any such transformation of $X$.

\begin{theorem}[Distribution of a univariate transformation]
    Let $X$ be a continuous random variable with probability density function $f_X(x)$ with support $\chi = [a, b]$.
    Let $Y = y(X)$ where $y(\cdot)$ is a monotonic,
    invertible and differentiable function over $\chi$ with inverse $X = x(Y) = y ^ {-1}(Y)$.
    Then $Y$ has a probability density function $f_Y(y)$ given by
    \[
    f_Y(y) = f_X(x(y))\left|\frac{\mbf{d}x(y)}{\mbf{d}y}\right|,
    \]
    for $y$ in $\mathcal{Y} = [y(a), y(b)]$.

    \begin{proof}
        Using the cdf method to prove this.
        If $Y = y(X)$,
        then the cdf of $Y$ is:
        \[
        F_Y(y) = \P(Y \leq y) = \P(y(X) \leq y).
        \]
        We are given that $y(\cdot)$ is monotonic,
        which leads to two cases:
        $y(\cdot)$ is monotonically increasing,
        $y(\cdot)$ is monotonically decreasing.

        Assume $y(\cdot)$ is increasing,
        then
        \[
        \P(y(X) \leq y) = \P(X \leq y ^ {-1}(y)) = F_X(x(y)),
        \]
        by the definition of the cdf of $X$.
        Now,
        get the pdf by differentiating
        \[
        f_Y(y) = \frac{d}{dy}F_y(y) = \frac{d}{dy}F_X(x(y)) = f_X(x(y))\frac{dx(y)}{dy},
        \]
        by the chain rule.

        For $y(\cdot)$ decreasing,
        we find
        \[
        F_Y(y) = \P(X \geq x(y)) = 1 - F_X(x(y))
        \]
        giving the pdf as
        \[
        f_Y(y) = -f_X(x(y))\frac{dx(y)}{dy},
        \]
        since $x(y)$ is a decreasing function we get $\frac{d}{dy}x(y) < 0$,
        which cancels the minus sign.
        So we can combine the two equations to get the result
        \[
        f_Y(y) = f_X(x(y))\left|\frac{\mbf{d}x(y)}{\mbf{d}y}\right|.
        \]
    \end{proof}
\end{theorem}

\subsubsection{Multivariate Transforms}

\begin{theorem}[Distribution of a multivariate transformation]
    Suppose we have a random $p$-vector $\mbf{X} = (X_1, \dotsc, X_p) ^ T$ with joint continuous pdf $f_{\mbf{X}}(x_1, \dotsc, x_p)$.
    Suppose we create a transformed random variable $\mbf{Y} = (Y_1, \dotsc, Y_p) ^ T$ where each $Y_i$ is a transformation of the $\mbf{X}\,:\, Y_1 = y_1(\mbf{X}), \dotsc, Y_p = y_p(\mbf{X})$.
    So,
    $\mbf{Y} = y(\mbf{X})$,
    where we assume that $y(\cdot)$ is an invertible differentiable transformation with inverse
    \[
    \mbf{X} = y ^ {-1}(\mbf{Y}) = x(\mbf{Y}) = (x_1(\mbf{Y}), \dotsc, x_p(\mbf{Y})).
    \]
    Let the support of the pdf of $X$ be $\chi \subseteq \R ^ p$ then as $X$ varies over $\chi$,
    $Y$ will vary over some set $\mathcal{Y}$ thus $\mathcal{Y}$ is the image of $\chi$.
    Finally,
    let us assume that all of the partial derivatives $\frac{\partial x_i}{\partial y_j}$ exist for all $i, j$ and $y \in \mathcal{Y}$.

    Given $\mbf{X}, \mbf{Y}, y(\mbf{X}), x(\mbf{Y})$ as above,
    then the joint density for $\mbf{Y}$ is given by:
    \[
    f_{\mbf{Y}}(\mbf{y}) = \begin{cases}
        f_{\mbf{X}}(x(\mbf{y}))|\det(\mbf{J})|, & y \in \mathcal{Y} \\
        0, & \text{otherwise}
    \end{cases}
    \]
    where $\mbf{J}$ is the Jacobian matrix of the transformation:
    \[
    \mbf{J} = \begin{bmatrix}
        \frac{\partial x_1}{\partial y_1} & \dotsi & \frac{\partial x_1}{\partial y_p} \\
        \vdots & \ddots & \vdots \\
        \frac{\partial x_p}{\partial y_1} & \dotsi & \frac{\partial x_p}{\partial y_p}
    \end{bmatrix}
    = \left[\frac{\partial x_i}{\partial y_j}\right]_{ij} = \left[\frac{\partial\mbf{x}}{\partial\mbf{y}}\right].
    \]
\end{theorem}

\subsection{The multivariate normal distribution}
\begin{definition}[Multivariate standard normal distribution]
    The random vector $Z = (Z_1, \dotsc, Z_p) ^ T \in \R ^ p$ is multivariate standard normal
    (MVN),
    written $\mbf{Z} \sim \mathcal{N}_p(\mbf{0}, \mbf{I}_p)$,
    if it has pdf:
    \[
    f(\mbf{z}) = \frac{1}{(2\pi) ^ {\frac{p}{2}}}\exp\left\{-\frac{1}{2}\mbf{z} ^ T\mbf{z}\right\}.
    \]
\end{definition}

\begin{proposition}[The multivariate normal
(MVN)
distribution]
    The random $p$-vector $\mbf{X}$ has a multivariate normal
    (MVN)
    distribution with mean $\mbf{\mu}$ and variance $\mbf{\Sigma}$,
    written $\mbf{X} \sim \mathcal{N}_p(\mbf{\mu}, \mbf{\sigma})$ if and only if it has pdf:
    \[
    f(\mbf{x}) = \frac{1}{(2\pi) ^ {\frac{p}{2}}(\det(\mbf{\Sigma)) ^ {\frac{1}{2}}}}\exp\left\{-\frac{1}{2}(\mbf{x} - \mbf{\mu}) ^ T\mbf{\Sigma} ^ {-1}(\mbf{x} - \mbf{\mu})\right\}.
    \]
\end{proposition}

\subsubsection{Distribution of a linear transformation}
\begin{theorem}[Linear transformation of the MVN]
    Suppose $\mbf{X} \sim \mathcal{N}_p(\mbf{\mu}, \mbf{\Sigma})$ and let $\mbf{Y} = \mbf{BX} + \mbf{c}$ where $\mbf{B} \in \R ^ {1 \times p}$ and $\mbf{c} \in \R ^ q$,
    then
    \[
    \mbf{Y} \sim \mathcal{N}_q(\mbf{B\mu} + \mbf{c}, \mbf{B\Sigma B} ^ T)
    \]
    
    \begin{proof}
        From our definition of the MVN,
        we know that we can write the MVN random vector $\mbf{X}$ as $\mbf{X} = \mbf{AZ} + \mbf{\mu}$ where $\mbf{A}$ is $p \times r$ and $\mbf{\mu} \in \R ^ p$,
        so
        \begin{align*}
            \mbf{Y} &= \mbf{BX} + \mbf{c} = \mbf{B}(\mbf{AZ} + \mbf{\mu}) + \mbf{c} \\
            &= \underset{(r \times q)}{(\mbf{BA})}\mbf{Z} + \underset{(q \times 1)}{(\mbf{B\mu} + \mbf{c})}
        \end{align*}
        so,
        by the definition of MVN,
        since $\mbf{Z}$ is a standard MVN then $\mbf{Y}$ must also be MVN with mean $(\mbf{B\mu} + \mbf{c})$ and variance $(\mbf{BA})(\mbf{BA}) ^ T = \mbf{BAA} ^ T\mbf{B} ^ T = \mbf{B\Sigma B} ^ T$,
        i.e.
        \[
        \mbf{Y} \sim \mathcal{N}_q(\mbf{B\mu} + \mbf{c}, \mbf{B\Sigma B} ^ T).
        \]
    \end{proof}
\end{theorem}

\subsubsection{Shape of the density function}
\begin{proposition}
    If $\mbf{Z} = (Z_1, \dotsc, Z_p) ^ T \sim \mathcal{N}_p(\mbf{0}, \mbf{I}_p)$,
    then $\mbf{Z} ^ T\mbf{Z} \sim \chi_p ^ 2$.

    \begin{proof}
        By the previous theorem each $Z_i \sim \mathcal{N}(0, 1)$.
        Furthermore,
        since $\Cov(Z_i, Z_j) = 0$ for $i \neq j$,
        each of the components of $\mbf{Z}$ must be independent.

        Second,
        we can write $\mbf{Z} ^ T\mbf{Z} = \sum_{i = 1}^{p}Z_i ^ 2 = \sum_{i = 1}^{p}U_{i}$,
        where each $U_i$ is $\chi_1 ^ 2$ and the $U_i$ are also independent of each other.
        Then by the properties of the chi-squared distribution,
        $\sum_{i = 1}^{p}U_i \sim \chi_p ^ 2$ as required.
    \end{proof}
\end{proposition}

\section{Multivariate Statistics and Estimation}

\subsection{Multivariate Statistics}

\begin{definition}[Data matrix]
    The measurement of the $j$th variable on the $i$th individual is denotes $x_{ij}$,
    and is stored as the $(i, j)$the element of the $n \times p$ data matrix
    \[
    \mbf{X} = \begin{bmatrix}
        x_{11} & x_{12} & \dotsi & x_{1p} \\
        x_{21} & x_{22} & \dotsi & x_{2p} \\
        \vdots & \vdots & \ddots & \vdots \\
        x_{n1} & x_{n2} & \dotsi & x_{1p}
    \end{bmatrix}
    \]
\end{definition}
The $i$th row of $\mbf{X}$ corresponds to the observation vector of all $p$ variables for the individual $i$.
Observation vectors are column vectors,
we write
\[
\mbf{x}_i = [x_{i1}, x_{i2}, \dotsc, x_{ip}] ^ T = \begin{bmatrix}
    x_{i1} \\
    x_{i2} \\
    \vdots \\
    x_{ip}
\end{bmatrix}.
\]
The $j$th column of $\mbf{X}$ corresponds to all the observations of variable $j$ across the $n$ individuals:
\[
\mbf{x} ^ {(j)} = \begin{bmatrix}
    x_{1j} \\
    x_{2j} \\
    \vdots \\
    x_{nj}
\end{bmatrix}.
\]

We now want to be able to summarise the data about the observations,
we can use the sample mean vector and the sample variance matrix
(details in the following)
to give us some information about the observations.

\begin{definition}[Sample mean vector]
    The sample mean vector of the data matrix $\mbf{X}$ is the $p$-vector given by the sample mean of the observation vectors:
    \[
    \bar{\mbf{x}} = \frac{1}{n}\sum_{i = 1}^{n}\mbf{x}_i,
    \]
    which can be expressed as $\bar{\mbf{x}} = \frac{1}{n}\mbf{X} ^ T\mbf{1}_n$,
    where $\mbf{1}_n$ is the $n$-vector of ones.
\end{definition}

\begin{definition}[Sample variance matrix]
    The sample variance matrix of the data matrix $\mbf{X}$ is a $p \times p$ matrix defined by:
    \begin{align*}
        \mbf{S}_{X} &= \frac{1}{n - 1}\sum_{i = 1}^{n}(\mbf{x}_i - \bar{\mbf{x}})(\mbf{x}_i - \bar{\mbf{x}}) ^ T \\
        &= \frac{1}{n - 1}\left(\sum_{i = 1}^{n}\mbf{x}_i\mbf{x}_i ^ T - n\bar{\mbf{x}}\bar{\mbf{x}} ^ T\right) \\
        &= \frac{1}{n - 1}\left(\mbf{X} ^ T\mbf{X} - n\bar{\mbf{x}}\bar{\mbf{x}} ^ T\right)
    \end{align*}
    which has the sample covariance between $(X_j, X_j)$ for element $(j, k)$,
    $j, k = 1, \dotsc, p$:
    \[
    s_{jk} = \frac{1}{n - 1}\sum_{i = 1}^{n}(x_{ij} - \bar{x}_j)(x_{ik} - \bar{x}_k).
    \]
\end{definition}

\subsection{Point Estimation}

Estimation is where we want to use a statistic $S$ computed from a sample $\mbf{X}$ to say something about a parameter $\theta$ of the population or model.

\begin{definition}[Estimate and estimator]
    An estimator of a population or model parameter $\theta$ is a function of the random sample,
    \[
    \hat{\Theta} = \hat{\Theta}(X_1, \dotsc, X_n),
    \]
    which takes values in the parameter space $\Omega$.

    Once the sample is observed,
    we evaluate its value as the estimate,
    \[
    \hat{\theta} = \hat{\Theta}(x_1, \dotsc, x_n).
    \]
    A point estimator is an estimator which returns a single value as our estimate for $\theta$,
    i.e. a single point in $\Omega$.
\end{definition}

\subsubsection{Evaluating an estimator}

We use the following to assess the quality of the estimate we make:
\begin{enumerate}[label = (\roman*)]
    \item Expectation $\E(\hat{\Theta})$ -
    informative for the location of the estimate,
    tells us what value the possible estimates are centred around if we consider all the possible different samples we might make.

    \item Variance $\Var(\hat{\Theta})$ -
    informative about the spread of the estimates,
    tells us how consistent or variable the estimates are across the different possible samples.

    \item Sampling distribution $f(\hat{\Theta})$ -
    describes the entire probabilistic behaviour of the estimates.
\end{enumerate}

When we take the expectation over all of the possible random samples $\mbf{X}$ we might observe.
This leads to the following definition of bias
\begin{definition}[Bias and Unbiased]
    The bias of an estimator is
    \[
    \mathrm{bias}(\hat{\Theta}, \theta) = \E(\hat{\Theta} - \theta) = \E(\hat{\Theta}) - \theta,
    \]
    where we take expectations with respect to the random sample $\mbf{X}$.

    An estimator is unbiased if $\mathrm{bias}(\hat{\Theta}, \theta) = 0$,
    i.e. if
    \[
    \E(\hat{\Theta}) = \theta.
    \]
\end{definition}

Bias tells us whether our estimator is on average near the parameter we are trying to estimate,
but we also would want to assess and quantify the variability of the possible estimates we could obtain from our random samples.

So,
an unbiased estimate with a high variance would be most likely less useful than a biased one with a small variance.
\begin{definition}[Standard error]
    The standard error of an estimator $\hat{\Theta}$ is the standard deviation of $\hat{\Theta}$:
    \[
    \mathrm{SE}(\hat{\Theta}) = \mathrm{sd}(\hat{\Theta}) = \sqrt{\Var(\hat{\Theta})}.
    \]
\end{definition}

Another measure of the quality of an estimator is the mean-square error:
\begin{definition}[Mean-square error
    (MSE)]
    The mean-square error
    (MSE)
    of an estimator $\hat{\Theta}$ for the parameter $\theta$ is
    \[
    \mathrm{MSE}(\hat{\Theta}, \theta) = \E((\hat{\Theta} - \theta) ^ 2) = \Var(\hat{\Theta}) + \mathrm{bias}(\hat{\Theta}, \theta) ^ 2.
    \]
\end{definition}

Where the standard error measures the average distance of the estimator from its mean $\E(\hat{\Theta})$.
the mean-square error measures the average squared distance from the parameter value $\theta$ itself.
When $\E(\hat{\Theta}) = \theta$ our estimator is unbiased,
then these two measures of error are just a function of each other.

\subsection{Estimating means and variances}

\begin{theorem}[Unbiasedness of the sample mean vector]
    Suppose $\mbf{X}_i \sim [\mbf{\mu}, \mbf{\Sigma}]$ are $n$ independent realisation of a multivariate random $p$-vector,
    with $\mbf{\mu}$ and $\mbf{\Sigma}$ finite,
    then the sample mean vector $\bar{\mbf{X}}$ is unbiased for $\mbf{\mu}$:
    \[
    \E(\bar{\mbf{X}}) = \mbf{\mu}.
    \]

    \begin{proof}
        \[
        \E(\bar{X}) = \E\left(\frac{1}{n}\sum_{i = 1}^{n}X_i\right) = \frac{1}{n}\sum_{i = 1}^{n}\E(X_i) = \frac{1}{n}\sum_{i = 1}^{n}\mu = \mu.
        \]
    \end{proof}
\end{theorem}

\begin{theorem}[Variance of the sample mean]
    For $\mbf{X}_i$ as above
    \[
    \Var(\bar{\mbf{X}}) = \frac{1}{n}\mbf{\Sigma}.
    \]

    \begin{proof}
        \begin{align*}
            \Var(\bar{X}) &= \Var\left(\frac{1}{n}\sum_{i = 1}^{n}X_i\right) \\
            &= \frac{1}{n ^ 2}\sum_{i = 1}^{n}\Var(X_i) &(\text{independence}) \\
            &= \frac{1}{n ^ 2}n\mbf{\Sigma} = \frac{1}{n}\mbf{\Sigma}.
        \end{align*}
    \end{proof}
\end{theorem}

\begin{theorem}[Unbiasedness of the sample variance matrix]
    For $\mbf{X}_i$ as above
    \[
    \E(\mbf{S}_X) = \mbf{\Sigma}
    \]

    \begin{proof}
        We need to show that
        \[
        \E(\mbf{S}_X) = \E\left(\frac{1}{n - 1}\sum_{i = 1}^{n}(\mbf{X}_i - \bar{\mbf{X}})(\mbf{X}_i - \bar{\mbf{X}}) ^ T\right)
        \]
        is equal to the variance matrix $\mbf{\Sigma}$.

        \begin{align*}
            (\mbf{X}_i - \bar{\mbf{X}})(\mbf{X}_i - \bar{\mbf{X}}) ^ T &= (\mbf{X}_i - \mbf{\mu} + \mbf{\mu} - \bar{\mbf{X}})(\mbf{X}_i - \mbf{\mu} + \mbf{\mu} - \bar{\mbf{X}}) ^ T \\
            &= ((\mbf{X}_i - \mbf{\mu}) - (\bar{\mbf{X}} - \mbf{\mu}))((\mbf{X}_i - \mbf{\mu}) - (\bar{\mbf{X}} - \mbf{\mu})) ^ T \\
            &= (\mbf{X}_i - \mbf{\mu})(\mbf{X}_i - \mbf{\mu}) ^ T + (\bar{\mbf{X}} - \mbf{\mu})(\bar{\mbf{X}} - \mbf{\mu}) ^ T \\
            &\phantom{==} -(\mbf{X}_i - \mbf{\mu})(\bar{\mbf{X}} - \mbf{\mu}) ^ T - (\bar{\mbf{X}} - \mbf{\mu})(\mbf{X}_i - \mbf{\mu}) ^ T
            \intertext{Now,
            if we take expectations we can see some terms on the right}
            \E\left((\mbf{X}_i - \bar{\mbf{X}})(\mbf{X}_i - \bar{\mbf{X}}) ^ T\right) &= \E\left((\mbf{X}_i - \mbf{\mu})(\mbf{X}_i - \mbf{\mu}) ^ T\right) + \E\left((\bar{\mbf{X}} - \mbf{\mu})(\bar{\mbf{X}} - \mbf{\mu}) ^ T\right) \\
            &\phantom{==} -\E\left((\mbf{X}_i - \mbf{\mu})(\bar{\mbf{X}} - \mbf{\mu}) ^ T\right) - \E\left((\bar{\mbf{X}} - \mbf{\mu})(\mbf{X}_i - \mbf{\mu}) ^ T\right) \\
            &= \Var(\mbf{X}_i) + \Var(\bar{\mbf{X}}) - \Cov(\mbf{X}_i, \bar{\mbf{X}}) - \Cov(\bar{\mbf{X}}, \mbf{X}_i).
        \end{align*}

        We already know the variances,
        and $\Cov(\bar{\mbf{X}}, \mbf{X}_i) = \Cov(\mbf{X}_i, \bar{\mbf{X}}) ^ T$ so
        \begin{align*}
            \Cov(\mbf{X}_i, \bar{\mbf{X}}) &= \Cov\left(\mbf{X}_i, \frac{1}{n}\sum_{j = 1}^{n}\mbf{X}_j\right) \\
            &= \frac{1}{n}\sum_{j = 1}^{n}\Cov(\mbf{X}_i, \mbf{X}_j) \\
            &= \frac{1}{n}\sum_{j = 1}^{n}\Cov(\mbf{X}_i, \mbf{X}_j) &(\mbf{X}_i \independent \mbf{X}_j \text{ for all } i \neq q) \\
            &= \frac{1}{n}\Var(\mbf{X}_i).
        \end{align*}

        Putting everything back into the expression for the expectation:
        \begin{align*}
            \E\left((\mbf{X}_i - \bar{\mbf{X}})(\mbf{X}_i - \bar{\mbf{X}}) ^ T\right) &= \mbf{\Sigma} + \frac{1}{n}\mbf{\Sigma} - \frac{1}{n}\mbf{\Sigma} - \left(\frac{1}{n}\mbf{\Sigma}\right) ^ T \\
            &= \mbf{\Sigma} - \frac{1}{n}\mbf{\Sigma} &(\mbf{\Sigma}\text{ is symmetric}) \\
            &= \frac{n - 1}{n}\mbf{\Sigma}.
        \end{align*}
        So we can say that
        \begin{align*}
            \E(\mbf{S}_X) &= \frac{1}{n - 1}\sum_{i = 1}^{n}\E\left((\mbf{X}_i - \bar{\mbf{X}})(\mbf{X}_i - \bar{\mbf{X}}) ^ T\right) \\
            &= \frac{1}{n - 1} \times n \times \frac{n - 1}{n}\mbf{\Sigma} \\
            &= \mbf{\Sigma}.
        \end{align*}
    \end{proof}
\end{theorem}

\begin{theorem}[Estimation of means and variances]
    Suppose $\mbf{X}_i$ are $n$ independent realisations of a multivariate random $p$-vector,
    with $\E(\mbf{X}_i) = \mbf{\mu}$ and $\Var(\mbf{X}_i) = \mbf{\Sigma}$.
    Assuming that the elements of $\mbf{\mu}$ and $\mbf{\Sigma}$ are all finite,
    then
    
    \begin{enumerate}[label = (\alph*)]
        \item The sample mean vector $\bar{\mbf{X}}$ is unbiased for $\mbf{\mu}$:
        $\E(\bar{\mbf{X}}) = \mbf{\mu}$.

        \item The variance of $\bar{\mbf{X}}$ is given by $\Var(\bar{\mbf{X}}) = \frac{1}{n}\mbf{\Sigma}$.

        \item The sample variance $\mbf{S}_X$ is unbiased for $\mbf{\Sigma}$:
        $\E(\mbf{S}_X) = \mbf{\Sigma}$.

        \item $\frac{1}{n}\mbf{S}_X$ is therefore an unbiased estimator of $\Var(\bar{X})$.
    \end{enumerate}
\end{theorem}

\subsection{Sampling distributions and confidence intervals}

\begin{definition}[Sampling distribution]
    The probability distribution for a statistic $S(\mbf{X})$ over all possible samples $\mbf{X}$ is known as its sampling distribution,
    $f(s)$,
    and depends on the functional form of $S$,
    the distribution of $\mbf{X}$,
    and the method of sampling.
\end{definition}

\begin{theorem}[Sampling distribution for $\bar{X}$]
    Let $X_1, \dotsc, X_n$ be an iid sample with mean $\mu$ and variance $\sigma ^ 2 > 0$ known.
    then for $n < 1$:

    \begin{enumerate}[label = \arabic*.]
        \item If $n$ is large,
        then by CLT $\bar{X} \sim \mathcal{N}(\mu, \sigma ^ 2 / n)$.

        \item If $X_i$ are iid $\mathcal{N}(\mu, \sigma ^ 2)$,
        then $\bar{X} \sim \mathcal{N}(\mu, \sigma ^ 2 / n)$.
    \end{enumerate}
\end{theorem}

\subsubsection{Confidence intervals}

\begin{definition}[Confidence interval]
    Let $\mbf{X} = (X_1, \dotsc, X_n) ^ T$ be a random sample from a probability distribution $f_X(x \mid \theta)$ with parameter $\theta$.
    A $100(1 - \alpha)\%$ confidence interval for $\theta$ is a random interval $\mathcal{C}(\mbf{X}) = (L(\mbf{X}), U(\mbf{X}))$ formed from the random sample via the random variables $L(\mbf{X})$ and $U(\mbf{X})$ such that
    \begin{align*}
        \P(L(\mbf{X}) < \theta < U(\mbf{X})) &= 1 - \alpha \\
        \P(\mathcal{C}(\mbf{X}) \ni \theta) &= 1 - \alpha.
    \end{align*}
    The probability $(1 - \alpha)$ is known as the confidence level or confidence coefficient,
    and is typically chosen to be close to $1$.
\end{definition}

The confidence interval should be interpreted as follows:
the random interval $[L(\mbf{X}), U(\mbf{X})]$ contains
(or 'covers')
the parameter $\theta$ with probability $1 - \alpha$.

The quantity $(1 - \alpha)$ is known as the confidence level or coverage probability.

\subsubsection{A distribution for \texorpdfstring{$S ^ 2$}{}}
Assuming all of our data $X_1, \dotsc, X_n$ are iid and normally distributed $\mathcal{N}(\mu, \sigma ^ 2)$.
We have the following lemma which helps us to derive a distribution for the sample variance.

\begin{lemma}
    For an iid sample $X_1, \dotsc, X_n$ from a Normal population,
    the random variable $\bar{X} = \frac{1}{n}\sum_{i}X_i$ and the vector of random variables $(X_1 - \bar{X}, \dotsc, X_n - \bar{X})$ are independent.
    Consequently,
    the random variables $\bar{X}$ and $S ^ 2 = \frac{1}{n - 1}\sum_{i}(X_i - \bar{X}) ^ 2$ are also independent.
\end{lemma}

Given the previous lemma,
we have enough information to say something about the random variable $S ^ 2$.

\begin{theorem}[Sampling distribution for $S ^ 2$]
    Let $X_1, \dotsc, X_n$ be an iid sample from a Normal population $\mathcal{N}(\mu, \sigma ^ 2)$ with $n > 1$ and $\sigma > 0$ then:
    \[
    \frac{(n - 1)S ^ 2}{\sigma ^ 2} \sim \chi_{n - 1}^{2}.
    \]

    \begin{proof}
        First note the sample values $X_i$ are iid $\mathcal{N}(\mu, \sigma ^ 2)$ for $i = 1, \dotsc, n$,
        then we have $(X_i - \mu) / \sigma \sim \mathcal{N}(0, 1)$.

        From the definition of a $\chi_{1}^{2}$ distribution we also have that
        \[
        U_i = \left[\frac{X_i - \mu}{\sigma}\right] ^ 2 \sim \chi_{1}^{2}.
        \]
        Since all the $X_i$ are independent of each other,
        then the $U_i$ are also independent of each other.
        Thus
        \[
        \sum_{i = 1}^{n}\left(\frac{X_i - \mu}{\sigma}\right) ^ 2 = \sum_{i = 1}^{n}U_i \sim \chi_{n}^{2}
        \]
        from the definition of the $\chi_{n}^{2}$ distribution.

        Now we will re-write this sum in terms of $S ^ 2$,
        which will allow us to deduce a distribution:
        \begin{align*}
            \sum_{i = 1}^{n}\left(\frac{X_i - \mu}{\sigma}\right) ^ 2 &= \frac{1}{\sigma ^ 2}\sum_{i = 1}^{n}(X_i - \mu) ^ 2 \\
            &= \frac{1}{\sigma ^ 2}\sum_{i = 1}^{n}[(X_i - \bar{X}) + (\bar{X} - \mu)] ^ 2 \\
            &= \frac{1}{\sigma ^ 2}\sum_{i = 1}^{n}(X_i - \bar{X}) ^ 2 + n\left(\frac{\bar{X} - \mu}{\sigma}\right) ^ 2 + \frac{1}{\sigma}\left(\frac{\bar{X} - \mu}{\sigma}\right)\underbrace{\sum_{i = 1}^{n}(X_i - \bar{X})}_{= 0} \\
            &= \frac{n - 1}{\sigma ^ 2}\left[\frac{1}{n - 1}\sum_{i = 1}^{n}(X_i - \bar{X}) ^ 2\right] + \left(\frac{\bar{X} - \mu}{\sigma / \sqrt{n}}\right) ^ 2 \\
            &= \frac{n - 1}{\sigma ^ 2}S ^ 2 + \left(\frac{\bar{X} - \mu}{\sigma / \sqrt{n}}\right) ^ 2.
        \end{align*}

        We have established in the first equation that the variable on the left must have a $\chi_{n}^{2}$ distribution,
        but we have decomposed it into two pieces.
        From the above corollary the two terms are independent,
        as only one involves $\bar{X}$ and the other only involves $S ^ 2$.
        We know that the second term in the sum is a $\chi_{1}^{2}$ random variable as it is the square of a standard Normal random variable.
        Therefore,
        for this sum to have a $\chi_{n}^{2}$ distribution then the first term of the sum must be $\chi_{n - 1}^{2}$ by the definition of a chi-square.
    \end{proof}
\end{theorem}

\begin{definition}[Pivotal statistic]
    A pivotal statistic,
    pivot,
    or pivotal quantity is a function of the data and parameters $G = G(\mbf{X}, \mbf{\theta})$ such that the distribution of $G$ does not depend on the parameters $\mbf{\theta}$.
\end{definition}

Pivotal statistics provide us with an easy mechanism to produce confidence intervals.
given a pivotal statistic $G(\mbf{X}, \mbf{\theta})$,
with sampling density function $f_{G}(g)$ we can simply write down the following probability statement
\[
\P(a \leq G(\mbf{X}, \mbf{\theta}) \leq b) = 1 - \alpha,
\]
use $f_{G}(g)$ to find the constants $a, b$ for a given level of confidence,
and then rearrange the inequalities inside the probability statement so that $\theta$ is sandwiched between an upper and lower limit.

\begin{proposition}[Confidence interval for $\sigma ^ 2$]
    For $X_1, \dotsc, X_n$ iid $\mathcal{N}(\mu, \sigma ^ 2)$,
    a $100(1 - \alpha)\%$ confidence interval for the population variance,
    $\sigma ^ 2$,
    is given by
    \[
    \left[\frac{(n - 1)S ^ 2}{\chi_{\alpha / 2, n - 1}^{*}}, \frac{(n - 1)S ^ 2}{\chi_{1 - \alpha / 2, n - 1}^{*}}\right] = (n - 1)S ^ 2\left[\frac{1}{\chi_{\alpha / 2, n - 1}^{*}}, \frac{1}{\chi_{1 - \alpha / 2, n - 1}^{*}}\right],
    \]
    where $\chi_{a, n - 1}^{*}$ corresponds to the point with probability $a$ lying above it for a $\chi ^ 2$ distribution with $(n - 1)$ degrees of freedom,
    i.e.
    $\P(X_{n - 1}^{2} > \chi_{a, n - 1}^{*}) = a$.

    \begin{proof}
        Using the sampling distribution for $S ^ 2$ we get
        \[
        \frac{(n - 1)S ^ 2}{\sigma ^ 2} \sim \chi_{n - 1}^{2},
        \]
        we can then write
        \begin{align*}
            1 - \alpha &= \P\left(\chi_{1 - \alpha / 2}^{*}(n - 1) < \frac{(n - 1)}{\sigma ^ 2}S ^ 2 < \chi_{\alpha / 2}^{*}(n - 1)\right) \\
            &= \P\left(\frac{1}{\chi_{\alpha / 2}^{*}(n - 1)} < \frac{\sigma ^ 2}{(n - 1)S ^ 2} < \frac{1}{\chi_{1 - \alpha / 2}^{*}(n - 1)}\right) \\
            &= \P\left(\frac{(n - 1)S ^ 2}{\chi_{\alpha / 2}^{*}(n - 1)} < \sigma ^ 2 < \frac{(n -  1)S ^ 2}{\chi_{1 - \alpha / 2}^{*}(n - 1)}\right).
        \end{align*}

        Note the positions of the critical values of the $\chi_{n - 1}^{2}$ in the confidence limits are reversed from what we might expect due to taking the reciprocal at the second step.
    \end{proof}
\end{proposition}

\subsubsection{A distribution for \texorpdfstring{$\bar{X}$}{} when \texorpdfstring{$\sigma ^ 2$}{} is unknown}

\begin{definition}[(Student's)
    $t$ distribution]
    If $Z \sim \mathcal{N}(0, 1)$ and $U \sim \chi_{k}^{2}$,
    and $Z$ is independent of $U$,
    then the variable
    \[
    T = \frac{Z}{\sqrt{U / k}}
    \]
    has a
    (Student's)
    $t$ distribution with $k$ degrees of freedom,
    $T \sim t_k$,
    with pdf
    \[
    f(t\mid k) = \frac{\Gamma\left(\frac{k + 1}{2}\right)}{\sqrt{k\pi}\Gamma\left(\frac{k}{2}\right)}\left(1 + \frac{x ^ 2}{k}\right) ^ {-\frac{k + 1}{2}}.
    \]
\end{definition}

\begin{proposition}[The $t$-statistic]
    For $X_1, \dotsc, X_n$ iid $\mathcal{N}(\mu, \sigma ^ 2)$,
    the random variable $T$ defined as
    \[
    T = \frac{\bar{X} - \mu}{S / \sqrt{n}}
    \]
    has a $t$ distribution with $(n - 1)$ degrees of freedom,
    $T \sim t_{n - 1}$.

    \begin{proof}
        We can rearrange the expression for $T$ to give us the right form for a $t$ distribution as follows:
        \begin{align*}
            \left(\frac{\bar{X} - \mu}{S / \sqrt{n}}\right) &= \left(\frac{\bar{X} - \mu}{S / \sqrt{n}}\right)\left(\frac{\sigma / \sqrt{n}}{\sigma / \sqrt{n}}\right) \\
            &= \left(\frac{(\bar{X} - \mu) / (\sigma / \sqrt{n})}{S / \sigma}\right) \\
            &= \frac{Z}{\sqrt{S ^ 2 / \sigma ^ 2}} = \frac{Z}{\sqrt{\frac{(n - 1)S ^ 2}{(n - 1)\sigma ^ 2}}} \\
            &= \frac{Z}{\sqrt{U / (n - 1)}} \\
            &= T
        \end{align*}
        where $Z = \frac{\bar{X} - \mu}{\sigma / \sqrt{n}} \sim \mathcal{N}(0, 1)$,
        $U = \frac{(n - 1)S ^ 2}{\sigma ^ 2} \sim \chi_{n - 1}^{2}$.
        So by the above definition we must have $T \sim t_{n - 1}$.
    \end{proof}
\end{proposition}

\begin{proposition}[Confidence interval for $\mu$ when $\sigma ^ 2$ is unknown]
    For $X_1, \dotsc, X_n$ iid $\mathcal{N}(\mu, \sigma ^ 2)$,
    the $100(1 - \alpha)\%$ confidence interval for $\mu$ when $\sigma ^ 2$ is unknown is
    \[
    \bar{X} \pm t_{n - 1, \alpha / 2}^{*}\frac{s}{\sqrt{n}}.
    \]
\end{proposition}

\subsubsection{The multivariate setting}

\begin{proposition}[Confidence regions for the multivariate Normal mean]
    For an iid sample $\mbf{X}_1, \dotsc, \mbf{X}_n$ from a multivariate Normal population $\mathcal{N}_p(\mbf{\mu}, \mbf{\Sigma})$ with $n > 1$ and known $\mbf{\Sigma} > 0$:
    \[
    \left\{\mbf{\mu}\mid (\bar{\mbf{x}} - \mbf{\mu}) ^ T\left(\frac{1}{n}\mbf{\Sigma}\right) ^ {-1}(\hat{\mbf{x}} - \mbf{\mu}) \leq \chi_{\alpha, p}^{*}\right\}
    \]
    is a $100(1 - \alpha)\%$ confidence region for $\mbf{\mu}$.

    \begin{proof}
        For the multivariate sample mean vector $\bar{\mbf{X}}$,
        and assuming a known variance,
        we have from the above results $\bar{\mbf{X}} \sim \mathcal{N}_p(\mbf{\mu}, \frac{1}{n}\mbf{\Sigma})$.
        We also established that if $\mbf{X} \sim \mathcal{N}_p(\mbf{\mu}, \mbf{V})$ then $\mbf{V} ^ {-\frac{1}{2}}(\mbf{X} - \mbf{\mu}) \sim \mathcal{N}_p(\mbf{0}, \mbf{I}_p)$ and if $\mbf{Z} \sum \mathcal{N}_p(\mbf{0}, \mbf{I}_p)$ then $\mbf{Z} ^ T\mbf{Z} \sum \chi_{p}^{2}$.

        Combining these results we get
        \begin{align*}
            \left[\frac{1}{\sqrt{n}}\mbf{\Sigma} ^ {-\frac{1}{2}}\right](\bar{X} - \mbf{\mu}) &\sim \mathcal{N}_p(\mbf{0}, \mbf{I}_p) \\
            \implies \left[\left(\frac{1}{\sqrt{n}}\mbf{\Sigma}\right) ^ {-\frac{1}{2}}(\bar{X} - \mbf{\mu})\right] ^ T\left[\left(\frac{1}{\sqrt{n}}\mbf{\Sigma}\right) ^ {-\frac{1}{2}}(\bar{X} - \mbf{\mu})\right] &= (\bar{X} - \mbf{\mu}) ^ T\left(\frac{1}{\sqrt{n}}\mbf{\Sigma}\right) ^ {-\frac{1}{2}}\left(\frac{1}{\sqrt{n}}\mbf{\Sigma}\right) ^ {-\frac{1}{2}}(\bar{X} - \mbf{\mu}) \sim \chi_{p}^{2} \\
            &= (\bar{X} - \mbf{\mu}) ^ T\left[\frac{1}{n}\mbf{\Sigma}\right] ^ {-1}(\bar{X} - \mbf{\mu}) \sim \chi_{p}^{2}.
        \end{align*}
        This multivariate statistic is a bit more unwieldy,
        it is still pivotal as $\chi_{p}^{2}$ doesn't involve any of the parameters.
        So,
        for a level of $(1 - \alpha)$ we can find the following region for $\bar{X}$ with the desired probability
        \[
        \P\left[\bar{X} \in \left\{\bar{x}\mid (\bar{x} - \mbf{\mu}) ^ T\left[\frac{1}{n}\mbf{\Sigma}\right] ^ {-1}(\bar{x} - \mbf{\mu}) \leq \chi_{\alpha, p}^{*}\right\}\right] = 1 - \alpha.
        \]
        Then,
        by the usual argument,
        a $100(1 - \alpha)\%$ confidence region for $\mbf{\mu}$ is
        \[
        \left\{\mbf{\mu} \mid (\bar{x} - \mbf{\mu}) ^ T\left[\frac{1}{n}\mbf{\Sigma}\right] ^ {-1}(\bar{x} - \mbf{\mu}) \leq \chi_{\alpha, p}^{*}\right\}.
        \]
    \end{proof}
\end{proposition}
