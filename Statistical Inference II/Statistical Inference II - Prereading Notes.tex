\section{Multivariate probability}

\begin{definition}[Random vector]
    If $X_1, \dotsc, X_p$ is a collection of real-valued random variables,
    then
    \[
    \mbf{X} = \begin{bmatrix}
        X_1 \\ \vdots \\ X_p
    \end{bmatrix} = (X_1, \dotsc, X_p) ^ T,
    \]
    is a $p$-dimensional random vector in $\R ^ p$,
    and $\mbf{x} = (x_1, \dotsc, x_p) ^ T$ is a particular realisation of $\mbf{X}$.
\end{definition}

\begin{definition}[PDF of a random vector]
    The pdf of a random vector $\mbf{X} = (X_1, \dotsc, X_p) ^ T$ is the joint probability density function
    (also just density or pdf)
    $f(X_1, \dotsc, X_p)$ from $\R ^ p$ to $\R_{0}^{+} = [0, \infty)$ such that:
    \begin{enumerate}[label = \roman*)]
        \item $f(\mbf{x}) \geq 0$ for all $\mbf{x} \in \R ^ p$.

        \item $\int_{\R ^ p}f(\mbf{x})\,d\mbf{x} = \int_{-\infty}^{\infty}\dotsi\int_{-\infty}^{\infty}f(X_1, \dotsc, X_p)\,dX_1\dotsc dX_p = 1$.

        \item For any subset $\mathcal{A}$ of $\R ^ p = \P(\mbf{X} \in \mathcal{A}) = \int_{\mathcal{A}}f(\mbf{x})\,d\mbf{x}$.
    \end{enumerate}
\end{definition}

\begin{definition}[CDF of a random vector]
    Similarly,
    the joint cdf is defined as
    \[
    F(\mbf{x}) = \P(X_1 \leq x_1, \dotsc, X_p \leq x_p) = \int_{-\infty}^{x_1} \dotsi \int_{-\infty}^{x_p}f(x_1', \dotsc, x_p')\,dx_p'\dotsc dx_1',
    \]
    and we can recover the pdf by
    (partial)
    differentiation
    \[
    f(\mbf{x}) = \frac{\partial ^ p}{\partial x_1, \dotsc, \partial x_p}F(\mbf{x}).
    \]
\end{definition}

\subsection{Key operations and relations}
\begin{example}
    Consider the bivariate
    $(p = 2)$ pdf:
    \[
    f_{X, Y}(x, y) = \begin{cases}
        \lambda ^ 2e ^ {-\lambda y}, &0 < x < y,\text{ and } \lambda > 0 \\
        0, &\text{otherwise}.
    \end{cases}
    \]
    Is this a valid pdf?

    \begin{solution}
        We know $\lambda ^ 2 > 0$ and $e ^ x > 0$ for all $x \in \R$,
        so we have $f_{X, Y}(x, y) \geq 0$ for all $(x, y) \in \R ^ 2$ hence we have positivity.

        Now we need to check that the integral integrates to $1$ over $\R ^ 2$:
        \begin{align*}
            \int_{-\infty}^{\infty}\int_{-\infty}^{\infty}f_{X, Y}(x, y)\,dxdy &= \int_{0}^{\infty}\int_{0}^{y}\lambda ^ 2e ^ {-\lambda y}\,dxdy \\
            &= \int_{0}^{\infty}\lambda ^ 2e ^ {-\lambda y}\int_{0}^{y}\,dxdy \\
            &= \int_{0}^{\infty}\lambda ^ 2e ^ {-\lambda y}y\,dy \\
            &= \left[-(\lambda y + 1)e ^ {-\lambda y}\right]_{0}^{\infty} \\
            &= 0 - (-1) = 1.
        \end{align*}
        Hence,
        the integral equals $1$ and we have positivity so this is a valid pdf.
    \end{solution}
\end{example}

\begin{definition}[Marginal pdf]
    A sub-collection $(X_1, \dotsc, X_q)$ of $(X_1, \dotsc, X_p)$,
    where $q < p$,
    has a marginal pdf defined as
    \[
    f(x_1, \dotsc, x_q) = \int_{-\infty}^{\infty}\dotsi\int_{-\infty}^{\infty}f(x_1, \dotsc, x_p)\,dx_{q + 1}\dotsc dx_p. 
    \]
    For discrete random variables,
    the same definition holds with integrals replaced by sums.
\end{definition}

\begin{definition}[Conditional pdf]
    If we know the values $X_{q + 1} = x_{q + 1}, \dotsc, X_p = x_p$,
    but not the values of $X_1, \dotsc, X_q$,
    then the $X_1, \dotsc, X_q$ are still random variables and have a conditional pdf
    \[
    f(x_1, \dotsc, x_q\mid x_{q + 1}, \dotsc, x_p) = \frac{f(x_1, \dotsc, x_q, x_{q + 1}, \dotsc, x_p)}{f(x_{q + 1}, \dotsc, x_p)}.
    \]
\end{definition}

\begin{definition}[Independence]
    A collection of random variables $(X_1, \dotsc, X_p)$ is said to be
    (mutually)
    independent if and only if their joint pdf factorises as
    \[
    f(x_1, x_2, \dotsc, x_p) = f_1(x_1)f_2(x_2)\dotsc f_p(x_p),
    \]
    for all $(x_1, x_2, \dotsc, x_p) \in \R ^ o$.
    We will sometimes write this as $X_1 \independent X_2 \dotsc \independent X_p$ or $X_i \independent X_j\ \forall i \neq j$,
    where the symbol $\independent$ means 'is independent of'.

    An equivalent statement,
    is that the conditional densities are equal to the marginal densities:
    \[
    f(x_i \mid x_1, \dotsc, x_{i - 1}, x_{i + 1}, \dotsc, x_p) = f(x_i),
    \]
    for all $i = 1, \dotsc, p$.
\end{definition}

\begin{definition}[Random sample,
    IID]
    The random variables $X_1, \dotsc, X_p$ are called independent and identically distributed
    (iid)
    random variables with pdf or pmf $f(x)$ if they are all independent of each other,
    and the marginal pdf or pmf of each $X_i$ is the same function $f(x)$.
    This means that the joint pdf simplifies to
    \[
    f(x_1, x_2, \dotsc, x_p) = f(x_1)f(x_2) \dotsc f(x_p) = \prod_{i = 1}^{p}f(x_i).
    \]
    Alternatively,
    $X_1, \dotsc, X_p$ are called a random sample of size $n$ from the distribution
    (or population)
    $f(x)$.
\end{definition}

\section{Expectations,
means,
and variances}

\subsection{Expectation of a scalar function of multiple random variables}

\begin{definition}[The honesty condition]
    The honesty condition for a pdf
    (or pmf)
    says that the integral
    (or sum)
    of a valid pdf over all possible values must equal $1$.
\end{definition}

\begin{remark}
    When working with probability densities if we can rearrange our integral
    (or part of it)
    to be the integral of a known valid pdf over its entire range,
    then we can immediately state that it must integrate to one.
\end{remark}

\subsection{Expectation and variance of a random vector}

\begin{definition}[Mean vector]
    The expectation $\E(\mbf{X})$ of a random vector $\mbf{X} = (X_1, \dotsc, X_p)$ is defined as:
    \[
    \E(\mbf{X}) = \int\mbf{x}f(\mbf{x})\,d\mbf{x} = \begin{bmatrix}
        \int_{-\infty}^{\infty}x_1f(\mbf{x})\,d\mbf{x} \\
        \vdots \\
        \int_{-\infty}^{\infty}x_pf(\mbf{x})\,d\mbf{x}
    \end{bmatrix} = \begin{bmatrix}
        \E(X_1) \\ \vdots \\ \E(X_p)
    \end{bmatrix} = \mbf{\mu},
    \]
    which is simply the $p$-dimensional vector whose components are the expectations of the individual components of $\mbf{x}$ and is known as the mean vector,
    $\mbf{\mu}$,
    of $\mbf{X}$.
\end{definition}

\begin{definition}[Variance matrix]
    The variance matrix of the random vector $\mbf{X}$ is defined to be the $p \times p$ matrix such that,
    for $i, j = 1, \dotsc, p$,
    the $(i, j)$ element of the matrix is $\Cov(X_i, X_j)$:
    \[
    \Var(\mbf{x}) = \E((\mbf{X} - \mbf{\mu})(\mbf{x} - \mbf{\mu}) ^ T) = \E(\mbf{X}\mbf{X} ^ T) - \mbf{\mu}\mbf{\mu} ^ T = \begin{bmatrix}
        \Sigma_{11} & \dotsi & \Sigma_{1p} \\
        \vdots & \ddots & \vdots \\
        \Sigma_{p1} & \dotsi & \Sigma_{pp}
    \end{bmatrix} = \mbf{\Sigma},
    \]
    where the components of the matrix comprise the individual variances and covariances of the random variables:
    \begin{align*}
        \Sigma_{jj} &= \E(X_j ^ 2) - \E(X_j) ^ 2 = \Var(X_j) = \sigma_j ^ 2 \\
        \Sigma_{ij} &= \E(X_iX_j) - \E(X_i)\E(X_j) = \Cov(X_i, X_j) = \Cov(X_j, X_i) = \Sigma_{ji}.
    \end{align*}
\end{definition}

\begin{definition}[Covariance and correlation]
    The covariance,
    $\Cov(X_i, X_j)$ between a pair of random variables $X_i$ and $X_j$ is a measure of the strength of linear association between the two variables,
    defined as:
    \begin{align*}
        \Cov(X_i, X_j) &= \E((X_i - \E(X_i))(X_j - \E(X_j))) = \Cov(X_j, X_i) \\
        &= \sigma_i\sigma_j\rho_{ij},
    \end{align*}
    where $\sigma_i, \sigma_j$ are the standard deviations of $X_i$ and $X_j$ respectively,
    and $\rho_{ij}$ is the correlation between $X_i$ and $X_j$.

    The correlation $\rho_{ij}$ between $X_i$ and $X_j$ is a unit-less measure of association which quantifies the extent to which two random variables are linearly related.
    The correlation is such that $\rho_{ij} \in [-1, 1]$,
    with values further from zero indicating stronger association.
    If $X_1$ and $X_2$ are independent then $\rho_{ij} = 0$,
    but the reverse is not true in general.
\end{definition}

\subsection{Data,
sample means and variances}

\begin{definition}[Data matrix]
    The measurement of the $j$th variable on the $i$th individual is denoted $x_{ij}$,
    and is stored as the $(i, j)$th element of the $n \times p$ data matrix
    \[
    \mbf{X} = \begin{bmatrix}
        x_{11} & x_{12} & \dotsi & x_{1p} \\
        x_{21} & x_{22} & \dotsi & x_{2p} \\
        \vdots & \vdots & \ddots & \vdots \\
        x_{n1} & x_{n2} & \dotsi & x_{np} \\
    \end{bmatrix}.
    \]
\end{definition}
The $i$th row of $\mbf{X}$ is the observation vector of all $p$ variables for individual $i$,
$\mbf{x}_i$,
and then the $j$th column of $\mbf{X}$ corresponds to all of the observation of variable $j$ across the $n$ individuals.

\section{Estimation}

\subsection{Point Estimation}
\begin{definition}[Estimate and estimator]
    An estimator or a population or model parameter $\theta$ is a random variable which is a function of the random sample,
    i.e. a statistic,
    \[
    \hat{\Theta} = \hat{\Theta}(X_1, \dotsc, X_n),
    \]
    which takes values in the parameter space $\Omega$.

    Once the sample is observed,
    the value of the estimator can be found and its value is the estimate,
    \[
    \hat{\theta} = \hat{\Theta}(x_1, \dotsc, x_n).
    \]
    A point estimator is any estimator which returns a single value as our estimate for $\theta$,
    i.e. a single point in $\Omega$.
\end{definition}

The frequentist approach to statistics is based on the following:
\begin{enumerate}[label = (\roman*)]
    \item A limiting frequency interpretation of probability.
    Probabilities are associated only with observable events,
    and are determined as the limit of relative frequencies in infinitely many trials.

    \item The parameter of interest,
    $\theta$,
    is the unknown but constant.
    The parameter is not an observable event,
    it cannot be viewed as a random variable,
    can not have a probability distribution,
    so we cannot make probability statements about $\theta$.

    \item Before observation the sample $\vec{X} = [X_1, \dotsc, X_n] ^ T$ is random,
    and can be considered to be any one of many possible samples $\vec{X} ^ {(1)}, \vec{X} ^ {(2)}, \dotsc$ consistent with $f(\vec{x})$.

    \item After observation the observed sample $\vec{x} = [x_1, \dotsc, x_n] ^ T$ is a constant realisation of the random $\vec{X}$.
    It gives a constant estimate $\hat{\theta} = \hat{\Theta}(\vec{x})$ as a single realisation of the random $\hat{\Theta}$.

    \item We equate the accuracy of $\hat{\theta}$ with the variability of $\hat{\Theta}$.
\end{enumerate}

\subsection{Evaluating an estimator}

\begin{definition}[Bias and Unbiased]
    The bias of an estimator is
    \[
    \mathrm{bias}(\hat{\Theta}, \theta) = \E(\hat{\Theta} - \theta) = \E(\hat{\Theta}) - \theta,
    \]
    where we take expectation with respect to the random sample $\vec{X}$.

    An estimator is unbiased if $\mathrm{bias}(\hat{\Theta}, \theta) = 0$,
    i.e. if
    \[
    \E(\hat{\Theta}) = \theta.
    \]
\end{definition}

\begin{definition}[Standard error]
    The standard error of an estimator $\hat{\Theta}$ is the standard deviation of $\hat{\Theta}$:
    \[
    \mathrm{SE}(\hat{\Theta}) = \mathrm{sd}(\hat{\Theta}) = \sqrt{\Var(\hat{\Theta})}.
    \]
\end{definition}
Standard error is used as a measure of the variability of the estimator.

Another measure of the quality of an estimator is the mean-square error:
\begin{definition}[Mean-square error
    (MSE)]
    The mean-square error
    (MSE)
    of an estimator $\hat{\Theta}$ for the parameter $\theta$ is
    \[
    \mathrm{MSE}(\hat{\Theta}, \theta) = \E((\hat{\Theta} - \theta) ^ 2) = \Var(\hat{\Theta}) + \mathrm{bias}(\hat{\Theta}, \theta) ^ 2.
    \]
\end{definition}
This is where the standard error measures the average distance of the estimator from its mean $\E(\hat{\Theta})$.

\subsection{The method of moments}

\begin{definition}[Central moments]
    The $k$th central moment of $X$ is:
    \[
    \mathbb{c}_k(\theta) = \begin{cases}
        \E(X) = \mu & k = 1 \\
        \E((X - \mu) ^ 2) = \Var(X) = \sigma ^ 2 & k = 2 \\
        \E((X - \mu) ^ k) & k = 3, 4, \dotsc
    \end{cases}
    \]
    The $k$th sample central moment of $X$ is:
    \[
    \hat{c}_k(\theta) = \begin{cases}
        \frac{1}{n}\sum_{i = 1}^{n}x_i = \bar{x} & k = 1 \\
        \frac{1}{n}\sum_{i = 1}^{n}(x_i - \bar{x}) ^ k & k = 2, 3, \dotsc
    \end{cases}
    \]
\end{definition}

\begin{definition}
    If $f(x; \theta)$ has $p$ parameters,
    $\vec{\theta} \in \R ^ p$,
    the method of moments
    (MoM)
    estimators of $\vec{\theta}$ are obtained by equating:
    \[
    \mathbb{c}_k(\hvec{\theta}) = \hat{c}_{\vec{X}}(,)
    \]
    for $k = 1, \dotsc, p$,
    and solving for $\hvec{\theta}$.
\end{definition}

\section{Estimating means and variances}

\subsection{Properties of the sample mean and variance}

\begin{proposition}[Unbiasedness of the sample mean vector]
    Suppose $\vec{X}_i \sim [\vec{\mu}, \vec{\Sigma}]$ are $n$ independent realisation of a multivariate random $p$-vector,
    with $\vec{\mu}$ and $\vec{\Sigma}$ finite,
    then the sample mean vector $\bar{\vec{X}}$ is unbiased for $\vec{\mu}$:
    \[
    \E(\bar{\vec{X}}) = \vec{\mu}.
    \]

    \begin{proof}
        \begin{align*}
            \E(\bar{X}) &= \E\left(\frac{1}{n}\sum_{i = 1}^{n}X_i\right) \\
            &= \frac{1}{n}\sum_{i = 1}^{n}\E(X_i) \\
            &= \frac{1}{n}\sum_{i = 1}^{n}\mu = \mu.
        \end{align*}
    \end{proof}
\end{proposition}

\begin{proposition}[Variance of the sample mean]
    For $\vec{X}_i$ as above
    \[
    \Var(\bar{\vec{X}}) = \frac{1}{n}\vec{\Sigma}.
    \]

    \begin{proof}
        \begin{align*}
            \Var(\bar{X}) &= \Var\left(\frac{1}{n}\sum_{i = 1}^{n}X_i\right) \\
            \intertext{Since the $X_i$ are all independent:}
            \Var(\bar{X}) &= \frac{1}{n ^ 2}\sum_{i = 1}^{n}\Var(X_i) \\
            &= \frac{1}{n ^ 2}n\vec{\Sigma} \\
            &= \frac{1}{n}\vec{\Sigma}.
        \end{align*}
    \end{proof}
\end{proposition}

\begin{proposition}[Unbiasedness of the sample variance matrix]
    For $\vec{X}_i$ as above
    \[
    \E(\vec{S}_X) = \vec{\Sigma}.
    \]
\end{proposition}

\begin{definition}[Standard error]
    The standard error of an estimator $\hat{\Theta}$ is the standard deviation of $\hat{\Theta}$.
    \[
    \mathrm{SE}(\hat{\Theta}) = \mathrm{sd}(\hat{\Theta}) = \sqrt{\Var(\hat{\Theta})}.
    \]
\end{definition}

\begin{definition}[Mean-square error
    (MSE)]
    The mean-square error
    (MSE)
    of an estimator $\hat{\Theta}$ for the parameter $\theta$ is
    \[
    \mathrm{MSE}(\hat{\Theta}, \theta) = \E((\hat{\Theta} - \theta) ^ 2) = \Var(\hat{\Theta}) + \bias(\hat{\Theta}, \theta) ^ 2.
    \]
\end{definition}

When $\E(\hat{\Theta}) = \theta$ the estimator is unbiased.

how can we find a suitable estimate $\hvec{\theta}$ or $\theta$?

\begin{definition}[Central moments]
    The $k$th central moment of $X$ is:
    \[
    \mathbb{c}_k(\theta) = \begin{cases}
        \E(X) = \mu & k = 1 \\
        \E((X - \mu) ^ 2) = \Var(X) = \sigma ^ 2 & k = 2 \\

\end{definition}

\begin{definition}
    If $f(x; \theta)$ has $p$ parameters,
    $\vec{\theta} \in \R ^ p$,

    (MoM)
    estimators of $\vec{\theta}$ are obtained by equating:

\section{Sampling distributions and confidence intervals}

\subsection{Sampling distributions}

\begin{definition}[Sampling distribution]
    The probability for a statistic $S(\vec{X})$ over all possible sample $\vec{X}$ is known as its sampling distribution,
    $f(s)$,
    and depends on the functional form of $S$,
    the distribution of $\vec{X}$,
    and the method of sampling.
\end{definition}

\begin{proposition}[Sampling Distribution for $\bar{X}$]
    Let $X_1, \dotsc, X_n$ be an IID sample with mean $\mu$ and variance $\sigma ^ 2 > 0$ known.
    Then for $n \geq 1$:
    \begin{enumerate}[label = (\roman*)]
        \item If $n$ is large,
        then by CLT $\bar{X} \sum \mathcal{N}(\mu, \sigma ^ 2 / n)$.

        \item If $X_i$ are IID $\mathcal{N}(\mu, \sigma ^ 2)$,
        then $\bar{X} \sim \mathcal{N}(\mu, \sigma ^ 2 / n)$ exactly regardless of $n$.
    \end{enumerate}
\end{proposition}

\subsection{Confidence intervals}
\begin{definition}[Confidence interval]
    Let $\vec{X} = [X_1, \dotsc, X_n] ^ T$ be a random sample from a probability distribution $f_{X}(\vec{x}; \theta)$ with parameter $\theta$.

    A $100(1 - \alpha)\%$ confidence interval for $\theta$ is a random interval,
    $\mathcal{C}(\vec{X}) = [\mathcal{L}(\vec{X}), \mathcal{U}(\vec{X})]$,
    formed from a random sample,
    $\vec{X}$.
    via the random variables $\mathcal{L}(\vec{X})$ and $\mathcal{U}(\vec{X})$ such that
    \begin{align*}
        \P(\mathcal{L}(\vec{X}) < \theta < \mathcal{U}(\vec{X})) &= 1 - \alpha \\
        \P(\mathcal{C}(\vec{X}) \ni \theta) &= 1 - \alpha.
    \end{align*}
    The probability $(1 - \alpha)$ is known as the confidence level,
    and is typically chosen to be close to $1$.
\end{definition}

\subsection{Pivotal statistics}

\begin{definition}[Pivotal statistic]
    A pivotal statistic,
    pivot,
    or pivotal quantity is a function of the data and parameters $G = G(\vec{X}, \vec{\theta})$ such that the distribution of $g$ does not depend on the parameters $\vec{\theta}$.
\end{definition}

Given a pivotal statistic $G(\vec{X}, \vec{\theta})$,
with sampling density function $f_{G}(g)$ we can simply write down the following probability statement
\[
\P(a \leq G(\vec{X}, \vec{\theta}) \leq b) = 1 - \alpha,
\]
use $f_{G}(g)$ to find the constants $a$ and $b$ for a given level of confidence,
then rearrange the inequalities inside the probability statement so that $\theta$ is sandwiched between an upper and lower limit.

\section{Distributions in Normal sampling}

\subsection{A distribution for $S ^ 2$}

\begin{proposition}
    For an IID sample $X_1, \dotsc, X_n$ from a Normal population,
    the random variable $\bar{X} = \frac{1}{n}\sum_{i}X_i$ and the vector of random variables $(X_1 - \bar{X}, \dotsc, X_n - \bar{X})$ are independent.

    Consequently,
    the random variables $\bar{X}$ and $S ^ 2 = \frac{1}{n - 1}\sum_{i}(X_i - \bar{X}) ^ 2$ are also independent.
\end{proposition}

\begin{proposition}[Sampling distribution for $S ^ 2$]
    Let $X_1, \dotsc, X_n$ be an IID sample from a Normal population $\mathcal{N}(\mu, \sigma ^ 2)$ with $n > 1$ and $\sigma > 0$ then:
    \[
    \frac{(n - 1)S ^ 2}{\sigma ^ 2} \sim \chi_{n - 1}^{2}.
    \]

    \begin{proof}
        First note that since the sample values $X_i$ are IID $\mathcal{N}(\mu, \sigma ^ 2)$ for $i = 1, \dotsc, n$,
        then we have that $(X_i - \mu) / \sigma \sim \mathcal{N}(0, 1)$.
        Consequently,
        from the definition of a $\chi_{1}^{2}$ distribution,
        we also have that
        \[
        \mathcal{U}_i = \left[\frac{X_i - \mu}{\sigma}\right] ^ 2 \sim \chi_{1}^{2}.
        \]
        Since all the $X_i$ are independent of each other,
        then the $\mathcal{U}_i$ are also independent of each other.
        Thus
        \[
        \sum_{i = 1}^{n}\left(\frac{X_i - \mu}{\sigma}\right) ^ 2 = \sum_{i = 1}^{n}\mathcal{U}_i \sim \chi_{n}^{2}
        \]
        from the definition of the $\chi_{n}^{2}$ distribution.

        We re-write this sum in terms of $S ^ 2$,
        which will allow us to deduce a distribution:
        \begin{align*}
            \sum_{i = 1}^{n}\left(\frac{X_i - \mu}{\sigma}\right) ^ 2 &= \frac{1}{\sigma ^ 2}\sum_{i = 1}^{n}(X_i - \mu) ^ 2 \\
            &= \frac{1}{\sigma ^ 2}\sum_{i = 1}^{n}\left[(X_i - \bar{X}) + (\bar{X} - \mu)\right] ^ 2 \\
            &= \frac{1}{\sigma ^ 2}\sum_{i = 1}^{n}(X_i - \bar{X}) ^ 2 + n\left(\frac{\bar{X} - \mu}{\sigma}\right) + \frac{1}{\sigma}\left(\frac{\bar{X} - \mu}{\sigma}\right)\sum_{i = 1}(X_i - \bar{X})
        \end{align*}
    \end{proof}
\end{proposition}
