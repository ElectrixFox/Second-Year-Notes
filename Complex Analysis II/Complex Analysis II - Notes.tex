\section{The Complex Plane and Complex Functions}

\subsection{Complex Numbers}

Complex numbers are of the form $z = x + iy$ where $i$ is the imaginary unit,
$i = \sqrt{-1}$.
The set of complex numbers is denoted $\C$.
We use an Argand diagram to visualise complex numbers $x + iy$ with $(x, y)$.

For $z = x + iy$ we have the real part of $z$,
$\mathrm{Re}(z) = x$,
and the imaginary part of $z$,
$\Im(z) = x$.

Addition of complex numbers,
$z_1 = x_1 + iy_1$,
$z_2 = x_2 + iy_2$
\[
z_1 \pm z_2 = (x_1 \pm x_2) + i(y_1 \pm y_2)
\]
the real parts are added together and the imaginary parts are added together.

Multiplication of complex numbers,
$z_1 = x_1 + iy_1$,
$z_2 = x_2 + iy_2$
\[
z_1z_2 = (x_1 + iy_1)(x_2 + iy_2) = (x_1x_2 - y_1y_2) + i(x_1y_2 + x_2y_1)
\]
just expand and use $i ^ 2 = -1$.

Looking at
\[
(x + iy)(x - iy) = x ^ 2 + ixy - ixy - i ^ 2y ^ 2 = x ^ 2 + y ^ 2 \in \R.
\]
We call the conjugate of a complex number as follows,
given $z = x + iy$ the conjugate is
\[
\bar{z} = x - iy = \mathrm{Re}(z) - i\mathrm{Im}(z).
\]
We can see that
\[
z\bar{z} = \mathrm{Re}(z) ^ 2 + \Im(z) ^ 2 \in \R
\]
which leads us to
\[
|z| = \sqrt{z\bar{z}} = \sqrt{\mathrm{Re}(z) ^ 2 + \Im(z) ^ 2}
\]
the modulus
(or absolute value)
of $z$.

Using the conjugate we can see
\[
\frac{z_1}{z_2} = \frac{x_1 + iy_1}{x_2 + iy_2} = \frac{(x_1 + iy_1)(x_2 - iy_2)}{(x_2 + iy_2)(x_2 - iy_2)} = \frac{x_1x_2 + y_1y_2}{x_2 ^ 2 + y_2 ^ 2} + i\frac{x_2y_1 - x_1y_2}{x_2 ^ 2 + y_2 ^ 2}.
\]

\begin{example}
    Find $\frac{3}{1 + i} - (3 + 2i)$.

    \begin{solution}
        \[
        \frac{3}{1 + i} - (3 + 2i) = \frac{3(1 - i)}{(1 + i)(1 - i)} - (3 + 2i) = \frac{3 - 3i}{2} - 3 - 2i = -\frac{3}{2} - \frac{7}{2}i.
        \]
    \end{solution}
\end{example}

Conjugation is nothing but reflection with respect to the real axis.

\begin{lemma}[Important Properties of Complex Numbers]
    \begin{enumerate}[label = (\roman*)]
        \item $z_1z_2 = 0 \iff z_1 = 0\text{ or } z_2 = 0$.

        \item $|z| = \sqrt{z\bar{z}}$.

        \item $\overline{(zw)} = \bar{z}\bar{w}$.

        \item $\Re(z) = \frac{z + \bar{z}}{2}$ and $\Im(z) = \frac{z - \bar{z}}{2i}$.

        \item $z ^ {-1} = \frac{\bar{z}}{|z| ^ 2}$.
    \end{enumerate}
\end{lemma}

Using polar coordinates and the Argand diagram we can write complex numbers in polar coordinates.
We define for any $z \in \C$
\[
r \coloneqq \sqrt{x ^ 2 + y ^ 2} = \sqrt{\Re(z) ^ 2+ \Im(z) ^ 2} = |z|
\]
and any $z \neq 0$
\[
\theta \coloneqq \begin{cases}
    \arctan(y / x) & x > 0, \\
    \pi / 2 & x = 0, y > 0, \\
    -\pi / 2 & x = 0, y < 0, \\
    \arctan(y / x) + \pi &x < 0, y > 0, \\
    \arctan(y / x) - \pi &x < 0, y < 0.
\end{cases}
\]
$\theta$ is the anticlockwise angle measured from the real axis,
$\theta$ is called the argument of $z$,
denoted $\arg(z)$.

We have the following polar coordinates for $z$:
\[
z = x + iy = r\cos(\theta) + ir\sin(\theta) = r(\cos(\theta) + i\sin(\theta)).
\]

\begin{example}
    $z = i$.

    \[
    r = |z| = \sqrt{0 ^ 2 + 1 ^ 2} = 1
    \]
    \[
    \theta = \pi / 2.
    \]
    \[
    i = 1(\cos(\pi / 2) + i\sin(\pi / 2)) = \cos(\pi / 2) + i\sin(\pi / 2).
    \]

    $z = 1 + i$
    \[
    |z| = \sqrt{1 ^ 2 + 1 ^ 2} = \sqrt{2},
    \]
    $\theta = \frac{\pi}{4}$,
    \[
    z = \sqrt{2}(\cos(\pi / 4) + i\sin(\pi / 4)).
    \]
\end{example}

\textbf{Important:}
$\arg(z)$ is only defined up to multiples of $2\pi$.

$\arg(i) = \frac{\pi}{2} + 2\pi k$,
$k \in \Z$,
not a function.

To define an argument we need a range of angles.

The principal value of $\arg(z)$ is the value in the interval $(-\pi, \pi]$ denoted $\Arg(z)$.

For example:
$\Arg(i) = \frac{\pi}{2}$,
$\Arg(-1) = \pi$.

\begin{lemma}[Properties of argument]
    We have the following properties of the argument:
    
    \begin{enumerate}[label = (\roman*)]
        \item $\arg(z_1z_2) = \arg(z_1) + \arg(z_2)\mod{2\pi}$.

        \item $\arg(1 / z) = -\arg(z)\mod{2\pi}$.

        \item $\arg(\bar{z}) = -\arg(z)\mod{2\pi}$.
    \end{enumerate}
\end{lemma}
\textit{Note:
the proof relies on why we went to polar coordinates:
multiplication is clear.}

\begin{lemma}
    Geometrically,
    multiplication in $\C$ is given by a dilated rotation;
    i.e.,
    if $z_1 = r_1(\cos(\theta_1) + i\sin(\theta_1))$ and $z_2 = r_2(\cos(\theta_2) + i\sin(\theta_2))$ then
    \[
    z_1z_2 = r_1r_2(\cos(\theta_1 + \theta_2) + i\sin(\theta_1 + \theta_2)).
    \]
    In particular,
    multiplying by $z_2$ constitutes an
    (anticlockwise)
    rotation of $z_1$ by $\theta_2$ degrees,
    followed by a dilation with factor $r_2$.

    \begin{proof}
        \begin{align*}
            z_1z_2 &= r_1(cos(\theta_1) + i\sin(\theta_2))r_2(\cos(\theta_2) + i\sin(\theta_2)) \\
            &= (r_1\cos(\theta_1) + ir_1\sin(\theta_1))(r_2\cos(\theta_2) + ir_2\sin(\theta_2)) \\
            &= (r_1r_2\cos(\theta_1)\cos(\theta_2) - r_1r_2\sin(\theta_1)\sin(\theta_2)) \\
            &+ i(r_1r_2\sin(\theta_1)\cos(\theta_2) + r_1r_2\cos(\theta_1)\sin(\theta_2)) \\
            &= r_1r_2[(\cos(\theta_1)\cos(\theta_2) - \sin(\theta_1)\sin(\theta_2)) + i(\sin(\theta_1)\cos(\theta_2) + \cos(\theta_1)\sin(\theta_2))] \\
            &= r_1r_2(\cos(\theta_1 + \theta_2) + i\sin(\theta_1 + \theta_2)).
        \end{align*}
    \end{proof}
\end{lemma}
\textit{Note:
this proves $\arg(z_1z_2) = \arg(z_1) + \arg(z_2)\mod{2\pi}$,
but also:
$\frac{1}{z} = \frac{\bar{z}}{|z| ^ 2}$ so $\arg(1 / z) = \arg(\bar{z})\mod{2\pi}$,
or $0 = \arg(z \cdot 1 / z) = \arg(z) + \arg(1 / z)\mod{2\pi}$.}

Useful notation:
write $\cos(\theta) + i\sin(\theta) \coloneqq e ^ {i\theta}$.

Motivated by
\[
e ^ {i\theta_1}e ^ {i\theta_2} = (\cos(\theta_1) + i\sin(\theta_1))(\cos(\theta_2) + i\sin(\theta_2)) = \cos(\theta_1 + \theta_2) + i\sin(\theta_1 + \theta_2) = e ^ {i(\theta_1 + \theta_2)}
\]

\begin{corollary}\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item $|z_1z_2| = |z_1||z_2|$.
        
        \item De Moivre's formula:
        $(\cos(\theta) + i\sin(\theta)) ^ n = \cos(n\theta) + i\sin(n\theta)$.
    \end{enumerate}

    \begin{proof}
        \begin{enumerate}[label = (\roman*)]
            \item Exercise.

            \item Done with induction.
        \end{enumerate}
    \end{proof}
\end{corollary}

The modulus has the following properties.
\begin{enumerate}[label = (\roman*)]
    \item
    (Triangle inequality)
    $|z_1 + z_2| \leq |z_1| + |z_2|$.

    \item $|z| \geq 0$ and $|z| = 0 \iff z = 0$.

    \item $\max(|\Re(z)|, |\Im(z)|) \leq |z| \leq |\Re(z)| + |\Im(z)|$.
\end{enumerate}

We can use $\arg(z)$,
$|z|$ to describe geometric sets.
$\mathbb{D} \coloneqq \{z \in \C\,:\, |z| < 1\}$ -
Unit disc.

$\mathbb{H} = \{z \in \C\,:\, \Im(z) > 0\}$ -
Upper half plane
(without the $x$ axis).

$\mathbb{H}_{R} \coloneqq \{z \in \C,\:\, \Re(z) > 0\}$ -
Right half plane
(without the $y$ axis).

$\mathbb{H}_{L} \coloneqq \{z \in \C,\:\, \Re(z) < 0\}$ -
Left half plane
(without the $y$ axis)

$\arg(z) = \frac{\pi}{4}$ -
Ray of angle $\pi / 4$.

$\{z \in \C\,:\, |z - i| = 4\}$ -
Circle centred at $i$ with radius $4$.

\begin{example}
    What is the set $|z - i| = |z + i|$?

    \begin{solution}
        The real line.

        $z = x + iy$
        \begin{align*}
            |x + i(y - 1)| = |x + i(y + 1)| &\iff |x + i(y - 1)| ^ 2 = |x + i(y + 1)| ^ 2 \\
            &\iff x ^ 2 + (y - 1) ^ 2 = x ^ 2 + (y + 1) ^ 2 \\
            &\iff 4y = 0 \implies y = 0.
        \end{align*}

        With $z$:
        one can show
        \[
        |z - \omega| ^ 2 = |z| ^ 2 - 2\Re(z\bar{\omega}) + |\omega| ^ 2.
        \]
    \end{solution}
\end{example}

\subsection{Complex Functions}

Complex functions are $f : \C \to \C$.
To visualise the graph of complex functions we need four dimensions.
We can always write
\[
f(z) = \Re(f(z)) + i\Im(f(z)).
\]
This is not enough to visualise it.

To understand complex functions we usually try to see how they take sets from $\C$ to sets in $\C$.
We will explore
"elementary"
functions like polynomials,
exponentials,
trigonometric functions,
logarithms and powers.

\subsection{The function \texorpdfstring{$f(z) = z ^ n$}{} with \texorpdfstring{$n \in \N$}{}}
To understand $z ^ n$ we will use polar representation:
if $z = re ^ {i\theta}$.
By our lemma,
\[
z ^ n = r ^ ne ^ {in\theta}
\]
if we fix $r$,
i.e. we are on a circle $z ^ n$ dilates our radius by power of $n$
($r \mapsto r ^ n$) and multiplies our argument by a factor of $n$
($\theta \to n\theta\mod{2\pi}$).

A segment of angular length $\eta$ on a circle of radius $r$ becomes an angular segment of length $n\eta\mod{2\pi}$ if $n\eta < 2\pi$ and the full circle if $n\eta \geq 2\pi$ with radius $r ^ n$.

If $n\eta > 2\pi$,
$z ^ n$ will not be injective.

We can slice the world to wedges of size $2\pi / n$ on which $z ^ n$ is injective.
For instance
\[
S_{k}^{(n)} = \{z \in \C\,:\,-\pi < 2\pi k / n < \Arg(z) \leq -\pi + 2\pi(k + 1) / n\}
\]
where $k = 0, \dotsc, n - 1$.
Or
\[
R_{k}^{(n)} = \{z \in \C\,:\, 2\pi k / n\mod{2\pi} < \Arg(z) \leq 2\pi(k + 1) / n\mod{2\pi}\}
\]
$k = 0, \dotsc, n - 1$.

\textit{Note:
when $n$ is even $R ^ {(n)}$s and $S ^ {(n)}$s will overlap
(not true if $n$ is odd).}

We now want to fix $\theta$.
Geometrically that a fixed $\theta$ is a ray
(half line)
without the origin.

The ray of angle $\theta$ is the set
\[
\{z \in \C \setminus \{0\}\,:\,\Arg(z) = \theta\}
\]
The closed ray of angle $\theta$ is the way of angle $\theta$ together with the origin $z = 0$,
i.e.
\[
\{z \in \C \setminus \{0\}\,:\,\Arg(z) = \theta \cup \{0\}\}.
\]

If $\theta$ is fixed $z ^ n = r ^ ne ^ {in\theta}$ which takes a segment on the ray $R_{\theta}$
($0 \leq a < r < b$)
will go to a segment on the ray $R_{n\theta\bmod{2\pi}}$
($a ^ n < r ^ n < b ^ n$).

\begin{lemma}
    \begin{itemize}
        \item The map $z ^ n$ injectively takes an angular segment of length $\frac{2\pi}{n}$ which is open at one end and closed at the other from a circle of radius $r$ to the entire circle of radius $r ^ n$.
        If the above segment is closed,
        or its size is larger than $\frac{2\pi}{n}$ the image is no longer injective.

        \item The map $z ^ n$ injectively takes a ray of angle $\theta$ to a ray of angle $n\theta\bmod{2\pi}$.
        Consequently,
        if $n|\theta_1 - \theta_2| \leq 2\pi$ the map $z ^ n$ injectively takes the wedge
        \[
        W = \{z \in \C\,:\, \theta_1 < \Arg(z) < \theta_2\}
        \]
        to the wedge
        \[
        W = \{z \in \C\,:\, n\theta_1\bmod{2\pi} < \Arg(z) <n\theta_2\bmod{2\pi}\}
        \]
        with the convention that for any $-\pi < \nu < \xi \leq \pi$
        \[
       \left\{z\in\mathbb{C}\;:\; \xi<\mathrm{Arg}(z)<\upsilon\right\} = \left\{ z\in\mathbb{C}\;:\; \xi<\mathrm{Arg}(z)\leq \pi\right\} \cup \left\{ z\in\mathbb{C}\;:\; -\pi<\mathrm{Arg}(z)<\upsilon\right\}. 
        \]
        If we replace one strict inequality
        (i.e. $<$)
        with an inequality
        (i.e. $\leq$)
        the above claim remains true.
        In this case,
        if $n|\theta_1 - \theta_2| = 2\pi$ the image of the map will be $\C \setminus \{0\}$.
        We can get the entire complex plane by adding $z = 0$ to $W$.
        When $n|\theta_1 - \theta_2| > 2\pi$ the image of $W$ by $z ^ n$ will be $\C \setminus \{0\}$ but not injectively.

        \item We can define $n$ different $n$th roots which are the inverses of the map $z ^ n$.
        We can write them all in the form
        \[
        z^{\frac{1}{n}} = \left\lvert z\right\rvert^{\frac{1}{n}}e^{i\left( \frac{\mathrm{Arg}\left( z\right)}{n} + \frac{2\pi k }{n}\right)}
        \]
        with $k = 0, \dotsc, n - 1$
        (we remove the $k$ index from the map,
        but one need to specify which $k$ we are talking about).
        note that for a fixed $k$ the $n$th root of $z$ takes $\C$ into a anti-clockwise rotation by $\frac{\pi}{n}$ of $R_{k - 1}^{(n)}$.
    \end{itemize}
\end{lemma}

Notation:
we want
\[
e ^ z = e ^ {x + iy} = e ^ x\cdot e ^ {iy}
\]
defined $e ^ {iy}$ earlier.

We define the following:
\begin{definition}[Complex exponential]
    We define the complex exponential function $\exp : \C \to \C$ by
    \[
    \exp(z) \coloneqq e ^ x(\cos{y} + i\sin{y}).\qquad (z = x + iy)
    \]
    Shorthand $\exp(z) = e ^ z$.
\end{definition}

One can show
(later)
that with this definition
\[
e ^ z = \sum_{n = 0}^{\infty}\frac{z ^ n}{n!}.
\]

\begin{proposition}
    We have the following complex exponential function properties:
    
    \begin{enumerate}[label = (\roman*)]
        \item $e ^ z \neq 0$ for all $z \in \C$.

        \item $e ^ {z_1 + z_2} = e ^ {z_1}e ^ {z_2}$.

        \item $e ^ z = 1$ if and only if $z = 2\pi ik$ for some $k \in \Z$.

        \item $e ^ {-z} = 1 / e ^ z$.

        \item $|e ^ z| = e ^ {\Re(z)}$.
    \end{enumerate}

    \begin{proof}
        By definition
        \[
        e ^ z = e ^ {x}e ^ {iy} = e ^ x(\cos{y} + i\sin{y})
        \]
        recall that any $\omega \in \C$ can be written uniquely 
        ($\bmod{2\pi}$)
        as $w = \underbrace{r}_{|\omega|}e ^ {i\overbrace{\theta}^{\Arg(w)}}$ so
        \[
        |e ^ z| = e ^ x\qquad \Arg(e ^ z) = y\bmod{2\pi}.
        \]
        $|e ^ z| \neq 0$ since $e ^ x \neq 0$.
        Since $\omega = 0 \iff |\omega| = 0$ we conclude that $e ^ z \neq 0$
        (i.e. $1$).
        \[
        e ^ z = 1 \iff e ^ xe ^ {iy} = 1 \cdot e ^ {i0}
        \]
        so $e ^ x = 1$ and $y = 0 + 2\pi k$ so $x = 0$ and $y = 2\pi k$,
        $k \in \Z$
        (shows (iii)).

        \[
        e ^ {z_1 + z_2} = e ^ {x_1 + x_2}e ^ {i(y_1 + y_2)} = e ^ {x_1}e ^ {x_2}e ^ {iy_1}e ^ {iy_2} = \left(e ^ {x_1}e ^ {iy_1}\right)\left(e ^ {x_2}e ^ {iy_2}\right) = e ^ {z_1}e ^ {z_2}.
        \]
        \[
        1 = e ^ 0 = e ^ {0 + i0} = e ^ {z - z} =  e ^ ze ^ {-z} \implies e ^ {-z} = \frac{1}{e ^ z}.
        \]
    \end{proof}
\end{proposition}

\begin{corollary}
    We have $\exp(2\pi i) = 1$ and $\exp(\pi i) = -1$.
    Latter is Euler's formula.
\end{corollary}

\begin{corollary}
    The complex exponential function is $2\pi i$-periodic,
    that is $\exp(z + 2k\pi i) = \exp(z)$ for any $k \in \Z$.
\end{corollary}

How does the exponential operate?
Recall $e ^ z - e ^ xe ^ {iy}$ and $|e ^ z| = e ^ x$ and $\Arg(e ^ z) = y\bmod{2\pi}$.
For fixed $x$,
i.e. $x = c$ for some $c \in \R$ we get image of a circle of radius $e ^ c$.
Segments on $x = c$ go to angular segments on the concentric circle with radius $e ^ c$.
If $y$ ranges over a segment of length more than $2\pi$ we will get the full circle.
When we fix $y$,
we will end up on a ray of angle $y\bmod{2\pi}$.
A segment on the line $y = c$ goes under $e ^ z$ to a segment on the ray $R_{c\bmod{2\pi}}$ if $a < x < b$ then we get the part of the ray $e ^ a < |\omega| < e ^ b$,
$\Arg(\omega) = c\bmod{2\pi}$.

\begin{lemma}
    \begin{itemize}
        \item The map $e ^ z$ injectively takes a segment of length $2\pi$ on the line $x = c$ which is open at one end and closed at the other to the entire concentric circle of radius $e ^ c$.
        If the above segment is closed,
        or its size is larger than $2\pi$ the image is no longer injective.

        \item The map $e ^ z$ injectively takes the line $y = c$ to the ray of angle $c$
        (without the origin).

        \item The map $e ^ z$ injectively takes the set $\{z \in \C\,:\, \theta < \Im(z) \leq \theta + 2\pi\}$ to $\C ^ {*}$ where
        \[
        \C ^ {*} =  \C \setminus \{0\}.
        \]
    \end{itemize}
\end{lemma}
With $e ^ z$ we can define trig functions.

Recall we defined $e ^ {ix} = \cos{x} + i\sin{x}$ when $x \in \R$.
\begin{align*}
    \cos{x} &= \frac{e ^ {ix} - e ^ {-ix}}{2}&\sin{x} &= \frac{e ^ {ix} - e ^ {-ix}}{2i} \\
    \cosh{x} &= \frac{e ^ x + e ^ {-x}}{2}& \sinh{x} &= \frac{e ^ x - e ^ {-x}}{2}
\end{align*}

\begin{definition}[Trigonometric functions]
    \begin{align*}
        \cos{z} &= \frac{e ^ {iz} - e ^ {-iz}}{2}&\sin{z} &= \frac{e ^ {iz} - e ^ {-iz}}{2i} \\
        \cosh{z} &= \frac{e ^ z + e ^ {-z}}{2}& \sinh{z} &= \frac{e ^ z - e ^ {-z}}{2}
    \end{align*}
\end{definition}
\textbf{Remember,
complex $\cos$ and $\sin$ are not bounded as they have $x$ and $y$ in $x + iy$.}


\begin{lemma}
    We have that for $z = x + iy$:
    \begin{align*}
        \sin(z) &= \sin(x)\cosh(y) + i\cos(x)\sinh(y) \\
        \cos(z) &= \cos(x)\cosh(y) - i\sin(x)\sinh(y) \\
        \sinh(z) &= -i\sin(iz) = \sinh(x)\cos(y) + i\cosh(x)\sin(y) \\
        \cosh(z) &= \cos(iz) = \cosh(x)\cos(y) + i\sinh(x)\sin(y).
    \end{align*}
    In addition we have that for all $z \in \C$
    \[
    \sin ^ 2(z) + \cos ^ 2(z) = 1,\qquad \cosh ^ 2(z) - \sinh ^ 2(z) = 1.
    \]

    \begin{proof}
        Exercise.

        \begin{align*}
            \sin{z} &= \frac{e ^ {iz} - e ^ {-iz}}{2i} \\
            &= \frac{e ^ {i(x + iy)} - e ^ {-i(x + iy)}}{2i} \\
            &= \frac{e ^ {-y}e ^ {ix} - e ^ {y}e ^ {-ix}}{2i} \\
            &= \frac{e ^ {-y}(\cos(x) + i\sin(x)) - e ^ {y}(\cos(x) - i\sin(x))}{2i}
        \end{align*}
    \end{proof}
\end{lemma}

In general complex $\cos$ and $\sin$ are bounded if and only if $y$ is bounded in $z = x + iy$.

\subsection{Logarithms and complex powers}


The real exponential $e ^ x$ has an inverse
($\log{x}$).
This,
in general,
will not work for $e ^ z$ since it is not injective.

Can we get an inverse when we restrict our domain?

\begin{lemma}[Inverting the exponential function]
    For every $w \in \C ^ {*} = \C \setminus \{0\}$,
    the equation
    \[
    e ^ z = w
    \]
    has a solution $z$.
    Furthermore,
    if we write $w = |w|e ^ {i\varphi}$ with $\varphi = \Arg(w)$,
    then all the solutions to $e ^ z = w$ are given by
    \[
    z = \log|w| + i(\varphi + 2\pi k)
    \]
    for $k \in \Z$.
    Here,
    $\log|w|$ is the usual natural logarithm of the real number $|w|$.
    \textit{Note that there are infinitely many solutions}.

    \begin{proof}
        \begin{align*}
            e ^ z = \omega &\iff e ^ xe ^ {iy} = |w|e ^ {iy} \\
            &\iff e ^ x = |\omega|\text{ and } y = y + 2\pi k\quad k \in \Z
        \end{align*}
        \[
        \]
    \end{proof}
\end{lemma}

From this we see a candidate to complex logarithm:
Just fix k.

We can try and define
\[
\log{z} = \log|z| + i\arg(z)
\]
where $\arg(z)$ goes in a fixed given range of angles $(\theta_1, \theta_2]$.
This definition is missing some
"nice"
properties.

To illustrate this choose the angles to be $(-\pi, \pi]$.
$\log(-1) = \log|-1| + i\pi = i\pi$.
Let's try to get closer to $-1$ from below,
choosing $\theta = -\pi + \delta$,
$\delta > 0$ gives us $z\delta = e ^ {i(-\pi + \delta)}$ close to $z = -1$.
\[
\log(z\delta) = \log\left|e ^ {i(-\pi + \delta)}\right| + i(-\pi + \delta) = -i(\pi - \delta) \approx -i\pi
\]
we have a discontinuity.

It is as if when we complete a full circle we
"climb"
by $2\pi$.

To avoid this issue we must disallow the ability to do full circles.
We do that by removing a closed ray $R_{\theta} = \{z = re ^ {i\theta}\,:\,r \geq 0\}$.

\begin{definition}[Complex logarithm functions]
    For any two real numbers $\theta_1 < \theta_2$ with $\theta_2 - \theta_1 = 2\pi$,
    let $\arg$ be the choice of argument function with values in $(\theta_1, \theta_2]$.
    Then the function
    \[
    \log(z) \coloneqq \log|z| + i\arg(z)
    \]
    is called a branch of logarithm.
    It has a jump discontinuity along the ray $R_{\theta_1} = R_{\theta_2}$.
    this ray is called a branch cut.
    If we choose $\arg(z) = \Arg(z) \in (-\pi, \pi]$,
    then we obtain a branch of logarithm called the principal branch of $\log$.
    We write $\Log$ for this principal branch:
    it is given by the formula
    \[
    \Log(z) \coloneqq \log|z| + i\Arg(z).
    \]
    The principal branch of logarithm has a
    "jump discontinuity"
    along the ray given by the non-positive real axis $\R_{\leq 0}$.
\end{definition}

Comments:
\begin{itemize}
    \item We must specify a branch
    ($\theta_1 < \theta_2$)
    to discuss a logarithm.

    \item For future:
    any branch of logarithm is continuous on $\C \setminus \R_{\theta}$.

    \item For real positive numbers
    \[
    \log{x} = \Log{x}.
    \]
\end{itemize}

\begin{lemma}[Properties of logarithms]
    We have the following properties when using any given branch of logarithm:
    \begin{enumerate}[label = (\roman*)]
        \item $e ^ {\log{z}} = z$ for any $z \in \C \setminus \{0\}$.

        \item In general
        \[
        \log(zw) \neq \log{z} + \log{w}.
        \]

        \item In general
        \[
        \log(e ^ z) \neq z.
        \]
    \end{enumerate}
\end{lemma}

\begin{lemma}\phantom{}
    \begin{itemize}
        \item A branch of the map $\log(z)$ injectively takes a ray of angle $\theta$ without the origin,
        to the line $y = \theta$ where $\theta$ is measured with respect to the branch cut.
    
        \item A branch of the map $\log(z)$ takes a concentric circle of radius $r$,
        minus the intersection of this circle with the branch cut,
        to an open segment of length $2\pi$ on the line $x = \log{r}$ with its lowest point given by the angle that defines the branch cut.
    \end{itemize}
\end{lemma}

\begin{definition}[Complex powers]
    For $w \in \C$ fixed by choosing any branch of $\log$ we can define a branch of the function $z \mapsto z ^ w$ by the equation
    \[
    z ^ w \coloneqq e ^ {w\log{z}}.
    \]
\end{definition}

For example,
if $w = 1 / n$ and use the principal branch we get
\[
z ^ {1 / n} = e ^ {\frac{\log|z|}{n} + i\frac{\Arg(z)}{n}} = |z| ^ {1 / n}e ^ {i\frac{\Arg(z)}{n}}.
\]

\begin{remark}
    This definition is consistent with $z ^ n = z \dotsi z$ with $n \in \N$.
    Also,
    for $n \in \N$ one can show that all the branches of $z ^ {1 / n}$ with argument in $(-\pi, \pi)$
    (branch cut is $\R_{\leq 0}$)
    are
    \[
    |z| ^ {1 / n}e ^ {i / n(\Arg(z) + 2\pi k)}
    \]
    $k = 0, \dotsc, n - 1$.
\end{remark}

\begin{example}
    Find $(1 - i) ^ {1 / 2}$ using the principal branch of the logarithm.

    \begin{solution}
        \[
        (1 - i) ^ {1 / 2} = e ^ {1 / 2\Log(1 - i)} = e ^ {1 / 2(\log|1 - i| + i\Arg(1 - i))} = e ^ {1 / 2(\log(\sqrt{2}) - i\pi / 4)} = e ^ {1 / 4\log(2)}e ^ {-i\pi / 8} = 2 ^ {1 / 4}e ^ {-i\pi / 8}.
        \]
    \end{solution}
\end{example}

\begin{example}
    Find $2 ^ {1 / 2}$ for all possible branches.

    \begin{solution}
        We only have two branches
        ($k = 0, k = 1$ from above)
        which will give $\sqrt{2}, -\sqrt{2}$.
    \end{solution}
\end{example}

\section{Topology and Continuity}
On $\R$ these notions are related to intervals
(open,
closed,
etc).
These studies can be extended to sets which are called open,
closed,
compact.
Such sets tell us a lot about our space and form its topology.

\subsection{Basic topology on the complex plane -
open and closed sets}

\begin{definition}[Balls in $\C$]\phantom{}
    \begin{itemize}
        \item The open ball $B_r(z_0)$ of radius $r$ centred at $z_0$ is
        \[
        B_r(z_0) \coloneqq \{z \in \C\,:\,|z - z_0| < r\}.
        \]

        \item The closed ball $\bar{B}_r(z_0)$ of radius $r$ centred at $z_0$ is
        \[
        \bar{B}_r(z_0) \coloneqq \{z \in \C\,:\, |z - z_0| \leq r\}.
        \]
    \end{itemize}
\end{definition}

\begin{definition}[Open/closed sets in $\C$]\phantom{}
    \begin{itemize}
        \item A subset $U \subseteq \C$ is open
        (in $\C$)
        if for every point $z_0 \in U$ there exists $\varepsilon > 0$ such that $B_{\varepsilon}(z_0) \subset U$.

        \item A subset $U \subseteq \C$ is closed
        (in $\C$)
        if its complement $U ^ c \coloneqq \C \setminus U$ is open.
    \end{itemize}
\end{definition}

\begin{lemma}[Open balls are open]
    For any $z_0 \in \C$ and any $r > 0$ the open ball $B_r(z_0)$ is an open set.

    \begin{proof}
        Let $z \in B_r(z_0)$.
        Define
        \[
        \varepsilon = \frac{r - |z - z_0|}{2} > 0\qquad (|z - z_0| < r).
        \]
        Claim:
        $B_{\varepsilon}(z) \subset B_r(z_0)$.
        Indeed,
        if $w \in B_{\varepsilon}(z)$
        \begin{align*}
            |w - z_0| &= |w - z + z - z_0| \\
            &\leq |w - z| + |z - z_0| \\
            &< \varepsilon + |z - z_0| \\
            &= \frac{r - |z - z_0|}{2} + |z - z_0| \\
            &= \frac{r + |z - z_0|}{2} < r.
        \end{align*}
    \end{proof}
\end{lemma}

\begin{remark}
    It can also be shown that for any $z_0 \in \C$ and any $r > 0$ the closed ball $\bar{B}_r(z_0)$ is closed in $\C$.
\end{remark}

\begin{example}
    The following sets are open:

    \begin{enumerate}[label = (\roman*)]
        \item $\mathbb{H} = \{z \in \C\,:\,\Im(z) > 0\}$.
        
        \item $\mathbb{H}_{-} = \{z \in \C\,:\,\Im(z) < 0\}$.
        
        \item $\mathbb{H}_{L} = \{z \in \C\,:\,\Re(z) < 0\}$.
        
        \item $\mathbb{H}_{R} = \{z \in \C\,:\,\Re(z) > 0\}$.
        
        \item $\mathbb{D} = \{z \in \C\,:\,|z| < 1\}$.
        
        \item $\Omega_1 = \{z \in \C\,:\,\Re(z) > 0, \Im(z) > 0\}$.
    \end{enumerate}

    Lets show that $\Omega_1$ is open.
    Given $z_0 \in \Omega_1$,
    define $\varepsilon = \frac{\min\{\Re(z_0), \Im(z_0)\}}{2}$.
    Claim:
    $B_{\varepsilon}(z) \subset \Omega_1$.
    Indeed if $w \in B_{\varepsilon}(z_0)$ then
    \begin{align*}
        \Re(w) &= \Re(w - z_0  + z_0) \\
        &= \Re(w - z_0) + \Re(z_0) \\
        \intertext{$|\Re(\xi)| \leq |\xi|$}
        &\geq -|w - z_0| + \Re(z_0) \\
        &> -\varepsilon + \Re(z_0) \\
        &= -\frac{\min(\Re(z_0), \Im(z_0))}{2} + \Re(z_0) \\
        &\geq \frac{\Re(z_0)}{2} > 0.
    \end{align*}
    Similarly $\Im(z_0) > 0$.
\end{example}

\textbf{Careful!}
Not all sets are open or closed.
$A - \{z \in \C\,:\,\Re(z) > 0, \Im(z) \geq 0\}$.
$A$ is not open
($1 - i\varepsilon \in B_{\varepsilon}(1)$,
but $1 \in A$,
$B_{\varepsilon}(1) \not\subset A$ for any $\varepsilon > 0$).
$A ^ c$ is not open,
i.e. $A$ is not closed.

\begin{lemma}[Unions and intersections of open sets]\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item Arbitrary unions of open sets are open;
        that is $\bigcup_{i \in I}U_i$ is open for any
        (possible infinite)
        collection of open sets $U_i$.

        \item Finite intersections of open sets are open;
        that is $\bigcap_{i = 1}^{n}U_i$ is open for any finite collection of open sets $U_i$.
    \end{enumerate}

    \begin{proof}\phantom{}
        \begin{enumerate}[label = (\roman*)]
            \item Let $z_0 \in \bigcup_{i \in I}U_i$,
            $\exists i_0 \in I$ s.t. $z_0 \in U_{i_0}$ since $U_{i_0}$ is open $\exists \varepsilon > 0$ s.t. $B_{\varepsilon}(z_0) \subset U_i \subseteq \bigcup_{i \in I}U_i$.
            $z_0$ was arbitrary so $\bigcup_{i \in I}U_i$ is open.

            \item Let $z_0 \in \bigcap_{i = 1}^{n}U_i$.
            This means that $z_0 \in U_i$ $\forall i = 1, \dotsc, n$.
            Since $U_i$ is open $\exists \varepsilon_i > 0$ s.t. $B_{\varepsilon_i}(z_0) \subset U_i$.
            Define $\varepsilon = \min(\varepsilon_1, \dotsc, \varepsilon_n)$ then $z_0 \in B_{\varepsilon}(z_0) \subseteq B_{\varepsilon_i}(z_0) \subset U_i$ $\forall i = 1, \dotsc, n$ this implies $B_{\varepsilon_0}(z) \subset \bigcap_{i = 1}^{n}U_i$.
        \end{enumerate}
    \end{proof}
\end{lemma}

\begin{corollary}[Unions and intersections of closed sets]\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item Finite unions of closed sets are closed.

        \item Arbitrary intersections of closed sets are closed.
    \end{enumerate}

    \begin{proof}\phantom{}
        \begin{enumerate}[label = (\roman*)]
            \item Follows from De Morgan's law
            \[
            \left(\bigcup_{i \in I}A_i\right) ^ c = \bigcap_{i \in I}A_i ^ c, \qquad \left(\bigcap_{i \in I}A_i\right) ^ c = \bigcup_{i \in I}A_i ^ c.
            \]
        \end{enumerate}
    \end{proof}
\end{corollary}

Given a set $A$ we can define special open and closed sets associated to $A$.

\begin{definition}[Interior points,
closure,
boundary]
    Let $A$ be a subset of $\C$.

    \begin{itemize}
        \item The interior $A ^ 0$ of $A$ is defined by
        \[
        A ^ 0 \coloneqq \{x \in A\,:\,\text{there exists an open set $U \subseteq A$ such that $x \in U$}\}
        \]
        (this can be replaced with $B_{\varepsilon}(x) \subset A$ for some $\varepsilon > 0$).
        
        \item The closure $\bar{A}$ of $A$ is the complement of the interior of the complement:
        \[
        \bar{A} \coloneqq \left(\left(A ^ c\right) ^ 0\right) ^ c = \{x \in X\,:\, U \cap A \neq \emptyset\text{ for every open set $U$ with $x \in U$}\}
        \]
        ($U$ can be replaced with $B_{\varepsilon}(x)$ for all $\varepsilon > 0$).
    
        \item The boundary $\partial A$ of $A$ is the closure without the interior:
        \[
        \partial A \coloneqq \bar{A} \setminus A ^ 0\qquad \left[=\left(A ^ 0\right) ^ c \cap \left(\left(A ^ c\right) ^ 0\right) ^ c = \left(A ^ 0 \cup \left(A ^ c\right) ^ 0\right) ^ c\right].
        \]
    \end{itemize}
\end{definition}

\begin{example}
    Consider the set
    \[
    A = \{z \in \C\,:\, 1 < |z| \leq 3\}.
    \]
    Then $A ^ 0$ is the area inside the inner and outer circle not inclusive of the boundary.
    The closure is the area inside the inner and outer circles including the boundary.
    The boundary is only the boundary of both the inner and outer circles.
\end{example}

\begin{remark}\phantom{}
    \begin{itemize}
        \item Don't confuse closure with conjugation.

        \item $A ^ 0$ is always open.
        $\bar{A}$ is always closed.
        $\partial A$ is always closed.

        \item Intuitively boundary is the edge of the set.
        Interior points are points in the set that are not on the edge and the closure is attained by adding all the edge points.
    \end{itemize}
\end{remark}

\textit{Note:
$A = \{z_0\}$,
$A ^ 0 = \emptyset$,
$\bar{A} = \{z_0\}$,
$\partial A = \{z_0\}$}

\textbf{Important properties:}
\begin{enumerate}[label = (\roman*)]
    \item $A$ is open $\iff \partial A \cap A = \emptyset \iff A = A ^ 0$.
    Moreover,
    \[
    A ^ 0 = \bigcup_{U \subseteq A, U\text{ open}}U.
    \]
    In other words:
    $A ^ 0$ is the largest open set that is contained in $A$.

    \item $A$ is closed $\iff \partial A \subseteq A \iff A = \bar{A}$.
    Moreover,
    \[
    \bar{A} = \bigcap_{A \subseteq F, F\text{ closed}}F.
    \]
    In other words:
    $\bar{A}$ is the smallest closed set that contains $A$.

    \item $\partial A = \{x \in X\,:\,\text{for all open sets $U$ containing $x$,
    there exist $y, z \in U$ with $y \in A$ and $z \in A ^ c$}\}$.
\end{enumerate}

\subsection{Convergence and continuity}

The notion of limits and convergence is paramount in Analysis.

\begin{definition}[Limits and convergence of sequences in $\C$]
    We say a sequence $\{z_n\}_{n \in \N} \subset \C$ converges to $z \in \C$ if
    \[
    \lim_{n \to \infty}|z_n - z| = 0.
    \]
    That is if for any $\varepsilon > 0$ there exists $N \in \N$ such that for every $n > N$
    \[
    |z_n - z| < \varepsilon.
    \]
\end{definition}

We notice that if $\{z_n\}_{n \in \N}$ is a sequence in $\C$ then $\{\Re(z_n)\}_{n \in \N}$ and $\{\Im(z_n)\}_{n \in \N}$ are sequences in $\R$.
Is there a connection between the convergence of these sequences.

\begin{lemma}
    A sequence of complex numbers $\{z_n\}_{n \in \N} \subset \C$ converges if and only if the sequences $\{\Re(z_n)\} \subset \R$ and $\{\Im(z_n)\} \subset \R$ converge in $\R$.
    Moreover,
    in that case
    \[
    \lim_{n \to \infty}z_n = \lim_{z \to \infty}\Re(z_n) + i\lim_{n \to \infty}\Im(z).
    \]

    \begin{proof}
        Denote $\Re(z_n) = x_n$,
        $\Im(z_n) = y_n$,
        i.e. $z_n = x_n + iy_n$ for $x_n, y_n \in \R$.
        Assume $z_n \to z = x + iy$.
        Then
        \[
        0 \leq |x_n - x| \leq \sqrt{(x_n - x) ^ 2 + (y_n - y) ^ 2} = |z_n| \to 0
        \]
        by assumption.
        Then $|x_n - x| \to 0$ as $n \to \infty$ by squeezing theorem.
        \[
        \lim_{n \to \infty}|x_n - x| < 0 \implies \lim_{n \to \infty}x_n = x.
        \]
        Similarly $y_n$ must converge to $y$ as $n \to \infty$.

        Assume $x_n \to x$ as $n \to \infty$,
        $y_n \to y$ as $n \to \infty$ then
        \begin{align*}
            |z_n - z| &= |(x_n - x) + i(y_n - y)| \\
            &\leq |x_n - x| + |i||y_n - y| \\
            &\implies \\
            0 \leq |z_n - z| &\leq |x_n - x| + |y_n - y|
        \end{align*}
        then $|z_n - z| \to 0$ as $n \to \infty$.
    \end{proof}
\end{lemma}

\begin{corollary}
    A sequence in $\C$ can have at most one limit.

    \begin{proof}
        This holds since it is true in $\R$.
        Indeed,
        if $\{z_n\}$ converges then $\{\Re(z_n)\}$ converges.
        It can converge to a unique limit,
        say $x$.
        Similarly $\{\Im(z_n)\}$ converges to a unique limit $y$,
        and consequently $\{z_n\}$ converges to the unique limit $x + iy$.
    \end{proof}
\end{corollary}

\begin{example}
    Show $z_n = \frac{1}{n} + i\left(1 - \frac{1}{n}\right) ^ n$ converges in $\C$.

    \begin{solution}
        $x_n = \frac{1}{n}$,
        $y_n = (1 - 1 / n) ^ n$.
        Since $x_n \to 0$ as $n \to \infty$ and $y_n \to e ^ {-1}$ as $n \to \infty$,
        $z_n$ must converge as well to $\frac{i}{e}$.
    \end{solution}
\end{example}

\textbf{Important:}
COLT rules follow in the complex plane
(where $i$ is treated as a constant).

We can formulate convergence of sequences in a topological sense.

\begin{lemma}
    Let $\{z_n\}_{n \in \N} \subset \C$ be a given sequence in $\C$.
    Then $\lim_{n \to \infty}z_n = z$ if and only if for any open set $U \subseteq \C$ such that $z \in U$ there exists $N \in \N$ such that for any $n > N$ we have that $z_n \in U$.

    \begin{proof}
        We start by noticing that $z_n \to z$ as $n \to \infty$ is equivalent to $\forall \varepsilon > 0$,
        $\exists N \in \N$ s.t. if $n > N$ $|z_n - z| < \varepsilon$ which is equivalent to $\forall \varepsilon > 0$,
        $\exists N \in \N$ s.t. if $n > N$,
        $z_n \in B_{\varepsilon}(z)$.

        "$\implies$".
        Let $U$ be an open set s.t. $z \in U$.
        By definition.
        $\exists \varepsilon > 0$ s.t. $B_{\varepsilon}(z) \subset U$.
        Since $z_n \to z$ as $n \to \infty$ we can find $N \in \N$ s.t. for all $n > N$ $z_n \in B_{\varepsilon}(z) \subset U$,
        which is the required result.

        "$\impliedby$"
        Given $\varepsilon > 0$ the set $B_{\varepsilon}(z)$ is open.
        By assumption,
        $\exists N \in \N$ s.t. if $n > N$ $z_n \in B_{\varepsilon}(z)$.
        Since this holds for any $\varepsilon > 0$ $z_n \to z$ as $n \to \infty$.
    \end{proof}
\end{lemma}

Sequences give excellent intrinsic criterion to test closedness of sets.

\begin{lemma}[Closedness criterion]
    The set $A \subseteq \C$ is a closed set if and only if for any sequence $\{z_n\}_{n \in \N}$ in $A$ that converges to an element $z \in \C$ we have that $z \in A$.
\end{lemma}

\begin{definition}[Continuity]
    Let $A$ be a subset of $\C$.
    A map $f : A \to \C$
    (or $\R$)
    is called continuous at $z_0 \in A$ if for any $\varepsilon > 0$ there exists $\delta > 0$ such that if $|z - z_0| < \delta$ and $z \in A$ then
    \[
    |f(z) - f(z_0)| < \varepsilon.
    \]
    We say a function $f$ is continuous on a set $A \subseteq \C$ if it is continuous at every point in $A$.
\end{definition}

Just like before we have the following

\begin{lemma}[Continuity via sequences]
    A function $f : A \to \C$
    (or $\R$)
    is continuous at $z_0 \in A$ if and only if
    \[
    \lim_{n \to \infty}f(z_n) = f(z_0)
    \]
    for every sequence $\{z_n\}_{n \in \N}$ in $A$ such that $\lim_{n \to \infty}z_n = z_0$
\end{lemma}

We use this usually to show lack of continuity
(and in fact,
lack of limit):
If we find $\{z_n\}, \{w_n\}$ s.t. $z_n \to z_0$ as $n \to \infty$,
$w_n \to z_0$ as $n \to \infty$.
$f(z_n) \to \ell_1$ as $n \to \infty$,
$f(w_n) \to \ell_2$ as $n \to \infty$ and $\ell_1 \neq \ell_2$ then $f$ is not continuous at $z_0$
(in fact there is no limit).

\begin{example}
    \[
    f(z) = f(x + iy) = \begin{cases}
        \frac{xy}{x ^ 2 + y ^ 2} &(x, y) \neq (0, 0) \\
        0 &(x, y) = (0, 0)
    \end{cases}
    \]
    is not continuous at $z = 0$.
    Indeed,
    note that for $y = kx$,
    $k \in \R$
    \[
    f(x + ikx) = \frac{k}{1 + k ^ 2}
    \]
    when $x \neq 0$.
    Choosing $z_n = \frac{1}{n}$ and $w_n = \frac{1}{n} + \frac{i}{n}$,
    we have that $z_n \to 0$,
    $w_n \to 0$
    \[
    f(z_n) = 0 \to 0\text{ as $n \to \infty$}
    \]
    \[
    f(w_n) = \frac{1}{2} \to \frac{1}{2}\text{ as $n \to \infty$}
    \]
    and as such there is no limit.
\end{example}

\begin{lemma}[Basic properties of continuous functions]\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item Products,
        sums,
        quotients of real/complex valued functions on $A \subseteq \C$ are continuous on $A$
        (where defined).

        \item Composition of continuous functions are continuous.
    \end{enumerate}
\end{lemma}

The following functions are continuous on $\C$:
\begin{itemize}[label = -]
    \item $f(z) = z$.

    \item $f(z) = \text{const}$.

    \item $f(z) = \Re(z)$,
    $f(z) = \Im(z)$,
    $f(z) = |z|$.

    \item $f(z) = \bar{z}$.

    \item $f(z) = z ^ n$,
    $n \in \N$.
    Consequently,
    polynomials are continuous.

    \item $f(z) = e ^ z$ and consequently $\sin(z)$,
    $\cos(z)$,
    $\sinh(z)$,
    $\cosh(z)$.
\end{itemize}
For any $\theta_1 < \theta_2$ s.t. $\theta_2 - \theta_1 = 2\pi$ the argument associated to $(\theta_1, \theta_2]$ is not continuous on $C ^ {*}$
($=\C \setminus\{0\}$).
This is due to a jump discontinuity at $R_{\theta_1}$
(ray of $\theta_1$).
It is continuous on $\C \setminus R_{\theta_1}$.

Any branch of $\log{z}$ is continuous on $\C \setminus R_{\theta_1}$
(the branch cut).

Much like sequences we can recast continuity in topological sets.

\begin{theorem}[Continuity via open sets]
    Let $A \subseteq \C$ be an open set and let $f : A \to \C$
    (or $\R$)
    be a given map.
    Then the following are equivalent.

    \begin{enumerate}[label = (\roman*)]
        \item $f$ is continuous.

        \item $f ^ {-1}(U)$ is an open set for every open set $U \subseteq \C$.

        \item $f ^ {-1}(F)$ is a closed set for every closed set $F \subseteq \C$.
    \end{enumerate}

    \begin{proof}
        Motivation:
        $f$ is continuous at an interior point $z_0$ if and only if for any $\varepsilon > 0$,
        there exists $\delta > -$ such that if $z \in B_{\delta}(z)$ then $f(z) \in B_{\varepsilon}(f(z_0))$.
        Equivalently for any $\varepsilon > 0$,
        there exists $\delta > 0$ such that
        \[
        B_{\delta}(z_0) \subset f ^ {-1}(B_{\varepsilon}f(z_0)).
        \]
    \end{proof}
\end{theorem}

We can use this theorem to classify sets.
For any $z_0 \in \C$,
$\{z_0\}$ is closed which implies if $f$ is continuous then $f ^ {-1}(\{z_0\})$ is closed.
For example
\[
\{z \in \C \mid |z| = 1\}
\]
for $f(z) = |z|$.

The set
\[
U = \{z = x + iy \in \C \mid (x ^ 2 + y ^ 2)\sin(\sqrt{x ^ 2 + 2}) > 2\}
\]
is open since $U = f ^ {-1}((2, \infty))$ where $f(x + iy) = (x ^ 2 + y ^ 2)\sin\sqrt{x ^ 2 + 2}$ continuous.

\subsection{Sequential Compactness}
A fundamental theorem in real analysis is Bolzano-Weierstrass that states that bounded sequences have converging subsequences.
This implies many results for continuous functions on closed bounded intervals or more generally compact sets.

\begin{definition}[Sequential Compactness]
    A non-empty subset $K$ of $\C$ is called sequentially compact if for any sequence $\{z_n\}_{n \in \N}$ in $K$ there exists a convergent subsequence,
    $\{z_{n_k}\}_{k \in \N}$,
    whose limit lies in $K$.
\end{definition}

\begin{example}
    $B_1(0) = \{z \in \C \mid |z| < 1\}$ is not sequentially compact.
    \[
    z_n = 1 - \frac{1}{n} \in B_1(0)
    \]
    $z_n \to 1$ as $n \to \infty$ but $1 \notin B_1(0)$.
\end{example}

There is a simple criteria for sequential compactness in $\C$.

\begin{theorem}[Heine-Borel for $\C$]    
    A subset $K$ of $\C$ is sequentially compact if and only if it is closed and bounded.

    \begin{proof}
        We'll show that if $K$ is bounded and closed,
        it is compact.
        Let $\{z_n\}_{n \in \N} \in K$ be given.
        We write
        \[
        z_n = x_n + iy_n\qquad x_n, y_n \in \R
        \]
        and see that $\{x_n\}_{n \in \N}$,
        $\{y_n\}_{n \in \N}$ are real sequences s.t.
        \[
        \max(|x_n|, |y_n|) \leq |z_n| \leq U
        \]
        i.e. $\{x_n\}$,
        $\{y_n\}$ are bounded.
        Using Bolzano-Weierstrass we can find a subsequence $\{x_{n_k}\}_{k \in \N}$ that converges to some $x \in \R$.
        \[
        z_{n_k} = x_{n_k} + iy_{n_k}
        \]
        $\{y_{n_k}\}_{k \in \N}$ is a subsequence of $\{y_n\}_{n \in \N}$ that is bounded.
        By Bolzano-Weierstrass there exists a subsequence of $\{y_{n_k}\}_{k \in \N}$,
        $\{y_{n_{k_j}}\}_{j \in \N}$,
        that converges to some $y \in \R$
        \[
        z_{n_{k_j}} = x_{n_{k_j}} + iy_{n_{k_j}}
        \]
        $x_{n_{k_j}}$ is a subsequence of $\{x_{n_k}\}_{k \in \N}$ which converges which implies it converges to $x$ too.

        We conclude that $\{z_{n_{k_j}}\}_{j \in \N}$ is a subsequence of $\{z_n\}_{n \in \N}$ that converges to $z = x + iy$
        (since $\Re(z_{n_{k_j}}) \to x$ and $\Im(z_{n_{k_j}}) \to y$).
        Since $K$ is closed,
        $\{z_{n_{k_j}}\}_{j \in \N} \subset K$ and $z_{n_{k_j}} \to z$ we must have that $z \in K$.
        The result is shown.
    \end{proof}
\end{theorem}
The above theorem holds in $\R ^ n$ and $\C ^ n$ for any $n \in \N$.
It is not true in general metric spaces.

\begin{example}\phantom{}
    \begin{itemize}
        \item $B_1(0)$ is not closed,
        so it is not compact.

        \item $\C$ is not compact since it is not bounded.

        \item $\{z \in \C\,:\, 1 \leq \Re(z) \leq 3, 0 \leq \Im(z) \leq 4\}$ is closed and bounded,
        and as such compact.
    \end{itemize}
\end{example}

\begin{theorem}[Extreme Value Theorem]
    Let $K$ be a sequentially compact subset of $\C$ and let $f : K \to \R$ be continuous.
    Then $f$ attains its minimum and maximum on $K$.
\end{theorem}

\begin{corollary}
    \begin{itemize}
        \item If $K$ is a closed and bounded set in $\C$ then $|z|$,
        $\Re(z)$,
        and $\Im(z)$ attain their minimum and maximum on $K$.

        \item If $K$ is a closed and bounded set in $\C$ and $f : K \to \C$ or $\R$ is continuous then $|f(z)|$ attains its minimum and maximum on $K$.
    \end{itemize}
\end{corollary}

\section{Complex Differentiation}

\subsection{Complex differentiability}

\begin{definition}[Complex differentiability]
    A function $f : U \to \C$ defined on an open set $U$ is $\C$ is
    (complex)
    differentiable at $z_0 \in U$ if
    \[
    \lim_{z \to z_0}\frac{f(z) - f(z_0)}{z - z_0}\text{ exists}.
    \]
    We call this limit the derivative of $f$ at $z_0$ and write $f'(z_0)$ for the limit,
    i.e.
    \[
    f'(z_0) = \lim_{h \to 0}\frac{f(z_0 + h) - f(z_0)}{h}.
    \]
\end{definition}

\begin{remark}
    While $f'(z_0)$ looks like the derivative in $\R$ we need to remember that $z \to z_0$ or $h \to 0$ is in the complex plane,
    i.e. we have infinitely many ways for $z$
    (or $h$)
    to approach $z_0$
    (or $0$).
    This is a two dimensional limit.

    One can show that if $f$ is complex differentiable at $z_0$,
    then it is continuous at $z_0$.
\end{remark}

\begin{example}
    Show that $f(z) = z ^ 2$ is complex differentiable at $z = 0$.

    \begin{solution}
        We must use the definition:
        \[
        \frac{f(z) - f(z_0)}{z - z_0} = \frac{z ^ 2 - 0 ^ 2}{z - 0} = z
        \]
        so
        \[
        \lim_{z \to 0}\frac{f(z) - f(0)}{z - 0} = \lim_{z \to 0}z = 0.
        \]
    \end{solution}
\end{example}

\begin{example}
    At which points is $f(z) = \bar{z}$ complex differentiable.

    \begin{solution}
        Let $z_0 \in \C$ be given
        \[
        \frac{f(z_0 + h) - f(z_0)}{h} = \frac{\bar{z}_0 + \bar{h} - \bar{z}_0}{h} = \frac{\bar{h}}{h}
        \]
        writing $h = re ^ {i\theta}$ we have
        \[
        h \to 0 \iff r \to 0\text{ and } \frac{\bar{h}}{h} = \frac{re ^ {-i\theta}}{re ^ {i\theta}} = e ^ {-2i\theta}
        \]
        this implies $\lim\frac{\bar{h}}{h}$ doesn't exist so $f(z)$ is not complex differentiable at any point.
    \end{solution}
\end{example}

\begin{lemma}[COLT for complex derivatives and the chain rule]\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item Let $f, g : \C \to \C$ be complex differentiable at $z_0 \in \C$.
        Then:
        \begin{itemize}
            \item $f + g$ is complex differentiable at $z_0$ and
            \[
            (f + g)'(z_0) = f'(z_0) + g'(z_0).
            \]

            \item (product rules)
            $fg$ is complex differentiable at $z_0$ and
            \[
            (fg)'(z_0) = f'(z_0)g(z_0) + g'(z_0)g(z_0).
            \]

            \item (quotient rule)
            if $g(z_0) \neq 0$ then $f / g$ complex differentiable at $z_0$ and
            \[
            \left(\frac{f}{g}\right)' = \frac{f'(z_0)g(z_0) - g'(z_0)f(z_0)}{(g(z_0)) ^ 2}
            \]
        \end{itemize}

        \item (chain rule)
        same as in $\R$.
    \end{enumerate}
\end{lemma}

\subsection{Cauchy-Riemann equations}
We want to simplify the method of showing that a function is complex differentiable.
Recall we can write
\[
f(z) = \Re(f(z)) + i\Im(f(z))
\]
with both functions being functions from $\C \to \R$.
We can identify sets in $\C$ as sets in $\R ^ 2$ and write
\[
f(z) = f(x + iy) = u(x, y) + iv(x, y)
\]
with $u, v$ as functions of two real variables to $\R$.

\begin{example}
    Find the real and imaginary parts of $f(z) = z ^ 2$.

    \begin{solution}
        \[
        f(z) = f(x + iy) = (x + iy) ^ 2 = x ^ 2 + 2ixy + i ^ 2y ^ 2 = (x ^ 2 - y ^ 2) + i(2xy)
        \]
        so $\Re(f(x + iy)) = u(x, y) = x ^ 2 - y ^ 2$,
        $\Im(f(x + iy)) = v(x, y) = 2xy$.
    \end{solution}
\end{example}

For functions from $\R ^ 2$ to $\R$ we have partial derivatives.

Is there a connection between the complex derivative of $f$ and partial derivatives of $\Re(f)$,
$\Im(f)$.

\begin{proposition}[Cauchy-Riemann equations]
    Let $f = u + iv$ be complex differentiable at $z_0 = x_0 + iy_0$.
    Then the real partial derivatives $u_x, u_y, v_x, v_y$ exist at $(x_0, y_0)$ and satisfy the Cauchy-Riemann equations:
    \[
    u_x(x_0, y_0) = v_y(x_0, y_0)\qquad u_y(x_0, y_0) = -v_x(x_0, y_0).
    \]
    Furthermore,
    the derivative of $f$ at $z_0$ can be written as
    \begin{align*}
        f'(z_0) &= u_x(z_0) + iv_x(z_0) = v_y(z_0) - iu_y(z_0) \\
        &= u_x(z_0) - iu_y(z_0) = v_y(z_0) + iv_x(z_0)
    \end{align*}
    where we have used the shorthand notation $w(z_0) = w(x_0, y_0)$ for functions from $\R ^ 2$ to $\R$.

    \begin{proof}
        The proof relies on the realisation that since $\lim_{n \to 0}\frac{f(z_0 + h_c) - f(z_0)}{h_c}$ exists $h_c$ can go to $0$ in any
        (complex)
        way.

        Let's consider two ways,
        we consider $h_c = ih$,
        $h \in \R$,
        $h \to 0$
        (in $\R$)
        \begin{align*}
            f'(z_0) &= \lim_{h \to 0}\frac{f(z_0 + ih) - f(z_0)}{ih} \\
            &= \lim_{h \to 0}\frac{f(x_0 + i(y_0 + h)) - f(x_0 + iy_0)}{ih} \\
            &= \lim_{h \to 0}\frac{(u(x_0, y_0 + h) + iv(x_0, y_0 + h)) - (u(x_0, y_0) + iv(x_0, y_0))}{ih} \\
            &= \lim_{h \to 0}\left(\frac{v(x_0, y_0 + h) - v(x_0, y_0)}{h}\right) - i\left(\frac{u(x_0, y_0 + h) - u(x_0, y_0)}{h}\right).
        \end{align*}
        Since this limit exists
        \[
        \Re(f'(z)) = \lim_{h \to 0}\frac{v(x_0, y_0 + h) - v(x_0, y_0)}{h} = v_y
        \]
        exists and
        \[
        \Im(f'(z_0)) = -\lim_{h \to 0}\frac{u(x_0, y_0 + h) - u(x_0, y_0)}{h} = -u_y
        \]
        so we conclude $v_y(x_0, y_0)$ and $u_y(x_0, y_0)$ exist and $f'(z_0) = v(x_0, y_0) - iu_y(x_0, y_0)$.
        
        Similarly for $h_c = h$.
        \[
        f'(z_0) = \lim_{h \to 0}\frac{f(z_0 + h) - f(z)}{h} = \lim_{h \to 0}\frac{f(x_0 + h + iy_0) - f(x_0 + iy_0)}{h}
        \]
        continuing as above we get that $u_x(x_0, y_0), v_x(x_0, y_0)$ exist and $\Re(f'(z_0)) = u_x(x_0, y_0)$,
        $\Im(f'(z_0)) = v_x(x_0, y_0)$,
        i.e.
        \[
        f'(z_0) = u_x(x_0, y_0), + iv_x(x_0, y_0) = v_y(x_0, y_0) - iu_y(x_0, y_0)
        \]
        which implies $u_x(x_0, y_0) = v_y(x_0, y_0)$ and $v_x(x_0, y_0) = -u_y(x_0, y_0)$.
    \end{proof}
\end{proposition}

\textit{Note:
The theorem doesn't tell us when $f$ is complex differentiable.
It tells us that if the Cauchy-Riemann equations do not hold,
$f$ is not complex differentiable.}

We need a converse
\begin{theorem}
    Let $f = u + iv$ be defined on an open subset $U$ of $\C$.
    Then,
    $f$ is complex differentiable at $z_0 = x_0 + iy_0 \in U$ if and only if $u$ and $v$ are
    (real)
    differentiable at $(x_0, y_0)$ and the Cauchy-Riemann equations are satisfied at that point.

    \begin{proof}
        Omitted.
    \end{proof}
\end{theorem}
We use $U$ as a subset of $\C$ and $\R ^ 2$ interchangeably.

In particular:
\begin{corollary}
    Let $f = u + iv$ be defined on an open subset $U$ of $\C$.
    Assume the partial derivatives $u_x, u_y, v_x, v_y$ exist,
    are continuous,
    and satisfy the Cauchy-Riemann equations at $z_0 \in U$.
    Then $f$ is complex differentiable at $z_0$.
\end{corollary}

Consider $f(z) = e ^ z = e ^ xe ^ {iy} = e ^ x\cos(y) + ie ^ x\sin(y)$,
$u(x, y) = e ^ x\cos(y)$,
$v(x, y) = e ^ x\sin(y)$.
$u, v \in C ^ {1}(\R ^ 2)$.
Consequently,
$f(z)$ is differentiable at $z_0 = x_0 + iy_0$ if and only if the Cauchy-Riemann equations hold at $(x_0, y_0)$,
\begin{align*}
    u_x(x, y) &= e ^ x\cos(y) & u_y(x, y) &= -e ^ x\sin(y) \\
    v_x(x, y) &= e ^ x\sin(y) & v_y(x, y) &= e ^ x\cos(y)
\end{align*}
$u_x = v_y$ and $v_x = -u_y$ so Cauchy-Riemann equations satisfied at any $(x, y)$ so $e ^ z$ is complex differentiable at any $z \in \C$.
Moreover,
\[
f'(z) = u_x(z) + iv_x(x) = e ^ x\cos(y) + ie ^ x\sin(y) = e ^ xe ^ {iy} = e ^ z.
\]

$f(z) = e ^ {iz}$ is complex differentiable on $\C$ by the chain rule and $f'(z) = ie ^ {iz}$ so
\[
\sin(z) = \frac{e ^ {iz} - e ^ {-iz}}{2i}
\]
is complex differentiable by COLT
\[
(\sin(z))' = \cos(z).
\]
Similarly $(\cos(z))' = -\sin(z)$.
$(\cosh(z))' = \sinh(z)$ and $(\sinh(z))' = \cosh(z)$ by the same method.

Polynomials are complex differentiable on $\C$ with
\[
(a_nz ^ n + \dotsi + a_1z + a_0)' = na_nz ^ {n - 1} + \dotsi + a_1.
\]
Any branch of $\log(z)$ on $\C \setminus R_{\theta_1}$ is complex differentiable on $\C \setminus R_{\theta_1}$ and $(\log(z))' = \frac{1}{z}$.

\begin{example}
    Let $f(z) = f(x + iy) = (x ^ 3 + 3x ^ 2y - y ^ 3 - x ^ 2 - 2y ^ 2) + i(-x ^ 3 + 3xy ^ 2 - y ^ 3 + 4xy + 3y)$.
    At which points is $f$ complex differentiable?

    \begin{solution}
        \begin{align*}
            u(x, y) &= x ^ 3 + 3x ^ 2y - y ^ 3 - x ^ 2 - 2y ^ 2 \\
            v(x, y) &= -x ^ 3 + 3xy ^ 2 - y ^ 3 + 4xy + 3y
        \end{align*}
        $u, v \in C ^ 2(\R ^ 2)$ so $f$ is complex differentiable at $z_0 = x_0 + iy_0$ if and only if the Cauchy-Riemann equations hold at $(x_0, y_0)$.

        If you do this,
        you get the the Cauchy Riemann equations hold when
        \[
        3x ^ 2 - 2x = -3y ^ 2 + 4x + 3
        \]
        which can be written as
        \[
        (x - 1) ^ 2 + y ^ 2 = 2.
        \]
    \end{solution}
\end{example}

\subsection{Holomorphicity}
Differentiability at one point is nice,
but usually not enough.
Most theorems in Real Analysis want differentiability on an interval.
A good candidate to replace the interval with is an open ball or in general,
an open set.

\begin{definition}[Holomorphic functions]
    A function $f : U \to \C$ defined on an open set $U \subseteq \C$ is holomorphic on $U$ if it is complex differentiable at every point in $U$.
    
    We say $f$ is holomorphic at $z_0$ if it is holomorphic on an open ball $B_{\varepsilon}(z_0)$ for some $\varepsilon > 0$.
\end{definition}

A function which is holomorphic on $\C$ is called an entire function.

$e ^ z, \sin(z), \cos(z), \sinh(z), \cosh(z)$,
polynomials are entire.

Any branch of $\log(z)$ on $\C \setminus R_{\theta}$ is holomorphic.

The function
\[
f(z) = f(x + iy) = (x ^ 3 + 3x ^ 2y - y ^ 3 - x ^ 2 - 2y ^ 2) + i(-x ^ 3 + 3xy ^ 2 - y ^ 3 + 4xy + 3y)
\]
is complex differentiable on 
\[
(x - 1) ^ 2 + y ^ 2 = 2
\]
no point of differentiability has a full open ball around it where $f$ is complex differentiable.
So $f$ is not holomorphic at any point.

Open sets are not good enough even in $\R$.
For instance we know that $f' = 0$ on $(a, b)$ implies $f$ is constant but
\[
f(x) = \begin{cases}
    1 & x \in (1, 3) \\
    2 & x \in (4, 5)
\end{cases}
\]
is $C ^ 1$ on $U = (1, 3) \cup (4, 5)$,
$f' = 0$ but $f$ is not constant.

\begin{definition}[Connectedness]
    A set $A \subseteq \C$ is said to be connected if $A \subseteq U_1 \cup U_2$ for some open sets $U_1$ and $U_2$ such that $U_1 \cap U_2 = \emptyset$ implies that $A \subseteq U_1$ or $A \subseteq U_2$.
    In other words,
    we cannot separate $A$ into two non-empty parts which lie in two disjoint open sets.
\end{definition}

Connectedness can be hard to show,
but intuitively if we can find a nice "path" between any two points in the set,
we think this will imply connectedness.

\begin{definition}[Paths and path connectedness]
    A path
    (or curve)
    in $\C$ is a continuous function $\gamma : [0, 1] \to \C$.
    We say that the path starts at $z \in \C$ and ends at $w \in \C$ if $\gamma(0) = z$ and $\gamma(1) = w$.
    A set $A \subseteq \C$ is said to be path connected if for any two points $z, w \in A$ there exists a path $\gamma$ that starts at $z$ and ends at $w$ such that $\gamma(t) \in A$ for all $t \in [0, 1]$
    (i.e. the path's image lies entirely inside $A$).
\end{definition}

\begin{theorem}
    If $A \subseteq \C$ is path connected then it is connected.
\end{theorem}

\begin{remark}
    The converse to the above theorem is not true in general:
    there are connected sets that are not path connected.
\end{remark}

A path $\gamma : [0, 1] \to \C$ can be thought of as two functions from $[0, 1]$ to $\R$.
indeed $\gamma(t) = \Re(\gamma(t)) + i\Im(\gamma(t))$.
We can use results from Real Analysis.

\begin{definition}\phantom{}
    \begin{itemize}
        \item A path
        (or curve)
        in $\C$ is said to be continuously differentiable,
        or $C ^ 1$,
        if its real part or imaginary part are continuously differentiable on $[0, 1]$.
        At the end points $0$ and $1$ this means that the real and imaginary parts have right-sided derivative at $0$ and left-sided derivative $1$,
        and that the derivatives are continuous from the right at $0$ and from the left at $1$.
        In that case we define
        \[
        \gamma'(t) \coloneqq (\Re(\gamma(t)))' + i(\Im(\gamma(t)))'.
        \]

        \item A path
        (or curve)
        in $\C$ is said to be a piecewise $C ^ 1$ path
        (or curve)
        or a contour if there exist
        \[
        0 = a_0 < a_1 < a_2 < \dotsi < a_{n - 1} < a_n = 1
        \]
        such that the paths $\gamma_i : [a_{i - 1}, a_i] \to \C$,
        with $i = 1, 2, \dotsc, n$ defined by
        \[
        \gamma_i(t) \coloneqq \gamma(t),\qquad t \in [a_{i - 1}, a_i]
        \]
        are $C ^ 1$ paths.

        \item We say a subset $A \subseteq \C$ is $C ^ 1$ path connected if for every pair of points $z, w \in A$ there exists a $C ^ 1$ path starts at $z$ and ends at $w$ such that $\gamma(t) \in A$ for all $t \in [0, 1]$.

        \item We say that $A$ is piecewise $C ^ 1$ path connected if for every pair of points $z, w \in A$ there exists a contour that starts at $z$ and ends at $w$ such that $\gamma(t) \in A$ for all $t \in [0, 1]$.
    \end{itemize}
\end{definition}

As any $C ^ 1$ or piecewise $C ^ 1$ path are paths $C ^ 1$ path connectedness or piecewise $C ^ 1$ path connectedness implies path connectedness.

Paths are actually continuous functions $\gamma : [a, b] \to \C$.

\begin{example}\phantom{}
    \begin{itemize}
        \item Open and closed balls are $C ^ 1$ path connected
        (every two points can be connected by a straight line inside the set).

        \item $\bar{\mathbb{H}} = \{z \in \C\,:\, \Im(z) \geq 0\}$ is $C ^ 1$ path connected
        (same as above).

        \item The annulus $A = \{z \in \C\,:\, 1 < |z| < 4\}$ is $C ^ 1$ path connected
        (but not every two points can be connected with a line).
    \end{itemize}
\end{example}

\begin{definition}[Domains]
    A domain
    (sometimes called regions)
    $D$ is an open,
    connected subset of $\C$
\end{definition}

The geometry of $\C$ provides.

\begin{theorem}
    Any domain in $\C$ is piecewise $C ^ 1$ path connected.
\end{theorem}

\begin{example}\phantom{}
    \begin{itemize}
        \item Open ball is a domain.

        \item Closed ball is not open and as such not a domain.

        \item $\C \setminus R_{\leq 0}$
        (a branch of log)
        is a domain.

        \item Any branch of $\log(z)$ and $z ^ {w}$ is defined on a domain $\C \setminus R_{\theta}$.

        \item The set $A = \{z \in \C\,:\, |z| \neq 1\}$ is not a domain
        (it is open but not connected)
    \end{itemize}
\end{example}

\begin{theorem}[Zero derivative theorem]\label{companalyII:thm:zerodivthm}
    Let $f : D \to \C$ be holomorphic on a domain $D \subseteq \C$.
    If $f'(z) = 0$ for every $z \in D$ then $f$ is constant on $\mathbb{D}$.
\end{theorem}

We will need the following in the proof:
\begin{lemma}[Chain rule revisited
(for paths)]
    Let $U \subset \C$ be an open set and let $f : U \to \C$ be a holomorphic function on $U$.
    Let $\gamma : [0, 1] \to U$ be a differentiable path
    (i.e. its real and imaginary parts are differentiable).
    Then for any $t_0 \in [0, 1]$ we have that
    \[
    (f \circ \gamma)'(t_0) = f'(\gamma(t_0))\gamma'(t_0).
    \]

    \begin{proof}
        Like in Real Analysis.
    \end{proof}
\end{lemma}

\begin{theorem}[continues = companalyII:thm:zerodivthm]
    \begin{proof}
        It is enough to show that for any $z_1, z_2 \in D$
        \[
        f(z_1) = f(z_2).
        \]

        Since $D$ is a domain we can find a contour $\gamma : [0, 1] \to \C$ s.t. $\gamma(0) = z_1$,
        $\gamma(1) = z_2$.

        We can find
        \[
        0 = a_0 < a_1 < \dotsi < a_n = 1
        \]
        s.t. $\gamma_i = \gamma|_{[a_{i - 1}, a_i]}$ is $C ^ 1$.

        If we'll show that
        \[
        f(\gamma_i(a_{i - 1})) = f(\gamma_i(a_i))\qquad \forall i
        \]
        then
        \[
        f(z_1) = f(\gamma(a_0)) = f(\gamma(a_1)) = f(\gamma(a_2)) = \dotsi = f(\gamma(a_n)) = f(z_2).
        \]
        Indeed:
        \[
        f \circ \gamma_i : [a_{i - 1}, a_i] \to \C
        \]
        is continuous and by the chain rule for paths it is differentiable on $(a_{i - 1}, a_i)$ with $(f \circ \gamma_i)'(t) = f'(\gamma_i(t))\cdot\gamma_i'(t) = 0$ so $(\Re(f \circ \gamma_i))' = 0$,
        $(\Im(f\circ\gamma_i))' = 0$ on $(a_{i - 1}, a_i)$ so by zero derivative theorem on $\R$.
        $\Re(f\circ \gamma_i)$ is constant and $\Im(f \circ \gamma_i)$ is constant on $[a_{i - 1}, a_i]$,
        completing the proof.
    \end{proof}
\end{theorem}

The beauty of complex analysis is that to have a constant holomorphic function we only need that $f$ doesn't fill the space
($\Re(f)$ is constant,
$(\Re(f)) ^ 2 + (\Im(f)) ^ 2 = 1$).

\subsection{The angle preserving properties of holomorphic functions}
There is almost a purely geometric description to being holomorphic.
For any curve $\gamma(t) : [0, 1] \to \C$ we can consider the tangent vector to $\gamma$ at $t_0$ which is
\[
\gamma'(t_0) = (\Re(\gamma))'(t_0) + i(\Im(\gamma))'(t_0).
\]

\begin{definition}[Conformal maps]
    We say that a
    (real differentiable)
    map $f : D \to \C$ on a domain $D \subseteq \C$ is conformal at $z_0$ if it preserves the angle and orientation between any two tangent vectors at $z_0$,
    where by orientation we mean the relative position of one vector with respect to another
    (i.e. if the tangent vector of a curve $\gamma_1$ was to the left of the tangent vector of the curve $\gamma_2$,
    the tangent vector of the image $\gamma_1$ by $f$ will remain to the left of the tangent vector of the image of $\gamma_2$ by $f$).
\end{definition}

We say that $f$ is conformal on $D$ if it is conformal at all points in $D$.

\begin{example}\phantom{}
    \begin{itemize}
        \item Rotations are conformal
        \[
        f(z) = e ^ {i\theta}z\qquad \theta \in \R.
        \]

        \item Dialation is conformal
        \[
        f(z) = \lambda z\qquad \lambda > 0.
        \]

        \item $f(z) = \bar{z}$ is not conformal.
        We'll only do it at $z_0$.
        The $C ^ 1$ paths $\gamma_1(t) = t$,
        $\gamma_2(s) = is$,
        $t \in [0, 1]$,
        $s \in [0, 1]$ meet at $z = 0$
        ($t = 0$,
        $s = 0$)
        with an angle $\pi / 2$.
        However,
        $f(\gamma_1(t)) = t$,
        $f(\gamma_2(s)) = -is$ meet at $z = 0$ with an angle $-\pi / 2$.
    \end{itemize}
\end{example}

\begin{lemma}[Holomorphic maps are conformal when their derivative is non-zero]
    Let $f$ be a holomorphic map at $z_0$.
    If $f'(z_0) \neq 0$ then $f$ is conformal at $z_0$.

    \begin{proof}
        Let $\gamma : [0, 1] \to \C$ be a differentiable curve and let $t_0 \in [0, 1]$ be such that $\gamma(t_0) = z_0$.
        What does $f$ do to the tangent vector.
        By the chain rule for paths
        \[
        (f \circ \gamma)'(t_0) = f'(\gamma / t_0) \cdot \gamma'(t_0) = f'(z_0) \cdot \gamma'(t_0).
        \]
        Since $f'(z_0) \neq 0$.
        The tangent vector of $f \circ \gamma$ at $t_0$ is nothing but dialation by $|f'(z_0)|$ and rotation by $\Arg(f'(z_0))$ of the tangent $\gamma'(t_0)$.
        Since it is the same rotation and dialation for all $\gamma$'s,
        $f$ is conformal.
    \end{proof}
\end{lemma}

\begin{remark}
    The condition $f'(z_0) \neq 0$ is necessary.
    $f(z) = z ^ 2$ satisfies $f'(0) = 0$.
    $f$ is not conformal at $z = 0$
    (the angle between the curves is not always preserved).
\end{remark}

Do we have a converse?

\begin{proposition}[Conformal maps are holomorphic]
    Let $D$ be a domain.
    If $f$ is conformal at $z_0 \in D$ then $f$ is complex differentiable at $z_0$ and $f'(z_0) \neq 0$.
    Therefore,
    if $f$ is conformal on $D$,
    then $f$ is holomorphic on $D$ and $f'(z) \neq 0$ for all $z \in D$.
\end{proposition}

\begin{corollary}
    Let $D$ be a domain.
    Then $f$ is conformal on $D$ if and only if $f$ is holomorphic on $D$ and $f'(z) \neq 0$ for all $z \in D$.
\end{corollary}

\begin{corollary}
    Any conformal map maps orthogonal grids in the $(x, y)$-plane to orthogonal grids.
\end{corollary}

\begin{example}
    $f(z) = z ^ 2$ and consider $z \neq 0$.
    $x = c$,
    $(u, v) = (c ^ 2 - y ^ 2, 2cy)$.
    i.e. $u = c ^ 2 - \frac{v ^ 2}{4c ^ 2}$ a parabola.
    $y = d$
    $(u, v) = (x ^ 2 - d ^ 2, 2dx)$,
    $u = \frac{v ^ 2}{4d ^ 2} - d ^ 2$.
\end{example}
