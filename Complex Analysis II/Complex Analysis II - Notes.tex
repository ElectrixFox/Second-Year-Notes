\section{The Complex Plane and Complex Functions}

\subsection{Complex Numbers}

Complex numbers are of the form $z = x + iy$ where $i$ is the imaginary unit,
$i = \sqrt{-1}$.
The set of complex numbers is denoted $\C$.
We use an Argand diagram to visualise complex numbers $x + iy$ with $(x, y)$.

For $z = x + iy$ we have the real part of $z$,
$\mathrm{Re}(z) = x$,
and the imaginary part of $z$,
$\Im(z) = x$.

Addition of complex numbers,
$z_1 = x_1 + iy_1$,
$z_2 = x_2 + iy_2$
\[
z_1 \pm z_2 = (x_1 \pm x_2) + i(y_1 \pm y_2)
\]
the real parts are added together and the imaginary parts are added together.

Multiplication of complex numbers,
$z_1 = x_1 + iy_1$,
$z_2 = x_2 + iy_2$
\[
z_1z_2 = (x_1 + iy_1)(x_2 + iy_2) = (x_1x_2 - y_1y_2) + i(x_1y_2 + x_2y_1)
\]
just expand and use $i ^ 2 = -1$.

Looking at
\[
(x + iy)(x - iy) = x ^ 2 + ixy - ixy - i ^ 2y ^ 2 = x ^ 2 + y ^ 2 \in \R.
\]
We call the conjugate of a complex number as follows,
given $z = x + iy$ the conjugate is
\[
\bar{z} = x - iy = \mathrm{Re}(z) - i\mathrm{Im}(z).
\]
We can see that
\[
z\bar{z} = \mathrm{Re}(z) ^ 2 + \Im(z) ^ 2 \in \R
\]
which leads us to
\[
|z| = \sqrt{z\bar{z}} = \sqrt{\mathrm{Re}(z) ^ 2 + \Im(z) ^ 2}
\]
the modulus
(or absolute value)
of $z$.

Using the conjugate we can see
\[
\frac{z_1}{z_2} = \frac{x_1 + iy_1}{x_2 + iy_2} = \frac{(x_1 + iy_1)(x_2 - iy_2)}{(x_2 + iy_2)(x_2 - iy_2)} = \frac{x_1x_2 + y_1y_2}{x_2 ^ 2 + y_2 ^ 2} + i\frac{x_2y_1 - x_1y_2}{x_2 ^ 2 + y_2 ^ 2}.
\]

\begin{example}
    Find $\frac{3}{1 + i} - (3 + 2i)$.

    \begin{solution}
        \[
        \frac{3}{1 + i} - (3 + 2i) = \frac{3(1 - i)}{(1 + i)(1 - i)} - (3 + 2i) = \frac{3 - 3i}{2} - 3 - 2i = -\frac{3}{2} - \frac{7}{2}i.
        \]
    \end{solution}
\end{example}

Conjugation is nothing but reflection with respect to the real axis.

\begin{lemma}[Important Properties of Complex Numbers]
    \begin{enumerate}[label = (\roman*)]
        \item $z_1z_2 = 0 \iff z_1 = 0\text{ or } z_2 = 0$.

        \item $|z| = \sqrt{z\bar{z}}$.

        \item $\overline{(zw)} = \bar{z}\bar{w}$.

        \item $\Re(z) = \frac{z + \bar{z}}{2}$ and $\Im(z) = \frac{z - \bar{z}}{2i}$.

        \item $z ^ {-1} = \frac{\bar{z}}{|z| ^ 2}$.
    \end{enumerate}
\end{lemma}

Using polar coordinates and the Argand diagram we can write complex numbers in polar coordinates.
We define for any $z \in \C$
\[
r \coloneqq \sqrt{x ^ 2 + y ^ 2} = \sqrt{\Re(z) ^ 2+ \Im(z) ^ 2} = |z|
\]
and any $z \neq 0$
\[
\theta \coloneqq \begin{cases}
    \arctan(y / x) & x > 0, \\
    \pi / 2 & x = 0, y > 0, \\
    -\pi / 2 & x = 0, y < 0, \\
    \arctan(y / x) + \pi &x < 0, y > 0, \\
    \arctan(y / x) - \pi &x < 0, y < 0.
\end{cases}
\]
$\theta$ is the anticlockwise angle measured from the real axis,
$\theta$ is called the argument of $z$,
denoted $\arg(z)$.

We have the following polar coordinates for $z$:
\[
z = x + iy = r\cos(\theta) + ir\sin(\theta) = r(\cos(\theta) + i\sin(\theta)).
\]

\begin{example}
    $z = i$.

    \[
    r = |z| = \sqrt{0 ^ 2 + 1 ^ 2} = 1
    \]
    \[
    \theta = \pi / 2.
    \]
    \[
    i = 1(\cos(\pi / 2) + i\sin(\pi / 2)) = \cos(\pi / 2) + i\sin(\pi / 2).
    \]

    $z = 1 + i$
    \[
    |z| = \sqrt{1 ^ 2 + 1 ^ 2} = \sqrt{2},
    \]
    $\theta = \frac{\pi}{4}$,
    \[
    z = \sqrt{2}(\cos(\pi / 4) + i\sin(\pi / 4)).
    \]
\end{example}

\textbf{Important:}
$\arg(z)$ is only defined up to multiples of $2\pi$.

$\arg(i) = \frac{\pi}{2} + 2\pi k$,
$k \in \Z$,
not a function.

To define an argument we need a range of angles.

The principal value of $\arg(z)$ is the value in the interval $(-\pi, \pi]$ denoted $\Arg(z)$.

For example:
$\Arg(i) = \frac{\pi}{2}$,
$\Arg(-1) = \pi$.

\begin{lemma}[Properties of argument]
    We have the following properties of the argument:
    
    \begin{enumerate}[label = (\roman*)]
        \item $\arg(z_1z_2) = \arg(z_1) + \arg(z_2)\mod{2\pi}$.

        \item $\arg(1 / z) = -\arg(z)\mod{2\pi}$.

        \item $\arg(\bar{z}) = -\arg(z)\mod{2\pi}$.
    \end{enumerate}
\end{lemma}
\textit{Note:
the proof relies on why we went to polar coordinates:
multiplication is clear.}

\begin{lemma}
    Geometrically,
    multiplication in $\C$ is given by a dilated rotation;
    i.e.,
    if $z_1 = r_1(\cos(\theta_1) + i\sin(\theta_1))$ and $z_2 = r_2(\cos(\theta_2) + i\sin(\theta_2))$ then
    \[
    z_1z_2 = r_1r_2(\cos(\theta_1 + \theta_2) + i\sin(\theta_1 + \theta_2)).
    \]
    In particular,
    multiplying by $z_2$ constitutes an
    (anticlockwise)
    rotation of $z_1$ by $\theta_2$ degrees,
    followed by a dilation with factor $r_2$.

    \begin{proof}
        \begin{align*}
            z_1z_2 &= r_1(cos(\theta_1) + i\sin(\theta_2))r_2(\cos(\theta_2) + i\sin(\theta_2)) \\
            &= (r_1\cos(\theta_1) + ir_1\sin(\theta_1))(r_2\cos(\theta_2) + ir_2\sin(\theta_2)) \\
            &= (r_1r_2\cos(\theta_1)\cos(\theta_2) - r_1r_2\sin(\theta_1)\sin(\theta_2)) \\
            &+ i(r_1r_2\sin(\theta_1)\cos(\theta_2) + r_1r_2\cos(\theta_1)\sin(\theta_2)) \\
            &= r_1r_2[(\cos(\theta_1)\cos(\theta_2) - \sin(\theta_1)\sin(\theta_2)) + i(\sin(\theta_1)\cos(\theta_2) + \cos(\theta_1)\sin(\theta_2))] \\
            &= r_1r_2(\cos(\theta_1 + \theta_2) + i\sin(\theta_1 + \theta_2)).
        \end{align*}
    \end{proof}
\end{lemma}
\textit{Note:
this proves $\arg(z_1z_2) = \arg(z_1) + \arg(z_2)\mod{2\pi}$,
but also:
$\frac{1}{z} = \frac{\bar{z}}{|z| ^ 2}$ so $\arg(1 / z) = \arg(\bar{z})\mod{2\pi}$,
or $0 = \arg(z \cdot 1 / z) = \arg(z) + \arg(1 / z)\mod{2\pi}$.}

Useful notation:
write $\cos(\theta) + i\sin(\theta) \coloneqq e ^ {i\theta}$.

Motivated by
\[
e ^ {i\theta_1}e ^ {i\theta_2} = (\cos(\theta_1) + i\sin(\theta_1))(\cos(\theta_2) + i\sin(\theta_2)) = \cos(\theta_1 + \theta_2) + i\sin(\theta_1 + \theta_2) = e ^ {i(\theta_1 + \theta_2)}
\]

\begin{corollary}\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item $|z_1z_2| = |z_1||z_2|$.
        
        \item De Moivre's formula:
        $(\cos(\theta) + i\sin(\theta)) ^ n = \cos(n\theta) + i\sin(n\theta)$.
    \end{enumerate}

    \begin{proof}
        \begin{enumerate}[label = (\roman*)]
            \item Exercise.

            \item Done with induction.
        \end{enumerate}
    \end{proof}
\end{corollary}

The modulus has the following properties.
\begin{enumerate}[label = (\roman*)]
    \item
    (Triangle inequality)
    $|z_1 + z_2| \leq |z_1| + |z_2|$.

    \item $|z| \geq 0$ and $|z| = 0 \iff z = 0$.

    \item $\max(|\Re(z)|, |\Im(z)|) \leq |z| \leq |\Re(z)| + |\Im(z)|$.
\end{enumerate}

We can use $\arg(z)$,
$|z|$ to describe geometric sets.
$\mathbb{D} \coloneqq \{z \in \C\,:\, |z| < 1\}$ -
Unit disc.

$\mathbb{H} = \{z \in \C\,:\, \Im(z) > 0\}$ -
Upper half plane
(without the $x$ axis).

$\mathbb{H}_{R} \coloneqq \{z \in \C,\:\, \Re(z) > 0\}$ -
Right half plane
(without the $y$ axis).

$\mathbb{H}_{L} \coloneqq \{z \in \C,\:\, \Re(z) < 0\}$ -
Left half plane
(without the $y$ axis)

$\arg(z) = \frac{\pi}{4}$ -
Ray of angle $\pi / 4$.

$\{z \in \C\,:\, |z - i| = 4\}$ -
Circle centred at $i$ with radius $4$.

\begin{example}
    What is the set $|z - i| = |z + i|$?

    \begin{solution}
        The real line.

        $z = x + iy$
        \begin{align*}
            |x + i(y - 1)| = |x + i(y + 1)| &\iff |x + i(y - 1)| ^ 2 = |x + i(y + 1)| ^ 2 \\
            &\iff x ^ 2 + (y - 1) ^ 2 = x ^ 2 + (y + 1) ^ 2 \\
            &\iff 4y = 0 \implies y = 0.
        \end{align*}

        With $z$:
        one can show
        \[
        |z - \omega| ^ 2 = |z| ^ 2 - 2\Re(z\bar{\omega}) + |\omega| ^ 2.
        \]
    \end{solution}
\end{example}

\subsection{Complex Functions}

Complex functions are $f : \C \to \C$.
To visualise the graph of complex functions we need four dimensions.
We can always write
\[
f(z) = \Re(f(z)) + i\Im(f(z)).
\]
This is not enough to visualise it.

To understand complex functions we usually try to see how they take sets from $\C$ to sets in $\C$.
We will explore
"elementary"
functions like polynomials,
exponentials,
trigonometric functions,
logarithms and powers.

\subsection{The function \texorpdfstring{$f(z) = z ^ n$}{} with \texorpdfstring{$n \in \N$}{}}
To understand $z ^ n$ we will use polar representation:
if $z = re ^ {i\theta}$.
By our lemma,
\[
z ^ n = r ^ ne ^ {in\theta}
\]
if we fix $r$,
i.e. we are on a circle $z ^ n$ dilates our radius by power of $n$
($r \mapsto r ^ n$) and multiplies our argument by a factor of $n$
($\theta \to n\theta\mod{2\pi}$).

A segment of angular length $\eta$ on a circle of radius $r$ becomes an angular segment of length $n\eta\mod{2\pi}$ if $n\eta < 2\pi$ and the full circle if $n\eta \geq 2\pi$ with radius $r ^ n$.

If $n\eta > 2\pi$,
$z ^ n$ will not be injective.

We can slice the world to wedges of size $2\pi / n$ on which $z ^ n$ is injective.
For instance
\[
S_{k}^{(n)} = \{z \in \C\,:\,-\pi < 2\pi k / n < \Arg(z) \leq -\pi + 2\pi(k + 1) / n\}
\]
where $k = 0, \dotsc, n - 1$.
Or
\[
R_{k}^{(n)} = \{z \in \C\,:\, 2\pi k / n\mod{2\pi} < \Arg(z) \leq 2\pi(k + 1) / n\mod{2\pi}\}
\]
$k = 0, \dotsc, n - 1$.

\textit{Note:
when $n$ is even $R ^ {(n)}$s and $S ^ {(n)}$s will overlap
(not true if $n$ is odd).}

We now want to fix $\theta$.
Geometrically that a fixed $\theta$ is a ray
(half line)
without the origin.

The ray of angle $\theta$ is the set
\[
\{z \in \C \setminus \{0\}\,:\,\Arg(z) = \theta\}
\]
The closed ray of angle $\theta$ is the way of angle $\theta$ together with the origin $z = 0$,
i.e.
\[
\{z \in \C \setminus \{0\}\,:\,\Arg(z) = \theta \cup \{0\}\}.
\]

If $\theta$ is fixed $z ^ n = r ^ ne ^ {in\theta}$ which takes a segment on the ray $R_{\theta}$
($0 \leq a < r < b$)
will go to a segment on the ray $R_{n\theta\bmod{2\pi}}$
($a ^ n < r ^ n < b ^ n$).

\begin{lemma}
    \begin{itemize}
        \item The map $z ^ n$ injectively takes an angular segment of length $\frac{2\pi}{n}$ which is open at one end and closed at the other from a circle of radius $r$ to the entire circle of radius $r ^ n$.
        If the above segment is closed,
        or its size is larger than $\frac{2\pi}{n}$ the image is no longer injective.

        \item The map $z ^ n$ injectively takes a ray of angle $\theta$ to a ray of angle $n\theta\bmod{2\pi}$.
        Consequently,
        if $n|\theta_1 - \theta_2| \leq 2\pi$ the map $z ^ n$ injectively takes the wedge
        \[
        W = \{z \in \C\,:\, \theta_1 < \Arg(z) < \theta_2\}
        \]
        to the wedge
        \[
        W = \{z \in \C\,:\, n\theta_1\bmod{2\pi} < \Arg(z) <n\theta_2\bmod{2\pi}\}
        \]
        with the convention that for any $-\pi < \nu < \xi \leq \pi$
        \[
       \left\{z\in\mathbb{C}\;:\; \xi<\mathrm{Arg}(z)<\upsilon\right\} = \left\{ z\in\mathbb{C}\;:\; \xi<\mathrm{Arg}(z)\leq \pi\right\} \cup \left\{ z\in\mathbb{C}\;:\; -\pi<\mathrm{Arg}(z)<\upsilon\right\}. 
        \]
        If we replace one strict inequality
        (i.e. $<$)
        with an inequality
        (i.e. $\leq$)
        the above claim remains true.
        In this case,
        if $n|\theta_1 - \theta_2| = 2\pi$ the image of the map will be $\C \setminus \{0\}$.
        We can get the entire complex plane by adding $z = 0$ to $W$.
        When $n|\theta_1 - \theta_2| > 2\pi$ the image of $W$ by $z ^ n$ will be $\C \setminus \{0\}$ but not injectively.

        \item We can define $n$ different $n$th roots which are the inverses of the map $z ^ n$.
        We can write them all in the form
        \[
        z^{\frac{1}{n}} = \left\lvert z\right\rvert^{\frac{1}{n}}e^{i\left( \frac{\mathrm{Arg}\left( z\right)}{n} + \frac{2\pi k }{n}\right)}
        \]
        with $k = 0, \dotsc, n - 1$
        (we remove the $k$ index from the map,
        but one need to specify which $k$ we are talking about).
        note that for a fixed $k$ the $n$th root of $z$ takes $\C$ into a anti-clockwise rotation by $\frac{\pi}{n}$ of $R_{k - 1}^{(n)}$.
    \end{itemize}
\end{lemma}

Notation:
we want
\[
e ^ z = e ^ {x + iy} = e ^ x\cdot e ^ {iy}
\]
defined $e ^ {iy}$ earlier.

We define the following:
\begin{definition}[Complex exponential]
    We define the complex exponential function $\exp : \C \to \C$ by
    \[
    \exp(z) \coloneqq e ^ x(\cos{y} + i\sin{y}).\qquad (z = x + iy)
    \]
    Shorthand $\exp(z) = e ^ z$.
\end{definition}

One can show
(later)
that with this definition
\[
e ^ z = \sum_{n = 0}^{\infty}\frac{z ^ n}{n!}.
\]

\begin{proposition}
    We have the following complex exponential function properties:
    
    \begin{enumerate}[label = (\roman*)]
        \item $e ^ z \neq 0$ for all $z \in \C$.

        \item $e ^ {z_1 + z_2} = e ^ {z_1}e ^ {z_2}$.

        \item $e ^ z = 1$ if and only if $z = 2\pi ik$ for some $k \in \Z$.

        \item $e ^ {-z} = 1 / e ^ z$.

        \item $|e ^ z| = e ^ {\Re(z)}$.
    \end{enumerate}

    \begin{proof}
        By definition
        \[
        e ^ z = e ^ {x}e ^ {iy} = e ^ x(\cos{y} + i\sin{y})
        \]
        recall that any $\omega \in \C$ can be written uniquely 
        ($\bmod{2\pi}$)
        as $w = \underbrace{r}_{|\omega|}e ^ {i\overbrace{\theta}^{\Arg(w)}}$ so
        \[
        |e ^ z| = e ^ x\qquad \Arg(e ^ z) = y\bmod{2\pi}.
        \]
        $|e ^ z| \neq 0$ since $e ^ x \neq 0$.
        Since $\omega = 0 \iff |\omega| = 0$ we conclude that $e ^ z \neq 0$
        (i.e. $1$).
        \[
        e ^ z = 1 \iff e ^ xe ^ {iy} = 1 \cdot e ^ {i0}
        \]
        so $e ^ x = 1$ and $y = 0 + 2\pi k$ so $x = 0$ and $y = 2\pi k$,
        $k \in \Z$
        (shows (iii)).

        \[
        e ^ {z_1 + z_2} = e ^ {x_1 + x_2}e ^ {i(y_1 + y_2)} = e ^ {x_1}e ^ {x_2}e ^ {iy_1}e ^ {iy_2} = \left(e ^ {x_1}e ^ {iy_1}\right)\left(e ^ {x_2}e ^ {iy_2}\right) = e ^ {z_1}e ^ {z_2}.
        \]
        \[
        1 = e ^ 0 = e ^ {0 + i0} = e ^ {z - z} =  e ^ ze ^ {-z} \implies e ^ {-z} = \frac{1}{e ^ z}.
        \]
    \end{proof}
\end{proposition}

\begin{corollary}
    We have $\exp(2\pi i) = 1$ and $\exp(\pi i) = -1$.
    Latter is Euler's formula.
\end{corollary}

\begin{corollary}
    The complex exponential function is $2\pi i$-periodic,
    that is $\exp(z + 2k\pi i) = \exp(z)$ for any $k \in \Z$.
\end{corollary}

How does the exponential operate?
Recall $e ^ z - e ^ xe ^ {iy}$ and $|e ^ z| = e ^ x$ and $\Arg(e ^ z) = y\bmod{2\pi}$.
For fixed $x$,
i.e. $x = c$ for some $c \in \R$ we get image of a circle of radius $e ^ c$.
Segments on $x = c$ go to angular segments on the concentric circle with radius $e ^ c$.
If $y$ ranges over a segment of length more than $2\pi$ we will get the full circle.
When we fix $y$,
we will end up on a ray of angle $y\bmod{2\pi}$.
A segment on the line $y = c$ goes under $e ^ z$ to a segment on the ray $R_{c\bmod{2\pi}}$ if $a < x < b$ then we get the part of the ray $e ^ a < |\omega| < e ^ b$,
$\Arg(\omega) = c\bmod{2\pi}$.

\begin{lemma}
    \begin{itemize}
        \item The map $e ^ z$ injectively takes a segment of length $2\pi$ on the line $x = c$ which is open at one end and closed at the other to the entire concentric circle of radius $e ^ c$.
        If the above segment is closed,
        or its size is larger than $2\pi$ the image is no longer injective.

        \item The map $e ^ z$ injectively takes the line $y = c$ to the ray of angle $c$
        (without the origin).

        \item The map $e ^ z$ injectively takes the set $\{z \in \C\,:\, \theta < \Im(z) \leq \theta + 2\pi\}$ to $\C ^ {*}$ where
        \[
        \C ^ {*} =  \C \setminus \{0\}.
        \]
    \end{itemize}
\end{lemma}
With $e ^ z$ we can define trig functions.

Recall we defined $e ^ {ix} = \cos{x} + i\sin{x}$ when $x \in \R$.
\begin{align*}
    \cos{x} &= \frac{e ^ {ix} - e ^ {-ix}}{2}&\sin{x} &= \frac{e ^ {ix} - e ^ {-ix}}{2i} \\
    \cosh{x} &= \frac{e ^ x + e ^ {-x}}{2}& \sinh{x} &= \frac{e ^ x - e ^ {-x}}{2}
\end{align*}

\begin{definition}[Trigonometric functions]
    \begin{align*}
        \cos{z} &= \frac{e ^ {iz} - e ^ {-iz}}{2}&\sin{z} &= \frac{e ^ {iz} - e ^ {-iz}}{2i} \\
        \cosh{z} &= \frac{e ^ z + e ^ {-z}}{2}& \sinh{z} &= \frac{e ^ z - e ^ {-z}}{2}
    \end{align*}
\end{definition}
\textbf{Remember,
complex $\cos$ and $\sin$ are not bounded as they have $x$ and $y$ in $x + iy$.}


\begin{lemma}
    We have that for $z = x + iy$:
    \begin{align*}
        \sin(z) &= \sin(x)\cosh(y) + i\cos(x)\sinh(y) \\
        \cos(z) &= \cos(x)\cosh(y) - i\sin(x)\sinh(y) \\
        \sinh(z) &= -i\sin(iz) = \sinh(x)\cos(y) + i\cosh(x)\sin(y) \\
        \cosh(z) &= \cos(iz) = \cosh(x)\cos(y) + i\sinh(x)\sin(y).
    \end{align*}
    In addition we have that for all $z \in \C$
    \[
    \sin ^ 2(z) + \cos ^ 2(z) = 1,\qquad \cosh ^ 2(z) - \sinh ^ 2(z) = 1.
    \]

    \begin{proof}
        Exercise.

        \begin{align*}
            \sin{z} &= \frac{e ^ {iz} - e ^ {-iz}}{2i} \\
            &= \frac{e ^ {i(x + iy)} - e ^ {-i(x + iy)}}{2i} \\
            &= \frac{e ^ {-y}e ^ {ix} - e ^ {y}e ^ {-ix}}{2i} \\
            &= \frac{e ^ {-y}(\cos(x) + i\sin(x)) - e ^ {y}(\cos(x) - i\sin(x))}{2i}
        \end{align*}
    \end{proof}
\end{lemma}

In general complex $\cos$ and $\sin$ are bounded if and only if $y$ is bounded in $z = x + iy$.

\subsection{Logarithms and complex powers}


The real exponential $e ^ x$ has an inverse
($\log{x}$).
This,
in general,
will not work for $e ^ z$ since it is not injective.

Can we get an inverse when we restrict our domain?

\begin{lemma}[Inverting the exponential function]
    For every $w \in \C ^ {*} = \C \setminus \{0\}$,
    the equation
    \[
    e ^ z = w
    \]
    has a solution $z$.
    Furthermore,
    if we write $w = |w|e ^ {i\varphi}$ with $\varphi = \Arg(w)$,
    then all the solutions to $e ^ z = w$ are given by
    \[
    z = \log|w| + i(\varphi + 2\pi k)
    \]
    for $k \in \Z$.
    Here,
    $\log|w|$ is the usual natural logarithm of the real number $|w|$.
    \textit{Note that there are infinitely many solutions}.

    \begin{proof}
        \begin{align*}
            e ^ z = \omega &\iff e ^ xe ^ {iy} = |w|e ^ {iy} \\
            &\iff e ^ x = |\omega|\text{ and } y = y + 2\pi k\quad k \in \Z
        \end{align*}
        \[
        \]
    \end{proof}
\end{lemma}

From this we see a candidate to complex logarithm:
Just fix k.

We can try and define
\[
\log{z} = \log|z| + i\arg(z)
\]
where $\arg(z)$ goes in a fixed given range of angles $(\theta_1, \theta_2]$.
This definition is missing some
"nice"
properties.

To illustrate this choose the angles to be $(-\pi, \pi]$.
$\log(-1) = \log|-1| + i\pi = i\pi$.
Let's try to get closer to $-1$ from below,
choosing $\theta = -\pi + \delta$,
$\delta > 0$ gives us $z\delta = e ^ {i(-\pi + \delta)}$ close to $z = -1$.
\[
\log(z\delta) = \log\left|e ^ {i(-\pi + \delta)}\right| + i(-\pi + \delta) = -i(\pi - \delta) \approx -i\pi
\]
we have a discontinuity.

It is as if when we complete a full circle we
"climb"
by $2\pi$.

To avoid this issue we must disallow the ability to do full circles.
We do that by removing a closed ray $R_{\theta} = \{z = re ^ {i\theta}\,:\,r \geq 0\}$.

\begin{definition}[Complex logarithm functions]
    For any two real numbers $\theta_1 < \theta_2$ with $\theta_2 - \theta_1 = 2\pi$,
    let $\arg$ be the choice of argument function with values in $(\theta_1, \theta_2]$.
    Then the function
    \[
    \log(z) \coloneqq \log|z| + i\arg(z)
    \]
    is called a branch of logarithm.
    It has a jump discontinuity along the ray $R_{\theta_1} = R_{\theta_2}$.
    this ray is called a branch cut.
    If we choose $\arg(z) = \Arg(z) \in (-\pi, \pi]$,
    then we obtain a branch of logarithm called the principal branch of $\log$.
    We write $\Log$ for this principal branch:
    it is given by the formula
    \[
    \Log(z) \coloneqq \log|z| + i\Arg(z).
    \]
    The principal branch of logarithm has a
    "jump discontinuity"
    along the ray given by the non-positive real axis $\R_{\leq 0}$.
\end{definition}

Comments:
\begin{itemize}
    \item We must specify a branch
    ($\theta_1 < \theta_2$)
    to discuss a logarithm.

    \item For future:
    any branch of logarithm is continuous on $\C \setminus \R_{\theta}$.

    \item For real positive numbers
    \[
    \log{x} = \Log{x}.
    \]
\end{itemize}

\begin{lemma}[Properties of logarithms]
    We have the following properties when using any given branch of logarithm:
    \begin{enumerate}[label = (\roman*)]
        \item $e ^ {\log{z}} = z$ for any $z \in \C \setminus \{0\}$.

        \item In general
        \[
        \log(zw) \neq \log{z} + \log{w}.
        \]

        \item In general
        \[
        \log(e ^ z) \neq z.
        \]
    \end{enumerate}
\end{lemma}

\begin{lemma}\phantom{}
    \begin{itemize}
        \item A branch of the map $\log(z)$ injectively takes a ray of angle $\theta$ without the origin,
        to the line $y = \theta$ where $\theta$ is measured with respect to the branch cut.
    
        \item A branch of the map $\log(z)$ takes a concentric circle of radius $r$,
        minus the intersection of this circle with the branch cut,
        to an open segment of length $2\pi$ on the line $x = \log{r}$ with its lowest point given by the angle that defines the branch cut.
    \end{itemize}
\end{lemma}

\begin{definition}[Complex powers]
    For $w \in \C$ fixed by choosing any branch of $\log$ we can define a branch of the function $z \mapsto z ^ w$ by the equation
    \[
    z ^ w \coloneqq e ^ {w\log{z}}.
    \]
\end{definition}

For example,
if $w = 1 / n$ and use the principal branch we get
\[
z ^ {1 / n} = e ^ {\frac{\log|z|}{n} + i\frac{\Arg(z)}{n}} = |z| ^ {1 / n}e ^ {i\frac{\Arg(z)}{n}}.
\]

\begin{remark}
    This definition is consistent with $z ^ n = z \dotsi z$ with $n \in \N$.
    Also,
    for $n \in \N$ one can show that all the branches of $z ^ {1 / n}$ with argument in $(-\pi, \pi)$
    (branch cut is $\R_{\leq 0}$)
    are
    \[
    |z| ^ {1 / n}e ^ {i / n(\Arg(z) + 2\pi k)}
    \]
    $k = 0, \dotsc, n - 1$.
\end{remark}

\begin{example}
    Find $(1 - i) ^ {1 / 2}$ using the principal branch of the logarithm.

    \begin{solution}
        \[
        (1 - i) ^ {1 / 2} = e ^ {1 / 2\Log(1 - i)} = e ^ {1 / 2(\log|1 - i| + i\Arg(1 - i))} = e ^ {1 / 2(\log(\sqrt{2}) - i\pi / 4)} = e ^ {1 / 4\log(2)}e ^ {-i\pi / 8} = 2 ^ {1 / 4}e ^ {-i\pi / 8}.
        \]
    \end{solution}
\end{example}

\begin{example}
    Find $2 ^ {1 / 2}$ for all possible branches.

    \begin{solution}
        We only have two branches
        ($k = 0, k = 1$ from above)
        which will give $\sqrt{2}, -\sqrt{2}$.
    \end{solution}
\end{example}

\section{Topology and Continuity}
On $\R$ these notions are related to intervals
(open,
closed,
etc).
These studies can be extended to sets which are called open,
closed,
compact.
Such sets tell us a lot about our space and form its topology.

\subsection{Basic topology on the complex plane -
open and closed sets}

\begin{definition}[Balls in $\C$]\phantom{}
    \begin{itemize}
        \item The open ball $B_r(z_0)$ of radius $r$ centred at $z_0$ is
        \[
        B_r(z_0) \coloneqq \{z \in \C\,:\,|z - z_0| < r\}.
        \]

        \item The closed ball $\bar{B}_r(z_0)$ of radius $r$ centred at $z_0$ is
        \[
        \bar{B}_r(z_0) \coloneqq \{z \in \C\,:\, |z - z_0| \leq r\}.
        \]
    \end{itemize}
\end{definition}

\begin{definition}[Open/closed sets in $\C$]\phantom{}
    \begin{itemize}
        \item A subset $U \subseteq \C$ is open
        (in $\C$)
        if for every point $z_0 \in U$ there exists $\varepsilon > 0$ such that $B_{\varepsilon}(z_0) \subset U$.

        \item A subset $U \subseteq \C$ is closed
        (in $\C$)
        if its complement $U ^ c \coloneqq \C \setminus U$ is open.
    \end{itemize}
\end{definition}

\begin{lemma}[Open balls are open]
    For any $z_0 \in \C$ and any $r > 0$ the open ball $B_r(z_0)$ is an open set.

    \begin{proof}
        Let $z \in B_r(z_0)$.
        Define
        \[
        \varepsilon = \frac{r - |z - z_0|}{2} > 0\qquad (|z - z_0| < r).
        \]
        Claim:
        $B_{\varepsilon}(z) \subset B_r(z_0)$.
        Indeed,
        if $w \in B_{\varepsilon}(z)$
        \begin{align*}
            |w - z_0| &= |w - z + z - z_0| \\
            &\leq |w - z| + |z - z_0| \\
            &< \varepsilon + |z - z_0| \\
            &= \frac{r - |z - z_0|}{2} + |z - z_0| \\
            &= \frac{r + |z - z_0|}{2} < r.
        \end{align*}
    \end{proof}
\end{lemma}

\begin{remark}
    It can also be shown that for any $z_0 \in \C$ and any $r > 0$ the closed ball $\bar{B}_r(z_0)$ is closed in $\C$.
\end{remark}

\begin{example}
    The following sets are open:

    \begin{enumerate}[label = (\roman*)]
        \item $\mathbb{H} = \{z \in \C\,:\,\Im(z) > 0\}$.
        
        \item $\mathbb{H}_{-} = \{z \in \C\,:\,\Im(z) < 0\}$.
        
        \item $\mathbb{H}_{L} = \{z \in \C\,:\,\Re(z) < 0\}$.
        
        \item $\mathbb{H}_{R} = \{z \in \C\,:\,\Re(z) > 0\}$.
        
        \item $\mathbb{D} = \{z \in \C\,:\,|z| < 1\}$.
        
        \item $\Omega_1 = \{z \in \C\,:\,\Re(z) > 0, \Im(z) > 0\}$.
    \end{enumerate}

    Lets show that $\Omega_1$ is open.
    Given $z_0 \in \Omega_1$,
    define $\varepsilon = \frac{\min\{\Re(z_0), \Im(z_0)\}}{2}$.
    Claim:
    $B_{\varepsilon}(z) \subset \Omega_1$.
    Indeed if $w \in B_{\varepsilon}(z_0)$ then
    \begin{align*}
        \Re(w) &= \Re(w - z_0  + z_0) \\
        &= \Re(w - z_0) + \Re(z_0) \\
        \intertext{$|\Re(\xi)| \leq |\xi|$}
        &\geq -|w - z_0| + \Re(z_0) \\
        &> -\varepsilon + \Re(z_0) \\
        &= -\frac{\min(\Re(z_0), \Im(z_0))}{2} + \Re(z_0) \\
        &\geq \frac{\Re(z_0)}{2} > 0.
    \end{align*}
    Similarly $\Im(z_0) > 0$.
\end{example}

\textbf{Careful!}
Not all sets are open or closed.
$A - \{z \in \C\,:\,\Re(z) > 0, \Im(z) \geq 0\}$.
$A$ is not open
($1 - i\varepsilon \in B_{\varepsilon}(1)$,
but $1 \in A$,
$B_{\varepsilon}(1) \not\subset A$ for any $\varepsilon > 0$).
$A ^ c$ is not open,
i.e. $A$ is not closed.

\begin{lemma}[Unions and intersections of open sets]\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item Arbitrary unions of open sets are open;
        that is $\bigcup_{i \in I}U_i$ is open for any
        (possible infinite)
        collection of open sets $U_i$.

        \item Finite intersections of open sets are open;
        that is $\bigcap_{i = 1}^{n}U_i$ is open for any finite collection of open sets $U_i$.
    \end{enumerate}

    \begin{proof}\phantom{}
        \begin{enumerate}[label = (\roman*)]
            \item Let $z_0 \in \bigcup_{i \in I}U_i$,
            $\exists i_0 \in I$ s.t. $z_0 \in U_{i_0}$ since $U_{i_0}$ is open $\exists \varepsilon > 0$ s.t. $B_{\varepsilon}(z_0) \subset U_i \subseteq \bigcup_{i \in I}U_i$.
            $z_0$ was arbitrary so $\bigcup_{i \in I}U_i$ is open.

            \item Let $z_0 \in \bigcap_{i = 1}^{n}U_i$.
            This means that $z_0 \in U_i$ $\forall i = 1, \dotsc, n$.
            Since $U_i$ is open $\exists \varepsilon_i > 0$ s.t. $B_{\varepsilon_i}(z_0) \subset U_i$.
            Define $\varepsilon = \min(\varepsilon_1, \dotsc, \varepsilon_n)$ then $z_0 \in B_{\varepsilon}(z_0) \subseteq B_{\varepsilon_i}(z_0) \subset U_i$ $\forall i = 1, \dotsc, n$ this implies $B_{\varepsilon_0}(z) \subset \bigcap_{i = 1}^{n}U_i$.
        \end{enumerate}
    \end{proof}
\end{lemma}

\begin{corollary}[Unions and intersections of closed sets]\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item Finite unions of closed sets are closed.

        \item Arbitrary intersections of closed sets are closed.
    \end{enumerate}

    \begin{proof}\phantom{}
        \begin{enumerate}[label = (\roman*)]
            \item Follows from De Morgan's law
            \[
            \left(\bigcup_{i \in I}A_i\right) ^ c = \bigcap_{i \in I}A_i ^ c, \qquad \left(\bigcap_{i \in I}A_i\right) ^ c = \bigcup_{i \in I}A_i ^ c.
            \]
        \end{enumerate}
    \end{proof}
\end{corollary}

Given a set $A$ we can define special open and closed sets associated to $A$.

\begin{definition}[Interior points,
closure,
boundary]
    Let $A$ be a subset of $\C$.

    \begin{itemize}
        \item The interior $A ^ 0$ of $A$ is defined by
        \[
        A ^ 0 \coloneqq \{x \in A\,:\,\text{there exists an open set $U \subseteq A$ such that $x \in U$}\}
        \]
        (this can be replaced with $B_{\varepsilon}(x) \subset A$ for some $\varepsilon > 0$).
        
        \item The closure $\bar{A}$ of $A$ is the complement of the interior of the complement:
        \[
        \bar{A} \coloneqq \left(\left(A ^ c\right) ^ 0\right) ^ c = \{x \in X\,:\, U \cap A \neq \emptyset\text{ for every open set $U$ with $x \in U$}\}
        \]
        ($U$ can be replaced with $B_{\varepsilon}(x)$ for all $\varepsilon > 0$).
    
        \item The boundary $\partial A$ of $A$ is the closure without the interior:
        \[
        \partial A \coloneqq \bar{A} \setminus A ^ 0\qquad \left[=\left(A ^ 0\right) ^ c \cap \left(\left(A ^ c\right) ^ 0\right) ^ c = \left(A ^ 0 \cup \left(A ^ c\right) ^ 0\right) ^ c\right].
        \]
    \end{itemize}
\end{definition}

\begin{example}
    Consider the set
    \[
    A = \{z \in \C\,:\, 1 < |z| \leq 3\}.
    \]
    Then $A ^ 0$ is the area inside the inner and outer circle not inclusive of the boundary.
    The closure is the area inside the inner and outer circles including the boundary.
    The boundary is only the boundary of both the inner and outer circles.
\end{example}

\begin{remark}\phantom{}
    \begin{itemize}
        \item Don't confuse closure with conjugation.

        \item $A ^ 0$ is always open.
        $\bar{A}$ is always closed.
        $\partial A$ is always closed.

        \item Intuitively boundary is the edge of the set.
        Interior points are points in the set that are not on the edge and the closure is attained by adding all the edge points.
    \end{itemize}
\end{remark}

\textit{Note:
$A = \{z_0\}$,
$A ^ 0 = \emptyset$,
$\bar{A} = \{z_0\}$,
$\partial A = \{z_0\}$}

\textbf{Important properties:}
\begin{enumerate}[label = (\roman*)]
    \item $A$ is open $\iff \partial A \cap A = \emptyset \iff A = A ^ 0$.
    Moreover,
    \[
    A ^ 0 = \bigcup_{U \subseteq A, U\text{ open}}U.
    \]
    In other words:
    $A ^ 0$ is the largest open set that is contained in $A$.

    \item $A$ is closed $\iff \partial A \subseteq A \iff A = \bar{A}$.
    Moreover,
    \[
    \bar{A} = \bigcap_{A \subseteq F, F\text{ closed}}F.
    \]
    In other words:
    $\bar{A}$ is the smallest closed set that contains $A$.

    \item $\partial A = \{x \in X\,:\,\text{for all open sets $U$ containing $x$,
    there exist $y, z \in U$ with $y \in A$ and $z \in A ^ c$}\}$.
\end{enumerate}

\subsection{Convergence and continuity}

The notion of limits and convergence is paramount in Analysis.

\begin{definition}[Limits and convergence of sequences in $\C$]
    We say a sequence $\{z_n\}_{n \in \N} \subset \C$ converges to $z \in \C$ if
    \[
    \lim_{n \to \infty}|z_n - z| = 0.
    \]
    That is if for any $\varepsilon > 0$ there exists $N \in \N$ such that for every $n > N$
    \[
    |z_n - z| < \varepsilon.
    \]
\end{definition}

We notice that if $\{z_n\}_{n \in \N}$ is a sequence in $\C$ then $\{\Re(z_n)\}_{n \in \N}$ and $\{\Im(z_n)\}_{n \in \N}$ are sequences in $\R$.
Is there a connection between the convergence of these sequences.

\begin{lemma}
    A sequence of complex numbers $\{z_n\}_{n \in \N} \subset \C$ converges if and only if the sequences $\{\Re(z_n)\} \subset \R$ and $\{\Im(z_n)\} \subset \R$ converge in $\R$.
    Moreover,
    in that case
    \[
    \lim_{n \to \infty}z_n = \lim_{z \to \infty}\Re(z_n) + i\lim_{n \to \infty}\Im(z).
    \]

    \begin{proof}
        Denote $\Re(z_n) = x_n$,
        $\Im(z_n) = y_n$,
        i.e. $z_n = x_n + iy_n$ for $x_n, y_n \in \R$.
        Assume $z_n \to z = x + iy$.
        Then
        \[
        0 \leq |x_n - x| \leq \sqrt{(x_n - x) ^ 2 + (y_n - y) ^ 2} = |z_n| \to 0
        \]
        by assumption.
        Then $|x_n - x| \to 0$ as $n \to \infty$ by squeezing theorem.
        \[
        \lim_{n \to \infty}|x_n - x| < 0 \implies \lim_{n \to \infty}x_n = x.
        \]
        Similarly $y_n$ must converge to $y$ as $n \to \infty$.

        Assume $x_n \to x$ as $n \to \infty$,
        $y_n \to y$ as $n \to \infty$ then
        \begin{align*}
            |z_n - z| &= |(x_n - x) + i(y_n - y)| \\
            &\leq |x_n - x| + |i||y_n - y| \\
            &\implies \\
            0 \leq |z_n - z| &\leq |x_n - x| + |y_n - y|
        \end{align*}
        then $|z_n - z| \to 0$ as $n \to \infty$.
    \end{proof}
\end{lemma}

\begin{corollary}
    A sequence in $\C$ can have at most one limit.

    \begin{proof}
        This holds since it is true in $\R$.
        Indeed,
        if $\{z_n\}$ converges then $\{\Re(z_n)\}$ converges.
        It can converge to a unique limit,
        say $x$.
        Similarly $\{\Im(z_n)\}$ converges to a unique limit $y$,
        and consequently $\{z_n\}$ converges to the unique limit $x + iy$.
    \end{proof}
\end{corollary}

\begin{example}
    Show $z_n = \frac{1}{n} + i\left(1 - \frac{1}{n}\right) ^ n$ converges in $\C$.

    \begin{solution}
        $x_n = \frac{1}{n}$,
        $y_n = (1 - 1 / n) ^ n$.
        Since $x_n \to 0$ as $n \to \infty$ and $y_n \to e ^ {-1}$ as $n \to \infty$,
        $z_n$ must converge as well to $\frac{i}{e}$.
    \end{solution}
\end{example}

\textbf{Important:}
COLT rules follow in the complex plane
(where $i$ is treated as a constant).

We can formulate convergence of sequences in a topological sense.

\begin{lemma}
    Let $\{z_n\}_{n \in \N} \subset \C$ be a given sequence in $\C$.
    Then $\lim_{n \to \infty}z_n = z$ if and only if for any open set $U \subseteq \C$ such that $z \in U$ there exists $N \in \N$ such that for any $n > N$ we have that $z_n \in U$.

    \begin{proof}
        We start by noticing that $z_n \to z$ as $n \to \infty$ is equivalent to $\forall \varepsilon > 0$,
        $\exists N \in \N$ s.t. if $n > N$ $|z_n - z| < \varepsilon$ which is equivalent to $\forall \varepsilon > 0$,
        $\exists N \in \N$ s.t. if $n > N$,
        $z_n \in B_{\varepsilon}(z)$.

        "$\implies$".
        Let $U$ be an open set s.t. $z \in U$.
        By definition.
        $\exists \varepsilon > 0$ s.t. $B_{\varepsilon}(z) \subset U$.
        Since $z_n \to z$ as $n \to \infty$ we can find $N \in \N$ s.t. for all $n > N$ $z_n \in B_{\varepsilon}(z) \subset U$,
        which is the required result.

        "$\impliedby$"
        Given $\varepsilon > 0$ the set $B_{\varepsilon}(z)$ is open.
        By assumption,
        $\exists N \in \N$ s.t. if $n > N$ $z_n \in B_{\varepsilon}(z)$.
        Since this holds for any $\varepsilon > 0$ $z_n \to z$ as $n \to \infty$.
    \end{proof}
\end{lemma}

Sequences give excellent intrinsic criterion to test closedness of sets.

\begin{lemma}[Closedness criterion]
    The set $A \subseteq \C$ is a closed set if and only if for any sequence $\{z_n\}_{n \in \N}$ in $A$ that converges to an element $z \in \C$ we have that $z \in A$.
\end{lemma}

\begin{definition}[Continuity]
    Let $A$ be a subset of $\C$.
    A map $f : A \to \C$
    (or $\R$)
    is called continuous at $z_0 \in A$ if for any $\varepsilon > 0$ there exists $\delta > 0$ such that if $|z - z_0| < \delta$ and $z \in A$ then
    \[
    |f(z) - f(z_0)| < \varepsilon.
    \]
    We say a function $f$ is continuous on a set $A \subseteq \C$ if it is continuous at every point in $A$.
\end{definition}

Just like before we have the following

\begin{lemma}[Continuity via sequences]
    A function $f : A \to \C$
    (or $\R$)
    is continuous at $z_0 \in A$ if and only if
    \[
    \lim_{n \to \infty}f(z_n) = f(z_0)
    \]
    for every sequence $\{z_n\}_{n \in \N}$ in $A$ such that $\lim_{n \to \infty}z_n = z_0$
\end{lemma}

We use this usually to show lack of continuity
(and in fact,
lack of limit):
If we find $\{z_n\}, \{w_n\}$ s.t. $z_n \to z_0$ as $n \to \infty$,
$w_n \to z_0$ as $n \to \infty$.
$f(z_n) \to \ell_1$ as $n \to \infty$,
$f(w_n) \to \ell_2$ as $n \to \infty$ and $\ell_1 \neq \ell_2$ then $f$ is not continuous at $z_0$
(in fact there is no limit).

\begin{example}
    \[
    f(z) = f(x + iy) = \begin{cases}
        \frac{xy}{x ^ 2 + y ^ 2} &(x, y) \neq (0, 0) \\
        0 &(x, y) = (0, 0)
    \end{cases}
    \]
    is not continuous at $z = 0$.
    Indeed,
    note that for $y = kx$,
    $k \in \R$
    \[
    f(x + ikx) = \frac{k}{1 + k ^ 2}
    \]
    when $x \neq 0$.
    Choosing $z_n = \frac{1}{n}$ and $w_n = \frac{1}{n} + \frac{i}{n}$,
    we have that $z_n \to 0$,
    $w_n \to 0$
    \[
    f(z_n) = 0 \to 0\text{ as $n \to \infty$}
    \]
    \[
    f(w_n) = \frac{1}{2} \to \frac{1}{2}\text{ as $n \to \infty$}
    \]
    and as such there is no limit.
\end{example}

\begin{lemma}[Basic properties of continuous functions]\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item Products,
        sums,
        quotients of real/complex valued functions on $A \subseteq \C$ are continuous on $A$
        (where defined).

        \item Composition of continuous functions are continuous.
    \end{enumerate}
\end{lemma}

The following functions are continuous on $\C$:
\begin{itemize}[label = -]
    \item $f(z) = z$.

    \item $f(z) = \text{const}$.

    \item $f(z) = \Re(z)$,
    $f(z) = \Im(z)$,
    $f(z) = |z|$.

    \item $f(z) = \bar{z}$.

    \item $f(z) = z ^ n$,
    $n \in \N$.
    Consequently,
    polynomials are continuous.

    \item $f(z) = e ^ z$ and consequently $\sin(z)$,
    $\cos(z)$,
    $\sinh(z)$,
    $\cosh(z)$.
\end{itemize}
For any $\theta_1 < \theta_2$ s.t. $\theta_2 - \theta_1 = 2\pi$ the argument associated to $(\theta_1, \theta_2]$ is not continuous on $C ^ {*}$
($=\C \setminus\{0\}$).
This is due to a jump discontinuity at $R_{\theta_1}$
(ray of $\theta_1$).
It is continuous on $\C \setminus R_{\theta_1}$.

Any branch of $\log{z}$ is continuous on $\C \setminus R_{\theta_1}$
(the branch cut).

Much like sequences we can recast continuity in topological sets.

\begin{theorem}[Continuity via open sets]
    Let $A \subseteq \C$ be an open set and let $f : A \to \C$
    (or $\R$)
    be a given map.
    Then the following are equivalent.

    \begin{enumerate}[label = (\roman*)]
        \item $f$ is continuous.

        \item $f ^ {-1}(U)$ is an open set for every open set $U \subseteq \C$.

        \item $f ^ {-1}(F)$ is a closed set for every closed set $F \subseteq \C$.
    \end{enumerate}

    \begin{proof}
        Motivation:
        $f$ is continuous at an interior point $z_0$ if and only if for any $\varepsilon > 0$,
        there exists $\delta > -$ such that if $z \in B_{\delta}(z)$ then $f(z) \in B_{\varepsilon}(f(z_0))$.
        Equivalently for any $\varepsilon > 0$,
        there exists $\delta > 0$ such that
        \[
        B_{\delta}(z_0) \subset f ^ {-1}(B_{\varepsilon}f(z_0)).
        \]
    \end{proof}
\end{theorem}

We can use this theorem to classify sets.
For any $z_0 \in \C$,
$\{z_0\}$ is closed which implies if $f$ is continuous then $f ^ {-1}(\{z_0\})$ is closed.
For example
\[
\{z \in \C \mid |z| = 1\}
\]
for $f(z) = |z|$.

The set
\[
U = \{z = x + iy \in \C \mid (x ^ 2 + y ^ 2)\sin(\sqrt{x ^ 2 + 2}) > 2\}
\]
is open since $U = f ^ {-1}((2, \infty))$ where $f(x + iy) = (x ^ 2 + y ^ 2)\sin\sqrt{x ^ 2 + 2}$ continuous.

\subsection{Sequential Compactness}
A fundamental theorem in real analysis is Bolzano-Weierstrass that states that bounded sequences have converging subsequences.
This implies many results for continuous functions on closed bounded intervals or more generally compact sets.

\begin{definition}[Sequential Compactness]
    A non-empty subset $K$ of $\C$ is called sequentially compact if for any sequence $\{z_n\}_{n \in \N}$ in $K$ there exists a convergent subsequence,
    $\{z_{n_k}\}_{k \in \N}$,
    whose limit lies in $K$.
\end{definition}

\begin{example}
    $B_1(0) = \{z \in \C \mid |z| < 1\}$ is not sequentially compact.
    \[
    z_n = 1 - \frac{1}{n} \in B_1(0)
    \]
    $z_n \to 1$ as $n \to \infty$ but $1 \notin B_1(0)$.
\end{example}

There is a simple criteria for sequential compactness in $\C$.

\begin{theorem}[Heine-Borel for $\C$]    
    A subset $K$ of $\C$ is sequentially compact if and only if it is closed and bounded.

    \begin{proof}
        We'll show that if $K$ is bounded and closed,
        it is compact.
        Let $\{z_n\}_{n \in \N} \in K$ be given.
        We write
        \[
        z_n = x_n + iy_n\qquad x_n, y_n \in \R
        \]
        and see that $\{x_n\}_{n \in \N}$,
        $\{y_n\}_{n \in \N}$ are real sequences s.t.
        \[
        \max(|x_n|, |y_n|) \leq |z_n| \leq U
        \]
        i.e. $\{x_n\}$,
        $\{y_n\}$ are bounded.
        Using Bolzano-Weierstrass we can find a subsequence $\{x_{n_k}\}_{k \in \N}$ that converges to some $x \in \R$.
        \[
        z_{n_k} = x_{n_k} + iy_{n_k}
        \]
        $\{y_{n_k}\}_{k \in \N}$ is a subsequence of $\{y_n\}_{n \in \N}$ that is bounded.
        By Bolzano-Weierstrass there exists a subsequence of $\{y_{n_k}\}_{k \in \N}$,
        $\{y_{n_{k_j}}\}_{j \in \N}$,
        that converges to some $y \in \R$
        \[
        z_{n_{k_j}} = x_{n_{k_j}} + iy_{n_{k_j}}
        \]
        $x_{n_{k_j}}$ is a subsequence of $\{x_{n_k}\}_{k \in \N}$ which converges which implies it converges to $x$ too.

        We conclude that $\{z_{n_{k_j}}\}_{j \in \N}$ is a subsequence of $\{z_n\}_{n \in \N}$ that converges to $z = x + iy$
        (since $\Re(z_{n_{k_j}}) \to x$ and $\Im(z_{n_{k_j}}) \to y$).
        Since $K$ is closed,
        $\{z_{n_{k_j}}\}_{j \in \N} \subset K$ and $z_{n_{k_j}} \to z$ we must have that $z \in K$.
        The result is shown.
    \end{proof}
\end{theorem}
The above theorem holds in $\R ^ n$ and $\C ^ n$ for any $n \in \N$.
It is not true in general metric spaces.

\begin{example}\phantom{}
    \begin{itemize}
        \item $B_1(0)$ is not closed,
        so it is not compact.

        \item $\C$ is not compact since it is not bounded.

        \item $\{z \in \C\,:\, 1 \leq \Re(z) \leq 3, 0 \leq \Im(z) \leq 4\}$ is closed and bounded,
        and as such compact.
    \end{itemize}
\end{example}

\begin{theorem}[Extreme Value Theorem]
    Let $K$ be a sequentially compact subset of $\C$ and let $f : K \to \R$ be continuous.
    Then $f$ attains its minimum and maximum on $K$.
\end{theorem}

\begin{corollary}
    \begin{itemize}
        \item If $K$ is a closed and bounded set in $\C$ then $|z|$,
        $\Re(z)$,
        and $\Im(z)$ attain their minimum and maximum on $K$.

        \item If $K$ is a closed and bounded set in $\C$ and $f : K \to \C$ or $\R$ is continuous then $|f(z)|$ attains its minimum and maximum on $K$.
    \end{itemize}
\end{corollary}

\section{Complex Differentiation}

\subsection{Complex differentiability}

\begin{definition}[Complex differentiability]
    A function $f : U \to \C$ defined on an open set $U$ is $\C$ is
    (complex)
    differentiable at $z_0 \in U$ if
    \[
    \lim_{z \to z_0}\frac{f(z) - f(z_0)}{z - z_0}\text{ exists}.
    \]
    We call this limit the derivative of $f$ at $z_0$ and write $f'(z_0)$ for the limit,
    i.e.
    \[
    f'(z_0) = \lim_{h \to 0}\frac{f(z_0 + h) - f(z_0)}{h}.
    \]
\end{definition}

\begin{remark}
    While $f'(z_0)$ looks like the derivative in $\R$ we need to remember that $z \to z_0$ or $h \to 0$ is in the complex plane,
    i.e. we have infinitely many ways for $z$
    (or $h$)
    to approach $z_0$
    (or $0$).
    This is a two dimensional limit.

    One can show that if $f$ is complex differentiable at $z_0$,
    then it is continuous at $z_0$.
\end{remark}

\begin{example}
    Show that $f(z) = z ^ 2$ is complex differentiable at $z = 0$.

    \begin{solution}
        We must use the definition:
        \[
        \frac{f(z) - f(z_0)}{z - z_0} = \frac{z ^ 2 - 0 ^ 2}{z - 0} = z
        \]
        so
        \[
        \lim_{z \to 0}\frac{f(z) - f(0)}{z - 0} = \lim_{z \to 0}z = 0.
        \]
    \end{solution}
\end{example}

\begin{example}
    At which points is $f(z) = \bar{z}$ complex differentiable.

    \begin{solution}
        Let $z_0 \in \C$ be given
        \[
        \frac{f(z_0 + h) - f(z_0)}{h} = \frac{\bar{z}_0 + \bar{h} - \bar{z}_0}{h} = \frac{\bar{h}}{h}
        \]
        writing $h = re ^ {i\theta}$ we have
        \[
        h \to 0 \iff r \to 0\text{ and } \frac{\bar{h}}{h} = \frac{re ^ {-i\theta}}{re ^ {i\theta}} = e ^ {-2i\theta}
        \]
        this implies $\lim\frac{\bar{h}}{h}$ doesn't exist so $f(z)$ is not complex differentiable at any point.
    \end{solution}
\end{example}

\begin{lemma}[COLT for complex derivatives and the chain rule]\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item Let $f, g : \C \to \C$ be complex differentiable at $z_0 \in \C$.
        Then:
        \begin{itemize}
            \item $f + g$ is complex differentiable at $z_0$ and
            \[
            (f + g)'(z_0) = f'(z_0) + g'(z_0).
            \]

            \item (product rules)
            $fg$ is complex differentiable at $z_0$ and
            \[
            (fg)'(z_0) = f'(z_0)g(z_0) + g'(z_0)g(z_0).
            \]

            \item (quotient rule)
            if $g(z_0) \neq 0$ then $f / g$ complex differentiable at $z_0$ and
            \[
            \left(\frac{f}{g}\right)' = \frac{f'(z_0)g(z_0) - g'(z_0)f(z_0)}{(g(z_0)) ^ 2}
            \]
        \end{itemize}

        \item (chain rule)
        same as in $\R$.
    \end{enumerate}
\end{lemma}

\subsection{Cauchy-Riemann equations}
We want to simplify the method of showing that a function is complex differentiable.
Recall we can write
\[
f(z) = \Re(f(z)) + i\Im(f(z))
\]
with both functions being functions from $\C \to \R$.
We can identify sets in $\C$ as sets in $\R ^ 2$ and write
\[
f(z) = f(x + iy) = u(x, y) + iv(x, y)
\]
with $u, v$ as functions of two real variables to $\R$.

\begin{example}
    Find the real and imaginary parts of $f(z) = z ^ 2$.

    \begin{solution}
        \[
        f(z) = f(x + iy) = (x + iy) ^ 2 = x ^ 2 + 2ixy + i ^ 2y ^ 2 = (x ^ 2 - y ^ 2) + i(2xy)
        \]
        so $\Re(f(x + iy)) = u(x, y) = x ^ 2 - y ^ 2$,
        $\Im(f(x + iy)) = v(x, y) = 2xy$.
    \end{solution}
\end{example}

For functions from $\R ^ 2$ to $\R$ we have partial derivatives.

Is there a connection between the complex derivative of $f$ and partial derivatives of $\Re(f)$,
$\Im(f)$.

\begin{proposition}[Cauchy-Riemann equations]
    Let $f = u + iv$ be complex differentiable at $z_0 = x_0 + iy_0$.
    Then the real partial derivatives $u_x, u_y, v_x, v_y$ exist at $(x_0, y_0)$ and satisfy the Cauchy-Riemann equations:
    \[
    u_x(x_0, y_0) = v_y(x_0, y_0)\qquad u_y(x_0, y_0) = -v_x(x_0, y_0).
    \]
    Furthermore,
    the derivative of $f$ at $z_0$ can be written as
    \begin{align*}
        f'(z_0) &= u_x(z_0) + iv_x(z_0) = v_y(z_0) - iu_y(z_0) \\
        &= u_x(z_0) - iu_y(z_0) = v_y(z_0) + iv_x(z_0)
    \end{align*}
    where we have used the shorthand notation $w(z_0) = w(x_0, y_0)$ for functions from $\R ^ 2$ to $\R$.

    \begin{proof}
        The proof relies on the realisation that since $\lim_{n \to 0}\frac{f(z_0 + h_c) - f(z_0)}{h_c}$ exists $h_c$ can go to $0$ in any
        (complex)
        way.

        Let's consider two ways,
        we consider $h_c = ih$,
        $h \in \R$,
        $h \to 0$
        (in $\R$)
        \begin{align*}
            f'(z_0) &= \lim_{h \to 0}\frac{f(z_0 + ih) - f(z_0)}{ih} \\
            &= \lim_{h \to 0}\frac{f(x_0 + i(y_0 + h)) - f(x_0 + iy_0)}{ih} \\
            &= \lim_{h \to 0}\frac{(u(x_0, y_0 + h) + iv(x_0, y_0 + h)) - (u(x_0, y_0) + iv(x_0, y_0))}{ih} \\
            &= \lim_{h \to 0}\left(\frac{v(x_0, y_0 + h) - v(x_0, y_0)}{h}\right) - i\left(\frac{u(x_0, y_0 + h) - u(x_0, y_0)}{h}\right).
        \end{align*}
        Since this limit exists
        \[
        \Re(f'(z)) = \lim_{h \to 0}\frac{v(x_0, y_0 + h) - v(x_0, y_0)}{h} = v_y
        \]
        exists and
        \[
        \Im(f'(z_0)) = -\lim_{h \to 0}\frac{u(x_0, y_0 + h) - u(x_0, y_0)}{h} = -u_y
        \]
        so we conclude $v_y(x_0, y_0)$ and $u_y(x_0, y_0)$ exist and $f'(z_0) = v(x_0, y_0) - iu_y(x_0, y_0)$.
        
        Similarly for $h_c = h$.
        \[
        f'(z_0) = \lim_{h \to 0}\frac{f(z_0 + h) - f(z)}{h} = \lim_{h \to 0}\frac{f(x_0 + h + iy_0) - f(x_0 + iy_0)}{h}
        \]
        continuing as above we get that $u_x(x_0, y_0), v_x(x_0, y_0)$ exist and $\Re(f'(z_0)) = u_x(x_0, y_0)$,
        $\Im(f'(z_0)) = v_x(x_0, y_0)$,
        i.e.
        \[
        f'(z_0) = u_x(x_0, y_0), + iv_x(x_0, y_0) = v_y(x_0, y_0) - iu_y(x_0, y_0)
        \]
        which implies $u_x(x_0, y_0) = v_y(x_0, y_0)$ and $v_x(x_0, y_0) = -u_y(x_0, y_0)$.
    \end{proof}
\end{proposition}

\textit{Note:
The theorem doesn't tell us when $f$ is complex differentiable.
It tells us that if the Cauchy-Riemann equations do not hold,
$f$ is not complex differentiable.}

We need a converse
\begin{theorem}
    Let $f = u + iv$ be defined on an open subset $U$ of $\C$.
    Then,
    $f$ is complex differentiable at $z_0 = x_0 + iy_0 \in U$ if and only if $u$ and $v$ are
    (real)
    differentiable at $(x_0, y_0)$ and the Cauchy-Riemann equations are satisfied at that point.

    \begin{proof}
        Omitted.
    \end{proof}
\end{theorem}
We use $U$ as a subset of $\C$ and $\R ^ 2$ interchangeably.

In particular:
\begin{corollary}
    Let $f = u + iv$ be defined on an open subset $U$ of $\C$.
    Assume the partial derivatives $u_x, u_y, v_x, v_y$ exist,
    are continuous,
    and satisfy the Cauchy-Riemann equations at $z_0 \in U$.
    Then $f$ is complex differentiable at $z_0$.
\end{corollary}

Consider $f(z) = e ^ z = e ^ xe ^ {iy} = e ^ x\cos(y) + ie ^ x\sin(y)$,
$u(x, y) = e ^ x\cos(y)$,
$v(x, y) = e ^ x\sin(y)$.
$u, v \in C ^ {1}(\R ^ 2)$.
Consequently,
$f(z)$ is differentiable at $z_0 = x_0 + iy_0$ if and only if the Cauchy-Riemann equations hold at $(x_0, y_0)$,
\begin{align*}
    u_x(x, y) &= e ^ x\cos(y) & u_y(x, y) &= -e ^ x\sin(y) \\
    v_x(x, y) &= e ^ x\sin(y) & v_y(x, y) &= e ^ x\cos(y)
\end{align*}
$u_x = v_y$ and $v_x = -u_y$ so Cauchy-Riemann equations satisfied at any $(x, y)$ so $e ^ z$ is complex differentiable at any $z \in \C$.
Moreover,
\[
f'(z) = u_x(z) + iv_x(x) = e ^ x\cos(y) + ie ^ x\sin(y) = e ^ xe ^ {iy} = e ^ z.
\]

$f(z) = e ^ {iz}$ is complex differentiable on $\C$ by the chain rule and $f'(z) = ie ^ {iz}$ so
\[
\sin(z) = \frac{e ^ {iz} - e ^ {-iz}}{2i}
\]
is complex differentiable by COLT
\[
(\sin(z))' = \cos(z).
\]
Similarly $(\cos(z))' = -\sin(z)$.
$(\cosh(z))' = \sinh(z)$ and $(\sinh(z))' = \cosh(z)$ by the same method.

Polynomials are complex differentiable on $\C$ with
\[
(a_nz ^ n + \dotsi + a_1z + a_0)' = na_nz ^ {n - 1} + \dotsi + a_1.
\]
Any branch of $\log(z)$ on $\C \setminus R_{\theta_1}$ is complex differentiable on $\C \setminus R_{\theta_1}$ and $(\log(z))' = \frac{1}{z}$.

\begin{example}
    Let $f(z) = f(x + iy) = (x ^ 3 + 3x ^ 2y - y ^ 3 - x ^ 2 - 2y ^ 2) + i(-x ^ 3 + 3xy ^ 2 - y ^ 3 + 4xy + 3y)$.
    At which points is $f$ complex differentiable?

    \begin{solution}
        \begin{align*}
            u(x, y) &= x ^ 3 + 3x ^ 2y - y ^ 3 - x ^ 2 - 2y ^ 2 \\
            v(x, y) &= -x ^ 3 + 3xy ^ 2 - y ^ 3 + 4xy + 3y
        \end{align*}
        $u, v \in C ^ 2(\R ^ 2)$ so $f$ is complex differentiable at $z_0 = x_0 + iy_0$ if and only if the Cauchy-Riemann equations hold at $(x_0, y_0)$.

        If you do this,
        you get the the Cauchy Riemann equations hold when
        \[
        3x ^ 2 - 2x = -3y ^ 2 + 4x + 3
        \]
        which can be written as
        \[
        (x - 1) ^ 2 + y ^ 2 = 2.
        \]
    \end{solution}
\end{example}

\subsection{Holomorphicity}
Differentiability at one point is nice,
but usually not enough.
Most theorems in Real Analysis want differentiability on an interval.
A good candidate to replace the interval with is an open ball or in general,
an open set.

\begin{definition}[Holomorphic functions]
    A function $f : U \to \C$ defined on an open set $U \subseteq \C$ is holomorphic on $U$ if it is complex differentiable at every point in $U$.
    
    We say $f$ is holomorphic at $z_0$ if it is holomorphic on an open ball $B_{\varepsilon}(z_0)$ for some $\varepsilon > 0$.
\end{definition}

A function which is holomorphic on $\C$ is called an entire function.

$e ^ z, \sin(z), \cos(z), \sinh(z), \cosh(z)$,
polynomials are entire.

Any branch of $\log(z)$ on $\C \setminus R_{\theta}$ is holomorphic.

The function
\[
f(z) = f(x + iy) = (x ^ 3 + 3x ^ 2y - y ^ 3 - x ^ 2 - 2y ^ 2) + i(-x ^ 3 + 3xy ^ 2 - y ^ 3 + 4xy + 3y)
\]
is complex differentiable on 
\[
(x - 1) ^ 2 + y ^ 2 = 2
\]
no point of differentiability has a full open ball around it where $f$ is complex differentiable.
So $f$ is not holomorphic at any point.

Open sets are not good enough even in $\R$.
For instance we know that $f' = 0$ on $(a, b)$ implies $f$ is constant but
\[
f(x) = \begin{cases}
    1 & x \in (1, 3) \\
    2 & x \in (4, 5)
\end{cases}
\]
is $C ^ 1$ on $U = (1, 3) \cup (4, 5)$,
$f' = 0$ but $f$ is not constant.

\begin{definition}[Connectedness]
    A set $A \subseteq \C$ is said to be connected if $A \subseteq U_1 \cup U_2$ for some open sets $U_1$ and $U_2$ such that $U_1 \cap U_2 = \emptyset$ implies that $A \subseteq U_1$ or $A \subseteq U_2$.
    In other words,
    we cannot separate $A$ into two non-empty parts which lie in two disjoint open sets.
\end{definition}

Connectedness can be hard to show,
but intuitively if we can find a nice "path" between any two points in the set,
we think this will imply connectedness.

\begin{definition}[Paths and path connectedness]
    A path
    (or curve)
    in $\C$ is a continuous function $\gamma : [0, 1] \to \C$.
    We say that the path starts at $z \in \C$ and ends at $w \in \C$ if $\gamma(0) = z$ and $\gamma(1) = w$.
    A set $A \subseteq \C$ is said to be path connected if for any two points $z, w \in A$ there exists a path $\gamma$ that starts at $z$ and ends at $w$ such that $\gamma(t) \in A$ for all $t \in [0, 1]$
    (i.e. the path's image lies entirely inside $A$).
\end{definition}

\begin{theorem}
    If $A \subseteq \C$ is path connected then it is connected.
\end{theorem}

\begin{remark}
    The converse to the above theorem is not true in general:
    there are connected sets that are not path connected.
\end{remark}

A path $\gamma : [0, 1] \to \C$ can be thought of as two functions from $[0, 1]$ to $\R$.
indeed $\gamma(t) = \Re(\gamma(t)) + i\Im(\gamma(t))$.
We can use results from Real Analysis.

\begin{definition}\phantom{}
    \begin{itemize}
        \item A path
        (or curve)
        in $\C$ is said to be continuously differentiable,
        or $C ^ 1$,
        if its real part or imaginary part are continuously differentiable on $[0, 1]$.
        At the end points $0$ and $1$ this means that the real and imaginary parts have right-sided derivative at $0$ and left-sided derivative $1$,
        and that the derivatives are continuous from the right at $0$ and from the left at $1$.
        In that case we define
        \[
        \gamma'(t) \coloneqq (\Re(\gamma(t)))' + i(\Im(\gamma(t)))'.
        \]

        \item A path
        (or curve)
        in $\C$ is said to be a piecewise $C ^ 1$ path
        (or curve)
        or a contour if there exist
        \[
        0 = a_0 < a_1 < a_2 < \dotsi < a_{n - 1} < a_n = 1
        \]
        such that the paths $\gamma_i : [a_{i - 1}, a_i] \to \C$,
        with $i = 1, 2, \dotsc, n$ defined by
        \[
        \gamma_i(t) \coloneqq \gamma(t),\qquad t \in [a_{i - 1}, a_i]
        \]
        are $C ^ 1$ paths.

        \item We say a subset $A \subseteq \C$ is $C ^ 1$ path connected if for every pair of points $z, w \in A$ there exists a $C ^ 1$ path starts at $z$ and ends at $w$ such that $\gamma(t) \in A$ for all $t \in [0, 1]$.

        \item We say that $A$ is piecewise $C ^ 1$ path connected if for every pair of points $z, w \in A$ there exists a contour that starts at $z$ and ends at $w$ such that $\gamma(t) \in A$ for all $t \in [0, 1]$.
    \end{itemize}
\end{definition}

As any $C ^ 1$ or piecewise $C ^ 1$ path are paths $C ^ 1$ path connectedness or piecewise $C ^ 1$ path connectedness implies path connectedness.

Paths are actually continuous functions $\gamma : [a, b] \to \C$.

\begin{example}\phantom{}
    \begin{itemize}
        \item Open and closed balls are $C ^ 1$ path connected
        (every two points can be connected by a straight line inside the set).

        \item $\bar{\mathbb{H}} = \{z \in \C\,:\, \Im(z) \geq 0\}$ is $C ^ 1$ path connected
        (same as above).

        \item The annulus $A = \{z \in \C\,:\, 1 < |z| < 4\}$ is $C ^ 1$ path connected
        (but not every two points can be connected with a line).
    \end{itemize}
\end{example}

\begin{definition}[Domains]
    A domain
    (sometimes called regions)
    $D$ is an open,
    connected subset of $\C$
\end{definition}

The geometry of $\C$ provides.

\begin{theorem}
    Any domain in $\C$ is piecewise $C ^ 1$ path connected.
\end{theorem}

\begin{example}\phantom{}
    \begin{itemize}
        \item Open ball is a domain.

        \item Closed ball is not open and as such not a domain.

        \item $\C \setminus R_{\leq 0}$
        (a branch of log)
        is a domain.

        \item Any branch of $\log(z)$ and $z ^ {w}$ is defined on a domain $\C \setminus R_{\theta}$.

        \item The set $A = \{z \in \C\,:\, |z| \neq 1\}$ is not a domain
        (it is open but not connected)
    \end{itemize}
\end{example}

\begin{theorem}[Zero derivative theorem]\label{companalyII:thm:zerodivthm}
    Let $f : D \to \C$ be holomorphic on a domain $D \subseteq \C$.
    If $f'(z) = 0$ for every $z \in D$ then $f$ is constant on $\mathbb{D}$.
\end{theorem}

We will need the following in the proof:
\begin{lemma}[Chain rule revisited
(for paths)]
    Let $U \subset \C$ be an open set and let $f : U \to \C$ be a holomorphic function on $U$.
    Let $\gamma : [0, 1] \to U$ be a differentiable path
    (i.e. its real and imaginary parts are differentiable).
    Then for any $t_0 \in [0, 1]$ we have that
    \[
    (f \circ \gamma)'(t_0) = f'(\gamma(t_0))\gamma'(t_0).
    \]

    \begin{proof}
        Like in Real Analysis.
    \end{proof}
\end{lemma}

\begin{theorem}[continues = companalyII:thm:zerodivthm]
    \begin{proof}
        It is enough to show that for any $z_1, z_2 \in D$
        \[
        f(z_1) = f(z_2).
        \]

        Since $D$ is a domain we can find a contour $\gamma : [0, 1] \to \C$ s.t. $\gamma(0) = z_1$,
        $\gamma(1) = z_2$.

        We can find
        \[
        0 = a_0 < a_1 < \dotsi < a_n = 1
        \]
        s.t. $\gamma_i = \gamma|_{[a_{i - 1}, a_i]}$ is $C ^ 1$.

        If we'll show that
        \[
        f(\gamma_i(a_{i - 1})) = f(\gamma_i(a_i))\qquad \forall i
        \]
        then
        \[
        f(z_1) = f(\gamma(a_0)) = f(\gamma(a_1)) = f(\gamma(a_2)) = \dotsi = f(\gamma(a_n)) = f(z_2).
        \]
        Indeed:
        \[
        f \circ \gamma_i : [a_{i - 1}, a_i] \to \C
        \]
        is continuous and by the chain rule for paths it is differentiable on $(a_{i - 1}, a_i)$ with $(f \circ \gamma_i)'(t) = f'(\gamma_i(t))\cdot\gamma_i'(t) = 0$ so $(\Re(f \circ \gamma_i))' = 0$,
        $(\Im(f\circ\gamma_i))' = 0$ on $(a_{i - 1}, a_i)$ so by zero derivative theorem on $\R$.
        $\Re(f\circ \gamma_i)$ is constant and $\Im(f \circ \gamma_i)$ is constant on $[a_{i - 1}, a_i]$,
        completing the proof.
    \end{proof}
\end{theorem}

The beauty of complex analysis is that to have a constant holomorphic function we only need that $f$ doesn't fill the space
($\Re(f)$ is constant,
$(\Re(f)) ^ 2 + (\Im(f)) ^ 2 = 1$).

\subsection{The angle preserving properties of holomorphic functions}
There is almost a purely geometric description to being holomorphic.
For any curve $\gamma(t) : [0, 1] \to \C$ we can consider the tangent vector to $\gamma$ at $t_0$ which is
\[
\gamma'(t_0) = (\Re(\gamma))'(t_0) + i(\Im(\gamma))'(t_0).
\]

\begin{definition}[Conformal maps]
    We say that a
    (real differentiable)
    map $f : D \to \C$ on a domain $D \subseteq \C$ is conformal at $z_0$ if it preserves the angle and orientation between any two tangent vectors at $z_0$,
    where by orientation we mean the relative position of one vector with respect to another
    (i.e. if the tangent vector of a curve $\gamma_1$ was to the left of the tangent vector of the curve $\gamma_2$,
    the tangent vector of the image $\gamma_1$ by $f$ will remain to the left of the tangent vector of the image of $\gamma_2$ by $f$).
\end{definition}

We say that $f$ is conformal on $D$ if it is conformal at all points in $D$.

\begin{example}\phantom{}
    \begin{itemize}
        \item Rotations are conformal
        \[
        f(z) = e ^ {i\theta}z\qquad \theta \in \R.
        \]

        \item Dialation is conformal
        \[
        f(z) = \lambda z\qquad \lambda > 0.
        \]

        \item $f(z) = \bar{z}$ is not conformal.
        We'll only do it at $z_0$.
        The $C ^ 1$ paths $\gamma_1(t) = t$,
        $\gamma_2(s) = is$,
        $t \in [0, 1]$,
        $s \in [0, 1]$ meet at $z = 0$
        ($t = 0$,
        $s = 0$)
        with an angle $\pi / 2$.
        However,
        $f(\gamma_1(t)) = t$,
        $f(\gamma_2(s)) = -is$ meet at $z = 0$ with an angle $-\pi / 2$.
    \end{itemize}
\end{example}

\begin{lemma}[Holomorphic maps are conformal when their derivative is non-zero]
    Let $f$ be a holomorphic map at $z_0$.
    If $f'(z_0) \neq 0$ then $f$ is conformal at $z_0$.

    \begin{proof}
        Let $\gamma : [0, 1] \to \C$ be a differentiable curve and let $t_0 \in [0, 1]$ be such that $\gamma(t_0) = z_0$.
        What does $f$ do to the tangent vector.
        By the chain rule for paths
        \[
        (f \circ \gamma)'(t_0) = f'(\gamma / t_0) \cdot \gamma'(t_0) = f'(z_0) \cdot \gamma'(t_0).
        \]
        Since $f'(z_0) \neq 0$.
        The tangent vector of $f \circ \gamma$ at $t_0$ is nothing but dialation by $|f'(z_0)|$ and rotation by $\Arg(f'(z_0))$ of the tangent $\gamma'(t_0)$.
        Since it is the same rotation and dialation for all $\gamma$'s,
        $f$ is conformal.
    \end{proof}
\end{lemma}

\begin{remark}
    The condition $f'(z_0) \neq 0$ is necessary.
    $f(z) = z ^ 2$ satisfies $f'(0) = 0$.
    $f$ is not conformal at $z = 0$
    (the angle between the curves is not always preserved).
\end{remark}

Do we have a converse?

\begin{proposition}[Conformal maps are holomorphic]
    Let $D$ be a domain.
    If $f$ is conformal at $z_0 \in D$ then $f$ is complex differentiable at $z_0$ and $f'(z_0) \neq 0$.
    Therefore,
    if $f$ is conformal on $D$,
    then $f$ is holomorphic on $D$ and $f'(z) \neq 0$ for all $z \in D$.
\end{proposition}

\begin{corollary}
    Let $D$ be a domain.
    Then $f$ is conformal on $D$ if and only if $f$ is holomorphic on $D$ and $f'(z) \neq 0$ for all $z \in D$.
\end{corollary}

\begin{corollary}
    Any conformal map maps orthogonal grids in the $(x, y)$-plane to orthogonal grids.
\end{corollary}

\begin{example}
    $f(z) = z ^ 2$ and consider $z \neq 0$.
    $x = c$,
    $(u, v) = (c ^ 2 - y ^ 2, 2cy)$.
    i.e. $u = c ^ 2 - \frac{v ^ 2}{4c ^ 2}$ a parabola.
    $y = d$
    $(u, v) = (x ^ 2 - d ^ 2, 2dx)$,
    $u = \frac{v ^ 2}{4d ^ 2} - d ^ 2$.
\end{example}

\subsection{Biholomorphic maps}
In many cases we would like to say that two domains are holomorphically equivalent.

\begin{definition}[Biholomorphic maps]
    Let $D$ and $D'$ be domains.
    We say that $f : D \to D'$ is biholomorphic if $f$ is holomorphic,
    a bijection,
    and the inverse $f ^ {-1} : D' \to D$ is also holomorphic.

    A biholomorphic map $f$ is called a biholomorphism.
    When $f$ as above exists,
    we say that the domains $D$ and $D'$ are biholomorphic and write $f : D \tilde{\to} D'$.
\end{definition}

\begin{remark}
    It is possible to show that if $f$ as above is holomorphic and bijective,
    then it is automatic that its inverse is holomorphic.
\end{remark}

\begin{example}\phantom{}
    \begin{itemize}
        \item For any $a, b \in \C$ with $a \neq 0$
        \[
        f(z) = az + b
        \]
        is a biholomorphism between $\C$ and $\C$.
        The inverse is
        \[
        f ^ {-1}(z) = \frac{z - b}{a}.
        \]

        \item $e ^ z$ is not a biholomorphism from $\C$ to $\C ^ {*}$ since it is not injective.
        $e ^ z$ is a biholomorphism from
        \[
        \{z \in \C : c < \Im(z) < c + 2\pi\}
        \]
        to $\C \setminus R_{C}$.
        Its inverse is the appropriate branch of $\log$.

        $e ^ z$ is a bijection from $\{z \in \C\,:\, -\pi < \Im(z) \leq \pi\}$ to $\C ^ {*}$,
        but has no holomorphic inverse.

        \item $z ^ 2$ is a biholomorphism from $\mathbb{H}_{R}$ to $\C \setminus R_{\leq 0}$
        (inverse $z ^ {1 / 2}$).
    \end{itemize}
\end{example}

We cannot have a biholomorphic map defined on the boundary that it maps $\partial D$ to $\partial D'$.
This is hard to prove.

\begin{example}
    $e ^ z$ takes $\{z \in \C\,:\, -\pi < \Im(z) < 2\pi\}$ to $\C ^ {*}$ not injectively.
    \[
    \partial D = \{z \in \C\,:\, \Im(z) = -\pi\} \cup \{z \in \C\,:\, \Im(z) = 2\pi\}.
    \]
    $f(\partial D) = R \setminus \{0\}$.
    $D' = f(D) = \C ^ {*}$,
    $\partial D' = \{0\}$.
\end{example}

\begin{example}
    $e ^ z$ takes $D = \{z \in \C\,:\, -\pi < \Im(z) < \pi\}$ biholomorphically to $\C \setminus \R_{\leq 0}$.
    Boundary is mapped to boundary but not injectively.
\end{example}

\begin{lemma}[Automorphism groups]
    Let $D \subset \C$ be a domain.
    The set of all biholomorphic maps $f : D \tilde{\to} D$ from $D$ to itself forms a group under composition.
    We call this group the automorphism group of $D$ and denote it by $\mathrm{Aut}(D)$.
\end{lemma}

\section{M\"{o}bius transforms}

\subsection{\mobius transformations}

\begin{definition}[\mobius transformations]
    Let $a, b, c, d \in \C$ be such that $ad - bc \neq 0$.
    The map
    \[
    M(z) \coloneqq \frac{az + b}{cz + d}.
    \]
    defined on $\C \setminus \left\{-\frac{d}{c}\right\}$ when $c \neq 0$ and $\C$ when $c = 0$,
    is called a \mobius transformation.
\end{definition}

Any \mobius transformation can be represented by a matrix $T = \begin{pmatrix}
    a & b \\ c & d
\end{pmatrix} \in \mathrm{GL}_2(\C)$ where
\[
\mathrm{GL}_2(\C) \coloneqq \left\{\begin{pmatrix}
    a & b \\ c & d
\end{pmatrix}\,:\, a, b, c, d \in \C, ad - bc \neq 0\right\}.
\]
In that case we write $M = M_T$.

\begin{remark}\phantom{}
    \begin{itemize}
        \item Why $ad - bc \neq 0$?
        If $ad - bc = 0$ the $(a\ b)$ is linearly dependent on $(c\ d)$ i.e. $\exists \alpha \in \C$ s.t. $a = \alpha c$ and $b = \alpha d$
        \[
        \frac{az + b}{cz + d} = \alpha
        \]
        constant.

        \item Note that for any $a, b, c, d \in \C$ s.t. $ad - bc \neq 0$ and any $k \in \C ^ {*}$
        \[
        (ka)(kd) - (kb)(kc) \neq 0.
        \]
        \[
        M_{kT}(z) = \frac{(ka)z + (kb)}{(kc)z + (kd)} = \frac{az + b}{cz + d} = M_T(z).
        \]
        The matrix representation is not unique.
    \end{itemize}
\end{remark}

\begin{lemma}
    The set of \mobius transformations form a group under composition.
    Furthermore,
    \begin{enumerate}[label = (\roman*)]
        \item $M_{T_1} \circ M_{T_2} = M_{T_1T_2}$.

        \item Any \mobius transformation is invertible with $(M_T) ^ {-1} = M_{T ^ {-1}}$.

        \item For any $k \in \C ^ {*}$ we have that $M_{kT} = M_T$.
        We conclude that $M_T = \mathrm{Id}$ if and only if
        \[
        T = k\begin{pmatrix}
            1 & 0 \\ 0 & 1
        \end{pmatrix}
        \]
        for some $k \in \C ^ {*}$.
    \end{enumerate}

    \begin{proof}
        Exercise.

        Note:
        if (i) holds then since $T \in \mathrm{GL}_2(\C) \implies T ^ {-1} \in \mathrm{GL}_2(\C)$ and
        \[
        M_T \circ M_{T ^ {-1}} = M_{T \cdot T ^ {-1}} = M_{\mathrm{Id}} = \mathrm{Id}.
        \]
    \end{proof}
\end{lemma}

We conclude that $T = \begin{pmatrix}
    a & b \\ c & d
\end{pmatrix}$
\begin{itemize}
    \item if $c \neq 0$
    \[
    M_T = \C \setminus \{-d / c\} \to \C \setminus \{a / c\}
    \]
    is a bijection.
    Where $(M_T) ^ {-1} = M_{T ^ {-1}}$ is defined $T ^ {-1} = \frac{1}{ad - bc}\begin{pmatrix}
        d & -b \\ -c & a
    \end{pmatrix}$.

    \item If $c = 0$ $M_T(z) = \frac{a}{d}z + \frac{b}{d}$ is a bijection from $\C \to \C$.
\end{itemize}

We can extend \mobius transformation to $\hat{\C} = \C \cup \{\infty\}$,
a circle with infinite radius.
If $c \neq 0$
\[
\lim_{z \to - d / c}\left|\frac{az + b}{cz + d}\right| = \infty.
\]
\[
\lim_{|z| \to \infty}\frac{az + b}{cz + d} = \frac{a}{c}.
\]
If $c = 0$
\[
\lim_{|z| \to \infty}\left|\frac{a}{d}z + \frac{b}{d}\right| = \infty.
\]

\begin{remark}
    We can extend the definition of \mobius transformation to bijections from $\hat{\C}$ to $\hat{C}$:
    Given $a, b, c, d \in \C$ such that $ad - bc \neq 0$ we define $M : \hat{\C} \to \hat{\C}$ by
    \[
    M(z) = \begin{cases}
        \frac{az + b}{cz + d}, &z \neq -\frac{d}{c}, \infty \\
        \infty, &z = -\frac{d}{c}, \\
        \frac{a}{c}, & z = \infty,
    \end{cases}
    \]
    when $c \neq 0$,
    and
    \[
    M(z) \coloneqq \begin{cases}
        \frac{az + b}{d}, &z \neq \infty \\
        \infty, &z = \infty,
    \end{cases}
    \]
    when $c = 0$.
\end{remark}

\begin{example}
    $f(z) = \frac{1}{z}$ corresponds to $T = \begin{pmatrix}
        0 & 1 \\ 1 & 0
    \end{pmatrix}$.

    Using polar coordinates we find that $f(z) = \frac{1}{r}e ^ {i\theta}$.
    Segments on a concentric circle of radius $r$ go to segments of same length but opposite direction on a concentric circle of radius $\frac{1}{r}$.
\end{example}

Similarly,
a segment on a ray of angle $\theta$ will go to a segment on a ray of angle $-\theta$.

\begin{lemma}\phantom{}
    \begin{itemize}
        \item The map $1 / z$ injectively takes the set $\mathbb{D} \setminus \{0\} = \{z \in \C\,:\, 0 < |z| < 1\}$ to $\mathbb{D} ^ c = \{z \in \C\,:\, |z| > 1\}$.
        When considering $1 / z$ over $\hat{\C}$ we find that it injectively takes $\mathbb{D}$ to $\mathbb{D} ^ c \cup \{\infty\}$.

        \item The map $1 / z$ injectively takes the ray at angle $\theta$
        (without the origin)
        to the ray at angle $-\theta$
        (without the origin).
        On $\hat{\C}$ the statement remain the same by adding the origin and $\infty$ to the rays.
    \end{itemize}
\end{lemma}

What about derivatives?

\begin{lemma}
    Let $M$ be a \mobius transform which is represented by the matrix $T = \begin{pmatrix}
        a & b \\ c & d
    \end{pmatrix} \in \mathrm{GL}_2(\C)$.
    If $c \neq 0$ the \mobius transformation $M = M_T$ is a biholomorphism
    \[
    M : \C \setminus \left\{-\frac{d}{c}\right\} \tilde{\to} \C \setminus \left\{\frac{a}{c}\right\}.
    \]
    and when $c = 0$ the \mobius transformation $M = M_T$ is a biholomorphism
    \[
    M : \C \tilde{\to} \C.
    \]

    \begin{proof}
        Already shown the case $c = 0$,
        i.e. $f(z) = \frac{a}{d}z + \frac{b}{d}$.

        If $c \neq 0$ we know that
        \[
        M : \C \setminus \{-d / c\} \to \C \setminus \{a / c\}
        \]
        is a bijection
        \[
        M(z) = \frac{az + b}{cz + d}
        \]
        is a quotient of complex differentiable functions.
        By COLT for derivatives for $z \neq -\frac{d}{c}$,
        $M'(z) = \frac{a(cz + d) - c(az + d)}{(cz + d) ^ 2} = \frac{ad - bc}{(cz + d) ^ 2}$.
        $M$ is holomorphic on $\C \setminus \{-d / c\}$.
        $M ^ {-1}$ is also \mobius so holomorphic on $\C \setminus \{a / c\}$.
        So $M$ is a biholomorphism between $\C \setminus \{-d / c\}$ and $\C \setminus \{a / c\}$.
    \end{proof}
\end{lemma}

Behold.
If
\[
M(z) = \frac{ax + b}{cz + d} = M_T(z)
\]
when we showed that
\[
M'(z) = \begin{cases}
    \frac{ad - bc}{(cz + d) ^ 2} & c \neq 0 \\
    \frac{a}{d} & c = 0.
\end{cases}
\]
$M'(z) \neq 0$ for any $z \in \C \setminus \{-d / c\}$ when $c \neq 0$ and any $z \in \C$ when $c = 0$.
Consequently

\begin{corollary}
    A \mobius transformation $M$ is conformal at all $z \in \C$ with $M(z) \neq \infty$.

    \begin{proof}
        $M$ is holomorphic and $M'(z) \neq 0$.
    \end{proof}
\end{corollary}

\mobius transformation truly capture the geometry of $\hat{\C}$.

\begin{proposition}
    A \mobius transformation $M$ is a biholomorphism of $\hat{\C}$ to itself.
    Moreover,
    any biholomorphism between $\hat{\C}$ and itself must be a \mobius transformation.

    \begin{proof}
        Omitted.
    \end{proof}
\end{proposition}

\subsection{The Three points theorem,
the cross ratio,
and fixed points for \mobius transformation}

To define a \mobius transformation we seem to need $a, b, c, d \in \C$ s.t. $ad - bc \neq 0$,
or alternatively $T \in \mathrm{GL}_2(\C)$ but since $kT$ represents the same \mobius transformation for $k \in \C ^ {*}$ we can always force one of $a, b, c$ or $d$,
to be one.
This means that in fact we need three points to give us a \mobius transformation

\begin{theorem}[Three points Theorem]
    Let $\{z_1, z_2, z_3\}$ and $\{w_1, w_2, w_3\}$ be two sets of three ordered distinct points in $\hat{\C}$.
    Then there exists a unique \mobius transformation $M$ such that $M(z_i) = w_i$ for $i = 1, 2, 3$.
\end{theorem}
The theorem includes two things:
existence,
uniqueness.
To show existence we have
\begin{definition}
    Given four distinct points $z_0, z_1, z_2, z_3 \in \C$ the cross-ratio of these points,
    which we will denote by $(z_0, z_1; z_2, z_3)$,
    is defined by
    \[
    (z_0, z_1; z_2, z_3) \coloneqq \frac{(z_0 - z_2)(z_1 - z_3)}{(z_0 - z_3)(z_1 - z_2)} = \frac{\frac{z_0 - z_2}{z_0 - z_3}}{\frac{z_1 - z_2}{z_1 - z_3}}.
    \]
\end{definition}

The cross ratio can be extended to $\hat{\C}$ with the convention that if one of the points is $\infty$ we remove all the terms that involve that point.
For example,
\[
(\infty, z_1; z_2, z_3) \coloneqq \frac{z_1 - z_3}{z_1 - z_2}.
\]

\textit{Proof of the existence of the \mobius transformation in the three points theorem}

For a given $z_1, z_2, z_3 \in \C$ distinct define
\[
F(z) = (z, z_1; z_2, z_3) = \frac{(z - z_2)(z_1 - z_3)}{(z - z_3)(z_1 - z_2)}
\]
$F$ is \mobius and
\[
F(z_1) = 1,\qquad F(z_2) = 0,\qquad F(z_3) = \infty.
\]
With
\[
G(w) = \frac{(w - w_2)(w_1 - w_3)}{(w - w_3)(w_1 - w_2)}.
\]
$G$ is \mobius so $G ^ {-1}$ exists and is \mobius.
So $G ^ {-1} \circ F$ is \mobius and
\[
G ^ {-1} \circ F(z_i) = w_i\qquad i = 1, 2, 3.
\]

What about uniqueness?

\textbf{Notation:}
For a given map $f : X \to X$ we say that $x \in X$ is a fixed point of $f$ if $f(x) = x$.

\begin{lemma}[Fixed point theorem for \mobius transformations]
    Let $M : \hat{\C} \to \hat{\C}$ be a \mobius transformation that is not the identity map.
    Then $M$ has at most $2$ fixed points in $\hat{\C}$.
    In other words,
    if a \mobius transformation has three fixed points in $\hat{\C}$,
    then it is the identity:

    \begin{proof}
        \[
        M(z) = \frac{az + b}{cz + d}
        \]
        then
        \[
        M(z) = z \iff az + b = (cz + d)z \iff cz ^ 2 + (d - a)z - b = 0.
        \]
        If $c \neq 0$ there exist at most two distinct solutions on $\C$.
        in this case $M(-d / c) = \infty$ so $\infty$ is not a fixed point.

        If $c = 0$,
        $(d - a)z - b = 0$,
        if $d \neq a$ there is a unique solution in $\C$ which is $z = \frac{b}{d - a}$.
        In this case $M(\infty) = \infty$ so we get two fixed points in $\hat{\C}$.

        If $d = a$,
        $-b = 0$,
        i.e. $b = c = 0$
        \[
        M(z) = \frac{az + 0}{0\cdot z + a} = z
        \]
        the identity.
    \end{proof}
\end{lemma}

\textit{Proof of the uniqueness of the \mobius transformation in the three points theorem}

Assume there exist $M_1, M_2$ \mobius s.t.
\[
M_1(z_i) = M_2(z_i) = w_i\qquad i = 1, 2, 3
\]
so $M = M_1 \circ M_2$ is \mobius s.t. $M(z_i) = z_i$ for $i = 1, 2, 3$,
i.e. $M$ has $3$ fixed points which implies $M = \mathrm{Id}$ i.e. $M_1 = M_2$.

\subsection{Geometric properties of \mobius transformation and how to find them explicitly}
Our existence part of the three point theorem gave us a recipe to find \mobius transformation that takes $(z_1, z_2, z_3)$ to $(w_1, w_2, w_3)$.
We want to simplify the process.

\begin{lemma}[Cross ratio is invariant under \mobius transformation]
    The cross ratio is invariant under \mobius transformations.
    In other words,
    if $M :\hat{\C} \to \hat{\C}$ is a \mobius transformation then
    \[
    (M(z_0), M(z_1); M(z_2), M(z_3)) = (z_0, z_1; z_2, z_3).
    \]

    \begin{proof}
        Proof relies on identifying building blocks of \mobius transformation.
    \end{proof}
\end{lemma}

\begin{lemma}[Building blocks of \mobius transformations]
    Any \mobius transformation is compositions of the following four types of maps:

    \begin{enumerate}[label = (\roman*)]
        \item Shift:
        $z \mapsto z + b$ for some $b \in \C$.

        \item Dilation
        (stretch/compression):
        $z \mapsto \lambda z$ for some $\lambda \in \R_{> 0}$.

        \item Rotation:
        $z \mapsto e ^ {i\theta}z$ for some $\theta \in (-\pi, \pi]$.

        \item Inversion:
        $z \mapsto \frac{1}{z}$.
    \end{enumerate}

    \begin{proof}
        For any $a \in \C$,
        $a \neq 0$,
        $b \in \C$ we have that
        \[
        az + b = |a|e ^ {i\Arg(a)}z + b
        \]
        well defined as $a \neq 0$.
        The map is a composition of rotation,
        dilation and shift.

        For any $a \in \C ^ {*}$
        ($\C ^ {*} = \C \setminus \{0\}$)
        and $b \in \C$
        \[
        \frac{1}{az + b}
        \]
        is a composition of inversion and the previous map
        (composition of rotation,
        dilation and shift).

        For any $a, \alpha \in \C ^ {*}$ and $b, \beta \in \C$
        \[
        \alpha \cdot \frac{1}{az + b} + \beta
        \]
        is a composition of rotation,
        dilation,
        shift and the previous map.
        If we show that any \mobius is one of these maps,
        we will conclude the proof.
        $M(z) = \frac{az + b}{cz + d}$,
        $ad - bc \neq 0$.
        If $c = 0$,
        $M(z) = \frac{a}{d}z + \frac{b}{d}$,
        first type of map.
        If $c \neq 0$,
        \[
        M(Z) = \frac{az + b}{cz + d} = \frac{\frac{a}{c}(cz + d) - \frac{ad}{c} + b}{cz + d} = \frac{a}{c} - \frac{ad - bc}{c(cz + d)}
        \]
        the third type of map.
    \end{proof}
\end{lemma}

\begin{corollary}
    Any \mobius transformation is compositions of the following two maps:

    \begin{enumerate}[label = (\roman*)]
        \item $z \mapsto az + b$ for some $a \in \C ^ {*}$ and $b \in \C$.

        \item $z \mapsto \frac{1}{z}$.
    \end{enumerate}
\end{corollary}

Since every \mobius transformation is a composition of maps of the form $f_1(z) = az + b$,
$a \in \C ^ {*}$,
$b \in \C$ and $f_2(z) = \frac{1}{z}$ it is enough to show that for any $z_0, z_1, z_2, z_3$,
\[
(f_1(z_0), f_1(z_1); f_1(z_2), f_1(z_2)) = (z_0, z_1; z_2, z_3) = (f_2(z_0), f_2(z_1); f_2(z_2), f_2(z_3))
\]
for instance
\[
\frac{f_1(z_0) - f_1(z_2)}{f_1(z_0) - f_1(z_3)} = \frac{a(z_0 - z_2)}{a(z_0 - z_3)} = \frac{z_0 - z_2}{z_0 - z_3}
\]
and similarly
\[
\frac{f_1(z_1) - f_1(z_2)}{f_1(z_1) - f_1(z_3)} = \frac{z_1 - z_2}{z_1 - z_3}.
\]

If we want to find a \mobius transformation $M$ s.t. $M(z_1) = w_1$,
$M(z_2) = w_2$,
$M(z_3) = w_3$ we see that due to the invariance of the cross-ratio for any $z \in \hat{\C}$
\[
(M(z), w_1; w_2, w_3) = (z, z_1; z_2, z_3).
\]

\begin{example}
    Find the \mobius transformation that takes the points $\{1, -1, i\}$ to $\{0,\infty, 1\}$ respectively.
    \[
    (M(z), 0; \infty, 1) = (z, 1; -1, i)
    \]
    \[
    \frac{1}{M(z) - 1}\cdot\frac{0 - 1}{1} = \frac{z - (-1)}{z - i}\cdot\frac{1 - i}{1 - (-1)}
    \]
    \[
    -\frac{1}{M(z) - 1} = \frac{z + 1}{z - i}\cdot\frac{1 - i}{2},\qquad M(z) = \frac{-iz + i}{z + 1}.
    \]
\end{example}

We now have another biholomorphic map.
We would like to explore which domains are
"natural"
for \mobius transformations.

\begin{lemma}
    Let $M : \hat{\C} \to \hat{\C}$ be a \mobius transformation.
    For any open set $D \subseteq \C$ we have that
    \[
    M\left(\partial D \setminus \left\{-\frac{d}{c}\right\}\right) = \partial M(D) \setminus \{\infty\}.
    \]
    When $c = 0$ we consider $-\frac{d}{c}$ as $\infty$ which is outside of $\C$ consequently $A \setminus \{\infty\} = A$ for any $A \subseteq \C$.
\end{lemma}

\begin{remark}
    We can extend the above to $\hat{\C}$ by including $-\frac{d}{c}$ and $\infty$.
\end{remark}

Boundary goes to boundary under \mobius.
We can have a method to find $M(D)$ for open $D$:
\begin{enumerate}[label = (\arabic*)]
    \item Find the image $M(\partial D)$ of the boundary of $D$.

    \item Find the image $M(z_0)$ of a point $z_0 \in D$.

    \item $M(D)$ is the set whose boundary is $M(\partial D)$ and contains $M(z_0)$.
    Moreover,
    \[
    M : D \tilde{\to} M(D)
    \]
\end{enumerate}

Do \mobius transformations have boundaries that are mapped to nice boundaries?

\begin{proposition}[Circles and lines Proposition]
    \mobius transformation map circles and lines in $\hat{\C}$ to circles and lines in $\hat{\C}$,
    where we consider any line to pass through infinity and where by circles in $\hat{\C}$ we simply mean circles in $\C$.
    Moreover,
    a circle or a line would be mapped to a line if and only if there is a point on it that is mapped to $\infty$.
\end{proposition}

\begin{lemma}[Equation of circles and lines in $\C$]
    Given $\gamma, \beta \in \R$ and $\alpha \in \C$,
    the equation
    \[
    \gamma z\bar{z} - \alpha\bar{z} - \bar{\alpha}z + \beta = 0
    \]
    describes a circle is $\gamma \neq 0$ and $|\alpha| ^ 2 - \beta\gamma > 0$,
    and a line if $\gamma = 0$ and $\alpha \neq 0$.
    Conversely,
    any circle or line can be described by an equation of this form.
\end{lemma}

The term circline is used to refer to an object that is either a circle or line.

\begin{example}
    The Cayley map is defined to be
    \[
    M_c(z) \coloneqq \frac{z - i}{z + i}.
    \]
    It is a \mobius transformation which is represented by a matrix $C = \begin{pmatrix}
        1 & -i \\ 1 & i
    \end{pmatrix}$.
    A straight forward calculation shows that its inverse is given by
    \[
    M_{C} ^ {-1}(z) \coloneqq \frac{iz + i}{1 - z}.
    \]
\end{example}

\begin{lemma}\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item The image of upper half plane $\mathbb{H}$ under the Cayley map is the unit disk,
        $\mathbb{D}$.

        \item The image of unit disk,
        $\mathbb{D}$,
        under the Cayley map is the left half plane,
        $\mathbb{H}_{L}$.

        \item The image of the first quadrant,
        \[
        \Omega_1 = \left\{z \in \C,\:\, 0 < \Arg(z) < \frac{\pi}{2}\right\} = \{z \in \C\mid \Re(z) > 0, \Im(z) > 0\},
        \]
        under the Cayley map is the lower unit disk
        \[
        \mathbb{D}_{-} = \{z \in \mathbb{D} \mid \Im(z) > 0\}
        \]

        \item The image of the upper unit disk,
        \[
        \mathbb{D}_{+} = \{z \in \mathbb{D} \mid \Im(z) > 0\}
        \]
        under the Cayley map is the left unit disk
        \[
        \mathbb{D}_{L} = \{z \in \mathbb{D} \mid \Re(z) < 0\}.
        \]
    \end{enumerate}

    \begin{proof}
        Showing (i) and (iii).

        $M_C$ is \mobius and as such takes boundary to boundary and we get $\partial M(\mathbb{H}) = M(\partial \mathbb{H})$.
        $\partial H$ is a line and as such will go to a line or a circle by a \mobius transformation.
        The image will be a line if and only if it contains $\infty$.
        $M_C$ takes $-i$ to $\infty$ and $-i \notin \partial \mathbb{H}$ so $M(\partial \mathbb{H})$ is a circle.
        To find the circle we'll find three points on it
        \[
        M_C(-1) = \frac{-1 - i}{-1 + i} = i ,\qquad M_C(0) = -1,\qquad M_C(1) = -i.
        \]
        This implies $M(\partial \mathbb{H})$ is the unit circle.
        To find if we get $\mathbb{D}$ or $\mathbb{D} ^ c$ we take an interior point to $\mathbb{H}$.
        $i \in \mathbb{H}$,
        $M_C(i) = 0$ which implies $M(\mathbb{H}) = \mathbb{D}$.
        \mobius will take a part of a circle or line to a part of a circle or line.
        $\Omega_1 = \mathbb{H} \cap \mathbb{H}_R$,
        $M_C$ is a bijection so $M(\Omega_1) = M(\mathbb{H}) \cap M(\mathbb{H}_R)$/
        Only need to find $M_C(\mathbb{H}_R)$.
        \[
        M_C(\partial \mathbb{H}_R) = \partial M(\mathbb{H}_R)
        \]
        which implies $\partial M_C(\mathbb{H}_R)$ is a line.
        $M_C(i) = 0$,
        $M_C(0) = -1$ so $\partial M_C(\mathbb{H}_R) = \text{real axis}$.
        $M_C(1 + i)  = \frac{2}{5} - \frac{i}{5}$ so $M_C(\mathbb{H}_R)$ which is the lower half plane.
    \end{proof}
\end{lemma}

\begin{corollary}
    Let $M_C$ be the Cayley map.
    Then
    \begin{itemize}
        \item $M_C ^ {-1}(\mathbb{D}) = \mathbb{H}$.

        \item $M_C ^ {-1}(\mathbb{H}_L) = \mathbb{D}$.

        \item $M_C ^ {-1}(\mathbb{D}_{-}) = \Omega_1$.
        
        \item $M_C ^ {-1}(\mathbb{D}_{L}) = \mathbb{D}_{+}$.
    \end{itemize}
\end{corollary}

\begin{proposition}[H$2$H]
    Every \mobius transformation mapping $\mathbb{H}$ to $\mathbb{H}$ is of the form $M_T$ with $T$ in the group
    \[
    \mathrm{SL}_2(\R) \coloneqq \left\{T = \begin{pmatrix}
        a & b \\ c & d
    \end{pmatrix}\,:\, a, b, c, d \in \R,\ \det{T} = ad - bc = 1\right\}.
    \]
    Conversely,
    every such \mobius transformation maps $\mathbb{H}$ to $\mathbb{H}$,
    and hence gives a biholomorphism from $\mathbb{H}$ to $\mathbb{H}$.

    \begin{proof}
        We will only show that if $M$ takes $\mathbb{H}$ to $\mathbb{H}$ then $M = M_t$ for $T \in \mathrm{SL}_2(\R)$.

        $M$ is \mobius so $\partial H$ must go to $\partial M(\mathbb{H}) = \partial \mathbb{H}$.
        There exist $x_1, x_2, x_3 \in \R \cup \{\infty\}$ s.t. $M(x_1) = 1$,
        $M(x_2) = 0$,
        $M(x_3) = \infty$.
        Assuming $x_1, x_2, x_3 \neq \infty$ we find that
        \[
        (M(z), 1; 0, \infty) = (z, x_1; x_2, x_3)
        \]
        \[
        M(z) = \frac{M(z) - 0}{1} \cdot \frac{1}{1 - 0} = \frac{z - x_2}{z - x_3}\cdot\frac{x_1 - x_3}{x_1 - x_2}
        \]
        so
        \[
        M(z) = \frac{(x_1 - x_3)z - x_2(x_1 - x_3)}{(x_1 - x_2)z - x_3(x_1 - x_2)}
        \]
        so $M = M_T$ for $T \in \mathrm{GL}_2(\R)$.
        same holds if one of the $x_i$s is $\infty$.
        Since $\mathbb{H}$ goes to $\mathbb{H}$,
        $M(i) \in \mathbb{H}$ i.e. $\Im(M(i)) > 0$.
        Writing $M(z) = \frac{az + b}{cz + d}$,
        $a, b, c, d \in \R$,
        $ad - bc \neq 0$ we get
        \[
        M(i) = \frac{ai + b}{ci + d} = \frac{(ai + b)(d - ci)}{c ^ 2 + d ^ 2} = \frac{db + ac}{c ^ 2 + d ^ 2} + i\frac{ad - bc}{c ^ 2 + d ^ 2}
        \]
        so $\Im(M(i)) > 0 \iff ad - bc > 0$.
        The matrix $\tilde{T} = \frac{1}{\sqrt{ad - bc}}T$ satisfies
        \[
        M_{\tilde{T}} = M_T
        \]
        and $\tilde{T} \in \mathrm{GL}_2(\R)$.
        In fact
        \[
        \det{\tilde{T}} = \left(\frac{1}{\sqrt{ad - bc}}\right) ^ 2\det{T} = 1.
        \]
    \end{proof}
\end{proposition}

\begin{proposition}[D$2$D]
    Every \mobius transformation from the unit disk $\mathbb{D}$ to itself is of the form $M_T$ with $T$ in the set
    \[
    \mathrm{SU}(1, 1) \coloneqq \left\{T = \begin{pmatrix}
        \alpha & \beta \\ \bar{\beta} & \bar{\alpha}
    \end{pmatrix}\,:\, \alpha, \beta \in \C, \det{T} = |\alpha| ^ 2 - |\beta| ^ 2 = 1\right\}.
    \]
    Conversely,
    every such \mobius transformation maps $\mathbb{D}$ to $\mathbb{D}$ and hence gives a biholomorphic automorphism of $\mathbb{D}$.

    \begin{proof}
        We recall that $M_C(\mathbb{H}) = \mathbb{D}$ and $(M_C) ^ {-1} = M_{C ^ {-1}}$ takes $\mathbb{D}$ to $\mathbb{H}$.
        Given \mobius $M$ from $\mathbb{D}$ to $\mathbb{D}$ the map
        \[
        f = (M_C) ^ {-1} \circ M \circ M_C.
        \]
        $f$ is \mobius
        (composition of \mobius\!\!)
        that takes $\mathbb{H}$ to $\mathbb{H}$.
        By $H2H$ theorem $\exists T \in \mathrm{SL}_2(\R)$ s.t. $f = M_t$,
        i.e. $M_{C ^ {-1}} \circ M \circ M_C = M_T$ i.e.
        \[
        M = M_C \circ M_T \circ M_{C ^ {-1}} = M_{CTC ^ {-1}}
        \]
        $T = \begin{pmatrix}
            a & b \\ c & d
        \end{pmatrix}$,
        $a, b, c, d \in \R$ $ad - bc = 1$,
        $C = \begin{pmatrix}
            1 & -i \\ 1 & i
        \end{pmatrix}$
        which implies
        \[
        CTC ^ {-1} = \frac{1}{2}\begin{pmatrix}
            (a + d) + i(b - c) & (a - d) - i(c + b) \\ (a - d) + i(c + b) & (a + d) - i(b - c)
        \end{pmatrix}.
        \]
    \end{proof}
\end{proposition}

\begin{example}
    Find a biholomorphic map that takes $\C \setminus \R_{\leq 0}$ to $\mathbb{D}$.

    \begin{solution}
        First the principal $z ^ {1 / 2}$.
        Rotate to $e ^ {i\pi / 2}z = iz$.
        Then $M_C$ the Cayley map to get the unit disc.

        Composing $z ^ {1 / 2}$,
        $e ^ {i\pi / 2}z$ and $M_C$ to get the unit disc.

        \[
        f(z) = M_C(iz ^ {1 / 2}) = \frac{iz ^ {1 / 2} - i}{iz ^ {1 / 2} + i} = \frac{z ^ {1 / 2} - 1}{z ^ {1 / 2} + 1}.
        \]
    \end{solution}
\end{example}

\section{Notions of Convergence in Complex Analysis and power series}
We are moving towards a fundamental result in Complex Analysis:
any holomorphic function in a domain can be written as a power series or extended power series.

To do this we need notions of convergence for sequences and series of functions.

\subsection{Pointwise and uniform convergence}

\begin{definition}[Pointwise convergence]
    Let $D \subseteq \C$ be a given set and let $\{f_n\}_{n \in \N}: D \to \C$ be a sequence of functions.
    We say that $\{f_n\}_{n \in \N}$ converges pointwise on $D$ to $f : D \to \C$ if for any $z \in D$ the limit function
    \[
    f(z) \coloneqq \lim_{n \to \infty}f_n(z)
    \]
    exists.

    For any $z \in D$ and any $\varepsilon > 0$,
    there exists $N(\varepsilon, z) \in \N$ such that if $n > N(\varepsilon, z)$
    \[
    |f_n(z) - f(z)| < \varepsilon.
    \]

    We say the series $\sum_{n = 1}^{\infty}f_n(z)$ converges pointwise on $D$ to $S : D \to \C$ if the sequence of functions $\{S_N\}_{N \in \N} : D \to \C$ defined by
    \[
    S_N(z) = \sum_{n = 1}^{N}f_n(z)
    \]
    converges pointwise to the function $S(z)$.
\end{definition}

\begin{example}
    $f_n(z) = z ^ n$ we have
    \[
    f_n(z) \underset{n \to \infty}{\to} \begin{cases}
        0 & |z| < 1, \\
        \infty & |z| > 1, \\
        1 & |z| = 1, \\
        \text{no limit} & z \neq 1, |z| = 1.
    \end{cases}
    \]

    $|f_n(z)| = |z ^ n| \to 0$ as $n \to \infty$ if $|z| < 1$ which implies
    \[
    |f_n(z) - 0| \to 0\text{ if } |z| < 1
    \]
    i.e. $f_n(z) \to 0$ as $n \to \infty$.
    Note:
    $|f_n(z) - 0| = |z| ^ k < \varepsilon \iff n > \frac{\log(\varepsilon)}{\log(|z|)}$
    when $|z| < 1$,
    $\varepsilon < 1$.
    $|f_n(z)| = |z| ^ n \to \infty$ as $n \to \infty$ if $|z| > 1$ i.e. $f_n(z) \to \infty$ as $n \to \infty$.
    $f_n(1) = 1 ^ n = 1 \to 1$ as $n \to \infty$.
    If $f_n(z) = z ^ n \to \xi$ as $n \to \infty$ then $|z| = 1$ but $z \neq 1$ then $f_{n + 1}(z) \to \xi$ as $n \to \infty$.

    $f_n(z)$ was continuous on $\C$ converges on $\mathbb{D} \cup \{1\}$ but the limit is not continuous
\end{example}

\begin{definition}[Uniform convergence]
    Let $D \subseteq \C$ be a given set and let $\{f_n\}_{n \in \N} : D \to \C$ be a sequence of functions.
    We say that $\{f_n\}_{n \in \N}$ converges uniformly to
    (the limit function)
    $f : D \to \C$ on $D$ if we have that
    \[
    \sup_{z \in D}|f_n(z) - f(z)| \to 0\text{ as } n \to \infty.
    \]
    In other words,
    for any $\varepsilon > 0$ there exists $N(\varepsilon) \in \N$ such that if $n > N(\varepsilon)$
    \[
    |f_n(z) - f(z)| < \varepsilon,\qquad \forall z \in D.
    \]

    We say that the series $\sum_{n = 1}^{\infty}f_n(z)$ converges uniformly to $S : D \to \C$ on $D$ if the sequence of functions $\{S_N\}_{N \in \N} : D \to \C$ defined by
    \[
    S_N(z) = \sum_{n = 1}^{N}f_n(z)
    \]
    converges uniformly to the function $S(z)$ on $D$.
\end{definition}

\begin{theorem}[Uniform limits of continuous functions are continuous]
    Let $\{f_n\}_{n \in \N} : D \to \C$  be a sequence of continuous functions that converges uniformly to $f$ on $D$.
    Then $f$ is continuous on $D$.
\end{theorem}

Do we have simple criteria to check uniform convergence?

\begin{lemma}[Criteria for uniform convergence and lack of uniform convergence for sequences]
    Let $\{f_n\}_{n \in \N} : D \to \C$ be a sequence of functions that converges pointwise to a limit function $f$ on $D$.

    \begin{enumerate}[label = (\roman*)]
        \item If there exists a non-negative sequence of real numbers $\{s_n\}_{n \in \N}$ that converges to zero and $n_0 \in \N$ such that for all $n \geq n_0$
        \[
        |f_n(z) - f(z)| \leq s_n,\qquad \forall z \in D,
        \]
        then the sequence of functions $\{f_n\}_{n \in \N}$ converges uniformly to $f$ on $D$.

        \item If there exists a non-negative sequence of real numbers $\{s_n\}_{n \in \N}$ that converges to $c > 0$,
        $n_0 \in \N$,
        and a sequence $\{z_n\}_{n \in \N} \subset D$ such that for all $n \geq n_0$
        \[
        |f_n(z_n) - f(z_n)| \geq s_n
        \]
        then the sequence of functions $\{f_n\}_{n \in \N}$ does not converge uniformly to $f$ on $D$.
    \end{enumerate}

    \begin{proof}
        \begin{enumerate}[label = (\roman*)]
            \item For any $z \in D$
            \[
            |f_n(z) - f(z)| \leq s_n
            \]
            \[
            0 \leq \sup_{z \in D}|f_n(z) - f(z)| \leq \sup_{z \in D}(s_n) = s_n \to 0
            \]
            as $n \to \infty$.
            By squeezing lemma
            \[
            \sup_{z \in D}|f_n(z) - f(z)| \to 0\text{ as } n \to \infty.
            \]

            \item We have that
            \[
            \sup_{z \in D}|f_n(z) - f(z)| \geq |f_n(z_n) - f(z_0)| \geq s_n
            \]
            if $f_n$ converged uniformly to $f$ then
            \[
            \sup_{z \in D}|f_n(z) - f(z)| \geq s_n
            \]
            then the first goes to $0$ but $s_n \to c > 0$ but this is a contradiction.
        \end{enumerate}
    \end{proof}
\end{lemma}

\begin{remark}\phantom{}
    \begin{itemize}
        \item In many cases we have $n_0 = 1$ in the conditions for the lemma above.

        \item Simpler version of second part of lemma is:
        If $\exists c > 0$ and $\{z_n\}_{n \in \N} \subset D$ such that $\forall n \in \N$
        \[
        |f_n(z_n) - f(z_n)| \geq c,
        \]
        then the sequence of functions $\{f_n\}_{n \in \N}$ does not converge uniformly to $f$ on $D$.
    \end{itemize}
\end{remark}

\begin{example}
    Show the sequence of functions $f_n(z) = e ^ z + \frac{1}{n}$ converges to $e ^ z$ uniformly on $\C$.

    \begin{solution}
        $|f_n(z) - f(z)| = \frac{1}{n} = s_n$,
        $s_n > 0$ $\forall n$,
        $s_n$ is $z$ independent and $s_n \to 0$ as $n \to \infty$ so by previous theorem the convergence is uniform.
    \end{solution}
\end{example}

\begin{example}
    Show that for any $R > 0$ the sequence of functions $f_n(z) = e ^ z + \frac{z}{n}$ converges to $e ^ z$ uniformly on $B_R(0)$.
    Does it converge uniformly on $\C$?

    \begin{solution}
        \[
        |f_n(z) - f(z)| = \left|\frac{z}{n}\right| = \frac{|z|}{n}.
        \]
        If $z \in B_R(0)$
        \[
        |f_n(z) - f(z)| \leq \frac{R}{n} = s_n(R)
        \]
        $s_n(R) > 0$,
        $S_n(R)$ is $z$ independent.
        $S_n(R) \to 0$ as $n \to \infty$.
        By our theorem $f_n \to f$ uniformly on $B_R(0)$.

        Note that $z_n = n$ satisfies
        \[
        |f_n(z_n) - f(z_n)| = \left|\frac{z_n}{n}\right| = 1\quad \forall n
        \]
        $\{z_n\} \subset \C$ so no uniform convergence.
    \end{solution}
\end{example}

\begin{theorem}[Criteria for uniform convergence of a series -
Weierstrass $M$-test]
    Let $\{f_n\}_{n \in \N} : D \to \C$ be a given sequence of functions.
    Assume that there exists a sequence of non-negative real numbers $\{M_n\}_{n \in \N}$ and $n_0 \in \N$ such that for all $n \geq n_0$
    \[
    |f_n(z)| \leq M_n,\quad \forall z \in D,
    \]
    and
    \[
    \sum_{n = n_0}^{\infty}M_n < \infty.
    \]
    Then $S_N(z) = \sum_{n = 1}^{N}f_n(z)$ converges uniformly on $D$ to some limit function $S : D \to \C$ which we denote by
    \[
    S(z) = \sum_{n = 1}^{\infty}f_n(z).
    \]
    In particular,
    if all functions $f_n(z)$ are continuous on $D$ then $S(z)$ is also continuous on $D$.
\end{theorem}

We also have
\begin{theorem}[Criteria for lack of uniform convergence of a series]
    Let $\{f_n\}_{n \in \N} : D \to \C$ be a given sequence of functions.
    If $\{f_n\}_{n \in \N}$ doesn't converge uniformly to $f(z) = 0$ on $D$ then the sequence $S_N(z) = \sum_{n = 1}^{N}f_n(z)$ does not converge uniformly on $D$.

    \begin{proof}
        \textit{Sketch}.
        If $S_n(z)$ converges uniformly to $S(z)$ then $f_n(z) = S_n(z) - S_{n - 1}(z)$ converges uniformly to $S(z) - S(z) = 0$.
    \end{proof}
\end{theorem}

\begin{example}
    Show that
    \[
    \sum_{n = 1}^{\infty}\frac{|2z| ^ {3n}}{3 ^ {2n}n ^ 2}
    \]
    converges uniformly to a continuous function on $\mathbb{D}$.

    \begin{solution}
        $f_n(z) = \frac{(2z) ^ {3n}}{3 3^ {2n}n ^ 2}$ is continuous on $\mathbb{D}$,
        \[
        |f_n(z)| = \frac{2 ^ {3n}|z| ^ {3n}}{3 ^ {2n}n ^ 2} \leq \frac{2 ^ {3n}}{3 ^ {2n}n ^ 2}\quad \forall z \in \mathbb{D}
        \]
        $M_n = \frac{2 ^ {3n}}{3 ^ {2n}n ^ 2} > 0$,
        $M_n$ is $z$ independent.
        \[
        M_n = \left(\frac{8}{9}\right) ^ n\frac{1}{n ^ 2} \leq \left(\frac{8}{9}\right) ^ n\quad \forall n
        \]
        as $\sum_{n = 1}^{\infty}\left(\frac{8}{9}\right) ^ n < \infty$,
        $\sum_{n = 1}^{\infty}M_n < \infty$ by Weierstrass $M$-test the series converges uniformly.
        As $f_n$ was continuous for all $n$ so is the series limit.
    \end{solution}
\end{example}

Uniform convergence is very hard to get.
The main problem with it is that it automatically includes the boundary of $\mathbb{D}$.

Consider $f_n(z) = z ^ n$ on $\mathbb{D}$
$f_n(z) \to 0$ as $n \to \infty$ on $\mathbb{D}$.
$f_n(z) \centernot\to 0$ uniformly.
This is since we can approach $z = 1$ by a bad sequence $\{z_n\}$.
Indeed,
let $z_n = \left(\frac{1}{2}\right) ^ {1 / n}$.
\[
|f_n(z_n) - f(z_n)| = \left(\left(\frac{1}{2}\right) ^ {1 / n}\right) ^ n = \frac{1}{2}\quad \forall n.
\]
By our theorem,
$f_n \centernot\to 0$ uniformly.

\subsection{Locally uniform convergence}

\begin{definition}[Locally uniform convergence]
    Let $D \subseteq \C$ be a given set and let $\{f_n\}_{n \in \N} : D \to \C$ be a sequence of functions.
    We say $\{f_n\}_{n \in \N}$ converges locally uniformly to
    (the limit function)
    $f$ on $D$,
    if for every $z_0 \in D$ there exists an open set $U_{z_0} \subset \C$
    (that can depend on $z_0$)
    which contains $z_0$ such that $\{f_n\}_{n \in \N}$ converges uniformly to $f$ on $U_{z_0} \cap D$.
\end{definition}

\begin{remark}
    We need to deal with the intersection of $U_{z_0}$ with out set $D$ to make sure we take into account all the different type of points in $D$
    (interior and boundary).
\end{remark}

\begin{example}
    Show that the sequence of functions $f_n(z) = z ^ n$ converges locally uniformly to $f(z) = 0$ on $\mathbb{D}$.

    \begin{solution}
        We claim that it is enough to show that $f_n(z)$ converges to $f(z)$ uniformly on $B_r(0)$ where $0 < r < 1$.
        $B_r(0)$ is a family of open sets in $\mathbb{D}$ s.t. any $z_0 \in \mathbb{D}$ belongs to one of these sets.

        Indeed,
        given $z_0 \in \mathbb{D}$,
        $z_0 \in B_{\frac{|z_0| + 1}{2}}(0) \subset \mathbb{D}$.
        On $B_r(0)$
        \[
        |f_n(z) - f(z)| = |z| ^ n \leq r ^ n = s_n
        \]
        $s_n > 0$,
        $s_n$ independent of $z$,
        $s_n \to 0$ as $n \to \infty$.
        So by our theorem $f_n(z)$ converges to $f(z)$ uniformly on $B_r(0)$,
        $0 < r < 1$.
    \end{solution}
\end{example}

\begin{theorem}
    Let $\{f_n\}_{n \in \N} : D \to \C$ be a sequence of continuous functions that converges locally uniformly to a limit function $f$ on $D$.
    Then $f : D \to \C$ is continuous on $D$.

    \begin{proof}
        Let $z_0 \in D$.
        There exists open $U_{z_0}$ s.t. $f_n$ converges to $f$ uniformly on $D \cap U_{z_0}$.
        Since $\{f_n\}$ continuous on $D \cap U_{z_0}$ so is $f$.
        Thus implies that $f$ is continuous at $z_0$.
        $z_0$ was arbitrary so $f$ is continuous on $D$.
    \end{proof}
\end{theorem}

\begin{lemma}[Criteria for local uniform convergence and lack of local uniform convergence for sequences]
    Let $\{f_n\}_{n \in \N} : D \to \C$ be a sequence of functions that converges pointwise to a limit function $f$ on $D$.

    \begin{enumerate}[label = (\roman*)]
        \item If for any $z_0 \in D$ there exist an open set $U_{z_0} \subset D$ which contains $z_0$,
        a non-negative sequence of real numbers $\{s_n(U_{z_0})\}_{n \in \N}$ that converges to zero and $n_0(U_{z_0}) \in \N$ that may depend on $U_{z_0}$ such that for all $n \geq n_0(U_{z_0})$
        \[
        |f_n(z) - f(z)| \leq s_n(U_{z_0}),\qquad \forall z \in U_{z_0} \cap D,
        \]
        then the sequence of functions $\{f_n\}_{n \in \N}$ converges locally uniformly to $f$ on $D$.

        \item If there exists a non-negative sequence of real numbers $\{s_n\}_{n \in \N}$ that converges to $c > 0$,
        $n_0 \in \N$,
        and a sequence $\{z_n\}_{n \in \N} \subset D$ that converges to $z_0 \in D ^ 0$ such that for all $n \geq n_0$
        \[
        |f_n(z_n) - f(z_n)| \geq s_n
        \]
        then the sequence of functions $\{f_n\}_{n \in \N}$ does not converge locally uniformly to $f$ on $D$.
    \end{enumerate}
\end{lemma}

\begin{example}
    Show that the series $\sum_{n = 1}^{\infty}\frac{(z + 1 / z) ^ n}{n!}$ converges locally uniformly to a continuous function on $\C ^ {*}$.

    \begin{solution}
        $f_n(z) = \frac{(z + 1 / z) ^ n}{n!}$ is continuous on $\C ^ {*}$ for any $n \in \N$.
        If we show locally uniform convergence,
        then the limit will be continuous by a previous theorem.
        \[
        |f_n(z)| = \frac{|z + 1 / z| ^ n}{n!} \leq \frac{\left(|z| + \frac{1}{|z|}\right) ^ n}{n!}.
        \]
        Consider the open sets
        \[
        A_{r, R} = \{z \in \C : r < |z| < R\}
        \]
        on $A_{r, R}$
        \[
        |f_n(z)| \leq \frac{(R + 1 / r) ^ n}{n!} = M_n(r, R).
        \]
        $M_n(r, R) > 0$,
        independent in $z$ and
        \[
        \sum_{n = 1}^{\infty}M_n(r, R) \leq \sum_{n = 0}^{\infty}\frac{(R + 1 / r) ^ n}{n!} = e ^ {R + 1 / r} < \infty.
        \]
        By Weierstrass $M$-test $\sum_{n = 1}^{\infty}\frac{(z + 1 / z) ^ n}{n!}$ converges uniformly in $A_{r, R}$ for any $0 < r < R$.

        We will show that any $z_0 \in \C ^ {*}$ belongs to some $A_{r, R}$.
        Indeed $z_0 \in A_{|z_0| / 2}$,
        $2|z_0|$.
    \end{solution}
\end{example}

\subsection{Complex power series}
\begin{definition}
    A
    (complex)
    power series is an expression of the form
    \[
    S(z) \coloneqq \sum_{n = 0}^{\infty}a_n(z - c) ^ n
    \]
    where $\{a_n\}_{n \in \N}$ is a sequence of complex numbers and $c \in \C$.
\end{definition}

\begin{theorem}
    For any sequence of complex numbers $\{a_n\}_{n \in \N}$ there exists $R \in \R_{\geq 0} \cup \{\infty\}$ such that
    \begin{itemize}
        \item If $R = 0$ then $S(z)$ converges only for $z = c$.
        In this case $S(c) = a_0$.

        \item If $R > 0$ then $S(z)$ converges absolutely for all $|z - c| < R$.
        When $R = \infty$ this condition holds for any $z$.

        \item If $0 < R < \infty$ then $S(z)$ diverges for $|z - c| > R$.
    \end{itemize}
    $R$ is called the radius of convergence of the power series and $B_R(c)$ is called the disc of convergence.
\end{theorem}

Radius of convergence can be found using
\[
R = \frac{1}{\limsup_{n \to \infty}\sqrt[n]{|a_n|}}.
\]
The following holds when the limit exist:
\[
R = \lim_{n \to \infty}\frac{|a_n|}{|a_{n + 1}|}.
\]

\begin{theorem}
    A power series $\sum_{n = 0}^{\infty}a_n(z - c) ^ n$ with radius of convergence $0 < R \leq \infty$ converges uniformly on any ball $B_r(c)$ with $0 < r < R$.
    This implies that the power series is locally uniformly convergent on its disk of convergence.
\end{theorem}

\begin{example}
    Show that the power series $\sum_{n = 0}^{\infty}\frac{z ^ n}{n!}$ converges locally uniformly on $\C$.

    \begin{solution}
        Using the above theorem it is enough to show $R = \infty$.
        $a_n = \frac{1}{n!}$,
        \[
        \left|\frac{a_n}{a_{n + 1}}\right| = n + 1 \to \infty\text{ as } n \to \infty
        \]
        so $R = \infty$.
    \end{solution}
\end{example}

\section{Complex Integration}

\subsection{Definition of contour integrals}
Now that we measured complex differentiability we ask ourselves if we can define a notation of integration that would lead to something like the FTC.
An immediate issue is that the FTC speaks about $\int_{a}^{x}f(t)\,dt$ which connects $a$ and $x$ via the interval $[a, x]$ which is the only option on $\R$.
On $\C$,
however there are infinitely many ways to connect $z_0$ and $z$.

\begin{definition}
    Consider a continuous function $f : [a, b] \to \C$ where $[a, b] \subset \R$.
    We define
    \[
    \int_{a}^{b}f(t)\,dt = \int_{a}^{b}(\Re(f(t)) + i\Im(f(t)))\,dt \coloneqq \int_{a}^{b}\Re(f(t))\,dt + i\int_{a}^{b}\Im(f(t))\,dt.
    \]
\end{definition}
By definition
\begin{align*}
    \Re\left(\int_{a}^{b}f(t)\,dt\right) &= \int_{a}^{b}\Re(f(t))\,dt \\
    \Im\left(\int_{a}^{b}f(t)\,dt\right) &= \int_{a}^{b}\Im(f(t))\,dt.
\end{align*}

\begin{example}
    Find $\int_{0}^{1}f(t)\,dt$ where $f(t) = t + it$.

    \begin{solution}
        By definition
        \begin{align*}
            \int_{0}^{1}f(t)\,dt &= \int_{0}^{1}(t + it)\,dt \\
            &= \int_{0}^{1}t\,dt + i\int_{0}^{1}t\,dt \\
            &= \left[\frac{t ^ 2}{2} + i\frac{t ^ 2}{2}\right]_{0}^{1} \\
            &= \frac{1}{2}(1 + i).
        \end{align*}
    \end{solution}
\end{example}

\begin{lemma}\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item Let $f_1$ and $f_2$ be continuous functions from $[a, b]$ to $\C$.
        Then
        \[
        \int_{a}^{b}(f_1(t) + f_2(t))\,dt = \int_{a}^{b}f_1(t)\,dt + \int_{a}^{b}f_2(t)\,dt.
        \]
    
        \item For any complex number $c \in \C$,
        and continuous function $f : [a, b] \to \C$
        \[
        \int_{a}^{b}cf(t)\,dt = c\int_{a}^{b}f(t)\,dt.
        \]
    \end{enumerate}

    \begin{proof}
        For part
        (ii)
        $c = \alpha + i\beta$,
        $\alpha, \beta \in \R$
        \[
        f(t) = u(t) + iv(t).
        \]
        \[
        cf(t) = (\alpha u(t) - \beta v(t)) + i(\alpha v(t) + \beta u(t)).
        \]
        By definition
        \begin{align*}
            \int_{a}^{b}cf(t)\,dt &= \int_{a}^{b}(\alpha u(t) - \beta v(t))\,dt + i\int_{a}^{b}(\alpha v(t) + \beta u(t))\,dt \\
            &= \alpha\int_{a}^{b}u(t)\,dt - \beta\int_{a}^{b}v(t)\,dt + i\left(\alpha\int_{a}^{b}v(t)\,dt + \beta\int_{a}^{b}u(t)\,dt\right) \\
            &= (\alpha + i\beta)\int_{a}^{b}(u(t) + iv(t))\,dt \\
            &= c\int_{a}^{b}f(t)\,dt.
        \end{align*}
    \end{proof}
\end{lemma}

We want to extend this to integrations over curves.

\begin{example}
    The path $\gamma : [0, 2\pi] \to \C$ given by $\gamma(\theta) = a + re ^ {i\theta}$ where $a \in \C$ and $r > 0$ are given describes a circle of radius $r$ centred at $a$.
    As $\theta$ increases we move in an anticlockwise direction.
    What is $\gamma'(\theta)$?
    \[
    \gamma(\theta) = (a_1 + ia_2) + (r\cos{\theta} + ir\sin{\theta}) = (a_1 + r\cos{\theta}) + i(a_2 + r\sin{\theta})
    \]
    \[
    \gamma'(\theta) = -r\sin{\theta} + ir\cos{\theta} = ir(\cos{\theta} + i\sin{\theta}) = ire ^ {i\theta}.
    \]

    $\gamma$ is $C ^ 1$.
\end{example}

\begin{example}
    The curve $\gamma : [0, 2\pi] \to \C$ given by $\gamma(\theta) = a + re ^ {-i\theta}$ where $a \in \C$ and $r > 0$ are given plots the same image but in a clockwise manner.

    $\gamma'(\theta) = -ire ^ {-i\theta}$.
    $\gamma$ is $C ^ 1$.
\end{example}

\begin{example}
    $\gamma : [0, 2] \to \C$ defined by
    \[
    \gamma(t) = \begin{cases}
        t + it & 0 \leq t \leq 1 \\
        t + i & 1 \leq t \leq 2.
    \end{cases}
    \]
    $\gamma$ is piecewise $C ^ 1$,
    \[
    \gamma'(t) = \begin{cases}
        1 + i & 0 < t < 1 \\
        1 & 1 < t < 2.
    \end{cases}
    \]
\end{example}

\begin{definition}
    Let $U \subseteq \C$ be an open set and let $f : U \to \C$ be a continuous function.
    Let $\gamma : [a, b] \to U$ be a $C ^ 1$-curve.
    Then we define the integral of $f$ along the curve $\gamma$ by
    \[
    \int_{\gamma}f(z)\,dz \coloneqq \int_{a}^{b}f(\gamma(t))\gamma'(t)\,dt.
    \]
    If $\gamma : [a. b] \to U$ is a contour with partition $a = a_0 < a_1 < a_2 < \dotsi < a_{n - 1} < a_n = b$ such that $\gamma_{[a_{i - 1}, a_i]} \to \C$ are $C ^ 1$ curves for $i = 1, \dotsc, n$ then we define
    \[
    \int_{\gamma}f(z)\,dz = \sum_{i = 1}^{n}\int_{\gamma_i}f(z)\,dz.
    \]
    Such integral is sometimes known as contour integral.
\end{definition}

\textit{Note:}
$f\circ \gamma$ and $\gamma'$ or $\gamma_i'$ are all continuous so the above is well defined.

\begin{lemma}[Basic properties]
    Assuming that $f_1, f_2$ and $f$ are continuous on $U$ we find that for any $C ^ 1$ curve $\gamma : [a, b] \to U$:

    \begin{enumerate}[label = (\roman*)]
        \item
        \[
        \int_{\gamma}(f_1(z) + f_2(z))\,dz = \int_{\gamma}f_1(z)\,dz + \int_{\gamma}f_2(z)\,dz.
        \]

        \item For any $c \in \C$
        \[
        \int_{\gamma}cf(z)\,dz = c\int_{\gamma}f(z)\,dz.
        \]

        \item Defining $(-\gamma) : [-b, -a] \to U$ by $(-\gamma)(t) = \gamma(-t)$ we have that
        \[
        \int_{\gamma}f(z)\,dz = -\int_{-\gamma}f(z)\,dz.
        \]
    \end{enumerate}
\end{lemma}

\begin{example}
    Consider the path $\gamma : [0, 2\pi] \to \C$ defined by $\gamma(\theta) = re ^ {i\theta}$ with $r > 0$.
    Find $\int_{\gamma}\,dz$ and $\int_{\gamma}\bar{z}\,dz$.

    \begin{solution}
        $\gamma'(\theta) = ire ^ {i\theta}$.
        \[
        \int_{\gamma}\,dz = \int_{\gamma}1\,dz = \int_{0}^{2\pi}1\cdot\gamma'(\theta)\,d\theta = \int_{0}^{2\pi}ire ^ {i\theta}\,d\theta = ir\left(\int_{0}^{2\pi}\cos{\theta}\,d\theta + i\int_{0}^{2\pi}\sin{\theta}\,d\theta\right) = 0.
        \]
        \[
        \int_{\gamma}\bar{z}\,dz = \int_{0}^{2\pi}\overline{\gamma(\theta)} \cdot \gamma'(\theta)\,d\theta = \int_{0}^{2\pi}re ^ {-i\theta}\cdot ire ^ {i\theta}\,d\theta = ir ^ 2\int_{0}^{2\pi}1\,d\theta = 2\pi ir ^ 2
        \]
    \end{solution}
\end{example}

\begin{example}
    Consider the path $\gamma : [0, 2\pi] \to \C$ defined by $\gamma(\theta) = a + re ^ {i\theta}$ with $a \in \C$ and $r > 0$.
    Find $\int_{\gamma}(z - a) ^ n\,dz$ for $n \in \Z$.

    \begin{solution}
        \begin{align*}
            \int_{\gamma}(z - a) ^ n\,dz &= \int_{0}^{2\pi}(\gamma(\theta) - a) ^ n\gamma'(\theta)\,d\theta \\
            &= \int_{0}^{2\pi}(re ^ {i\theta}) ^ n\cdot ire ^ {i\theta}\,d\theta \\
            &= \int_{0}^{2\pi}r ^ ne ^ {in\theta}\cdot ire ^ {i\theta}\,d\theta \\
            &= ir ^ {n + 1}\int_{0}^{3\pi}e ^ {i(n + 1)\theta}\,d\theta \\
            &= ir ^ {n + 1}\left(\int_{0}^{2\pi}(\cos((n + 1)\theta) + i\int_{0}^{2\pi}\sin((kn+ 1)\theta)\,d\theta)\right) \\
            &= \begin{cases}
                ir ^ 0\left(\int_{0}^{2\pi}\cos(0)\,d\theta + i\int_{0}^{2\pi}\sin(0)\,d\theta\right) & n + 1 = 0 \\
                ir ^ {n + 1}\left[\frac{\sin((n + 1)\theta)}{n + 1}\right]_{0}^{2\pi} - i\left[\frac{\cos((n + 1)\theta)}{n + 1}\right]_{0}^{2\pi} & n + 1 \neq 0
            \end{cases} \\
            &= \begin{cases}
                2\pi i & n = -1 \\
                0 & n \in \Z \setminus \{-1\}.
            \end{cases}
        \end{align*}
    \end{solution}
\end{example}

It appears that integration over $\gamma$ depends on the parameter.
However,
consider
\begin{align*}
    \gamma : &[0, 2\pi] \to \C & \delta : &[0, 1] \to \C \\
    \gamma(\theta) &= e ^ {i\theta} & \delta(t) &= e ^ {2\pi it}
\end{align*}
Seem to both give the unit circle almost bijectively
(besides $\theta = 2\pi$,
$t = 1$).
They
"feel"
similar.
Will they give the same integral?

\begin{lemma}[Reparameterisation of curves]
    Let $U \subset \C$ be an open set,
    $f : U \to \C$ be continuous,
    and let $\gamma : [a, b] \to U$ be a $C ^ 1$ curve.
    If $\varphi : [a', b'] \to [a, b]$ is continuously differentiable bijection with $\varphi(a') = a$ and $\varphi(b') = b$ and we define $\delta : [a', b'] \to \C$ by
    \[
    \delta(t) \coloneqq \gamma(\varphi(t)) = (\gamma \circ \varphi)(t)
    \]
    then
    \[
    \int_{\gamma}f(z)\,dz = \int_{\delta}f(z)\,dz.
    \]

    \begin{proof}
        \begin{align*}
            \int_{\gamma}f(z)\,dz &= \int_{a'}^{b'}f(\delta(t))\delta'(t)\,dt \\
            &= \int_{a'}^{b'}f(\delta(\varphi(t))\delta'(\varphi(t))\varphi'(t)\,dt \\
            &= \int_{\varphi(a')}^{\varphi(b')}f(\gamma(s))\gamma'(s)\,ds \\
            &= \int_{a}^{b}f(\gamma(s))\gamma'(s)\,ds \\
            &= \int_{\delta}f(z)\,dz.
        \end{align*}
    \end{proof}
\end{lemma}

\begin{example}
    $\gamma'(\theta) = e ^ {i\theta}$,
    $\theta \in [0, 2\pi]$,
    $\delta(t) = e ^ {2\pi it}$,
    $t \in [0, 1]$.

    $\delta(t) = \gamma(\varphi(t))$,
    $\varphi(t) = 2\pi t$.
\end{example}

Given a domain $D$ such that there exists a bijective contour $\gamma : [a, b] \to \partial D$ with a continuous inverse $\gamma ^ {-1} : \partial D \to [a, b]$ and such that $\gamma'(t) \neq 0$,
we define
\[
\int_{\partial D}f(z)\,dz = \int_{\gamma}f(z)\,dz
\]
where we assume we start at $\gamma(a)$ and end at $\gamma(b)$.
Notion well defined and doesn't depend on $\gamma$ due to reparametrisation of curves lemma.
When boundary has no end points,
such as a circle,
can extend above definition to case where $\gamma : [a, b) \to \partial D$ with previous properties $\gamma(a) = \gamma(b)$.
E.g.
when $D = B_r(a)$ with $a \in \C$ and $r > 0$ we have that
\[
\partial D = \{z \in \C\,:\,|z - a| = r\}
\]
bijective image of the $C ^ 1$ curve $\gamma : [0, 2\pi) \to \C$
\[
\gamma_{a, r}(\theta) = a + re ^ {i\theta}
\]
going in anticlockwise direction.
Consequently
\[
\int_{|z - a| = r}f(z)\,dz = \int_{\gamma_{a, r}}f(z)\,dz = \int_{0}^{2\pi}f(a + re ^ {i\theta})ire ^ {i\theta}\,d\theta.
\]
The last element of definition for integration is:

\begin{definition}
    Let $\gamma : [a, b] \to \C$ and $\delta : [c, d] \to \C$ be two contours such that $\gamma(b) = \delta(c)$.
    The addition of these contours,
    denoted by $\gamma \cup \delta$,
    is the contour defined by $\gamma \cup \delta : [a, b + d - c] \to \C$
    \[
    \gamma \cup \delta(t) \coloneqq \begin{cases}
        \gamma(t), &a \leq t \leq b, \\
        \delta(t + c - b), & b \leq t \leq b + d - c.
    \end{cases}
    \]
\end{definition}

\begin{lemma}
    Let $f : U \to \C$ be a continuous function and let $\gamma : [a, b] \to U$ and $\delta : [c, d] \to U$ be two contours such that $\gamma(b) = \delta(c)$.
    Then
    \[
    \int_{\gamma \cup \delta}f(z)\,dz = \int_{\gamma}f(z)\,dz + \int_{\delta}f(z)\,dz.
    \]
\end{lemma}

\subsection{The Fundamental Theorem of Calculus}
FTC on $\R$ has two parts:
\begin{itemize}
    \item If $f$ is continuous on $[a, b]$ then
    \[
    F(x) = \int_{a}^{x}f(t)\,dt
    \]
    is differentiable on $(a, b)$ and $F' = f$.

    \item If $f$ is continuous on $[a, b]$ and $F$ is the anti-derivative of $f$ then
    \[
    \int_{a}^{b}f(t)\,dt = F(b) - F(a).
    \]
    This is sometimes known as the Newton-Leibniz formula.
\end{itemize}

We want to show these in complex setting.

\begin{theorem}[Newton-Leibniz theorem]
    Let $U \subset \C$ be an open set and let $F : U \to \C$ be holomorphic function with continuous derivative $f$.
    Then for any contour $\gamma : [a, b] \to U$ we have
    \[
    \int_{\gamma}f(z)\,dz = F(\gamma(b)) - F(\gamma(a)).
    \]
    In particular if $\gamma$ is closed,
    that is $\gamma(a) = \gamma(b)$,
    then we have that
    \[
    \int_{\gamma}f(z)\,dz = 0.
    \]

    \begin{proof}
        Assume that $\gamma$ is $C ^ 1$.
        \begin{align*}
            \int_{\gamma}f(z)\,dz &= \int_{a}^{b}f(\gamma(t))\cdot\gamma'(t)\,dt \\
            &= \int_{a}^{b}F'(\gamma(t))\cdot\gamma'(t)\,dt \\
            &= \int_{a}^{b}\frac{d}{dt}F(\gamma(t))\,dt \\
            &= \int_{a}^{b}\Re(F(\gamma(t))'\,dt + i\int_{a}^{b}\Im(F(\gamma(t))'\,dt \\
            &= \Re(F(\gamma(t))\big|_{a}^{b} + i\Im(F(\gamma(t))\big|_{a}^{b} \\
            &= F(\gamma(b)) - F(\gamma(a)).
        \end{align*}

        Assume $\gamma$ is a contour with partition $a = a_0 < a_1 < \dotsi < a_n = b$.
        By definition
        \[
        \int_{\gamma}f(z)\,dz = \sum_{i = 1}^{n}\int_{\gamma|_{[a_{i - 1}, a_i]}}f(z)\,dz = \sum_{i = 1}^{n}(F(\gamma(a_i)) - F(\gamma(a_{i - 1}))) = F(\gamma(a_n)) - F(\gamma(a_0)) = F(\gamma(b)) - F(\gamma(a)).
        \]
    \end{proof}
\end{theorem}

\begin{remark}\phantom{}
    \begin{itemize}
        \item We will show that if $F$ is holomorphic it is infinitely many times differentiable and as such $f = F'$ is continuous.
        The continuity of $f$ is,
        in the end,
        redundant.

        \item If $\gamma_1, \gamma_2$ are contours such that $\gamma_1(a) = \gamma_2(a')$,
        $\gamma_1(b) = \gamma_2(b')$ then if $f$ is continuous and $f = F'$ where $F$ is holomorphic then
        \[
        0 = \int_{\gamma_1 \cup (-\gamma_2)}f(z)\,dz  = \int_{\gamma_1}f(z)\,dz + \int_{-\gamma_2}f(z)\,dz = \int_{\gamma_1}f(z)\,dz - \int_{\gamma_2}f(z)\,dz
        \]
        i.e.
        \[
        \int_{\gamma_1}f(z)\,dz = \int_{\gamma_2}f(z)\,dz.
        \]
    \end{itemize}
\end{remark}

\[
\int_{|z - a| = r}(z - a) ^ n\,dz = \int_{\gamma}(z - a) ^ n\,dz
\]
$\gamma(\theta) = a + re ^ {i\theta}$,
$\theta \in [0, 2\pi]$,
$n \in \Z$.
$(z - a) ^ n$ is holomorphic in $\C \setminus \{a\}$ for all $n \in \Z$.
Moreover,
\[
F(z) = \frac{(z - a) ^ {n + 1}}{n + 1},\qquad n \neq -1
\]
is a holomorphic function in $\C \setminus \{a\}$ and $F' = f$ which implies by Newton-Leibniz as $\{|z - a| = r\} \subset \C \setminus\{a\}$
\[
\int_{|z - a| = r}(z - a) ^ n\,dz = 0.
\]
Note that for $n = -1$ the anti-derivative for $\frac{1}{z - a}$ will be some form of logarithm.
But there exists no holomorphic logarithm in $\C \setminus \{a\}$.
We cannot use the Newton-Leibniz theorem.
