\section{The Complex Plane and Complex Functions}

\subsection{Complex Numbers}

Complex numbers are of the form $z = x + iy$ where $i$ is the imaginary unit,
$i = \sqrt{-1}$.
The set of complex numbers is denoted $\C$.
We use an Argand diagram to visualise complex numbers $x + iy$ with $(x, y)$.

For $z = x + iy$ we have the real part of $z$,
$\mathrm{Re}(z) = x$,
and the imaginary part of $z$,
$\Im(z) = x$.

Addition of complex numbers,
$z_1 = x_1 + iy_1$,
$z_2 = x_2 + iy_2$
\[
z_1 \pm z_2 = (x_1 \pm x_2) + i(y_1 \pm y_2)
\]
the real parts are added together and the imaginary parts are added together.

Multiplication of complex numbers,
$z_1 = x_1 + iy_1$,
$z_2 = x_2 + iy_2$
\[
z_1z_2 = (x_1 + iy_1)(x_2 + iy_2) = (x_1x_2 - y_1y_2) + i(x_1y_2 + x_2y_1)
\]
just expand and use $i ^ 2 = -1$.

Looking at
\[
(x + iy)(x - iy) = x ^ 2 + ixy - ixy - i ^ 2y ^ 2 = x ^ 2 + y ^ 2 \in \R.
\]
We call the conjugate of a complex number as follows,
given $z = x + iy$ the conjugate is
\[
\bar{z} = x - iy = \mathrm{Re}(z) - i\mathrm{Im}(z).
\]
We can see that
\[
z\bar{z} = \mathrm{Re}(z) ^ 2 + \Im(z) ^ 2 \in \R
\]
which leads us to
\[
|z| = \sqrt{z\bar{z}} = \sqrt{\mathrm{Re}(z) ^ 2 + \Im(z) ^ 2}
\]
the modulus
(or absolute value)
of $z$.

Using the conjugate we can see
\[
\frac{z_1}{z_2} = \frac{x_1 + iy_1}{x_2 + iy_2} = \frac{(x_1 + iy_1)(x_2 - iy_2)}{(x_2 + iy_2)(x_2 - iy_2)} = \frac{x_1x_2 + y_1y_2}{x_2 ^ 2 + y_2 ^ 2} + i\frac{x_2y_1 - x_1y_2}{x_2 ^ 2 + y_2 ^ 2}.
\]

\begin{example}
    Find $\frac{3}{1 + i} - (3 + 2i)$.

    \begin{solution}
        \[
        \frac{3}{1 + i} - (3 + 2i) = \frac{3(1 - i)}{(1 + i)(1 - i)} - (3 + 2i) = \frac{3 - 3i}{2} - 3 - 2i = -\frac{3}{2} - \frac{7}{2}i.
        \]
    \end{solution}
\end{example}

Conjugation is nothing but reflection with respect to the real axis.

\begin{lemma}[Important Properties of Complex Numbers]
    \begin{enumerate}[label = (\roman*)]
        \item $z_1z_2 = 0 \iff z_1 = 0\text{ or } z_2 = 0$.

        \item $|z| = \sqrt{z\bar{z}}$.

        \item $\overline{(zw)} = \bar{z}\bar{w}$.

        \item $\Re(z) = \frac{z + \bar{z}}{2}$ and $\Im(z) = \frac{z - \bar{z}}{2i}$.

        \item $z ^ {-1} = \frac{\bar{z}}{|z| ^ 2}$.
    \end{enumerate}
\end{lemma}

Using polar coordinates and the Argand diagram we can write complex numbers in polar coordinates.
We define for any $z \in \C$
\[
r \coloneqq \sqrt{x ^ 2 + y ^ 2} = \sqrt{\Re(z) ^ 2+ \Im(z) ^ 2} = |z|
\]
and any $z \neq 0$
\[
\theta \coloneqq \begin{cases}
    \arctan(y / x) & x > 0, \\
    \pi / 2 & x = 0, y > 0, \\
    -\pi / 2 & x = 0, y < 0, \\
    \arctan(y / x) + \pi &x < 0, y > 0, \\
    \arctan(y / x) - \pi &x < 0, y < 0.
\end{cases}
\]
$\theta$ is the anticlockwise angle measured from the real axis,
$\theta$ is called the argument of $z$,
denoted $\arg(z)$.

We have the following polar coordinates for $z$:
\[
z = x + iy = r\cos(\theta) + ir\sin(\theta) = r(\cos(\theta) + i\sin(\theta)).
\]

\begin{example}
    $z = i$.

    \[
    r = |z| = \sqrt{0 ^ 2 + 1 ^ 2} = 1
    \]
    \[
    \theta = \pi / 2.
    \]
    \[
    i = 1(\cos(\pi / 2) + i\sin(\pi / 2)) = \cos(\pi / 2) + i\sin(\pi / 2).
    \]

    $z = 1 + i$
    \[
    |z| = \sqrt{1 ^ 2 + 1 ^ 2} = \sqrt{2},
    \]
    $\theta = \frac{\pi}{4}$,
    \[
    z = \sqrt{2}(\cos(\pi / 4) + i\sin(\pi / 4)).
    \]
\end{example}

\textbf{Important:}
$\arg(z)$ is only defined up to multiples of $2\pi$.

$\arg(i) = \frac{\pi}{2} + 2\pi k$,
$k \in \Z$,
not a function.

To define an argument we need a range of angles.

The principal value of $\arg(z)$ is the value in the interval $(-\pi, \pi]$ denoted $\Arg(z)$.

For example:
$\Arg(i) = \frac{\pi}{2}$,
$\Arg(-1) = \pi$.

\begin{lemma}[Properties of argument]
    We have the following properties of the argument:
    
    \begin{enumerate}[label = (\roman*)]
        \item $\arg(z_1z_2) = \arg(z_1) + \arg(z_2)\mod{2\pi}$.

        \item $\arg(1 / z) = -\arg(z)\mod{2\pi}$.

        \item $\arg(\bar{z}) = -\arg(z)\mod{2\pi}$.
    \end{enumerate}
\end{lemma}
\textit{Note:
the proof relies on why we went to polar coordinates:
multiplication is clear.}

\begin{lemma}
    Geometrically,
    multiplication in $\C$ is given by a dilated rotation;
    i.e.,
    if $z_1 = r_1(\cos(\theta_1) + i\sin(\theta_1))$ and $z_2 = r_2(\cos(\theta_2) + i\sin(\theta_2))$ then
    \[
    z_1z_2 = r_1r_2(\cos(\theta_1 + \theta_2) + i\sin(\theta_1 + \theta_2)).
    \]
    In particular,
    multiplying by $z_2$ constitutes an
    (anticlockwise)
    rotation of $z_1$ by $\theta_2$ degrees,
    followed by a dilation with factor $r_2$.

    \begin{proof}
        \begin{align*}
            z_1z_2 &= r_1(cos(\theta_1) + i\sin(\theta_2))r_2(\cos(\theta_2) + i\sin(\theta_2)) \\
            &= (r_1\cos(\theta_1) + ir_1\sin(\theta_1))(r_2\cos(\theta_2) + ir_2\sin(\theta_2)) \\
            &= (r_1r_2\cos(\theta_1)\cos(\theta_2) - r_1r_2\sin(\theta_1)\sin(\theta_2)) \\
            &+ i(r_1r_2\sin(\theta_1)\cos(\theta_2) + r_1r_2\cos(\theta_1)\sin(\theta_2)) \\
            &= r_1r_2[(\cos(\theta_1)\cos(\theta_2) - \sin(\theta_1)\sin(\theta_2)) + i(\sin(\theta_1)\cos(\theta_2) + \cos(\theta_1)\sin(\theta_2))] \\
            &= r_1r_2(\cos(\theta_1 + \theta_2) + i\sin(\theta_1 + \theta_2)).
        \end{align*}
    \end{proof}
\end{lemma}
\textit{Note:
this proves $\arg(z_1z_2) = \arg(z_1) + \arg(z_2)\mod{2\pi}$,
but also:
$\frac{1}{z} = \frac{\bar{z}}{|z| ^ 2}$ so $\arg(1 / z) = \arg(\bar{z})\mod{2\pi}$,
or $0 = \arg(z \cdot 1 / z) = \arg(z) + \arg(1 / z)\mod{2\pi}$.}

Useful notation:
write $\cos(\theta) + i\sin(\theta) \coloneqq e ^ {i\theta}$.

Motivated by
\[
e ^ {i\theta_1}e ^ {i\theta_2} = (\cos(\theta_1) + i\sin(\theta_1))(\cos(\theta_2) + i\sin(\theta_2)) = \cos(\theta_1 + \theta_2) + i\sin(\theta_1 + \theta_2) = e ^ {i(\theta_1 + \theta_2)}
\]

\begin{corollary}\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item $|z_1z_2| = |z_1||z_2|$.
        
        \item De Moivre's formula:
        $(\cos(\theta) + i\sin(\theta)) ^ n = \cos(n\theta) + i\sin(n\theta)$.
    \end{enumerate}

    \begin{proof}
        \begin{enumerate}[label = (\roman*)]
            \item Exercise.

            \item Done with induction.
        \end{enumerate}
    \end{proof}
\end{corollary}

The modulus has the following properties.
\begin{enumerate}[label = (\roman*)]
    \item
    (Triangle inequality)
    $|z_1 + z_2| \leq |z_1| + |z_2|$.

    \item $|z| \geq 0$ and $|z| = 0 \iff z = 0$.

    \item $\max(|\Re(z)|, |\Im(z)|) \leq |z| \leq |\Re(z)| + |\Im(z)|$.
\end{enumerate}

We can use $\arg(z)$,
$|z|$ to describe geometric sets.
$\mathbb{D} \coloneqq \{z \in \C\,:\, |z| < 1\}$ -
Unit disc.

$\mathbb{H} = \{z \in \C\,:\, \Im(z) > 0\}$ -
Upper half plane
(without the $x$ axis).

$\mathbb{H}_{R} \coloneqq \{z \in \C,\:\, \Re(z) > 0\}$ -
Right half plane
(without the $y$ axis).

$\mathbb{H}_{L} \coloneqq \{z \in \C,\:\, \Re(z) < 0\}$ -
Left half plane
(without the $y$ axis)

$\arg(z) = \frac{\pi}{4}$ -
Ray of angle $\pi / 4$.

$\{z \in \C\,:\, |z - i| = 4\}$ -
Circle centred at $i$ with radius $4$.

\begin{example}
    What is the set $|z - i| = |z + i|$?

    \begin{solution}
        The real line.

        $z = x + iy$
        \begin{align*}
            |x + i(y - 1)| = |x + i(y + 1)| &\iff |x + i(y - 1)| ^ 2 = |x + i(y + 1)| ^ 2 \\
            &\iff x ^ 2 + (y - 1) ^ 2 = x ^ 2 + (y + 1) ^ 2 \\
            &\iff 4y = 0 \implies y = 0.
        \end{align*}

        With $z$:
        one can show
        \[
        |z - \omega| ^ 2 = |z| ^ 2 - 2\Re(z\bar{\omega}) + |\omega| ^ 2.
        \]
    \end{solution}
\end{example}

\subsection{Complex Functions}

Complex functions are $f : \C \to \C$.
To visualise the graph of complex functions we need four dimensions.
We can always write
\[
f(z) = \Re(f(z)) + i\Im(f(z)).
\]
This is not enough to visualise it.

To understand complex functions we usually try to see how they take sets from $\C$ to sets in $\C$.
We will explore
"elementary"
functions like polynomials,
exponentials,
trigonometric functions,
logarithms and powers.

\subsection{The function \texorpdfstring{$f(z) = z ^ n$}{} with \texorpdfstring{$n \in \N$}{}}
To understand $z ^ n$ we will use polar representation:
if $z = re ^ {i\theta}$.
By our lemma,
\[
z ^ n = r ^ ne ^ {in\theta}
\]
if we fix $r$,
i.e. we are on a circle $z ^ n$ dilates our radius by power of $n$
($r \mapsto r ^ n$) and multiplies our argument by a factor of $n$
($\theta \to n\theta\mod{2\pi}$).

A segment of angular length $\eta$ on a circle of radius $r$ becomes an angular segment of length $n\eta\mod{2\pi}$ if $n\eta < 2\pi$ and the full circle if $n\eta \geq 2\pi$ with radius $r ^ n$.

If $n\eta > 2\pi$,
$z ^ n$ will not be injective.

We can slice the world to wedges of size $2\pi / n$ on which $z ^ n$ is injective.
For instance
\[
S_{k}^{(n)} = \{z \in \C\,:\,-\pi < 2\pi k / n < \Arg(z) \leq -\pi + 2\pi(k + 1) / n\}
\]
where $k = 0, \dotsc, n - 1$.
Or
\[
R_{k}^{(n)} = \{z \in \C\,:\, 2\pi k / n\mod{2\pi} < \Arg(z) \leq 2\pi(k + 1) / n\mod{2\pi}\}
\]
$k = 0, \dotsc, n - 1$.

\textit{Note:
when $n$ is even $R ^ {(n)}$s and $S ^ {(n)}$s will overlap
(not true if $n$ is odd).}

We now want to fix $\theta$.
Geometrically that a fixed $\theta$ is a ray
(half line)
without the origin.

The ray of angle $\theta$ is the set
\[
\{z \in \C \setminus \{0\}\,:\,\Arg(z) = \theta\}
\]
The closed ray of angle $\theta$ is the way of angle $\theta$ together with the origin $z = 0$,
i.e.
\[
\{z \in \C \setminus \{0\}\,:\,\Arg(z) = \theta \cup \{0\}\}.
\]

If $\theta$ is fixed $z ^ n = r ^ ne ^ {in\theta}$ which takes a segment on the ray $R_{\theta}$
($0 \leq a < r < b$)
will go to a segment on the ray $R_{n\theta\bmod{2\pi}}$
($a ^ n < r ^ n < b ^ n$).

\begin{lemma}
    \begin{itemize}
        \item The map $z ^ n$ injectively takes an angular segment of length $\frac{2\pi}{n}$ which is open at one end and closed at the other from a circle of radius $r$ to the entire circle of radius $r ^ n$.
        If the above segment is closed,
        or its size is larger than $\frac{2\pi}{n}$ the image is no longer injective.

        \item The map $z ^ n$ injectively takes a ray of angle $\theta$ to a ray of angle $n\theta\bmod{2\pi}$.
        Consequently,
        if $n|\theta_1 - \theta_2| \leq 2\pi$ the map $z ^ n$ injectively takes the wedge
        \[
        W = \{z \in \C\,:\, \theta_1 < \Arg(z) < \theta_2\}
        \]
        to the wedge
        \[
        W = \{z \in \C\,:\, n\theta_1\bmod{2\pi} < \Arg(z) <n\theta_2\bmod{2\pi}\}
        \]
        with the convention that for any $-\pi < \nu < \xi \leq \pi$
        \[
       \left\{z\in\mathbb{C}\;:\; \xi<\mathrm{Arg}(z)<\upsilon\right\} = \left\{ z\in\mathbb{C}\;:\; \xi<\mathrm{Arg}(z)\leq \pi\right\} \cup \left\{ z\in\mathbb{C}\;:\; -\pi<\mathrm{Arg}(z)<\upsilon\right\}. 
        \]
        If we replace one strict inequality
        (i.e. $<$)
        with an inequality
        (i.e. $\leq$)
        the above claim remains true.
        In this case,
        if $n|\theta_1 - \theta_2| = 2\pi$ the image of the map will be $\C \setminus \{0\}$.
        We can get the entire complex plane by adding $z = 0$ to $W$.
        When $n|\theta_1 - \theta_2| > 2\pi$ the image of $W$ by $z ^ n$ will be $\C \setminus \{0\}$ but not injectively.

        \item We can define $n$ different $n$th roots which are the inverses of the map $z ^ n$.
        We can write them all in the form
        \[
        z^{\frac{1}{n}} = \left\lvert z\right\rvert^{\frac{1}{n}}e^{i\left( \frac{\mathrm{Arg}\left( z\right)}{n} + \frac{2\pi k }{n}\right)}
        \]
        with $k = 0, \dotsc, n - 1$
        (we remove the $k$ index from the map,
        but one need to specify which $k$ we are talking about).
        note that for a fixed $k$ the $n$th root of $z$ takes $\C$ into a anti-clockwise rotation by $\frac{\pi}{n}$ of $R_{k - 1}^{(n)}$.
    \end{itemize}
\end{lemma}

Notation:
we want
\[
e ^ z = e ^ {x + iy} = e ^ x\cdot e ^ {iy}
\]
defined $e ^ {iy}$ earlier.

We define the following:
\begin{definition}[Complex exponential]
    We define the complex exponential function $\exp : \C \to \C$ by
    \[
    \exp(z) \coloneqq e ^ x(\cos{y} + i\sin{y}).\qquad (z = x + iy)
    \]
    Shorthand $\exp(z) = e ^ z$.
\end{definition}

One can show
(later)
that with this definition
\[
e ^ z = \sum_{n = 0}^{\infty}\frac{z ^ n}{n!}.
\]

\begin{proposition}
    We have the following complex exponential function properties:
    
    \begin{enumerate}[label = (\roman*)]
        \item $e ^ z \neq 0$ for all $z \in \C$.

        \item $e ^ {z_1 + z_2} = e ^ {z_1}e ^ {z_2}$.

        \item $e ^ z = 1$ if and only if $z = 2\pi ik$ for some $k \in \Z$.

        \item $e ^ {-z} = 1 / e ^ z$.

        \item $|e ^ z| = e ^ {\Re(z)}$.
    \end{enumerate}

    \begin{proof}
        By definition
        \[
        e ^ z = e ^ {x}e ^ {iy} = e ^ x(\cos{y} + i\sin{y})
        \]
        recall that any $\omega \in \C$ can be written uniquely 
        ($\bmod{2\pi}$)
        as $w = \underbrace{r}_{|\omega|}e ^ {i\overbrace{\theta}^{\Arg(w)}}$ so
        \[
        |e ^ z| = e ^ x\qquad \Arg(e ^ z) = y\bmod{2\pi}.
        \]
        $|e ^ z| \neq 0$ since $e ^ x \neq 0$.
        Since $\omega = 0 \iff |\omega| = 0$ we conclude that $e ^ z \neq 0$
        (i.e. $1$).
        \[
        e ^ z = 1 \iff e ^ xe ^ {iy} = 1 \cdot e ^ {i0}
        \]
        so $e ^ x = 1$ and $y = 0 + 2\pi k$ so $x = 0$ and $y = 2\pi k$,
        $k \in \Z$
        (shows (iii)).

        \[
        e ^ {z_1 + z_2} = e ^ {x_1 + x_2}e ^ {i(y_1 + y_2)} = e ^ {x_1}e ^ {x_2}e ^ {iy_1}e ^ {iy_2} = \left(e ^ {x_1}e ^ {iy_1}\right)\left(e ^ {x_2}e ^ {iy_2}\right) = e ^ {z_1}e ^ {z_2}.
        \]
        \[
        1 = e ^ 0 = e ^ {0 + i0} = e ^ {z - z} =  e ^ ze ^ {-z} \implies e ^ {-z} = \frac{1}{e ^ z}.
        \]
    \end{proof}
\end{proposition}

\begin{corollary}
    We have $\exp(2\pi i) = 1$ and $\exp(\pi i) = -1$.
    Latter is Euler's formula.
\end{corollary}

\begin{corollary}
    The complex exponential function is $2\pi i$-periodic,
    that is $\exp(z + 2k\pi i) = \exp(z)$ for any $k \in \Z$.
\end{corollary}

How does the exponential operate?
Recall $e ^ z - e ^ xe ^ {iy}$ and $|e ^ z| = e ^ x$ and $\Arg(e ^ z) = y\bmod{2\pi}$.
For fixed $x$,
i.e. $x = c$ for some $c \in \R$ we get image of a circle of radius $e ^ c$.
Segments on $x = c$ go to angular segments on the concentric circle with radius $e ^ c$.
If $y$ ranges over a segment of length more than $2\pi$ we will get the full circle.
When we fix $y$,
we will end up on a ray of angle $y\bmod{2\pi}$.
A segment on the line $y = c$ goes under $e ^ z$ to a segment on the ray $R_{c\bmod{2\pi}}$ if $a < x < b$ then we get the part of the ray $e ^ a < |\omega| < e ^ b$,
$\Arg(\omega) = c\bmod{2\pi}$.

\begin{lemma}
    \begin{itemize}
        \item The map $e ^ z$ injectively takes a segment of length $2\pi$ on the line $x = c$ which is open at one end and closed at the other to the entire concentric circle of radius $e ^ c$.
        If the above segment is closed,
        or its size is larger than $2\pi$ the image is no longer injective.

        \item The map $e ^ z$ injectively takes the line $y = c$ to the ray of angle $c$
        (without the origin).

        \item The map $e ^ z$ injectively takes the set $\{z \in \C\,:\, \theta < \Im(z) \leq \theta + 2\pi\}$ to $\C ^ {*}$ where
        \[
        \C ^ {*} =  \C \setminus \{0\}.
        \]
    \end{itemize}
\end{lemma}
With $e ^ z$ we can define trig functions.

Recall we defined $e ^ {ix} = \cos{x} + i\sin{x}$ when $x \in \R$.
\begin{align*}
    \cos{x} &= \frac{e ^ {ix} - e ^ {-ix}}{2}&\sin{x} &= \frac{e ^ {ix} - e ^ {-ix}}{2i} \\
    \cosh{x} &= \frac{e ^ x + e ^ {-x}}{2}& \sinh{x} &= \frac{e ^ x - e ^ {-x}}{2}
\end{align*}

\begin{definition}[Trigonometric functions]
    \begin{align*}
        \cos{z} &= \frac{e ^ {iz} - e ^ {-iz}}{2}&\sin{z} &= \frac{e ^ {iz} - e ^ {-iz}}{2i} \\
        \cosh{z} &= \frac{e ^ z + e ^ {-z}}{2}& \sinh{z} &= \frac{e ^ z - e ^ {-z}}{2}
    \end{align*}
\end{definition}
\textbf{Remember,
complex $\cos$ and $\sin$ are not bounded as they have $x$ and $y$ in $x + iy$.}


\begin{lemma}
    We have that for $z = x + iy$:
    \begin{align*}
        \sin(z) &= \sin(x)\cosh(y) + i\cos(x)\sinh(y) \\
        \cos(z) &= \cos(x)\cosh(y) - i\sin(x)\sinh(y) \\
        \sinh(z) &= -i\sin(iz) = \sinh(x)\cos(y) + i\cosh(x)\sin(y) \\
        \cosh(z) &= \cos(iz) = \cosh(x)\cos(y) + i\sinh(x)\sin(y).
    \end{align*}
    In addition we have that for all $z \in \C$
    \[
    \sin ^ 2(z) + \cos ^ 2(z) = 1,\qquad \cosh ^ 2(z) - \sinh ^ 2(z) = 1.
    \]

    \begin{proof}
        Exercise.

        \begin{align*}
            \sin{z} &= \frac{e ^ {iz} - e ^ {-iz}}{2i} \\
            &= \frac{e ^ {i(x + iy)} - e ^ {-i(x + iy)}}{2i} \\
            &= \frac{e ^ {-y}e ^ {ix} - e ^ {y}e ^ {-ix}}{2i} \\
            &= \frac{e ^ {-y}(\cos(x) + i\sin(x)) - e ^ {y}(\cos(x) - i\sin(x))}{2i}
        \end{align*}
    \end{proof}
\end{lemma}

In general complex $\cos$ and $\sin$ are bounded if and only if $y$ is bounded in $z = x + iy$.

\subsection{Logarithms and complex powers}


The real exponential $e ^ x$ has an inverse
($\log{x}$).
This,
in general,
will not work for $e ^ z$ since it is not injective.

Can we get an inverse when we restrict our domain?

\begin{lemma}[Inverting the exponential function]
    For every $w \in \C ^ {*} = \C \setminus \{0\}$,
    the equation
    \[
    e ^ z = w
    \]
    has a solution $z$.
    Furthermore,
    if we write $w = |w|e ^ {i\varphi}$ with $\varphi = \Arg(w)$,
    then all the solutions to $e ^ z = w$ are given by
    \[
    z = \log|w| + i(\varphi + 2\pi k)
    \]
    for $k \in \Z$.
    Here,
    $\log|w|$ is the usual natural logarithm of the real number $|w|$.
    \textit{Note that there are infinitely many solutions}.

    \begin{proof}
        \begin{align*}
            e ^ z = \omega &\iff e ^ xe ^ {iy} = |w|e ^ {iy} \\
            &\iff e ^ x = |\omega|\text{ and } y = y + 2\pi k\quad k \in \Z
        \end{align*}
        \[
        \]
    \end{proof}
\end{lemma}

From this we see a candidate to complex logarithm:
Just fix k.

We can try and define
\[
\log{z} = \log|z| + i\arg(z)
\]
where $\arg(z)$ goes in a fixed given range of angles $(\theta_1, \theta_2]$.
This definition is missing some
"nice"
properties.

To illustrate this choose the angles to be $(-\pi, \pi]$.
$\log(-1) = \log|-1| + i\pi = i\pi$.
Let's try to get closer to $-1$ from below,
choosing $\theta = -\pi + \delta$,
$\delta > 0$ gives us $z\delta = e ^ {i(-\pi + \delta)}$ close to $z = -1$.
\[
\log(z\delta) = \log\left|e ^ {i(-\pi + \delta)}\right| + i(-\pi + \delta) = -i(\pi - \delta) \approx -i\pi
\]
we have a discontinuity.

It is as if when we complete a full circle we
"climb"
by $2\pi$.

To avoid this issue we must disallow the ability to do full circles.
We do that by removing a closed ray $R_{\theta} = \{z = re ^ {i\theta}\,:\,r \geq 0\}$.

\begin{definition}[Complex logarithm functions]
    For any two real numbers $\theta_1 < \theta_2$ with $\theta_2 - \theta_1 = 2\pi$,
    let $\arg$ be the choice of argument function with values in $(\theta_1, \theta_2]$.
    Then the function
    \[
    \log(z) \coloneqq \log|z| + i\arg(z)
    \]
    is called a branch of logarithm.
    It has a jump discontinuity along the ray $R_{\theta_1} = R_{\theta_2}$.
    this ray is called a branch cut.
    If we choose $\arg(z) = \Arg(z) \in (-\pi, \pi]$,
    then we obtain a branch of logarithm called the principal branch of $\log$.
    We write $\Log$ for this principal branch:
    it is given by the formula
    \[
    \Log(z) \coloneqq \log|z| + i\Arg(z).
    \]
    The principal branch of logarithm has a
    "jump discontinuity"
    along the ray given by the non-positive real axis $\R_{\leq 0}$.
\end{definition}

Comments:
\begin{itemize}
    \item We must specify a branch
    ($\theta_1 < \theta_2$)
    to discuss a logarithm.

    \item For future:
    any branch of logarithm is continuous on $\C \setminus \R_{\theta}$.

    \item For real positive numbers
    \[
    \log{x} = \Log{x}.
    \]
\end{itemize}

\begin{lemma}[Properties of logarithms]
    We have the following properties when using any given branch of logarithm:
    \begin{enumerate}[label = (\roman*)]
        \item $e ^ {\log{z}} = z$ for any $z \in \C \setminus \{0\}$.

        \item In general
        \[
        \log(zw) \neq \log{z} + \log{w}.
        \]

        \item In general
        \[
        \log(e ^ z) \neq z.
        \]
    \end{enumerate}
\end{lemma}

\begin{lemma}\phantom{}
    \begin{itemize}
        \item A branch of the map $\log(z)$ injectively takes a ray of angle $\theta$ without the origin,
        to the line $y = \theta$ where $\theta$ is measured with respect to the branch cut.
    
        \item A branch of the map $\log(z)$ takes a concentric circle of radius $r$,
        minus the intersection of this circle with the branch cut,
        to an open segment of length $2\pi$ on the line $x = \log{r}$ with its lowest point given by the angle that defines the branch cut.
    \end{itemize}
\end{lemma}

\begin{definition}[Complex powers]
    For $w \in \C$ fixed by choosing any branch of $\log$ we can define a branch of the function $z \mapsto z ^ w$ by the equation
    \[
    z ^ w \coloneqq e ^ {w\log{z}}.
    \]
\end{definition}

For example,
if $w = 1 / n$ and use the principal branch we get
\[
z ^ {1 / n} = e ^ {\frac{\log|z|}{n} + i\frac{\Arg(z)}{n}} = |z| ^ {1 / n}e ^ {i\frac{\Arg(z)}{n}}.
\]

\begin{remark}
    This definition is consistent with $z ^ n = z \dotsi z$ with $n \in \N$.
    Also,
    for $n \in \N$ one can show that all the branches of $z ^ {1 / n}$ with argument in $(-\pi, \pi)$
    (branch cut is $\R_{\leq 0}$)
    are
    \[
    |z| ^ {1 / n}e ^ {i / n(\Arg(z) + 2\pi k)}
    \]
    $k = 0, \dotsc, n - 1$.
\end{remark}

\begin{example}
    Find $(1 - i) ^ {1 / 2}$ using the principal branch of the logarithm.

    \begin{solution}
        \[
        (1 - i) ^ {1 / 2} = e ^ {1 / 2\Log(1 - i)} = e ^ {1 / 2(\log|1 - i| + i\Arg(1 - i))} = e ^ {1 / 2(\log(\sqrt{2}) - i\pi / 4)} = e ^ {1 / 4\log(2)}e ^ {-i\pi / 8} = 2 ^ {1 / 4}e ^ {-i\pi / 8}.
        \]
    \end{solution}
\end{example}

\begin{example}
    Find $2 ^ {1 / 2}$ for all possible branches.

    \begin{solution}
        We only have two branches
        ($k = 0, k = 1$ from above)
        which will give $\sqrt{2}, -\sqrt{2}$.
    \end{solution}
\end{example}

\section{Topology and Continuity}
On $\R$ these notions are related to intervals
(open,
closed,
etc).
These studies can be extended to sets which are called open,
closed,
compact.
Such sets tell us a lot about our space and form its topology.

\subsection{Basic topology on the complex plane -
open and closed sets}

\begin{definition}[Balls in $\C$]\phantom{}
    \begin{itemize}
        \item The open ball $B_r(z_0)$ of radius $r$ centred at $z_0$ is
        \[
        B_r(z_0) \coloneqq \{z \in \C\,:\,|z - z_0| < r\}.
        \]

        \item The closed ball $\bar{B}_r(z_0)$ of radius $r$ centred at $z_0$ is
        \[
        \bar{B}_r(z_0) \coloneqq \{z \in \C\,:\, |z - z_0| \leq r\}.
        \]
    \end{itemize}
\end{definition}

\begin{definition}[Open/closed sets in $\C$]\phantom{}
    \begin{itemize}
        \item A subset $U \subseteq \C$ is open
        (in $\C$)
        if for every point $z_0 \in U$ there exists $\varepsilon > 0$ such that $B_{\varepsilon}(z_0) \subset U$.

        \item A subset $U \subseteq \C$ is closed
        (in $\C$)
        if its complement $U ^ c \coloneqq \C \setminus U$ is open.
    \end{itemize}
\end{definition}

\begin{lemma}[Open balls are open]
    For any $z_0 \in \C$ and any $r > 0$ the open ball $B_r(z_0)$ is an open set.

    \begin{proof}
        Let $z \in B_r(z_0)$.
        Define
        \[
        \varepsilon = \frac{r - |z - z_0|}{2} > 0\qquad (|z - z_0| < r).
        \]
        Claim:
        $B_{\varepsilon}(z) \subset B_r(z_0)$.
        Indeed,
        if $w \in B_{\varepsilon}(z)$
        \begin{align*}
            |w - z_0| &= |w - z + z - z_0| \\
            &\leq |w - z| + |z - z_0| \\
            &< \varepsilon + |z - z_0| \\
            &= \frac{r - |z - z_0|}{2} + |z - z_0| \\
            &= \frac{r + |z - z_0|}{2} < r.
        \end{align*}
    \end{proof}
\end{lemma}

\begin{remark}
    It can also be shown that for any $z_0 \in \C$ and any $r > 0$ the closed ball $\bar{B}_r(z_0)$ is closed in $\C$.
\end{remark}

\begin{example}
    The following sets are open:

    \begin{enumerate}[label = (\roman*)]
        \item $\mathbb{H} = \{z \in \C\,:\,\Im(z) > 0\}$.
        
        \item $\mathbb{H}_{-} = \{z \in \C\,:\,\Im(z) < 0\}$.
        
        \item $\mathbb{H}_{L} = \{z \in \C\,:\,\Re(z) < 0\}$.
        
        \item $\mathbb{H}_{R} = \{z \in \C\,:\,\Re(z) > 0\}$.
        
        \item $\mathbb{D} = \{z \in \C\,:\,|z| < 1\}$.
        
        \item $\Omega_1 = \{z \in \C\,:\,\Re(z) > 0, \Im(z) > 0\}$.
    \end{enumerate}

    Lets show that $\Omega_1$ is open.
    Given $z_0 \in \Omega_1$,
    define $\varepsilon = \frac{\min\{\Re(z_0), \Im(z_0)\}}{2}$.
    Claim:
    $B_{\varepsilon}(z) \subset \Omega_1$.
    Indeed if $w \in B_{\varepsilon}(z_0)$ then
    \begin{align*}
        \Re(w) &= \Re(w - z_0  + z_0) \\
        &= \Re(w - z_0) + \Re(z_0) \\
        \intertext{$|\Re(\xi)| \leq |\xi|$}
        &\geq -|w - z_0| + \Re(z_0) \\
        &> -\varepsilon + \Re(z_0) \\
        &= -\frac{\min(\Re(z_0), \Im(z_0))}{2} + \Re(z_0) \\
        &\geq \frac{\Re(z_0)}{2} > 0.
    \end{align*}
    Similarly $\Im(z_0) > 0$.
\end{example}

\textbf{Careful!}
Not all sets are open or closed.
$A - \{z \in \C\,:\,\Re(z) > 0, \Im(z) \geq 0\}$.
$A$ is not open
($1 - i\varepsilon \in B_{\varepsilon}(1)$,
but $1 \in A$,
$B_{\varepsilon}(1) \not\subset A$ for any $\varepsilon > 0$).
$A ^ c$ is not open,
i.e. $A$ is not closed.

\begin{lemma}[Unions and intersections of open sets]\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item Arbitrary unions of open sets are open;
        that is $\bigcup_{i \in I}U_i$ is open for any
        (possible infinite)
        collection of open sets $U_i$.

        \item Finite intersections of open sets are open;
        that is $\bigcap_{i = 1}^{n}U_i$ is open for any finite collection of open sets $U_i$.
    \end{enumerate}

    \begin{proof}\phantom{}
        \begin{enumerate}[label = (\roman*)]
            \item Let $z_0 \in \bigcup_{i \in I}U_i$,
            $\exists i_0 \in I$ s.t. $z_0 \in U_{i_0}$ since $U_{i_0}$ is open $\exists \varepsilon > 0$ s.t. $B_{\varepsilon}(z_0) \subset U_i \subseteq \bigcup_{i \in I}U_i$.
            $z_0$ was arbitrary so $\bigcup_{i \in I}U_i$ is open.

            \item Let $z_0 \in \bigcap_{i = 1}^{n}U_i$.
            This means that $z_0 \in U_i$ $\forall i = 1, \dotsc, n$.
            Since $U_i$ is open $\exists \varepsilon_i > 0$ s.t. $B_{\varepsilon_i}(z_0) \subset U_i$.
            Define $\varepsilon = \min(\varepsilon_1, \dotsc, \varepsilon_n)$ then $z_0 \in B_{\varepsilon}(z_0) \subseteq B_{\varepsilon_i}(z_0) \subset U_i$ $\forall i = 1, \dotsc, n$ this implies $B_{\varepsilon_0}(z) \subset \bigcap_{i = 1}^{n}U_i$.
        \end{enumerate}
    \end{proof}
\end{lemma}

\begin{corollary}[Unions and intersections of closed sets]\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item Finite unions of closed sets are closed.

        \item Arbitrary intersections of closed sets are closed.
    \end{enumerate}

    \begin{proof}\phantom{}
        \begin{enumerate}[label = (\roman*)]
            \item Follows from De Morgan's law
            \[
            \left(\bigcup_{i \in I}A_i\right) ^ c = \bigcap_{i \in I}A_i ^ c, \qquad \left(\bigcap_{i \in I}A_i\right) ^ c = \bigcup_{i \in I}A_i ^ c.
            \]
        \end{enumerate}
    \end{proof}
\end{corollary}

Given a set $A$ we can define special open and closed sets associated to $A$.

\begin{definition}[Interior points,
closure,
boundary]
    Let $A$ be a subset of $\C$.

    \begin{itemize}
        \item The interior $A ^ 0$ of $A$ is defined by
        \[
        A ^ 0 \coloneqq \{x \in A\,:\,\text{there exists an open set $U \subseteq A$ such that $x \in U$}\}
        \]
        (this can be replaced with $B_{\varepsilon}(x) \subset A$ for some $\varepsilon > 0$).
        
        \item The closure $\bar{A}$ of $A$ is the complement of the interior of the complement:
        \[
        \bar{A} \coloneqq \left(\left(A ^ c\right) ^ 0\right) ^ c = \{x \in X\,:\, U \cap A \neq \emptyset\text{ for every open set $U$ with $x \in U$}\}
        \]
        ($U$ can be replaced with $B_{\varepsilon}(x)$ for all $\varepsilon > 0$).
    
        \item The boundary $\partial A$ of $A$ is the closure without the interior:
        \[
        \partial A \coloneqq \bar{A} \setminus A ^ 0\qquad \left[=\left(A ^ 0\right) ^ c \cap \left(\left(A ^ c\right) ^ 0\right) ^ c = \left(A ^ 0 \cup \left(A ^ c\right) ^ 0\right) ^ c\right].
        \]
    \end{itemize}
\end{definition}

\begin{example}
    Consider the set
    \[
    A = \{z \in \C\,:\, 1 < |z| \leq 3\}.
    \]
    Then $A ^ 0$ is the area inside the inner and outer circle not inclusive of the boundary.
    The closure is the area inside the inner and outer circles including the boundary.
    The boundary is only the boundary of both the inner and outer circles.
\end{example}

\begin{remark}\phantom{}
    \begin{itemize}
        \item Don't confuse closure with conjugation.

        \item $A ^ 0$ is always open.
        $\bar{A}$ is always closed.
        $\partial A$ is always closed.

        \item Intuitively boundary is the edge of the set.
        Interior points are points in the set that are not on the edge and the closure is attained by adding all the edge points.
    \end{itemize}
\end{remark}

\textit{Note:
$A = \{z_0\}$,
$A ^ 0 = \emptyset$,
$\bar{A} = \{z_0\}$,
$\partial A = \{z_0\}$}

\textbf{Important properties:}
\begin{enumerate}[label = (\roman*)]
    \item $A$ is open $\iff \partial A \cap A = \emptyset \iff A = A ^ 0$.
    Moreover,
    \[
    A ^ 0 = \bigcup_{U \subseteq A, U\text{ open}}U.
    \]
    In other words:
    $A ^ 0$ is the largest open set that is contained in $A$.

    \item $A$ is closed $\iff \partial A \subseteq A \iff A = \bar{A}$.
    Moreover,
    \[
    \bar{A} = \bigcap_{A \subseteq F, F\text{ closed}}F.
    \]
    In other words:
    $\bar{A}$ is the smallest closed set that contains $A$.

    \item $\partial A = \{x \in X\,:\,\text{for all open sets $U$ containing $x$,
    there exist $y, z \in U$ with $y \in A$ and $z \in A ^ c$}\}$.
\end{enumerate}
