\begin{problem}
    Find all complex numbers $z$ for which $|z| ^ 2 = z ^ 2$.

    \begin{solution}
        Recall $|z| ^ 2 = z\bar{z}$.
        We want to solve
        \[
        z \cdot \bar{z} = z ^ 2
        \]
        $z = 0$ solves the above,
        $z \neq 0$ divide by $z$ and get
        \[
        \bar{z} = z \iff z \in \R.
        \]
        Since $z = 0$ is a real number we conclude that
        \[
        |z| ^ 2 = z ^ 2 \iff z \in \R.
        \]
    \end{solution}
\end{problem}

\begin{problem}
    If $\omega = \frac{z - 1}{z + 1}$ show that $\Re(\omega) = \frac{|z| ^ 2 - 1}{|z| ^ 2 + 2\Re(z) + 1}$ and $\Im(\omega) = \frac{2\Im(z)}{|z| ^ 2 + 2\Re(z) + 1}$.

    \begin{solution}
        \begin{align*}
            \frac{z - 1}{z + 1} &= \frac{z - 1}{z + 1}\cdot\frac{\overline{z + 1}}{\overline{z + 1}} \\
            &= \frac{z - 1}{z + 1}\cdot\frac{\bar{z} + 1}{\bar{z} + 1} \\
            &= \frac{z\bar{z} - \bar{z} + \bar{z} - 1}{z\bar{z} + \bar{z} + z + 1} \\
            &= \frac{|z| ^ 2 + 2i\Im(z) - 1}{|z| ^ 2 + 2\Re(z) + 1} \\
            &= \frac{|z| ^ 2 - 1 + 2i\Im(z)}{|z| ^ 2 + 2\Re(z) + 1} \\
            &= \frac{|z| ^ 2 - 1}{|z| ^ 2 + 2\Re(z) + 1} + i\frac{2\Im(z)}{|z| ^ 2 + 2\Re(z) + 1}
        \end{align*}
    \end{solution}
\end{problem}

\begin{problem}
    Describe geometrically the subset of $\C$ specified by $|z + 2 - i| < |iz - 1 + 2i|$.

    \begin{solution}
        Useful formula
        \begin{align*}
            |z - \omega| ^ 2 &= (z - \omega)\overline{(z - \omega)} \\
            &= (z - \omega)(\bar{z} - \bar{\omega}) \\
            &= z\bar{z} - \omega\bar{z} - z\bar{\omega} + \omega\bar{\omega} \\
            &= |z| ^ 2 - 2\Re(\bar{z}\omega) + |\omega| ^ 2 \\
            &= |z| ^ 2 - 2\Re(\bar{\omega}z) + |\omega| ^ 2
        \end{align*}
        consequently
        \begin{align*}
            |z + 2 - i| < |iz - 1 + 2i| &\overset{\text{everything is non-negative}}{\iff} |z + (2 - i)| ^ 2 < |iz - 1 + 2i| ^ 2 \\
            &\iff |z - (i - 2)| ^ 2 < |iz - (1 - 2i)| ^ 2 \\
            &\iff |z| ^ 2 - 2\Re((-i -2)z) + |i - 2| ^ 2 < |iz| ^ 2 - 2\Re((1 + 2i)iz) + |1 - 2i| ^ 2 \\
            \intertext{$|iz| ^ 2 = (|i||z|) ^ 2 = |z| ^ 2$,
            $|i - 2| ^ 2 = 5$,
            $|1 - 2i| ^ 2 = 5$}
        \end{align*}
        so the inequality holds if and only if
        \[
        -2\Re((-i - 2)z) < -2\Re((1 + 2i)iz)
        \]
        $\Re(\alpha z) = \alpha\Re(z)$ if $\alpha \in \R$,
        $\Re(i\alpha z) = -\alpha\Im(z)$ if $\alpha \in \R$.
    \end{solution}
\end{problem}

\begin{problem}
    The real axis and the imaginary axis divide $\C$ into four quadrants as follows:
    \[
    \begin{array}{c|c}
         \Omega_2 & \Omega_1  \\
         \hline
         \Omega_3 & \Omega_4
    \end{array}
    \]
    Determine the images of $\Omega_1, \Omega_2, \Omega_3, \Omega_4$ under the exponential map $z \mapsto e ^ z$.

    \begin{solution}
        $e ^ z = \underbrace{e ^ x}_{> 0} \cdot e ^ {iy}$.
        By uniqueness of polar representation
        \[
        |e ^ z| = e ^ x\qquad \Arg(e ^ z) = y\bmod{2\pi}.
        \]
        For a fixed $x$ and $a < y < b$ we get that $e ^ z$ gives us a circle with angular segment of length $\min(b - a, 2\pi)$.
        The half line $x = c$,
        $0 < y < \infty$ goes to the full circle
        \[
        \{z \in \C\,:\,|z| = e ^ c\}.
        \]
        Same for $x = c$,
        $-\infty < y < 0$.
        \[
        \exp(\Omega_1) = \{z \in \C\,:\, e ^ 0 < |z| < e ^ {\infty}\} = \{z \in \C\,:\, 1 < |z| < \infty\} = \exp(\Omega_4).
        \]
        with $0, \infty$ minimal $x$ and
        "maximal"
        $x$ respectively.
        Similarly
        \[
        \exp(\Omega_2) = \{z \in \C\,:\,e ^ {-\infty} < |z| < e ^ 0\} = \{z \in \C\,:\, 0 < |z| < 1\} = \exp(\Omega_3).
        \]
    \end{solution}
\end{problem}

\begin{problem}
    Give examples to illustrate that,
    in general,
    for complex numbers $z, w$,
    \[
    \Log(zw) \neq \Log(z) + \Log(w).
    \]

    \begin{solution}
        \begin{align*}
            \Log(zw) &= \log|zw| + i\Arg(zw) \\
            &= \log(|z||w|) + i\Arg(zw) \\
            &= \log|z| + \log|w| + i\underbrace{\Arg(zw)}_{\Arg(z) + \Arg(w)\bmod{2\pi}}.
        \end{align*}
        On the other hand
        \begin{align*}
            \Log(z) + \Log(w) &= \log|z| + i\Arg(z) + \log|w| + i\Arg(w) \\
            &= \log|z| + \log|w| + i(\Arg(z) + \Arg(w)).
        \end{align*}
        So
        \[
        \Log(zw) = \Log(z) + \Log(w) \iff \Arg(zw) = \Arg(z) + \Arg(w).
        \]
        This doesn't always hold.
        Take $z = w = -i = e ^ {-i\pi / 2}$.
        $\Arg(z) = \Arg(w) = -\pi / 2$,
        $\Arg(zw) = \Arg(-1) = \pi$.
        
    \end{solution}
\end{problem}

\begin{problem}\phantom{}
    \begin{enumerate}[label = (\roman*)]
        \item Show that the set $\{z_0\}$,
        consisting of a single point $z_0 \in \C$,
        is closed.

        \item Show that the closed ball $\bar{B}_{r}(z_0) \coloneqq \{z \in \C\,:\,|z - z_0| \leq r\}$,
        of radius $r > 0$ centred at $z_0 \in \C$,
        is closed.
    \end{enumerate}

    \begin{solution}
        \begin{enumerate}[label = (\roman*)]
            \item To show that $\{z_0\}$ is closed we need to show that $\{z_0\} ^ c = \C \setminus \{z_0\}$ is open.
            Given $z \in \{z_0\} ^ c$ we define $\varepsilon = \frac{|z - z_0|}{2} > 0$.
            We claim that $B_{\varepsilon}(z) \subset \{z_0\} ^ c$.
            Indeed,
            if $w \in B_{\varepsilon}(z)$
            \begin{align*}
                |w - z_0| &= |(w - z) + (z - z_0)| \\
                &\geq ||w - z| - |z - z_0|| \\
                &= |z - z_0| - |w - z| \\
                &\geq |z - z_0| - \varepsilon \\
                &= \frac{|z - z_0|}{2} > 0
            \end{align*}
            so $w \neq z_0$.

            \item We will show that if $\{z_{n}\}_{n \in \N} \subset \bar{B}_r(z_n)$ and $z_n \to z$ as $n \to \infty$ then $z \in \bar{B}_r(z_0)$.
            We have that
            \[
            |z_n - z_0| \leq n
            \]
            and $z_n \to z$ as $n \to \infty$.
            We want to show that
            \[
            |z - z_0| \leq r
            \]
            since then $z \in \bar{B}_r(z_0)$.
            Recall that $f(z) = |z|$ is continuous and using COLT $z_n - z_0 \to z - z_0$ as $n \to \infty$.
            We conclude that
            \[
            |z_n - z_0| \to |z - z_0| \text{ as } n \to \infty.
            \]
            Since
            \[
            0 \leq |z_n - z_0| \leq r\quad \forall n
            \]
            we must have that its limit also satisfies
            \[
            0 \leq |z - z_0| \leq r.
            \]
        \end{enumerate}
    \end{solution}
\end{problem}

\begin{problem}
    In the space $C([a, b])$ of continuous functions defined on a closed interval $[a, b]$
    (for $a < b$),
    let
    \[
    d_1(f, g) \coloneqq \int_{a}^{b}|f(t) - g(t)|\,dt.
    \]
    Show that $d_1$ is a metric on $C([a, b])$.

    \begin{solution}
        Positivity:
        for any $f, g \in C([a, b])$ $|f(t) - g(t)| \geq 0$ $\forall t \in [a, b]$,
        so $\int_{a}^{b}|f(t) - g(t)|\,dt \geq 0$.
        Also
        \[
        d_1(f, g) = 0 \iff \int_{a}^{b}|f(t) - g(t)|\,dt = 0 \iff |f(t) - g(t)| = 0 \quad \forall t \in [a, b] \iff f \equiv g.
        \]
        Symmetry
        \[
        d_1(f, g) = \int_{a}^{b}|f(t) - g(t)|\,dt = \int_{a}^{b}|g(t) - f(t)|\,dt = d_1(g, f).
        \]
        Triangle inequality:
        want to show $\forall f, g, h \in C([a, b])$
        \[
        d_1(f, g) \leq d_1(f, h) + d_1(h, g)
        \]
        \begin{align*}
            \int_{a}^{b}|f(t) - g(t)|\,dt \leq \int_{a}^{b}|f(t) - h(t)|\,dt + \int_{a}^{b}|h(t) - g(t)|\,dt.
        \end{align*}

        Indeed,
        let $f, g, h \in C([a, b])$ then for any $t \in [a, b]$
        \[
        |f(t) - g(t)| \leq |f(t) - h(t)| + |h(t) - g(t)|
        \]
        integrating over $[a, b]$ gives the desired result.
    \end{solution}
\end{problem}

\begin{problem}
    Show that a map $f : \C \to \C$ is continuous at $z_0 \in \C$ if and only if
    \[
    \lim_{n \to \infty}f(z_n) = f(z_0)\quad\text{for every convergent sequence $\{z_n\}_{n \in \N}$ in $\C$ with $z_n \to z_0$}.
    \]

    \begin{solution}
        "$\implies$"
        Assume that $f$ is continuous at $z_0$,
        i.e. for any $\varepsilon > 0$,
        $\exists \delta > 0$ s.t. if $|z - z_0| < \delta$ then $|f(z) - f(z_0)| < \varepsilon$.
        Let $\{z_n\}_{n \in \N}$ be such that $z_n \to z_0$ as $n \to \infty$.
        We would like to show that for any $\varepsilon > 0$,
        $\exists N \in \N$ s.t. if $n > N$ then
        \[
        |f(z_n) - f(z_0)| < \varepsilon.
        \]
        Given $\varepsilon > 0$ we find $\delta > 0$ s.t. if $|z - z_0| < \delta$,
        $|f(z) - f(z_0)| < \varepsilon$.
        Since $z_n \to z_0$ there exists $N \in \N$ s.t. if $n > N$,
        $|z_n - z_0| < \delta$.
        Consequently if $n > N$,
        \[
        |f(z_n) - f(z_0)| < \varepsilon.
        \]
        As $\varepsilon > 0$ was arbitrary,
        this direction is proved.
        
        "$\impliedby$".
        We will show that if $f$ is not continuous at $z_0$,
        then there exists a sequence $\{z_n\}_{n \in \N}$ s.t. $z_n \to z_0$ as $n \to \infty$ but $f(z_n) \not\to f(z_0)$ as $n \to \infty$.
        $\exists \varepsilon_0 > 0$ s.t. $\forall \delta > 0$ $\exists z_{\delta}$ with $|z_{\delta} - z| < \delta$ s.t.
        \[
        |f(z_{\delta}) - f(z_0)| \geq \varepsilon_0.
        \]
        Let $\delta_n = \frac{1}{n}$ and denote $z_n = z\delta_n$ we have that
        \[
        0 \leq |z_n - z_0| < \delta_n = \frac{1}{n} \to 0
        \]
        by squeezing theorem $|z_n - z_0| \to 0$ as $n \to \infty$ i.e. $z_n \to z_0$ as $n \to \infty$.
        However,
        \[
        |f(z_n) - f(z_0)| \geq \varepsilon_0\qquad \forall n \in \N
        \]
        which implies that $f(z_n) \not\to f(z_0)$ as $n \to \infty$.
    \end{solution}
\end{problem}

\begin{problem}
    At which points is the function
    \[
    f(z) = x\cosh{y} + y\sin(iy)\cos{x}
    \]
    differentiable?
    At what point are they holomorphic?

    \begin{solution}
        \begin{align*}
            \sin(iy) &= \frac{e ^ {i(iy)} - e ^ {-i(iy)}}{2i} \\
            &= \frac{e ^ {-y} - e ^ {y}}{2i} \\
            &= i\frac{e ^ y - e ^ {-y}}{2} = i\sinh(y)
        \end{align*}
        which implies $f(x + iy) = \frac{x\cosh(y)}{u(x, y)} + i\frac{\sinh(y)\cos(x)}{v(x, y)}$.

        $u, v \in C ^ 1(z ^ 2)$ and as such by theorem from lectures $f$ is complex differentiable at $z_0 = x_0 + iy_0$ if and only if Cauchy-Riemann equations hold.
        \begin{align*}
            u_x(x, y) &= \cosh(y) & v_y(x, y) &= \cosh(y)\cos(x) \\
            u_y(x, y) &= x\sinh(y) & v_x(x, y) &= -\sinh(y)\sin(x)
        \end{align*}
        \begin{align*}
            u_x = v_y &\iff \cosh(y) = \cosh(y)\cos(x) \\
            &\iff \cosh(y) = 0\text{ or } 1 = \cos(x) \\
            &\iff \cos(x) = 1 &&(\text{as $\cosh(y) \geq 1$}) \\
            &\iff x = 2\pi k\quad k \in \Z.
        \end{align*}

        \begin{align*}
            u_y = -v_x &\iff x\sinh(y) = -(-\sinh(y)\cdot \sin(x)) \\
            &\iff x\sinh(y) = \sin(x)\sinh(y) \\
            &\iff \sinh(y) = 0\text{ or } x = \sin(x) \\
            &\iff y = 0\text{ or } x = \sin(x).
        \end{align*}
        In total $x = 2\pi k$ and $y = 0$ or $x = \sin(x)$ so we conclude $x = 2\pi k$ and $y = 0$ or $2\pi k = \sin(2\pi k) = 0$.
        We find that the set of points on which $f$ is differentiable is:
        \[
        \{z = 2\pi k, k \in \Z\} \cup \{\Re(z) = 0\}.
        \]
        Since there is no open ball around any point in $S$ that is contained in $S$,
        $f$ is not holomorphic at any point
    \end{solution}
\end{problem}

\begin{problem}
    Let $f(z)$ be a holomorphic function.
    Prove the following variants of the zero derivative theorem,
    which says that,
    if any one of the following conditions hold on a domain $D$ then $f(z)$ is constant on $D$
    \begin{enumerate}[label = (\roman*)]
        \item $f(z)$ is a real number for all $z \in D$.

        \item The real part of $f(z)$ is constant on $D$.

        \item The modulus of $f(z)$ is constant on $D$.
    \end{enumerate}

    \begin{solution}
        $\Im(f(z)) = 0$ for all $z \in D$,
        i.e. if $f = u + v$ then $v = 0$.
        Since $f$ is holomorphic the Cauchy-Riemann equations hold and $u_x = v_y = 0$,
        $u_y = -v_x = 0$.
        $u$ is differentiable on $D$ and $u_x = u_y = 0$ on $D$ which is a domain so $u$ is constant.
        This implies $f(z) = u(x, y) + iv(x, y)$ is constant.


        $\Re(f(z)) = c$ for some $c \in \R$ and all $z \in D$,
        i.e. $u(x, y) = c$ which implies $u_x = u_y = 0$ and the proof follows like the previous.


        If $f = u + iv$ we have that $\exists c \geq 0$ s.t. $|f(z)| ^ 2 = c ^ 2$,
        i.e.
        \[
        u(x, y) ^ 2 + v(x, y) ^ 2 = c ^ 2.
        \]
        If $c = 0$ then $u = v = 0$ $\forall z \in D$.
        Assume that $c > 0$.
        We can differentiate our equation with respect to $x$ to get
        \[
        2u(x, y)u_x(x, y) + 2v(x, y)v_x(x, y) = 0.
        \]
        Similarly,
        differentiating with respect to $y$ gives
        \[
        2u(x, y)u_y(x, y) + 2v(x, y)v_y(x, y) = 0.
        \]
        Using the Cauchy-Riemann equations we find that
        \begin{align*}
            uu_x - vu_y &= 0 \\
            uu_y + uu_x &= 0
        \end{align*}
        i.e.
        \[
        \underbrace{\begin{pmatrix}
            u(x, y)  & -v(x, y) \\ v(x, y) & u(x, y)
        \end{pmatrix}}_{A(x, y)}
        \begin{pmatrix}
            u_x(x, y) \\ u_y(x, y)
        \end{pmatrix} = \begin{pmatrix}
            0 \\ 0
        \end{pmatrix}
        \]
        $\det(A(x, y)) = u(x, y) ^ 2 + v(x, y) ^ 2 = c ^ 2 \neq 0$.
        $A(x, y)$ is invertible which implies $A(x, y)\begin{pmatrix}
            u_x \\ u_y
        \end{pmatrix} = \begin{pmatrix}
            0 \\ 0
        \end{pmatrix}$ if and only if $u_x = 0$,
        $u_y = 0$,
        this implies that $u$ is constant and similarly $v$ is constant.
    \end{solution}
\end{problem}

\begin{problem}
    Find the image of the unit circle and the unit disc under the \mobius transformation $f(z) = w = \frac{z + i}{z - i}$.

    \begin{solution}
        $f$ is \mobius and as such will take circlines to circlines as well as boundary to boundary.
        A circline will go to a line under $f$ if and only if it contains a point that $f$ takes to $\infty$.
        $f(i) = \infty$ and $i \in \partial \mathbb{D}$,
        so $f(\partial \mathbb{D})$ is a line.
        $-i \in \partial \mathbb{D}$ and $f(-i) = 0$.
        $1 \in \partial \mathbb{D}$ and
        \[
        f(1) = \frac{1 + i}{1 - i} = \frac{(1 + i)(1 + i)}{|1 - i| ^ 2} = \frac{1 ^ 2+ 2i + i ^ 2}{1 ^ 2 + (-1) ^ 2} = i.
        \]
        So $f(\partial \mathbb{D})$ is the imaginary axis.
        Since boundary goes to boundary $\partial f(\mathbb{D}) = f(\partial \mathbb{D})$ which is the imaginary axis,
        $f(\mathbb{D})$ will contain $f(\text{interior point})$ and be bounded by the imaginary axis
        \[
        f(0) = \frac{0 + i}{0 - i} = -1
        \]
        so $f(\mathbb{D}) = \mathbb{H}_L$.
    \end{solution}
\end{problem}

\begin{problem}
    Find the \mobius transformation taking the ordered set of points $\{-i, 1, i\}$ to the ordered set of points $\{-i, 0, i\}$.
    What is the image of the unit disc under this map?
    Which point is sent to $\infty$?

    \begin{solution}
        We will use the cross-ratio.
        \[
        (z_0, z_1; z_2, z_3) = \frac{z_0 - z_2}{z_0 - z_3}\cdot \frac{z_1 - z_3}{z_1 - z_2}.
        \]
        If one of the points is $\infty$ we remove the affected parts or write $1$ instead.
        \[
        (z_0, \infty; z_2, z_3) = \frac{z_0 - z_2}{z_0 - z_3}\cdot \frac{1}{1}.
        \]
        Cross ratio is invariant under \mobius so
        \begin{align*}
            (z, -i; -1, i) = (M(z), -i; 0, i) &\iff \frac{z - (-1)}{z - i}\cdot\frac{-i - i}{-i - (-1)} = \frac{M(z) - 0}{M(z) - i}\cdot \frac{-i - i}{-i - 0} \\
            &\iff (z + 1)(-i)(M(z) - i) = (z - i)M(z)(1 - i) \\
            &\iff M(z)(-iz - i) - (z + 1) = M(z)(z - iz - i - 1) \\
            &\iff M(z)(-iz - i - z + iz + i + 1) = z +  1 \\
            &\iff M(z) = \frac{z + 1}{1 - z}.
        \end{align*}
    \end{solution}
\end{problem}

    Consider the region $P \subset \C$ defined by
    \[
    P = \left\{z \in \mathbb{D}\,:\,-\frac{3\pi}{4} < \Arg{z} < \frac{3\pi}{4}\right\}.
    \]
    \begin{enumerate}[label = (\alph*)]
        \item Draw $P$.

        \item Find a biholomorphic map from $P$ onto the upper half-plane $\mathbb{H}$.
        
        \item Find a biholomorphic map from $P$ onto the lower half-plane $\mathbb{H}_{-} \coloneqq \{w \in \C\,:\,\Im(z) < 0\}$.
    \end{enumerate}

    \begin{solution}
        \begin{enumerate}[label = (\alph*)]
            \item Looks like a Pacman from the $-$ve imaginary axis to the $+$ve imaginary axis on the left hand side of the origin.

            \item Open to semicircle on the y-axis using $z ^ {2 / 3}$,
            branch on $\C \setminus \R_{\leq 0}$.

            \item Rotation by $e ^ {-\pi / 2}z$ then $M_C ^ {-1}$ then $z ^ 2$.
        \end{enumerate}
    \end{solution}
\end{problem}
