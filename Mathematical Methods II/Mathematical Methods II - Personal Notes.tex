\section{Line integrals}

\subsection{Curves}
A curve $C$ is a $1$d subset of $\R ^ n$,
for $n > 1$.

It is usually convenient to use parameterisation to describe a curve, $x(t)$,
which maps a real interval $[t_0, t_1]$ into $\R ^ n$.
This is a vector valued function
\[
\mbf{x}(t) = x_1(t)\mbf{e}_1 + x_2(t)\mbf{e}_2 + \ldots + x_n(t)\mbf{e}_n,
\]
where $\{\mbf{e}_1, \mbf{e}_2, \dotsc, \mbf{e}_n\}$ denote the Cartesian unit vectors.
When $n = 2$ or $n = 3$,
we will usually write $x_1 \equiv x$,
$x_2 \equiv y$ and $x_3 \equiv z$.

Think of $\mbf{x}(t)$ as the trajectory of a particle,
with the parameter $t$ being time.
\begin{example}
    Find a parameterisation $\mbf{x}(t)$ representing the circle,
    $x ^ 2 + y ^ 2 = a ^ 2$ in $\R ^ 2$.

    \begin{solution}
        We can change to polar coordinates $(r, \theta)$ with $x = r\cos{\theta}$ and $y = r\sin{\theta}$.

        Then
        \[
        x ^ 2 + y ^ 2 = a ^ 2 \implies r ^ 2(\sin ^ 2\theta + \cos ^ 2\theta) = a ^ 2 \implies r = a,
        \]
        so we can set $t = \theta$ and obtain the parameterisation
        \[
        \mbf{x}(t) = a\cos{t}\mbf{e}_1 + a\sin{t}\mbf{e}_2,\quad t \in [0, 2\pi].
        \]
    \end{solution}
\end{example}

If $\mbf{x}(t_1) = \mbf{x}(t_0)$ then the curve is closed.

If a curve does not cross itself it is called simple.

A parameterisation $\mbf{x}(t)$ is differentiable if it has a well-defined tangent vector,
\[
\frac{d\mbf{x}}{dt} = \lim_{dt \to 0}\frac{\mbf{x}(t + dt) - \mbf{x}(t)}{dt},
\]

If $\mbf{x}(t)$ and $\mbf{x}(\tau)$ are two different parametrisations of the same curve,
then the chain rule gives
\[
\frac{d\mbf{x}}{dt} = \frac{d\mbf{x}}{d\tau}\frac{d\tau}{dt}.
\]

A parameterisation is \textbf{regular} if $\frac{d\mbf{x}}{dt} \neq 0$ everywhere,
and a curve is regular if it has at least one regular parameterisation.

If the derivative $\frac{d\mbf{x}}{dt}$ vanishes then these singular points,
where the direction of the curve is not defined are known as cusps.

\subsection{Line integrals of scalar fields}
$\R ^ n$ is a map $f\,:\,\R ^ n \to \R$.
In other words,
it assigns a real number to every point in $\R ^ n$.

\begin{definition}[Line integral]
    The line integral of a scalar field $f(x)$ along a curve $C$ with parameterisation $\mbf{x}(t)$,
    for $t \in [t_0, t_1]$,
    is defined as
    \[
    \int_{C}f\,d\ell = \int_{t_0}^{t_1}f(x(t))\left|\frac{d\mbf{x}}{dt}\right|\,dt.
    \]
\end{definition}

\begin{proposition}
    The line integral of a scalar field is independent of the choice of parameterisation.

    \begin{proof}
        Suppose we have two parameterisations of the same curve:
        $\mbf{x}(t)$ for $t \in [t_0, t_1]$,
        and $\mbf{x}(\tau)$ for $\tau \in [\tau_0, \tau_1]$.
        By the chain rule,
        the tangent vectors are related by
        \[
        \frac{d\mbf{x}}{dt} = \frac{d\tau}{dt}\frac{d\mbf{x}}{d\tau} \implies \left|\frac{d\mbf{x}}{dt}\right| = \left|\frac{d\tau}{dt}\right|\left|\frac{d\mbf{x}}{d\tau}\right|.
        \]
        There are now two cases:

        $\mbf{x}(\tau_0) = \mbf{x}(t_0)$.
        meaning both parameterisations trace the curve in the same direction.
        Then
        \[
        \frac{d\tau}{dt} > 0 \implies \int_{t_0}^{t_1}f(\mbf{x}(t))\left|\frac{d\mbf{x}}{dt}\right|\,dt = \int_{t_0}^{t_1}f(\mbf{x}(t))\left|\frac{d\mbf{x}}{d\tau}\right|\frac{d\tau}{dt}\,dt = \int_{\tau_0}^{\tau_1}f(\mbf{x}(\tau))\left|\frac{d\mbf{x}}{d\tau}\right|\,d\tau.
        \]
        $\mbf{x}(\tau_0) = \mbf{x}(t_1)$,
        meaning they trace the curve in opposite directions.
        Now
        \[
        \frac{d\tau}{dt} < 0 \implies \int_{t_0}^{t_1}f(\mbf{x}(t))\left|\frac{d\mbf{x}}{dt}\right|\,dt = \int_{t_0}^{t_1}f(\mbf{x}(t))\left|\frac{d\mbf{x}}{d\tau}\right|\left(-\frac{d\tau}{dt}\right)\,dt = \int_{\tau_0}^{\tau_1}f(\mbf{x}(\tau))\left|\frac{d\mbf{x}}{d\tau}\right|\,d\tau.
        \]
        Both cases recover the original formula.
    \end{proof}
\end{proposition}

\begin{definition}[Arclength]
    For the special case $f(\mbf{x}) = 1$,
    the line integral is called the arclength of the curve,
    \[
    L = \int_{C}\,d\ell = \int_{t_0}^{t_1}\left|\frac{d\mbf{x}}{dt}\right|\,dt.
    \]
\end{definition}

\begin{definition}[Rectifiable]
    A curve which has finite length is called rectifiable.
\end{definition}

\subsection{Line integrals of vector fields}

\begin{definition}[Vector field]
    A vector field is a function $\mbf{f}\,:\,\R ^ n \to \R ^ n$ that assigns a vector to each point in $\R ^ n$.
    For $n = 2$,
    we can visualise $\mbf{f}$ as an arrow at each point in the plane.
\end{definition}

\begin{definition}
    The unit tangent vector of a parameterisation is
    \[
    \hat{\mbf{t}} = \frac{\frac{d\mbf{x}}{dt}}{\left|\frac{d\mbf{x}}{dt}\right|}.
    \]
\end{definition}

\begin{definition}[Oriented curve]
    An oriented curve is a curve together with a specified
    (consistent)
    choice of $\hat{\mbf{t}}$.
    We indicate this by adding an arrow to the curve.
\end{definition}

The line integral of a vector field $\mbf{f}(\mbf{x})$ along an oriented curve $C$ with parameterisation $\mbf{x}(t)$,
for $t \in [t_0, t_1]$,
is defined as the scalar line integral of the tangential component:
\[
\int_{C}\mbf{f}\cdot\hat{\mbf{t}}\,d\ell = \int_{t_0}^{t_1}\mbf{f}(\mbf{x}(t))\cdot\hat{\mbf{t}}\left|\frac{d\mbf{x}}{dt}\right|\,dt = \int_{t_0}^{t_1}\mbf{f}(\mbf{x}(t))\cdot\frac{d\mbf{x}}{dt}\,dt.
\]

\begin{definition}[Circulation]
    If the curve $C$ is closed,
    this is often emphasised by the notation
    \[
    \int_{C}\mbf{f}\cdot\hat{\mbf{t}}\,d\ell = \oint_{C}\mbf{f}\cdot\hat{\mbf{t}}\,d\ell,
    \]
    and the line integral is called the circulation of $\mbf{f}$ around $C$.
\end{definition}

\subsection{Curvilinear coordinates}
For solving problems in $\R ^ 3$ with particular symmetries,
it is useful to use non-Cartesian coordinates.

In a curvilinear coordinate system $(u, v, w)$,
the coordinate lines,
where two of the coordinates are constant and one varies,
may be curved.
the lines may be parameterised by the corresponding coordinate,
so are given by $x(u, v, w)$ with two of the coordinates fixed.

The corresponding tangent vectors are $\frac{\partial \mbf{x}}{\partial u}, \frac{\partial \mbf{x}}{\partial v}, \frac{\partial \mbf{x}}{\partial w}$.

The coordinate system is orthogonal if the tangent vectors are mutually orthogonal.
We can define an orthonormal basis $\{\mbf{e}_u, \mbf{e}_v, \mbf{e}_w\}$ by normalising
\[
\mbf{e}_u = \frac{1}{h_u}\frac{\partial \mbf{x}}{\partial u},\quad\mbf{e}_v = \frac{1}{h_v}\frac{\partial \mbf{x}}{\partial v},\quad\mbf{e}_w = \frac{1}{h_w}\frac{\partial \mbf{x}}{\partial w},
\]
where
\[
h_u = \left|\frac{\partial \mbf{x}}{\partial u}\right|,\quad h_v = \left|\frac{\partial \mbf{x}}{\partial u}\right|,\quad h_w = \left|\frac{\partial \mbf{x}}{\partial w}\right|
\]
are called the scale factors.

There are two non-Cartesian systems commonly used.

\textbf{Cylindrical coordinates}

useful for problems with axial symmetry:
\[
\mbf{x}(r, \theta, z) = r\cos(\theta)\mbf{e}_1 + r\sin(\theta)\mbf{e}_2 + z\mbf{e}_3,\quad r > 0,\quad \theta \in [0, 2\pi],\quad z \in \R,
\]
giving scale factors $h_r = 1, h_{\theta} = r, h_z = 1$.

\textbf{Spherical coordinates}

useful for problems with spherical symmetry:
\[
\mbf{x}(r, \theta, \phi) = r\sin(\theta)\cos(\phi)\mbf{e}_1 + r\sin(\theta)\sin(\phi)\mbf{e}_2 + r\cos(\theta)\mbf{e}_3,\quad r > 0,\quad \theta \in [0, \pi], \quad \phi \in [0, 2\pi],
\]
giving scale factors $h_r = 1, h_{\theta} = r, h_{\phi} = r\sin{\theta}$.



\textbf{Right-handed coordinate system}

A coordinate system is right handed if $\mbf{e}_u \times \mbf{e}_v$ points in the positive $\mbf{e}_w$ direction.
Equivalently,
if
\[
\frac{\partial \mbf{x}}{\partial w} \cdot \left(\frac{\partial \mbf{x}}{\partial u} \times \frac{\partial \mbf{x}}{\partial v}\right) > 0.
\]

\section{Surface and volume integrals}

\subsection{Surfaces}.

A $2$D surface embedded in $\R ^ 3$ can be defined parametrically as a map $\mbf{x}(u, v)$ from $\R ^ 2 \to \R ^ 3$,
for $u \in [u_0, u_1]$ and $v \in [v_0, v_1]$.

For each value $v$,
the position $\mbf{x}(u, v)$ describes a curve on the surface with tangent vector $\frac{\partial \mbf{x}}{\partial u}$.

If $(u, v)$ is a well-defined parameterisation,
then their tangent vectors cannot be collinear.

\begin{definition}[Unit normal and oriented surface]
    For a parameterisation $(u, v)$ well-defined,
    we define the unit normal to the surface as
    \[
    \hat{\mbf{n}} = \frac{\frac{\partial\mbf{x}}{\partial u} \times \frac{\partial \mbf{x}}{\partial v}}{\left|\frac{\partial\mbf{x}}{\partial u} \times \frac{\partial \mbf{x}}{\partial v}\right|}.
    \]

    A surface together with specified
    (consistent)
    choice of $\hat{\mbf{n}}$ is called an oriented surface.
\end{definition}

\begin{definition}[Smooth surface]
    A surface is smooth if $\hat{mbf{n}}$ varies continuously over a surface.
\end{definition}

\begin{definition}[Piecewise smooth surface]
    A piecewise smooth surface is one that can be divided into finitely many smooth portions.
\end{definition}

\begin{definition}[Open surface]
    A surface is open if it is bounded by a curve.
\end{definition}

\begin{definition}[Closed surface]
    A surface is closed if it has no boundary.
\end{definition}

\begin{definition}[Boundary]
    The boundary of an open surface $S$ is a curve,
    often written $\partial S$.
\end{definition}

A simple type of surface is an explicit surface given by the height of a two-dimensional function,
$z = h(x, y)$.

\subsection{Surface integrals of scalar fields}

If $f(\mbf{x}) = f(x, y, z)$ is a scalar field and $S$ is a surface with parameterisation $\mbf{x}(u, v)$ then we define the surface integral of $f$ over $S$ to be
\[
\int_{S}f\,dS = \int_{U}f(\mbf{x}(u, v))\left|\frac{\partial\mbf{x}}{\partial u} \times \frac{\partial \mbf{x}}{\partial v}\right|\,dudv,
\]
where $U$ is the preimage of $S$ in the $(u, v)$ plane.

Note that the area element $\left|\frac{\partial\mbf{x}}{\partial u} \times \frac{\partial \mbf{x}}{\partial v}\right|\,dudv$ is the area on the surface in $\R ^ 3$ corresponding to an infinitesimal rectangle with sides $(du, dv)$ in the $(u, v)$ plane.


\begin{proposition}
    If $(u, v)$ are
    (two of a set of)
    orthogonal curvilinear coordinates,
    then
    \[
    \left|\frac{\partial\mbf{x}}{\partial u} \times \frac{\partial \mbf{x}}{\partial v}\right| = h_uh_v.
    \]

    \begin{proof}
        Recall that $\frac{\partial\mbf{x}}{\partial u} = h_u\mbf{e}_u$ and $\frac{\partial\mbf{x}}{\partial v} = h_v\mbf{e}_v$.
        Since $h_u, h_v \geq 0$,
        we have that
        \[
        \left|\frac{\partial\mbf{x}}{\partial u} \times \frac{\partial \mbf{x}}{\partial v}\right| = h_uh_v|\mbf{e}_u \times \mbf{e}_v| = h_uh_v|\pm\mbf{e}_w| = h_uh_v.
        \]
    \end{proof}
\end{proposition}

\begin{proposition}\label{prep:mathmethII:prop:surfintscfldindpparam}
    The surface integral of a scalar field is independent of the choice of parameterisation.

    \begin{proof}
        Suppose $\mbf{x}(x, v)$ and $\mbf{x}(\mu, \nu)$ are two regular parametrisations of the same surface.
        By the chain rule,
        \[
        \frac{\partial \mbf{x}}{\partial \mu} = \frac{\partial\mbf{x}}{\partial \mu}\frac{\partial \mu}{\partial u} + \frac{\partial\mbf{x}}{\partial \nu}\frac{\partial \nu}{\partial u}, \qquad \frac{\partial\mbf{x}}{\partial v} = \frac{\partial \mbf{x}}{\partial \mu}\frac{\partial\mu}{\partial v} + \frac{\partial \mbf{x}}{\partial \nu}\frac{\partial \nu}{\partial v}.
        \]
        So
        \begin{align*}
            \frac{\partial\mbf{x}}{\partial u} \times \pd[\mbf{x}]{v} &= \left(\frac{\partial\mbf{x}}{\partial \mu}\frac{\partial \mu}{\partial u} + \frac{\partial\mbf{x}}{\partial \nu}\frac{\partial \nu}{\partial u}\right) \times \left(\frac{\partial \mbf{x}}{\partial \mu}\frac{\partial\mu}{\partial v} + \frac{\partial \mbf{x}}{\partial \nu}\frac{\partial \nu}{\partial v}\right) \\
            &= \left(\frac{\partial\mbf{x}}{\partial \mu}\frac{\partial \mu}{\partial u} + \frac{\partial\mbf{x}}{\partial \nu}\frac{\partial \nu}{\partial u}\right) + \left(\frac{\partial \mbf{x}}{\partial \nu}\frac{\partial\nu}{\partial u} \times \frac{\partial \mbf{x}}{\partial \mu}\frac{\partial \mu}{\partial v}\right) \\
            &= \left(\pd[\mu]{u}\pd[\nu]{v} - \pd[\mu]{v}\pd[\nu]{u}\right)\pd[\mbf{x}]{\mu} \times \frac{\partial \mbf{x}}{\partial \nu}.
        \end{align*}
        The means
        \[
        \left|\frac{\partial \mbf{x}}{\partial u} \times \pd[\mbf{x}]{v}\right| = |J|\left|\pd[\mbf{x}]{\mu} \times \pd[\mbf{x}]{\nu}\right|,
        \]
        where $J = \pd[\mu]{u}\pd[\nu]{v} - \pd[\mu]{v}\pd[\nu]{u}$ is the Jacobian for the change of variables from $(\mu, \nu)$ to $(u, v)$.
    \end{proof}
\end{proposition}

\subsection{Surface integrals of vector fields}
If $\mbf{f}\,:\,\R ^ 3 \to \R ^ 3$ is a vector field and $S$ is an oriented surface with unit normal $\hat{\mbf{n}}$,
we define the surface integral
(or flux)
of $\mbf{f}$ through $S$ as
\[
\int_{S}\mbf{f}\cdot\hat{\mbf{n}}\,dS.
\]

If the surface $S$ is closed,
this is often emphasised by writing
\[
\int_{S}\mbf{f} \cdot \hat{\mbf{n}}\,dS = \oint_{S}\mbf{f}\cdot\hat{\mbf{n}}\,dS.
\]


\subsection{Volume integrals}

\begin{definition}
    The volume integral of a scalar field $f : \R ^ 3 \to \R$ over a $3$D subregion $V \subset \R ^ 3$ is
    \[
    \int_{V}f\,dV = \int_{U}f(\mbf{x}(u, v, w))\left|\pd[\mbf{x}]{u}\cdot\left(\pd[\mbf{x}]{v}\pd[\mbf{x}]{w}\right)\right|\,du\,dv\,dw
    \]
    where $U$ is the preimage of $V$ in $(u, v, w)$ space
\end{definition}

For some well-defined coordinates $(u, v, w)$.
The factor $\left|\pd[\mbf{x}]{u}\cdot\left(\pd[\mbf{x}]{v}\pd[\mbf{x}]{w}\right)\right|\,du\,dv\,dw$ is the volume of a parallelepiped.

\begin{proposition}
    If $(u, v, w)$ are orthogonal curvilinear coordinates,
    then
    \[
    \left|\pd[\mbf{x}]{u}\cdot\left(\pd[\mbf{x}]{v}\pd[\mbf{x}]{w}\right)\right| = h_uh_vh_w.
    \]

    \begin{proof}
        Recall that $\pd[\mbf{x}]{u} = h_u\mbf{e}_u$,
        and similarly for the other two tangent vectors.
        Since $h_u, h_v, h_w \geq 0$,
        \[
        \left|\pd[\mbf{x}]{u}\cdot\left(\pd[\mbf{x}]{v}\pd[\mbf{x}]{w}\right)\right| = h_uh_vh_w|\mbf{e}_u \cdot (\mbf{e}_v \times \mbf{e}_w)| = h_uh_vh_w|\mbf{e}_u \cdot (\pm\mbf{e}_u)| = h_uh_vh_w.
        \]
    \end{proof}
\end{proposition}

\begin{remark}
    \[
    \left|\pd[\mbf{x}]{u}\cdot\left(\pd[\mbf{x}]{v}\pd[\mbf{x}]{w}\right)\right|
    \]
    is often called the Jacobian
    (as previously seen).

    \begin{proof}
        Look at the triple product as a determinant.
    \end{proof}
\end{remark}

\begin{proposition}
    The volume integral of a scalar field is independent of the choice of coordinates.

    \begin{proof}
        This is analogous to \autoref{prep:mathmethII:prop:surfintscfldindpparam},
        using the fact that a $3$D change of variables obeys
        \[
        dx\,dy\,dz = \left|\pd[\mbf{x}]{u}\cdot\left(\pd[\mbf{x}]{v}\pd[\mbf{x}]{w}\right)\right|\,du\,dv\,dw.
        \]
    \end{proof}
\end{proposition}

\section{Differential operators}

\subsection{Differentiating scalar fields}

\begin{definition}
    The gradient of a scalar field $f : \R ^ 3 \to \R$ is the operator
    \[
    \nabla f\Bigr\rvert_{\mbf{x}} = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}f\hat{\mbf{n}}\,dS,
    \]
    where $S$ is a closed surface around $\mbf{x}$ with outward normal $\hat{\mbf{n}}$ enclosed volumed $|V|$,
    and diameter $d(S) = \sup_{x, y \in S}|\mbf{x} - \mbf{y}|$.
\end{definition}

\begin{proposition}
    In Cartesian coordinates
    \[
    \mbf{\nabla}f = \pd[f]{x}\mbf{e}_1 + \pd[f]{y}\mbf{e}_2 + \pd[f]{z}\mbf{e}_3.
    \]
\end{proposition}

\begin{lemma}[Mean Value Theorem for Double Integrals]
    If a scalar field $f : \R ^ 2 \to \R$ is continuous on a closed,
    bounded and connected domain $A \subset \R ^ 2$ then there exists $\mbf{x}_{*} \in A$ such that
    \[
    \int_{A}f\,dS = |A|f(\mbf{x}_{*}).
    \]

    \begin{proof}
        Under the given conditions,
        we know $f$ attains both its maximum and minimum on $A$,
        say at some points $\mbf{x}_{\text{max}}, \mbf{x}_{\text{min}}$.
        We also know that
        \[
        |A|f(\mbf{x}_{\text{min}}) \leq \int_{A}f\,dS \leq |A|f(\mbf{x}_{\text{max}}).
        \]
        Now let $\mbf{x}(t)$ parameterise a curve in $A$ such that $\mbf{x}(t_0) = \mbf{x}_{\text{min}}$ and $\mbf{x}(t_1) = \mbf{x}_{\text{max}}$,
        consider the function
        \[
        g(t) = |A|f(\mbf{x}(t)).
        \]

        This is continuous with $g(t_0) = |A|f(\mbf{x}_{\text{min}})$ and $g(t_1) = |A|f(\mbf{x}_{\text{max}})$,
        so by the intermediate value theorem and the first equation there exists $t_{*}$ with
        \[
        g(t_{*}) = \int_{A}f\,dS.
        \]
        Then at $\mbf{x}_{*} = \mbf{x}(t_{*})$ we have $|A|f(\mbf{x}_{*}) = \int_{A}f\,dS$,
        as required.
    \end{proof}
\end{lemma}

\begin{proof}[Proof of the previous proposition]
    To find the $\mbf{e}_1$ component of $\mbf{\nabla}f$,
    let $V$ be a small cylinder of radius $\epsilon$ and length $h$,
    whose axis is the unit vector $\mbf{e}_1$.
    The volume of the cylinder is $|V| = \pi\epsilon ^ 2h$.

    Then
    \begin{align*}
        \mbf{e}_1 \cdot \mbf{\nabla}f &= \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}f\mbf{e}_1 \cdot \hat{\mbf{n}}\,dS \\
        &= \lim_{\epsilon, h \to 0}\frac{1}{\pi\epsilon ^ 2h}\left[\int_{S_1}f\,dS - \int_{S_0}f\,dS\right]\quad[\text{since $\mbf{e}_1 \cdot \hat{\mbf{n}} = 0$ on the curved surface}] \\
        &= \lim_{\epsilon, h \to 0}\frac{1}{\pi\epsilon ^ 2h}\left[\pi\epsilon ^ 2f(\mbf{x}_1) - \pi\epsilon ^ 2f(\mbf{x}_0)\right]
    \end{align*}
    for some $\mbf{x}_1 \in S$ and $\mbf{x}_0 \in S_0$ by the previous lemma.
    Taking the $\epsilon$ limit,
    these points tend to the centre of their faces by the squeezing theorem,
    so
    \[
    \mbf{e}_1 \cdot \mbf{\nabla}f = \lim_{h \to 0}\frac{f(\mbf{x} + h\mbf{e}_1) - f(\mbf{x})}{h} = \pd[f]{\mbf{x}}.
    \]
    The other components are obtained analogously.
\end{proof}

\begin{proposition}
    In curvilinear coordinates $(u, v, w)$,
    the gradient has the expression
    \[
    \mbf{\nabla}f = \frac{1}{h_u}\pd[f]{u}\mbf{e}_u + \frac{1}{h_v}\pd[f]{v}\mbf{e}_v + \frac{1}{h_w}\pd[f]{w}\mbf{e}_w.
    \]

    \begin{proof}
        This is an easy consequence of the previous proposition.
        Note
        \[
        \mbf{e}_u \cdot \mbf{\nabla}f = \left(\frac{1}{h_u}\pd[\mbf{x}]{u}\right)\cdot\mbf{\nabla}f = \frac{1}{h_u}\left(\pd[x]{u}\pd[f]{x} + \pd[y]{u}\pd[f]{y} + \pd[z]{u}\pd[f]{z}\right) = \frac{1}{h_u}\pd[f]{u},
        \]
        where the last step used the chain rule.
        The $\mbf{e}_v$ and $\mbf{e}_w$ components are analogous.
    \end{proof}
\end{proposition}

\subsection{Differentiating vector fields}

There are two standard ways to differentiate a vector field $\mbf{f} : \R ^ 3 \to \R ^ 3$,
defined in a similar way to the gradient.
$S$ is a closed surface enclosing $\mbf{x}$ with outward normal $\hat{\mbf{n}}$,
enclosed volume $|V|$,
and diameter $d(S)$.

\begin{definition}[Divergence]
    The divergence of $\mbf{f}$ is the scalar field
    \[
    \mbf{\nabla}\cdot\mbf{f}\Bigr\rvert_{\mbf{x}} = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hat{\mbf{n}}\cdot\mbf{f}\,dS.
    \]
\end{definition}

\begin{definition}[Curl]
    The curl of $\mbf{f}$ is the vector field
    \[
    \mbf{\nabla}\times\mbf{f}\Bigr\rvert_{\mbf{x}} = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hat{\mbf{n}}\times\mbf{f}\,dS.
    \]
\end{definition}

If the divergence of the vector field is $\mbf{\nabla} \cdot \mbf{f} > 0$ then it points outward,
if it is less than $0$ then it points inward.

The curl measures how much on average the vector field is rotating around any given axis.

\begin{lemma}
    If $\mbf{a} \in \R ^ 3$ is a constant vector and $f : \R ^ 3 \to \R$ is a $C ^ 1$ scalar field,
    then
    \begin{enumerate}[label = (\roman*)]
        \item
        \[
        \mbf{\nabla} \cdot (f\mbf{a}) = (\mbf{\nabla} f) \cdot\mbf{a},
        \]

        \item
        \[
        \mbf{\nabla} \times (f\mbf{a}) = (\mbf{\nabla} f) \times\mbf{a}.
        \]
    \end{enumerate}

    \begin{proof}
        These formulae follow from the integral definitions
        \[
        \mbf{\nabla} \cdot (f\mbf{a}) = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hat{\mbf{n}}\cdot(f\mbf{a})\,dS = \left(\lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}f\hat{\mbf{n}}\,dS\right) \cdot \mbf{a} = (\mbf{\nabla} f) \cdot\mbf{a},
        \]
        and
        \[
        \mbf{\nabla} \times (f\mbf{a}) = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hat{\mbf{n}}\times(f\mbf{a})\,dS = \left(\lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}f\hat{\mbf{n}}\,dS\right) \times \mbf{a} = (\mbf{\nabla} f) \times\mbf{a}.
        \]
    \end{proof}
\end{lemma}

\begin{proposition}
    In Cartesian coordinates,
    divergence and curl are given by
    \begin{enumerate}[label = (\roman*)]
        \item
        \[
        \mbf{\nabla} \cdot \mbf{f} = \pd[f_1]{x} + \pd[f_2]{y} + \pd[f_3]{z}.
        \]

        \item
        \[
        \mbf{\nabla} \times \mbf{f} = \begin{vmatrix}
            \mbf{e}_1 & \mbf{e}_2 & \mbf{e}_3 \\
            \pd{x} & \pd{y} & \pd{z} \\
            f_1 & f_2 & f_3
        \end{vmatrix}
        = \left(\pd[f_3]{y} - \pd[f_2]{z}\right)\mbf{e}_1 + \left(\pd[f_1]{z} - \pd[f_1]{x}\right)\mbf{e}_2 + \left(\pd[f_2]{x} - \pd[f_1]{y}\right)\mbf{e}_3.
        \]
    \end{enumerate}

    \begin{proof}
        Using the previous lemma,
        we have
        \begin{align*}
            \mbf{\nabla} \cdot \mbf{f} &= \mbf{\nabla} \cdot (f_1\mbf{e}_1 + f_2\mbf{e}_2 + f_3\mbf{e}_3) \\
            &= \mbf{\nabla} \cdot (f_1\mbf{e}_1) + \mbf{\nabla} \cdot (f_2\mbf{e}_2) + \mbf{\nabla} \cdot (f_3\mbf{e}_3) \\
            &= (\mbf{\nabla}f_1) \cdot \mbf{e}_1 + (\mbf{\nabla}f_2) \cdot \mbf{e}_2 + (\mbf{\nabla}f_3) \cdot \mbf{e}_3 \\
            &= \pd[f_1]{x} + \pd[f_2]{y} + \pd[f_3]{z}.
        \end{align*}

        We also have
        \[
        \mbf{\nabla} \times (f_1\mbf{e}_1) = (\mbf{\nabla}f_1) \times \mbf{e}_1 = \pd[f_1]{y}\mbf{e}_2 \times \mbf{e}_1 + \pd[f_1]{z}\mbf{e}_3 \times \mbf{e}_1 = -\pd[f_1]{y}\mbf{e}_3 + \pd[f_1]{z}\mbf{e}_2.
        \]
        The other terms are obtained similarly.
    \end{proof}
\end{proposition}

\begin{corollary}
    If $f : \R ^ 3 \to \R$ is a scalar field,
    and $\mbf{f} : \R ^ 3 \to \R ^ 3$ is a vector field,
    both with continuous second derivatives,
    then
    \begin{enumerate}[label = (\roman*)]
        \item $\mbf{\nabla} \cdot (\mbf{\nabla} \times \mbf{f}) = 0$,
        
        \item $\mbf{\nabla} \times (\mbf{\nabla}f) = \mbf{0}$.
    \end{enumerate}

    \begin{proof}
        Note the expressions are independent of any choice of coordinates.
        So we can prove the identities in Cartesian coordinates,
        so we can use the previous proposition.

        For
        (i)
        \begin{align*}
            \mbf{\nabla} \cdot (\mbf{\nabla} \times \mbf{f}) &= \pd{x}\left(\pd[f_3]{y} - \pd[f_2]{z}\right) + \pd{y}\left(\pd[f_1]{z} - \pd[f_1]{x}\right) + \pd{z}\left(\pd[f_2]{x} - \pd[f_1]{y}\right) \\
            &= \frac{\partial ^ 2f_1}{\partial y\partial z} - \frac{\partial ^ 2f_1}{\partial z\partial y} + \frac{\partial ^ 2f_2}{\partial z\partial x} - \frac{\partial ^ 2f_2}{\partial x\partial z} + \frac{\partial ^ 2f_3}{\partial x\partial y} - \frac{\partial ^ 2f_3}{\partial y\partial x} = 0.
        \end{align*}

        For
        (ii)
        \[
        \mbf{\nabla} \times (\mbf{\nabla}f) = \left(\frac{\partial ^ 2 f}{\partial y \partial z} - \frac{\partial ^ 2 f}{\partial z \partial y}\right)\mbf{e}_1 + \left(\frac{\partial ^ 2 f}{\partial z \partial x} - \frac{\partial ^ 2 f}{\partial x \partial z}\right)\mbf{e}_2 + \left(\frac{\partial ^ 2 f}{\partial x \partial y} - \frac{\partial ^ 2 f}{\partial y \partial x}\right)\mbf{e}_3 = \mbf{0}.
        \]
    \end{proof}
\end{corollary}


\subsection{Product rules}

\begin{proposition}[Product rules]
    If $f, g : \R ^ 3 \to \R$ are scalar fields,
    and $\mbf{f, g} : \R ^ 3 \to \R ^ 3$ are vector fields,
    all $C ^ {1}$,
    then
    \begin{enumerate}[label = (\roman*)]
        \item $\mbf{\nabla}(fg) \equiv g\mbf{\nabla}f + f\mbf{\nabla}g$,

        \item $\bnabla \cdot (g\mbf{f}) \equiv (\bnabla g) \cdot \mbf{f} + g\bnabla \cdot \mbf{f}$,

        \item $\bnabla \cdot (\mbf{f} \times \mbf{g}) \equiv \mbf{g}(\bnabla \times \mbf{f}) - \mbf{f} \cdot (\bnabla \times \mbf{g})$,

        \item $\bnabla \times (g\mbf{f}) \equiv g(\bnabla \times \mbf{f}) + (\bnabla g) \times \mbf{f}$,

        \item $\bnabla \times (\mbf{f} \times \mbf{g}) \equiv (\mbf{g} \cdot \bnabla)\mbf{f} - (\mbf{f} \cdot \bnabla)\mbf{g} + \mbf{f}(\bnabla \cdot \mbf{g}) - \mbf{g}(\bnabla \cdot \mbf{f})$.

        \item $\bnabla(\mbf{f} \cdot \mbf{g}) \equiv (\mbf{g} \cdot \bnabla) \mbf{f} + (\mbf{f} \cdot \bnabla)\mbf{g} + \mbf{f} \times (\bnabla \times \mbf{g}) + \mbf{g} \times (\bnabla \times \mbf{f})$.
    \end{enumerate}

    \begin{proof}
        Most of the proofs are long and pretty easy.
    \end{proof}
\end{proposition}

\subsection{Divergence and curl in curvilinear coordinates}

\begin{proposition}
    In orthogonal curvilinear coordinates $(u, v, w)$,
    the divergence and curl are given by
    \begin{enumerate}[label = (\roman*)]
        \item
        \[
        \bnabla \cdot \mbf{f} = \frac{1}{h_uh_vh_w}\left[\pd{u}(h_vh_wf_u) + \pd{v}(h_uh_wf_v) + \pd{w}(h_uh_vf_w)\right].
        \]

        \item
        \[
        \bnabla \times \mbf{f} = \frac{1}{h_uh_vh_w}\begin{vmatrix}
            h_u\mbf{e}_u & h_v\mbf{e}_v & h_w\mbf{e}_w \\
            \pd{u} & \pd{v} & \pd{w} \\
            h_uf_u & h_vf_v & h_wf_w
        \end{vmatrix}.
        \]
    \end{enumerate}

    \begin{proof}
        Note
        \[
        \bnabla v = \frac{1}{h_v}\mbf{e}_v\quad\mbf{e}_v = h_v\bnabla v\quad\mbf{e}_w = h_w\bnabla w.
        \]
        For
        (i),
        note using orthogonality followed by the previous proposition,
        we get
        \begin{align*}
            \bnabla  \cdot (f_u\mbf{e}_u) &= \bnabla \cdot (f_u\mbf{e}_v \times \mbf{e}_w) \\
            &= \bnabla \cdot (f_uh_vh_w \bnabla v \times \bnabla w) \\
            &= \bnabla(f_uh_vh_w) \cdot \bnabla v \times \bnabla w f_uh_vh_w\bnabla \cdot (\bnabla v \times \bnabla w) \\
            &= \bnabla(f_uh_vh_w) \cdot \bnabla v \times \bnabla w f_uh_vh_w\bnabla \cdot (\bnabla \times [v \bnabla w]) \\
            &= \bnabla(f_uh_vh_w) \cdot \bnabla v \times \bnabla w  \\
            &= \bnabla(f_uh_vh_w) \cdot \frac{\mbf{e}_v \times \mbf{e}_w}{h_vh_w} \\
            &= \frac{1}{h_vh_w}\mbf{e}_u \cdot \bnabla(f_uh_vh_w) \\
            &= \frac{1}{h_uh_vh_w}\pd{u} \cdot \bnabla(h_vh_wf_u),
        \end{align*}
        the other components are similar.

        For
        (ii),
        we use $\mbf{e}_u = h_u\bnabla u$ and the previous proposition to find
        \begin{align*}
            \bnabla\times(f_u\mbf{e}_u) &= \bnabla\times(f_uh_u\bnabla u)\\
            &= \bnabla(f_u h_u)\times\bnabla u + f_uh_u\bnabla\times\bnabla u\\
            &= \bnabla(f_u h_u)\times\bnabla u \quad \textrm{[using $\bnabla\times\bnabla f = 0$]}\\
            &= \frac{1}{h_u}\bnabla(f_uh_u)\times\mbf{e}_u\\
            &= \frac{1}{h_u}\bnabla(f_uh_u)\times(\mbf{e}_v\times\mbf{e}_w)\\
            &= \frac{1}{h_u}\mbf{e}_v[\mbf{e}_w\cdot\bnabla(f_uh_u)] - \mbf{e}_w[\mbf{e}_v\cdot\bnabla(f_uh_u)]\\
            &= \frac{1}{h_uh_w}\pd{w}(h_uf_u)\mbf{e}_v - \frac{1}{h_uh_v}\pd{v}(h_uf_u)\mbf{e}_w.
        \end{align*}
        The penultimate step used the vector triple product identity.
        Again,
        the other components are similar.
    \end{proof}
\end{proposition}

\section{Integral theorems}

\begin{theorem}[Fundamental Theorem of Calculus]
    If $f'(x)$ is continuous on $[a, b]$ then
    \[
    \int_{a}^{b}f'(t)\,dt = f(b) - f(a).
    \]
\end{theorem}

\begin{theorem}[Fundamental Theorem of Line Integrals]
    If $f : \R ^ n \to \R$ is a $C ^ {1}$ scalar field and $C$ is an oriented curve with parameterisation $x(t)$ for $t \in [t_0, t_1]$,
    then
    \[
    \int_{C}\bnabla\mbf{f} \cdot \hat{\mbf{t}}\,d\ell = f(\mbf{x}(t_1)) - f(\mbf{x}(t_0)).
    \]

    \begin{proof}
        Evaluating the line integral gives
        \begin{align*}
            \int_{C}\bnabla\mbf{f} \cdot \hat{\mbf{t}}\,d\ell &= \int_{t_0}^{t_1}\bnabla f\cdot\frac{d\mbf{x}}{dt}\,dt \\
            &= \int_{t_0}^{t_1}\left(\pd[f]{x}\mbf{e}_1 + \pd[f]{y}\mbf{e}_2 + \pd[f]{z}\mbf{e}_3\right) \cdot \left(\frac{dx}{dt}\mbf{e}_1 + \frac{dy}{dt}\mbf{e}_2 + \frac{dz}{dt}\mbf{e}_3\right)\,dt \\
            &= \int_{t_0}^{t_1}\left(\pd{f}[x]\frac{dx}{dt} + \pd[f]{y}\frac{dy}{dt} + \pd[f]{z}\frac{dz}{dt}\right)\,dt \\
            &= \int_{t_0}^{t_1}\frac{df}{dt}\,dt&(\text{By the Chain rule}) \\
            &= f(\mbf{x}(t_1)) - f(\mbf{x}(t_0))
        \end{align*}
        where the last line uses FTOC.
    \end{proof}
\end{theorem}

\begin{definition}[(scalar)
    potential]
    A vector field $\mbf{f} : \R ^ n \to \R ^ n$ is conservative if it can be written as $\mbf{f} = \bnabla g$ for some $C ^ {1}$ scalar field $g : \R ^ n \to \R$ called a
    (scalar)
    potential.
\end{definition}

\begin{proposition}[Path independence]\label{prep:mathmethII:prop:pathindep}
    A vector field is conservative if and only if it has path independent line integrals between every pair of points.

    \begin{proof}
        Firstly.
        assume $\mbf{f}$ is conservative,
        so $\mbf{f} = \bnabla g$.
        Let $C_1, C_2$ be two curves sharing both endpoints $\mbf{x}_a, \mbf{x}_b$.

        Then by FTOC for line integrals
        \[
        \int_{C_1}\mbf{f}\cdot\hat{\mbf{t}}\,d\ell = g(\mbf{x}_b) - g(\mbf{x}_a) = \int_{C_2}\mbf{f}\cdot\hat{\mbf{t}}\,d\ell.
        \]
        So the integral depends only on the endpoints,
        not the path.
        
        Conversely suppose that $\mbf{f}$ has path independent line integrals.
        Claim that a suitable scalar field is
        \[
        g(\mbf{x}) = \int_{C(\mbf{x})}\mbf{f}\cdot\hat{\mbf{t}}\,d\ell,
        \]
        where $C(\mbf{x})$ is
        (any)
        curve connecting $\mbf{0}$ to $\mbf{x}$.
        To see that $\bnabla g = \mbf{f}$,
        consider one component:
        \[
        \pd[g]{x} = \lim_{\delta \to 0}\frac{1}{\delta}\left[\int_{C(\mbf{x} + \delta \mbf{e}_1)}\mbf{f}\cdot\hat{\mbf{t}}\,d\ell - \int_{C(\mbf{x})}\mbf{f}\cdot\hat{\mbf{t}}\,d\ell\right].
        \]
        By path independence,
        this is the same as integrating along the straight-line segment $C_1$.
        So
        \begin{align*}
            \pd[g]{x} &= \lim_{\delta \to 0}\frac{1}{\delta}\int_{0}^{\delta}f_1(\mbf{x} + t\mbf{e}_1)\,dt \\
            &= \lim_{\delta \to 0}\delta f_1(\mbf{x} + t_{*}\mbf{e}_1)\text{ for some $t_{*} \in [0, \delta]$ by mean value theorem} \\
            &= f_1(x).
        \end{align*}
    \end{proof}
\end{proposition}

\subsection{Divergence Theorem}

\begin{theorem}[Divergence Theorem]
    If $\mbf{f} : \R ^ 3 \to \R ^ 3$ is a $C ^ {1}$ vector field and $S$ is a closed surface with outward normal $\hat{\mbf{n}}$ bounding a region $V$,
    then
    \[
    \oint_{S}\mbf{f}\cdot\hat{\mbf{n}}\,dS = \int_{V}(\bnabla \cdot \mbf{f})\,dV.
    \]

    \begin{proof}
        The idea is to subdivide $V$ into a disjoint union of small domains $V_i$ for $i = 1, \dotsc, m$,
        whose boundaries $S_i$ have normal vectors $\hat{\mbf{n}}_i$.


        Then observe that each segment of interior boundary is shared by two neighbouring domains with equal and opposite normals.
        So these interior surface integrals all cancel and we write
        
        \begin{align*}
            \oint_{S}\mbf{f} \cdot \hat{\mbf{n}}\,dS &= \sum_{i = 1}^{m}\oint_{S_i}\mbf{f} \cdot \hat{\mbf{n}}_i\,dS \\
            &= \sum_{i = 1}^{m}\left(\frac{1}{|V_i|}\oint_{S_i}\mbf{f}\cdot\hat{\mbf{n}}_i\,dS\right)|V_i| \\
            &= \lim_{m \to \infty}\sum_{i = 1}^{m}\left(\frac{1}{|V_i|}\oint_{S_i}\mbf{f} \cdot \hat{\mbf{n}}_i\,dS\right)|V_i| \\
            &= \int_{V}(\bnabla \cdot \mbf{f})\,dV.
        \end{align*}
    \end{proof}
\end{theorem}

\subsection{Stokes' Theorem}

\begin{theorem}[Stokes' Theorem]\label{prep:mathmethII:thm:stokesthm}
    If $\mbf{f} : \R ^ 3 \to \R ^ 3$ is a $C ^ {1}$ vector field and $C$ is an oriented closed curve bounding a surface $S$,
    then
    \[
    \oint_{C}\mbf{f}\cdot\hat{\mbf{t}}\,d\ell = \int_{S}(\bnabla \times \mbf{f})\cdot\hat{\mbf{n}}\,dS,
    \]
    where $\hat{\mbf{n}}$ is chosen such that $\hat{\mbf{t}} \times \hat{\mbf{n}}$ on $C$ is outward.
\end{theorem}

\begin{corollary}[Green's Theorem]
    If $\mbf{f} : \R ^ 2 \to \R ^ 2$ is a $C ^ {1}$ vector field and $C$ is a closed curve in $\R ^ 2$ oriented clockwise bounding a region $A$,
    then
    \[
    \oint_{C}\mbf{f} \cdot \hat{\mbf{t}}\,d\ell = \int_{A}\left(\pd[f_2]{x} - \pd[f_1]{y}\right)\,dxdy.
    \]
\end{corollary}

\begin{corollary}[Green's Theorem]
    If $\mbf{f} : \R ^ 3 \to \R ^ 3$ is a $C ^ {1}$ vector field with $\bnabla \times \mbf{f} = \mbf{0}$ throughout a simply connected region $V \subset \R ^ 3$,
    then $\mbf{f}$ is conservative in $V$.

    \begin{proof}
        From \autoref{prep:mathmethII:prop:pathindep} $\mbf{f} = \bnabla g$ if and only if $\mbf{f}$ has path independent line integrals.
        So consider two arbitrary curves $C_1, C_2$ between two arbitrary endpoints $\mbf{x}_a, \mbf{x}_b \in V$,
        and define $C = C_1 - C_2$,
        being the clockwise curve from $\mbf{x}_a$ to $\mbf{x}_b$ to $\mbf{x}_a$ again,
        not $\mbf{x}_a$ to $\mbf{x}_b$ along $C_1, C_2$.

        Since $V$ is simply connected,
        we can always find a surface $S$ with boundary $C$ that lies wholly within $V$.
        Then $\bnabla \times \mbf{f} = \mbf{0}$ everywhere on that surface so Stokes' Theorem gives
        \[
        \oint_{C}\mbf{f} \cdot \hat{\mbf{t}}\,d\ell = 0 \implies \int_{C_1}\mbf{f}\cdot\hat{\mbf{t}}\,d\ell = \int_{C_2}\mbf{f}\cdot\hat{\mbf{t}}\,d\ell.
        \]
    \end{proof}
\end{corollary}

\subsection{Proof of Stokes' Theorem}
To prove Stokes' theorem we need the following lemma.

\begin{lemma}
    Let $C$ be a curve in the plane having unit normal $\hat{\mbf{n}}$,
    with diameter $d(C)$,
    enclosed area $|A|$,
    and containing the point $\mbf{x}$.
    For any $C ^ {1}$ vector field $\mbf{f} : \R ^ 3 \to \R ^ 3$,
    \[
    \hat{\mbf{n}} \cdot(\bnabla \times \mbf{f})\Bigr\rvert_{\mbf{x}} = \lim_{d(C) \to 0}\frac{1}{|A|}\oint_{C}\mbf{f} \cdot \hat{\mbf{t}}\,d\ell,
    \]
    where $C$ is parameterised such that $\hat{\mbf{t}} = \hat{\mbf{n}} \times \hat{\mbf{n}}_{c}$,
    with $\hat{\mbf{n}}_c$ being the outward normal to the curve.

    \begin{proof}
        We defined $\bnabla \times \mbf{f}$ as a surface integral,
        so extrude $C$ by a
        (small)
        distance $h$ in the $\hat{\mbf{n}}$ direction to form a thin
        "pill"
        surface $S$ with normal $\hat{\mbf{n}}_{s}$.

        By definition of curl,
        \[
        \hat{\mbf{n}} \cdot (\bnabla \times \mbf{f})\Bigr\rvert_{\mbf{x}} = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hat{\mbf{n}} \cdot (\hat{\mbf{n}}_{s} \times \mbf{f})\,dS = \lim_{|A|, h \to 0}\frac{1}{|A|h}\int_{\tilde{S}}\hat{\mbf{f}} \cdot (\hat{\mbf{n}} \times \hat{\mbf{n}}_s)\,dS.
        \]
        The integrals on the top and bottom surfaces vanish because $\hat{\mbf{n}} \times \hat{\mbf{n}}_s = \mbf{0}$,
        leaving only the side
        "ribbon"
        surface $\tilde{S}$,
        on which $\hat{\mbf{n}}_{s} \equiv \hat{\mbf{n}}_{c}$.

        We can parameterise $\tilde{S}$ by $\tilde{\mbf{x}}(t, v) = \mbf{x}_{c}(t) + v\hat{\mbf{n}}$ for $v \in [0, h]$ and $t \in [t_0, t_1]$,
        where $\mbf{x}_{c}(t)$ is a parameterisation of $C$.
        The corresponding Jacobian is
        \[
        \left|\pd[\tilde{\mbf{x}}]{t} \times \pd[\tilde{\mbf{x}}]{v}\right| = \left|\frac{d\mbf{x}_{c}}{dt} \times \hat{\mbf{n}}\right| = \left|\frac{d\mbf{x}_c}{dt}\right|,
        \]
        since the tangent vector $\frac{d\mbf{x}_{c}}{dt}$ is perpendicular to $\hat{\mbf{n}}$.
        Evaluating the surface integral then gives
        \begin{align*}
            \int_{S}\hat{\mbf{f}} \cdot(\hat{\mbf{n}} \times \hat{\mbf{n}}_{c})\,dS &= \int_{t_0}^{t_1}(\hat{\mbf{n}} \times \hat{\mbf{n}}_{c}) \times \left(\int_{0}^{h}\mbf{f}\,dv\right)\left|\frac{d\mbf{x}_c}{dt}\right|\,dt \\
            &= h\int_{t_0}^{t_1}(\hat{\mbf{n}} \times \hat{\mbf{n}}_{c}) \times f(t, v_{*}(t))\left|\frac{d\mbf{x}_c}{dt}\right|\,dt,
        \end{align*}
        where we used the mean value theorem for integrals for each $t$,
        with all of the $v_{*}(t) \in [0, h]$.

        When we take the limit $h \to 0$,
        we have $v_{*}(t) \to 0$ for all $t$,
        so our first equation gives
        \[
        \hat{\mbf{n}} \cdot (\bnabla \times \mbf{f})\Bigr\rvert_{\mbf{x}} = \lim_{|A| \to 0}\frac{1}{|A|}\int_{t_0}^{t_1}(\hat{\mbf{n}} \times \hat{\mbf{n}}_{c}) \cdot \mbf{f}(t, 0)\left|\frac{d\mbf{x}_c}{dt}\right|\,dt = \lim_{d(C) \to 0}\frac{1}{|A|}\oint_{C}\hat{\mbf{t}}\cdot\mbf{f}\,d\ell,
        \]
        where the tangent vector is $\hat{\mbf{t}} = \hat{\mbf{n}} \times \hat{\mbf{n}}_{c}$.
    \end{proof}
\end{lemma}


\begin{theorem}[continues = prep:mathmethII:thm:stokesthm]
    \begin{proof}
        We subdivide $S$ into a disjoint union of small regions $S_i$ for $i = 1, \dotsc, m$,
        whose boundary curves have unit tangent vector $\hat{\mbf{t}}_i$.

        Then observe that each interior boundary segment is shared by two neighbouring regions with equal and opposite tangents.
        So these interior line integrals all cancel and we can write
        \begin{align*}
            \oint_{C}\mbf{f}\cdot\hat{\mbf{t}}\,d\ell &= \sum_{i = 1}^{m}\oint_{S_i}\mbf{f}\cdot\hat{\mbf{t}_i}\,d\ell \\
            &= \sum_{i = 1}^{m}\left(\frac{1}{|A_i|}\oint_{S_i}\mbf{f} \cdot \hat{\mbf{t}}_i\,d\ell\right)|A_i| \\
            &= \lim_{m \to \infty}\sum_{i = 1}^{m}\left(\frac{1}{|A_i|}\oint_{S_i}\mbf{f} \cdot \hat{\mbf{t}}_i\,d\ell\right)|A_i| \\
            &= \int_{S}(\bnabla \times \mbf{f}) \cdot \hat{\mbf{n}}\,dS,
        \end{align*}
        where in the last step we used the previous lemma.
    \end{proof}
\end{theorem}

\section{Index notation}

\subsection{Scalar products}

For simplifying vector equations,
there is a powerful method called index
(or suffix)
notation.

A vector equation such as
\begin{align*}
    \mbf{c} &= \mbf{a} + \mbf{b} \\
    \intertext{is written in index notation as}
    c_i &= a_i + b_i,
\end{align*}
where it is understood that the equation holds for $i = 1, \dotsc, n$,
meaning an arbitrary component in the standard
(Cartesian)
basis $\{\mbf{e}_1, \dotsc, \mbf{e}_n\}$.

\begin{definition}[Free index]
    An index that appears once in each term is called a free index.
\end{definition}

In index notation,
we write the scalar product as
\[
\mbf{a} \cdot \mbf{b} = a_jb_j,
\]
where the repeated index indicates that the term should be summed from $j = 1, \dotsc, n$.
The implied summation is called the
(Einstein)
summation convention.

\begin{definition}[Dummy index]
    An index that appears exactly twice in a term is called a dummy index.
\end{definition}

The scalar product $\mbf{a} \cdot \mbf{b}$ is as a quadratic form
\[
\mbf{a} \cdot \mbf{b} = (a_1, a_2, a_3)\begin{pmatrix}
    1 & 0 & 0 \\ 0 & 1 & 0 \\ 0 & 0 & 1
\end{pmatrix}\begin{pmatrix}
    b_1 \\ b_2 \\ b_3
\end{pmatrix} = a_1b_1 + a_2b_2 + a_3b_3.
\]
In index notation,
this may be written
\[
\mbf{a} \cdot \mbf{b} = \delta_{ij}a_ib_j,
\]
where the components of the identity matrix are given by the Kronecker delta,
defined as
\[
\delta_{ij} = \begin{cases}
    1 &\text{if } i = j, \\
    0 &\text{if } i \neq j.
\end{cases}
\]

\begin{proposition}
    Multiplying an expression with free index $i$ by $\delta_{ij}$ is equivalent to replacing $i \to j$.

    \begin{proof}
        Consider some expression $a_i$ with free index $i$.
        Then if we multiply by $\delta_{ij}$,
        our index $i$ becomes a dummy index,
        and
        \[
        a_i\delta_{ij} = a_1\delta_{1j} = a_2\delta_{2j} + \dotsc + a_n\delta_{nj}.
        \]
        Now $j$ is the free index.
        If $j = 1$,
        then
        \[
        a_i\delta_{i1} = a_1\delta{11} + a_2\delta{21} + \dotsc + a_n\delta{n1} = a_1(1) + a_2(0) + \dotsc + a_n(0) = a_1.
        \]
        Similarly for any $j$,
        we have $a_i\delta_{ij} = a_j$.
    \end{proof}
\end{proposition}

\subsection{Vector products}

In $\R ^ 3$,
we can write vector products in index notation by introducing the alternating tensor
(or Levi-Civita symbol),
\[
\epsilon_{ijk} = \begin{cases}
    0 &\text{if any of $i, j, k$ are equal}, \\
    +1 &\text{if $(i, j, k) = (1, 2, 3), (2, 3, 1)$ or $(3, 1, 2)$}, \\
    -1 &\text{if $(i, j, k) = (1, 3, 2), (2, 1, 3)$ or $(3, 2, 1)$}.
\end{cases}
\]

We can express this product in index notation as
\[
[\mbf{a} \times \mbf{b}]_{i} = \epsilon_{ijk}a_jb_k.
\]

\begin{proposition}[The $\epsilon\text{-}\delta$ identity]
    \[
    \epsilon_{ijk}\epsilon_{klm} = \delta_{il}\delta_{jm} - \delta_{im}\delta_{jl}.
    \]

    \begin{proof}
        The left hand side is
        \[
        \epsilon_{ijk}\epsilon_{klm} = \begin{cases}
            +1 &\text{if $(klm)$ is an even permutation of $(ijk)$}, \\
            -1 &\text{if $(klm)$ is an odd permutation of $(ijk)$}, \\
            0 &\text{if $(klm)$ is not a permutation of $(ijk)$}.
        \end{cases}
        \]
        So there are six possibilities:
        \[
        \epsilon_{ijk}\epsilon_{klm} = \delta_{ik}\delta_{jl}\delta_{km} - \delta_{ik}\delta_{jm}\delta_{kl} + \delta_{il}\delta_{jm}\delta_{kk} - \delta_{il}\delta_{jk}\delta_{km} + \delta_{im}\delta_{jk}\delta_{kl} - \delta_{im}\delta_{jl}\delta_{kk}.
        \]
        Using $\delta_{kk} = 3$ and the previous proposition,
        we find
        \begin{align*}
        \epsilon_{ijk}\epsilon_{klm} &= \delta_{im}\delta_{jl} - \delta_{il}\delta_{jm} + 3\delta_{il}\delta_{jm} - \delta_{il}\delta_{jm} + \delta_{im}\delta_{jl} - 3\delta_{im}\delta_{jl} \\
        &= \delta_{il}\delta_{jm} - \delta_{im}\delta_{jl}.
        \end{align*}
    \end{proof}
\end{proposition}
