\section{Line Integrals}

\subsection{Curves}

A curve $C$ is a $1$D subset of $\R ^ n$,
for $n > 1$.
Usually describe a curve by a parameterisation $\vec{x}(t)$:
\[
\vec{x}(t) = x_1(t)\vec{e}_1 + x_2(t)\vec{e}_2 + \dotsc + x_n(t)\vec{e}_n
\]
which maps an interval $[t_0, t_1]$ into $\R ^ n$.
\textit{Think of $\vec{x}(t)$ as "trajectory" of a particle with $t$ being time.}

\begin{example}
    Find a parameterisation $\vec{x}(t)$ for the circle $x ^ 2 + y ^ 2 = a ^ 2$ in $\R ^ 2$.

    \begin{solution}
        Recall
        \[
        \begin{cases}
            x = r\cos{\theta} \\
            y = r\sin{\theta}
        \end{cases}
        \]
        so
        \[
        x ^ 2 + y ^ 2 = a ^ 2 \implies r ^ 2(\cos ^ 2{\theta} + \sin ^ 2{\theta}) = a ^ 2 \implies r = a.
        \]
        So we can set $t = \theta$ and obtain
        \[
        \vec{x}(t) = a\cos(t)\vec{e}_1 + a\sin(t)\vec{e}_2\quad\text{for } t \in [0, 2\pi].
        \]
    \end{solution}
\end{example}

A curve is closed if $\vec{x}(t_1) = \vec{x}(t_0)$.

A curve is simple if it doesn't cross/intersect itself.

\begin{example}
    Is the helix $\vec{x}(t) = \cos(t)\vec{e}_1 + \sin(t)\vec{e}_2 + t\vec{e}_3$ for $t \in [0, 6\pi]$ closed?
    Or simple?

    \begin{solution}
        $\vec{x}(0) = \vec{e}_1 \neq \vec{e}_1 + 6\pi\vec{e}_2 = \vec{x}(6\pi)$ so it is not closed,
        i.e. open.
        No self-intersections so it is simple.
    \end{solution}
\end{example}

\begin{example}
    Is the "clover" $\vec{x}(t) = \cos(3t)\cos(t)\vec{e}_1 + \cos(3t)\sin(t)\vec{e}_2$ for $t \in [0, \pi]$ closed? Or simple?

    \begin{solution}
        $\vec{x}(0) = \vec{e}_1 = \vec{e}_1 = \vec{x}(\pi)$ so it is closed.
        It is not simple because of the triple intersection at $\vec{x} = \vec{0}$,
        $\vec{x}\left(\frac{\pi}{6}\right) = \vec{x}\left(\frac{\pi}{2}\right) = \vec{x}\left(\frac{5\pi}{6}\right) = \vec{0}$.
    \end{solution}
\end{example}

A parameterisation $\vec{x}(t)$ is differentiable if it has a well-defined tangent vector
\[
\frac{d\vec{x}}{dt} = \lim_{dt \to 0}\frac{\vec{x}(t + dt) - \vec{x}(t)}{dt}
\]
points along the curve.

If $\vec{x}(t)$ is a trajectory then $\frac{d\vec{x}}{dt}$ is its velocity,
and $\left|\frac{d\vec{x}}{dt}\right|$ its speed.

\textit{Note:
the parameterisation of a curve is not unique
(could have opposite direction,
or different speed).}

\begin{example}
    Find the tangent vectors for
    \begin{enumerate}[label = (\roman*)]
        \item $\vec{x}(t) = \vec{a} + t(\vec{b} - \vec{a})$ for $t \in [0, 1]$.

        \item $\vec{x}(\tau) = \vec{a} + (\vec{b} - \vec{a})\sin{\tau}$ for $\tau \in \left[0, \frac{\pi}{2}\right]$.
    \end{enumerate}

    \begin{solution}
        \begin{enumerate}[label = (\roman*)]
            \item
            \[
            \frac{d\vec{x}}{dt} = \vec{b} - \vec{a}.
            \]

            \item
            \[
            \frac{d\vec{x}}{d\tau} = (\vec{b} - \vec{a})\cos{\tau}.
            \]
        \end{enumerate}

        They both describe the same straight line from $\vec{a}$ to $\vec{b}$
        (but with different speeds)
        because $\frac{d\vec{x}}{dt}$ and $\frac{d\vec{x}}{dt}$ have the same direction.

        If $\vec{x}(t)$ and $\vec{x}(\tau)$ are two parameterisation of the same curve,
        the Chain Rule gives
        \[
        \frac{d\vec{x}}{dt} = \frac{d\vec{x}}{d\tau}\frac{d\tau}{dt}.
        \]
    \end{solution}
\end{example}

\begin{example}
    Verify the chain rule formula for the two parametrisation in the previous example.

    At a given point on the curve,
    \[
    \vec{x}(t) = \vec{x}(\tau) \iff \vec{a} + t(\vec{b} - \vec{a}) = \vec{a} + (\vec{b} - \vec{a})\sin{\tau} \implies t = \sin{\tau} \implies \frac{dt}{d\tau}\cos{\tau}.
    \]
    Then
    \[
    \frac{d\vec{x}}{dt} = \frac{d\vec{x}}{d\tau}\frac{1}{\frac{dt}{d\tau}} = \frac{1}{\cos{\tau}}\frac{d\vec{x}}{dt}.
    \]
\end{example}

\begin{remark}
    A parameterisation $\vec{x}(t)$ is called regular if $\frac{d\vec{x}}{dt} \neq \vec{0}$ everywhere
    (otherwise it is singular).
    
    A curve is called regular if it has at least one regular parameterisation.
\end{remark}

\begin{example}
    Hypocycloid $\vec{x}(t) = \cos ^ 3(t)\vec{e}_1 + \sin ^ 3(t)\vec{e}_2$ for $t \in [0, 2\pi]$.

    Tangent vector:
    \[
    \frac{d\vec{x}}{dt} = -3\cos ^ 2(t)\sin(t)\vec{e}_1 + 3\sin ^ 2(t)\cos(t)\vec{e}_2.
    \]
    Vanishes when either $\sin(t) = 0$ or $\cos(t) = 0$ so $t = 0, \pi / 2, \pi, 3\pi / 2$ and $\vec{x}(0) = (1, 0)$,
    $\vec{x}\left(\frac{\pi}{2}\right) = (0, 1)$,
    $\vec{x}(\pi) = (-1, 0)$,
    $\vec{x}\left(\frac{3\pi}{2}\right) = (0, -1)$.
\end{example}

\subsection{Line integrals of scalar fields}

Recall:
a scalar field on $\R ^ n$ is a map $f : \R ^ n \to \R$
(assigns a real number to every point).

The line integral of a scalar field $f(\vec{x})$ along a curve $C$ with parameterisation $\vec{x}(t)$ for $t \in [t_0, t_1]$ is
\[
\int_{C}f\,d\ell = \int_{t_0}^{t_1}f(\vec{x}(t))\left|\frac{d\vec{x}}{dt}\right|\,dt.
\]

\begin{example}
    Integrate $f(\vec{x}) = y + x ^ 2$ around the unit circle.

    Parameterisation:
    \[
    \vec{x}(t) = \cos(t)\vec{e}_1 + \sin(t)\vec{e}_2 \text{ for } t \in [0, 2\pi].
    \]

    Tangent vector:
    \[
    \frac{d\vec{x}}{dt} = -\sin(t)\vec{e}_1 + \cos(t)\vec{e}_2 \implies \left|\frac{d\vec{x}}{dt}\right| = \sqrt{\sin ^ 2 t + \cos ^ 2 t} = 1
    \]
    so
    \begin{align*}
        \int_{C}f\,d\ell &= \int_{0}^{2\pi}(y + x ^ 2)(1)\,dt \\
        &= \int_{0}^{2\pi}(\sin(t) + \cos ^ 2(t))\,dt \\
        &= \int_{0}^{2\pi}\cos ^ 2t\,dt \\
        &= \frac{1}{2}\int_{0}^{2\pi}(1 + \cos(2t))\,dt \\
        &= \frac{1}{2}\left[t\right]_{0}^{2\pi} = \pi.
    \end{align*}
    \[
    \]
\end{example}

\begin{proposition}
    The line integral of a scalar field is independent of the choice of parameterisation.

    \begin{proof}
        Suppose we have two parameterisations of the same curve
        \[
        \begin{cases}
            \vec{x}(t) &\text{for } t \in [t_0, t_1] \\
            \vec{x}(\tau) &\text{for } \tau \in [\tau_0, \tau_1].
        \end{cases}
        \]
        By the chain rule
        \[
        \frac{d\vec{x}}{dt} = \frac{d\tau}{dt}\frac{d\vec{x}}{dt} \implies \left|\frac{d\vec{x}}{dt}\right| = \left|\frac{d\tau}{dt}\right|\left|\frac{d\vec{x}}{d\tau}\right|.
        \]

        There are now two cases to consider:

        Case $1$:
        \[
        \vec{x}(t_0) = \vec{x}(\tau_0)
        \]
        i.e. both have same direction.
        Then $\frac{dt}{d\tau} > 0$,
        so
        \[
        \int_{t_0}^{t_1}f(\vec{x}(t))\left|\frac{d\vec{x}}{dt}\right|\,dt = \int_{t_0}^{t_1}f(\vec{x}(t))\frac{d\tau}{dt}\left|\frac{d\vec{x}}{dt}\right|\,dt = \int_{\tau_0}^{\tau_1}f(\vec{x}(\tau))\left|\frac{d\vec{x}}{d\tau}\right|\,d\tau.
        \]
        
        Case $2$:
        $\vec{x}(t_0) = \vec{x}(\tau_1)$ i.e. opposite directions.
        Then $\frac{dt}{d\tau} < 0$,
        so
        \begin{align*}
            \int_{t_0}^{t_1}f(\vec{x}(t))\left|\frac{d\vec{x}}{dt}\right|\,dt &= \int_{t_0}^{t_1}f(\vec{x}(t))\left(-\frac{d\tau}{dt}\right)\left|\frac{d\vec{x}}{d\tau}\right|\,dt \\
            &= -\int_{\tau_1}^{\tau_0}f(\vec{x}(\tau))\left|\frac{d\vec{x}}{d\tau}\right|\,d\tau \\
            &= \int_{\tau_0}^{\tau_1}f(\vec{x}(\tau))\left|\frac{d\vec{x}}{d\tau}\right|\,d\tau.
        \end{align*}
    \end{proof}
\end{proposition}

For the special case $f(\vec{x}) \equiv 1$,
the line integral is called the arclength of the curve,
$L = \int_{C}d\ell = \int_{t_0}^{t_1}\left|\frac{d\vec{x}}{dt}\right|\,dt$.

\begin{remark}
    There is always an arclength parameterisation $\vec{x}(s)$ where $\left|\frac{d\vec{x}}{ds}\right| \equiv 1$ all along the curve.
    Then $s$ is the distance and $L = \int_{0}^{L}\,ds$.
\end{remark}

\begin{example}
    Calculate the arclength of the circle $x ^ 2 + y ^ 2 = a ^ 2$.

    \begin{solution}
        Parameterisation:
        $\vec{x}(t) = a\cos(t)\vec{e}_1 + a\sin(t)\vec{e}_2$ for $t \in [0, 2\pi]$.
        \[
        \frac{d\vec{x}}{dt} = -a\sin(t)\vec{e}_1 + a\cos(t)\vec{e}_2 \implies \left|\frac{d\vec{x}}{dt}\right| = a.
        \]
        So
        \[
        L = \int_{0}^{2\pi}\left|\frac{d\vec{x}}{dt}\,dt\right| = \int_{0}^{2\pi}a\,dt = 2\pi a.
        \]
    \end{solution}
\end{example}

\begin{example}
    Calculate the arclength of the hypocycloid $\vec{x}(t) = \cos ^ 3(t)\vec{e}_1 + \sin ^ 2(t)\vec{e}_2$ for $t \in [0, 2\pi]$.
    \begin{align*}
        \frac{d\vec{x}}{dt} &= -3\cos ^ 2(t)\sin(t)\vec{e}_1 + 3\sin ^ 2(t)\cos(t)\vec{e}_2 \\
        \implies \left|\frac{d\vec{x}}{dt}\right| &= \sqrt{9\cos ^ 4(t)\sin ^ 2(t) + 9\sin ^ 4(t)\cos ^ 2(t)} \\
        &= 3\sqrt{\cos ^ 2(t)\sin ^ 2(t)}\sqrt{\cos ^ 2t + \sin ^ 2t} \\
        &= 3|\cos(t)\sin(t)|.
    \end{align*}
    By symmetry,
    we only need to integrate over $t \in [0, \pi / 2]$,
    so
    \begin{align*}
        L &= \int_{0}^{2\pi}3|\cos(t)\sin(t)|\,dt \\
        &= 4\int_{0}^{\pi / 2}3\cos(t)\sin(t)\,dt \\
        &= 6\int_{0}^{\pi / 2}2\cos(t)\sin(t)\,dt \\
        &= 6\int_{0}^{\pi / 2}\sin(2t)\,dt \\
        &= 6\left[-\frac{1}{2}\cos(2t)\right]_{0}^{\pi / 2} = 6.
    \end{align*}
\end{example}

\subsection{Line integrals of vector fields}
A vector field is a function $\vec{f} : \R ^ n \to \R ^ n$ that assigns a vector to each point in $\R ^ n$.

For a curve with parameterisation $\vec{x}(t)$ for $t \in [t_0, t_1]$,
the unit tangent vector is:
\[
\hvec{t} = \frac{\frac{d\vec{x}}{dt}}{\left|\frac{d\vec{x}}{dt}\right|}
\]
\textit{"Only depends on direction,
not speed".}

An oriented curve is a curve together with a specified
(consistent)
choice of $\hvec{t}$.

The line integral of a vector field $\vec{f}(\vec{x})$ along an oriented curve $C$ with parameterisation $\vec{x}(t)$ for $t \in [t_0, t_1]$ is defined as
\[
\int_{C}\vec{f}\cdot\hvec{t}\,d\ell = \int_{t_0}^{t_1}\vec{f}(\vec{x}(t))\cdot\hvec{t}\left|\frac{d\vec{x}}{dt}\right|\,dt = \int_{t_0}^{t_1}\vec{f}(\vec{x}(t))\frac{d\vec{x}}{dt}\,dt
\]
the scalar line integral of the tangential component of $\vec{f}$.

\textbf{Warning:}
Changing the orientation $\hvec{t} \to -\hvec{t}$ will flip the sign of the integral.

\begin{example}
    Evaluate $\int_{C}\vec{f}\cdot\hvec{t}d\ell$ for $\vec{f}(\vec{x}) = -\vec{x}$ and $C$ the straight line from $(1, 0)$ to $(2, 0)$.

    Parameterisation:
    $\vec{x}(t) = (1 + t)\vec{e}_1$ for $t \in [0, 1]$.
    \[
    \frac{d\vec{x}}{dt} = \vec{e}_1.
    \]
    Then
    \[
    \int_{C}\vec{f}\cdot\hvec{t}\,d\ell = \int_{0}^{1}(-x\vec{e}_1 - y\vec{e}_2) \cdot \vec{e}_1\,dt = \int_{0}^{1}x\,dt = -\int_{0}^{1}(1 + t)\,dt = -\left[t + \frac{1}{2}t ^ 2\right]_{0}^{1} = -\frac{3}{2}.
    \]
\end{example}

If $C$ is closed we usually write
\[
\int_{C}\vec{f}\cdot\hvec{t}\,d\ell = \oint_{C}\vec{f}\cdot\hvec{t}\,d\ell
\]
called the circulation of $\vec{f}$ around $C$.

\begin{example}
    Evaluate $\oint_{C}\vec{f}\cdot\hvec{t}\,d\ell$ for $\vec{f}(\vec{x}) = -\vec{x}$ and $C$ the anti-clockwise unit circle.
    
    \begin{solution}
        Parameterisation:
        \[
        \vec{x}(t) = \cos(t)\vec{e}_1 + \sin(t)\vec{e}_2\qquad\text{for } t \in [0, 2\pi].
        \]
        \[
        \frac{d\vec{x}}{dt} = -\sin(t)\vec{e}_1 + \cos(t)\vec{e}_2.
        \]
        So
        \begin{align*}
            \oint_{C}\vec{f}\cdot\hvec{t}\,d\ell &= \int_{0}^{2\pi}(-x\vec{e}_1 - y\vec{e}_2)\cdot \frac{d\vec{x}}{dt}\,dt \\
            &= \int_{0}^{2\pi}(-\cos(t)\vec{e}_1 - \sin(t)\vec{e}_2)\cdot(-\sin(t)\vec{e}_1 + \cos(t)\vec{e}_2)\,dt \\
            &= \int_{0}^{2\pi}(\cos(t)\sin(t) - \sin(t)\cos(t))\,dt = 0.
        \end{align*}
    \end{solution}
\end{example}

\begin{example}
    Evaluate $\oint_{C}\vec{g}\cdot\hvec{t}\,d\ell$ for $\vec{g}(\vec{x}) = (y - x)\vec{e}_1 - (x + y)\vec{e}_2$ and $C$ the anti-clockwise unit circle.

    \begin{solution}
        \begin{align*}
            \oint_{C}\vec{g}\cdot\hvec{t}\,d\ell &= \int_{0}^{2\pi}\left[(\sin{t} - \cos{t})\vec{e}_1 - (\cos{t} + \sin{t})\vec{e}_2\right] \cdot \left[-\sin(t)\vec{e}_1 + \cos(t)\vec{e}_2\right] \\
            &= \int_{0}^{2\pi}(-\sin ^ 2{t} + \cos(t)\sin(t) - \cos ^ 2{t} - \sin(t)\cos(t))\,dt \\
            &= -\int_{0}^{2\pi}\,dt = -2\pi.
        \end{align*}
    \end{solution}
\end{example}

\begin{remark}
    \begin{enumerate}[label = (\roman*)]
        \item Alternative notation for vector line integrals:
        \[
        \int_{C}\vec{f}\cdot\hvec{t}\,d\ell = \int_{C}\vec{f}\cdot\vec{dx}.
        \]

        \item A physical example is the work done by a particle moving in a force-field $\vec{F}(\vec{x})$ under $m\ddot{\vec{x}} = \vec{F}(\vec{x})$.
        \[
        [\text{work done}] = K(T) - K(0) = \int_{0}^{T}\frac{dK}{dt}\,dt = \int_{0}^{T}m\dot{\vec{x}}\cdot\ddot{\vec{x}}\,dt = \int_{0}^{T}\vec{F}\cdot\dot{\vec{x}}\,dt = \int_{C}\vec{F}\cdot\hvec{t}\,d\ell
        \]
        change in kinetic energy
        \[
        K = \frac{1}{2}m|\dot{\vec{x}}| ^ 2 = \frac{1}{2}m\dot{\vec{x}}\cdot\dot{\vec{x}}.
        \]
    \end{enumerate}
\end{remark}

\subsection{Curvilinear Coordinates}
A system of curvilinear coordinates $(u,, v, w)$ for $\R ^ 3$ has coordinate lines $\vec{x}(u, v_0, w_0)$,
$\vec{x}(u_0, v, w_0)$,
$\vec{x}(u_0, v_0, w)$
which are curves with tangent vectors
\[
\pd[\vec{x}]{u}, \pd[\vec{x}]{v}, \pd[\vec{x}]{w}.
\]
The coordinate system is called orthogonal if the three tangent vectors are mutually orthogonal everywhere.
Then we get an orthonormal basis
\[
\vec{e}_u = \frac{1}{\left|\pd[\vec{x}]{u}\right|}\pd[\vec{x}]{u}\qquad\vec{e}_v = \frac{1}{\left|\pd[\vec{x}]{v}\right|}\pd[\vec{x}]{v}\qquad\vec{e}_w = \frac{1}{\left|\pd[\vec{x}]{w}\right|}\pd[\vec{x}]{w}.
\]
The scale factors $h_u, h_v, h_w$ are the bottom parts of the fractions.

Geometrically,
the scale factors tell us how much an infinitesimal region in $(u, v, w)$-space is stretched at different points in $\R ^ 3$.

\begin{example}
    Find the unit basis vectors for
    (plane)
    polar coordinates $\vec{x}(r, \theta) = r\cos(\theta)\vec{e}_1 + r\sin(\theta)\vec{e}_2$.

    \begin{solution}
        Differentiate to find coordinate tangent vectors:
        \[
        \pd[\vec{x}]{r} = \cos(\theta)\vec{e}_1 + \sin(\theta)\vec{e}_2\qquad\pd[\vec{x}]{\theta} = -r\sin(\theta)\vec{e}_1 + r\cos(\theta)\vec{e}_2.
        \]
        Scale factors:
        \[
        h_r = \left|\pd[\vec{x}]{r}\right| = \sqrt{\cos ^ 2\theta + \sin ^ 2\theta} = 1.
        \]
        \[
        h_{\theta} = \left|\pd[\vec{x}]{\theta}\right| = \sqrt{r ^ 2\cos ^ 2\theta + r ^ 2\sin ^ 2\theta} = r.
        \]
        So the basis vectors are
        \begin{align*}
            \vec{e}_r &= \frac{1}{h_r}\pd[\vec{x}]{r} = \cos(\theta)\vec{e}_1 + \sin(\theta)\vec{e}_2 \\
            \vec{e}_{\theta} &= \frac{1}{h_{\theta}}\pd[\vec{x}]{\theta} = -\sin(\theta)\vec{e}_1 + \cos(\theta)\vec{e}_2.
        \end{align*}
        (These are normal to $r = \text{constant}$ and $\theta = \text{constant}$ respectively)
    \end{solution}
\end{example}

Need to learn the two most common non-Cartesian systems in $\R ^ 3$
\begin{enumerate}[label = (\roman*)]
    \item Cylindrical coordinates:
    ($r > 0$,
    $\theta \in [0, 2\pi]$,
    $z \in \R$)
    \[
    \vec{x}(r, \theta, z) = r\cos(\theta)\vec{e}_1 + r\sin(\theta)\vec{e}_2 + z\vec{e}_3.
    \]
    Scale factors:
    $h_r = 1$,
    $h_{\theta} = r$,
    $h_z = 1$.
    Orthogonal.

    \item Spherical coordinates:
    ($r > 0$,
    $\theta \in [0, \pi)$,
    $\phi \in [0, 2\pi]$)
    \[
    \vec{x}(r, \theta, \phi) = r\sin(\theta)\cos(\phi)\vec{e}_1 + r\sin(\theta)\sin(\phi)\vec{e}_2 + r\cos(\theta)\vec{e}_3.
    \]
    Scale factors:
    $h_r = 1$,
    $h_{\theta} = r$,
    $h_{\theta} = r\sin{\theta}$.

    To derive the parameterisation:

    Side view of sphere and draw $z$ from top to bottom then derive using trigonometry and the radius line.

    Then do a top view with $y$ drawn as usual,
    use trigonometry to derive the others.
\end{enumerate}

\begin{example}
    Write the vector field $\vec{g}(\vec{x}) = (y - x)\vec{e}_1 - (x + y)\vec{e}_2$ in the cylindrical basis $\{\vec{e}_r, \vec{e}_{\theta}, \vec{e}_z\}$.

    \begin{solution}
        We found
        \begin{align*}
            \vec{e}_r &= \cos(\theta)\vec{e}_1 + \sin(\theta)\vec{e}_2 \\
            \vec{e}_{\theta} &= -\sin(\theta)\vec{e}_1 + \cos(\theta)\vec{e}_2 \\
            \vec{e}_z &= \vec{e}_3.
        \end{align*}
        We get
        \begin{align*}
            \vec{e}_1 &= \cos(\theta)\vec{e}_r - \sin(\theta)\vec{e}_{\theta} \\
            \vec{e}_2 &= -\sin(\theta)\vec{e}_r + \cos(\theta)\vec{e}_{\theta} \\
            \vec{e}_3 &= \vec{e}_z.
        \end{align*}
        So
        \begin{align*}
            \vec{g}(\vec{x}) &= (r\sin{\theta} - r\cos{\theta})(\cos(\theta)\vec{e}_r - \sin(\theta)\vec{e}_{\theta}) - (r\cos{\theta} + r\sin{\theta})(\sin(\theta)\vec{e}_r + \cos(\theta)\vec{e}_{\theta}) \\
            &= -r(\vec{e}_r + \vec{e}_{\theta}).
        \end{align*}
    \end{solution}
\end{example}

\section{Surface and Volume Integrals}

\subsection{Surfaces}

A $2$D surface embedded in $\R ^ 3$ can be defined parametrically as a map $\vec{x}(u, r)$ from $R ^ 2 \to \R ^ 3$.

For fixed $v = v_0$,
position $\vec{x}(u, v_0)$ is a curve with tangent vector $\pd[\vec{x}]{u}$
(and vice versa).

If $\vec{x}(u, v)$ is a well-defined parameterisation then $\pd[\vec{x}]{u}, \pd[\vec{x}]{v}$ are not collinear so we define the unit normal
\[
\hvec{n} = \frac{\pd[\vec{x}]{u}\times\pd[\vec{x}]{r}}{\left|\pd[\vec{x}]{u}\times\pd[\vec{x}]{r}\right|}.
\]
The sign of $\hvec{n}$ depends on the ordering of $\pd[\vec{x}]{u}$,
$\pd[\vec{x}]{v}$.
A surface together with a specified
(consistent)
choice of $\hvec{n}$ is called oriented.

\begin{example}
    Find the outward unit normal for the sphere $x ^ 2 + y ^ 2 + z ^ 2 = a ^ 2$.

    From spherical coordinates,
    $\hvec{n} = \vec{e}_r$
    (the basis vector).
    (Could derive this from $\pd[\vec{x}]{\theta} \times \pd[\vec{x}]{\phi} = (h_{\theta}\vec{e}_{\theta}) \times (h_{\phi}\vec{e}_{\phi}) = h_{\theta}h_{\phi}(\vec{e}_{\theta} \times \vec{e}_{\phi}) = h_{\theta}h_{\phi}\vec{e}_r$.)
\end{example}

\begin{example}
    Find the outward unit normal for the torus
    \[
    \vec{x}(u, v) = (R + \cos{v})\cos(u)\vec{e}_1 + (R + \cos{v})\sin(u)\vec{e}_2 + \sin(v)\vec{e}_3\qquad\text{for } u, v \in [0, 2\pi]
    \]
    with $R > 1$ constant.


    Tangent vectors:
    \begin{align*}
        \pd[\vec{x}]{u} &= -(R + \cos{v})\sin(u)\vec{e}_1 + (R +\cos{v})\cos(u)\vec{e}_2 \\
        \pd[\vec{x}]{v} &= -\sin(v)\cos(u)\vec{e}_1 - \sin(v)\sin(u)\vec{e}_2 + \cos(v)\vec{e}_3
    \end{align*}
    \begin{align*}
        \pd[\vec{x}]{u} \times \pd[\vec{x}]{v} &= \begin{vmatrix}
            \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
            -(R + \cos{v})\sin{u} & (R + \cos{v})\cos{u} & 0 \\
            -\sin(v)\cos(u) & -\sin(v)\sin(u) & \cos{v}
        \end{vmatrix} \\
        &=\vec{e}_1((R + \cos{v})\cos(u)\cos(V) - 0) + \vec{e}_2(0 + (R + \cos{v})\sin(u)\cos(v)) \\
        &+ \vec{e}_3((R + \cos{v})\sin(v)\sin ^ 2(u) + (R + \cos{v})\cos ^ 2(u)\sin(v) \\
        &= (R + \cos{v})(\cos(u)\cos(v)\vec{e}_1 + \sin(u)\cos(v)\vec{e}_2 + (\sin ^ 2u + \cos ^ 2u)\sin(v)\vec{e}_3.
    \end{align*}
    Magnitude:
    \begin{align*}
        \left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| &= |R + \cos{v}|\sqrt{\cos ^ 2(u)\cos ^ 2(v) + \sin ^ 2(u)\cos ^ 2(v) + \sin ^ 2(v)} \\
        &= |R + \cos{v}|\sqrt{\cos ^ 2{v} + \sin ^ 2{v}} = |R + \cos{v}| = R + \cos{v}\footnotemark
    \end{align*}
    \footnotetext{No absolute as $R > 1$ and $|\cos{x}| \leq 1$}.

    So unit normal is
    \[
    \hvec{n} = \frac{\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}}{\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|} = \cos(u)\cos(v)\vec{e}_1 + \sin(u)\cos(v)\vec{e}_2 + \sin(v)\vec{e}_3.
    \]

    Putting in $u = 0$,
    $v = 0$,
    then
    $\hvec{n} = \vec{e}_1$
    (so outward $v$).
\end{example}

A surface is smooth if $\hvec{n}$ varies continuously over the surface.
piecewise-smooth if it can be divided into finitely-many smooth pieces.

\begin{example}
    Sketch the surface $\vec{x}(r, \theta) = r\cos(\theta)\vec{e}_1 + r\sin(\theta)\vec{e}_2 + r\vec{e}_3$ for $r \in [0, \infty)$,
    $\theta \in [0, 2\pi]$,
    and determine whether or not it is smooth.

    \begin{solution}
        To sketch the surface,
        note that $x ^ 2 + y ^ 2 = z ^ 2$.
        This is a cone with its point at $\vec{x} = \vec{0}$.

        \ifplots
        \begin{figure}[h!]
             \begin{tikzpicture}
                 \begin{axis}
                 [
                 view={130}{30},
                 z buffer=sort,
                 ticks=none,
                 ]
                 \addplot3[surf,shader=interp,
                     domain=0:2*pi,
                     y domain = 0:3,
                     samples=40,
                     samples y=20,
                     colormap/cool,
                     shader=interp,
                 ]
                 ({y*cos(deg(x))},
                 {y*sin(deg(x))},
                 {y});
                 \end{axis}
             \end{tikzpicture}
        \end{figure}
        \fi

        To check smoothness,
        compute the normal:
        \[
        \pd[\vec{x}]{r} \times \pd[\vec{x}]{\theta} = \begin{vmatrix}
            \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
            \cos{\theta} & \sin{\theta} & 1 \\
            -r\sin{\theta} & r\cos{\theta} & 0
        \end{vmatrix} = -r\cos(\theta)\vec{e}_1 - r\sin(\theta)\vec{e}_2 + r\vec{e}_3
        \]
        which vanishes at $r = 0$ which means $\hvec{n}$ is not defined at $r = 0$ so the surface is not smooth.
    \end{solution}
\end{example}

A surface is open if it is bounded by a curve
(written $\partial S$),
or closed if it has no boundary.

\begin{example}
    Find the upward unit normal to the surface $z = h(x, y)$.

    \begin{solution}
        Parameterisation:
        \[
        \vec{x}(x, y) = x\vec{e}_1 + y\vec{e}_1 + h(x, y)\vec{e}_3
        \]
        so
        \[
        \pd[\vec{x}]{x} \times \pd[\vec{x}]{y} = \begin{vmatrix}
            \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
            1 & 0 & \pd[h]{x} \\
            0 & 1 & \pd[h]{y}
        \end{vmatrix} = -\pd[h]{x}\vec{e}_1 - \pd[h]{y}\vec{e}_2 + \vec{e}_3
        \]
        therefore the unit normal
        \[
        \hvec{n} = \frac{-\pd[h]{x}\vec{e}_1 - \pd[h]{y}\vec{e}_2 + \vec{e}_3}{\sqrt{1 + \left(\pd[h]{x}\right) ^ 2 + \left(\pd[h]{y}\right) ^ 2}}
        \]
    \end{solution}
\end{example}

\subsection{Surface Integrals of Scalar Fields}
If $f : \R ^ 3 \to \R$ is a scalar field and $\mathcal{S}$ is a surface with parameterisation $\vec{x}(u, v)$ then we define the surface integral of $f$ over $\mathcal{S}$ to be
\[
\int_{\mathcal{S}}f\,d\mathcal{S} = \int_{\mathcal{U}}f(\vec{x}(u, v))\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|\,dudv
\]
$\mathcal{U}$ is the preimage of $\mathcal{S}$ in the $(u, v)$-plane.

Idea:
the area element $\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|\,dudv$ is the area on $\mathcal{S}$ corresponding to an infinitesimal rectangle with sides $(du, dv)$ in the $(u, v)$-plane.

\begin{example}
    Find the area element for the plane $z = ax + b$ when parameterised by $(x, y)$.

    \begin{solution}
        Parameterisation:
        \[
        \vec{x}(x, y) = x\vec{e}_1 + y\vec{e}_2 + (ax + b)\vec{e}_3
        \]
        with
        \[
        \pd[\vec{x}]{x} \times \pd[\vec{x}]{y} = -\pd{x}(ax + b)\vec{e}_1 - \pd{y}(ax + b)\vec{e}_2 + \vec{e}_3 = -a\vec{e}_1 + \vec{e}_3.
        \]
        So the area element is 
        \[
        \left|\pd[\vec{x}]{x} \times \pd[\vec{x}]{y}\right|\,dxdy = \sqrt{1 + a ^ 2}\,dxdy
        \]
    \end{solution}
\end{example}

\begin{proposition}
    If $(u, v)$ are two of a set of orthogonal curvilinear coordinates,
    then
    \[
    \left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| = h_uh_v.
    \]

    \begin{proof}
        Recall $\pd[\vec{x}]{u} = h_u\vec{e}_u$ and $\pd[\vec{x}]{v} = h_V\vec{e}_v$ so
        \[
        \left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| = |h_uh_v(\vec{e}_u \times \vec{e}_v)| = h_uh_v|\vec{e}_u \times \vec{e}_v| = h_uh_v|\pm\vec{e}_w| = h_uh_v.
        \]
    \end{proof}
\end{proposition}

\textit{Note:}
\begin{enumerate}[label = (\textit{\arabic*})]
    \item
    \textit{The value of $\int_{\mathcal{S}}f\,d\mathcal{S}$ does not depend on the sign of $\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}$,
    i.e. orientation of the surface doesn't matter.}
    
    \item
    \textit{The surface area of $\mathcal{S}$ is $|\mathcal{S}| = \int_{\mathcal{S}}\,d\mathcal{S}$
    ($f(\vec{x}) \equiv 1$)
    (similarly to arclength).}
\end{enumerate}

\begin{example}
    Find the surface area of the sphere $x ^ 2 + y ^ 2 + z ^ 2 = a ^ 2$.

    \begin{solution}
        Parametrise with spherical coordinates,
        since $r = a$
        (constant).
        \[
        \vec{x}(\theta, \phi) = a\sin(\theta)\cos(\phi)\vec{e}_1 + a\sin(\theta)\sin(\phi)\vec{e}_2 + a\cos(\theta)\vec{e}_3
        \]
        for $\theta \in [0, \pi]$,
        $\phi \in [0, 2\pi]$.

        Recall $h_{\theta} = a$,
        $h_{\phi} = a\sin{\theta}$.

        So we have
        \[
        \left|\pd[\vec{x}]{\theta} \times \pd[\vec{x}]{\phi}\right| = h_{\theta}h_{\phi} = a ^ 2\sin{\theta}.
        \]
        Therefore
        \begin{align*}
            |S| &= \int_{\mathcal{S}}\,d\mathcal{S} \\
            &= \int_{0}^{2\pi}\left(\int_{0}^{\pi}a ^ 2\sin{\theta}\,d\theta\right)\,d\phi \\
            &= a ^ 2 \int_{0}^{2\pi}\left[-\cos{\theta}\right]_{0}^{\pi}\,d\phi \\
            &= 2a ^ 2\int_{0}^{2\pi}\,d\phi = 4\pi a ^ 2.
        \end{align*}
    \end{solution}
\end{example}

\begin{example}
    Find the surface area of the torus from a previous example.
    We had $\vec{x}(u, v)$ for $u, v \in [0, 2\pi]$ and $\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| = R + \cos{v}$.
    So
    \begin{align*}
        |\mathcal{S}| &= \int_{0}^{2\pi}\left(\int_{0}^{2\pi}(R + \cos{v})\,du\right)\,dv \\
        &= 2\pi\int_{0}^{2\pi}(R + \cos{v})\,dv = 4\pi ^ 2R.
    \end{align*}
\end{example}

\begin{proposition}
    The surface integral of a scalar field is independent of the choice of parameterisation.

    \begin{proof}
        Suppose $\vec{x}(u, v)$ and $\vec{x}(\mu, \nu)$ parametrise the same surface.
        \begin{align*}
            \pd[\vec{x}]{u} \times \pd[\vec{x}]{v} &= \left(\pd[\vec{x}]{\mu}\pd[\mu]{u} + \pd[\vec{x}]{\nu}\pd[\nu]{u}\right) \times \left(\pd[\vec{x}]{\mu}\pd[\mu]{v} + \pd[\vec{x}]{\nu}\pd[\nu]{v}\right) \\
            &= \pd[\vec{x}]{\mu}\pd[\mu]{u} \times \pd[\vec{x}]{\nu}\pd[\nu]{v} + \pd[\vec{x}]{\nu}\pd[\nu]{\mu} \times \pd[\vec{x}]{\mu}\pd[\mu]{v}. \\
            &= \underbrace{\left(\pd[\mu]{u}\pd[\nu]{v} - \pd[\mu]{v}\pd[\nu]{u}\right)}_{J - \text{Jacobian for change of variables $(\mu, \nu) \to (u, v)$}}\pd[\vec{x}]{\mu} \times \pd[\vec{x}]{\nu}
        \end{align*}
        which leads to
        \[
        \left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| = |J|\left|\pd[\vec{x}]{\mu} \times \pd[\vec{x}]{\nu}\right|
        \]
        so
        \[
        \int_{\mathcal{U}}f(\vec{x}(u, v))\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|\,dudv = \int_{\mathcal{U}'}f(\vec{x}(\mu, \nu)|J|\left|\pd[\vec{x}]{\mu} \times \pd[\vec{x}]{\nu}\right|\frac{1}{|J|}\,d\mu d\nu.
        \]
    \end{proof}
\end{proposition}

\begin{remark}
    If $\mathcal{S} \in \R ^ 2$ is a flat surface in the $(x, y)$-plane,
    then our parameterisation $\vec{x}(u, v)$ is just a change of variables and
    \[
    \left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| = \left|\pd[x]{u}\pd[y]{v} - \pd[x]{v}\pd[y]{u}\right| = |J|
    \]
    only has a $\vec{e}_3$ component.
    $|J|$ is the Jacobian for the change of coordinates in the plane.
\end{remark}

\subsection{Surface Integrals of Vector Fields}

If $\vec{f} : \R ^ 3 \to \R ^ 3$ is a vector field and $\mathcal{S}$ is an oriented surface with unit normal $\hvec{n}$,
we define the surface integral
(or flux)
of $\vec{f}$ through $\mathcal{S}$ as
\[
\int_{\mathcal{S}}\vec{f}\cdot\hvec{n}\,d\mathcal{S} = \int_{U}\vec{f}(\vec{x}(u, v))\cdot\hvec{n}\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|\,dudv = \int_{U}\vec{f}(\vec{x}(u, v))\cdot\left(\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right)\,dudv.
\]
\textit{Note:
sometimes see the notation $\hvec{n}\,d\mathcal{S} = d\vec{\mathcal{S}}$.
Unlike a scalar surface integral,
the flux also depends on the sign of $\hvec{n}$.
A non-zero flux requires a component of $\vec{f}$
"through"
the surface.}

\begin{example}
    Calculate the outward flux of $\vec{f}(\vec{x}) = \cos ^ 7(\theta)\vec{e}_r$
    [in spherical coordinates]
    through the hemisphere $r = a$,
    $z \geq 0$.

    \begin{solution}
        Parametrise in spherical coordinates:
        $\vec{x}(\theta, \phi)$.
        Recall:
        $\hvec{n} = \vec{e}_r$ and $\left|\pd[\vec{x}]{\theta} \times \pd[\vec{x}]{\phi}\right| = h_{\theta}h_{\phi} = a ^ 2\sin{\theta}$.

        So
        \begin{align*}
            \int_{\mathcal{S}}\vec{f}\cdot\hvec{n}\,d\mathcal{S} &= \int_{0}^{2\pi}\left(\int_{0}^{\pi / 2}\vec{f}\cdot\hvec{n}\left|\pd[\vec{x}]{\theta}\times\pd[\vec{x}]{\theta}\right|\,d\theta\right)\,d\phi \\
            &= \int_{0}^{2\pi}\left(\int_{0}^{\pi / 2}(\cos ^ 7(\theta)\vec{e}_r)\cdot\vec{e}_r a ^ 2\sin(\theta)\,d\theta\right)\,d\phi \\
            &= a ^ 2\int_{0}^{2\pi}\left(\int_{0}^{\pi / 2}(\cos ^ 7(\theta)\sin(\theta)\,d\theta\right)\,d\phi \\
            &= a ^ 2\int_{0}^{2\pi}\left[-\frac{1}{8}\cos ^ 8(\theta)\right]_{0}^{\pi / 2}\,d\phi \\
            &= \frac{a ^ 2}{8}\int_{0}^{2\pi}\,d\phi = \frac{\pi a ^ 2}{4}
        \end{align*}
    \end{solution}
\end{example}

If the surface is closed,
we write $\int_{\mathcal{S}}\vec{f}\cdot\hvec{n}\,d\mathcal{S} = \oint_{S}\vec{f}\cdot\hvec{n}\,d\mathcal{S}$.

\begin{example}
    Repeat the previous example for the full sphere.

    \begin{solution}
        Now $\theta \in [0, \pi]$.
        But $\vec{f} = \cos ^ 7(\theta)\vec{e}_r$ which is anti-symmetric about the equator $\theta = \pi / 2$ therefore $\oint_{\mathcal{S}}\vec{f}\cdot\hvec{n}\,d\mathcal{S} = 0$.
    \end{solution}
\end{example}

We will also meet vector-valued surface integrals of the form:
\[
\int_{\mathcal{S}}\vec{f}\,d\mathcal{S}
\]
for some vector field $\vec{f} : \R ^ 3 \to \R ^ 3$.

\begin{example}
    Evaluate $\int_{\mathcal{S}}\vec{e}_r\,d\mathcal{S}$ where $\mathcal{S}$ is the hemisphere $r = a$,
    $z \geq 0$ and $\vec{e}_r$ is the spherical basis vector.

    \begin{solution}
        Important -
        $\vec{e}_r$ is not constant over $\mathcal{S}$ because its direction varies which means that it cannot be taken out of the integral.

        Instead,
        write it in the Cartesian basis:
        \[
        \vec{e}_r = \sin(\theta)\cos(\phi)\vec{e}_1 + \sin(\theta)\sin(\phi)\vec{e}_2 + \cos(\theta)\vec{e}_3.
        \]
        So
        \begin{align*}
            \int_{\mathcal{S}}\vec{e}_r\,d\mathcal{S} &= \vec{e}_1\int_{\mathcal{S}}\sin(\theta)\cos(\phi)\,d\mathcal{S} + \vec{e}_2\int_{\mathcal{S}}\sin(\theta)\sin(\phi)\,d\mathcal{S} + \vec{e}_3\int_{\mathcal{S}}\cos(\theta)\,d\mathcal{S} \\
            &= 0 + 0 + \vec{e}_3\int_{\mathcal{S}}\cos(\theta)\,d\mathcal{S} \\
            &= \vec{e}_3\int_{0}^{2\pi}\left(\int_{0}^{\pi / 2}\cos(\theta)\left|\pd[\vec{x}]{\theta}\times\pd[\vec{x}]{\phi}\right|\,d\theta\right)\,d\phi \\
            &= a ^ 2\vec{e}_3\int_{0}^{2\pi}\left(\int_{0}^{\pi / 2}\cos(\theta)\sin(\theta)\,d\theta\right)\,d\phi \\
            &= a ^ 2\vec{e}_3\int_{0}^{2\pi}\left[\frac{1}{2}\sin ^ 2(\theta)\right]_{0}^{\pi / 2}\,d\phi \\
            &= \frac{a ^ 2}{2}\vec{e}_3\int_{0}^{2\pi}\,d\phi = \pi a ^ 2\vec{e}_3.
        \end{align*}
    \end{solution}
\end{example}

\textit{Note:
This is not the same as $\vec{e}_r\int_{\mathcal{S}}\,d\mathcal{S} = \vec{e}_r|S| = 2\pi a ^ 2\vec{e}_r$.
Can think of $\frac{1}{|\mathcal{S}|}\int_{\mathcal{S}}\vec{e}_r\,d\mathcal{S}$ as the
"average"
of $\vec{e}_r$ over $\mathcal{S}$.
}

\subsection{Volume Integrals}
The volume integral of a scalar field $f : \R ^ 3 \to \R$ over a $3$D subregion $\mathcal{V} \subset \R ^ 3$ is
\[
\int_{V}f\,dV = \int_{U}f(\vec{x}(u, v, w))\left|\pd[\vec{x}]{u}\cdot\left(\pd[\vec{x}]{v} \times \pd[\vec{x}]{w}\right)\right|\,dudvdw.
\]
The "volume element" $\left|\pd[\vec{x}]{u} \cdot\left(\pd[\vec{x}]{v} \times \pd[\vec{x}]{w}\right)\right|\,dudvdw$ is the volume of an infinitesimal parallelepiped.

Scalar tripple product:
\[
\begin{vmatrix}
    \pd[x]{u} & \pd[x]{v} & \pd[x]{w} \\
    \pd[y]{u} & \pd[y]{v} & \pd[y]{w} \\
    \pd[z]{u} & \pd[z]{v} & \pd[z]{w}
\end{vmatrix}.
\]

\begin{proposition}
    If $(u, v, w)$ are orthogonal curvilinear coordinates,
    then
    \[
    \left|\pd[\vec{x}]{u} \cdot \left(\pd[\vec{x}]{r} \times \pd[\vec{x}]{w}\right)\right| = h_uh_vh_w.
    \]

    \begin{proof}
        Recall $\left|\pd[\vec{x}]{u}\right| = h_u$ which implies $\pd[\vec{x}]{u} = h_u\vec{e}_u$.
        Similarly $\pd[\vec{x}]{v} = h_v\vec{e}_v$,
        $\pd[\vec{x}]{w} = h_w\vec{e}_w$.
        \[
        \left|\pd[\vec{x}]{u} \cdot \left(\pd[\vec{x}]{v} \times \pd[\vec{x}]{w}\right)\right| = |h_uh_vh_w\vec{e}_u \cdot(\vec{e}_r \times \vec{e}_w)| = |\pm h_uh_vh_w| = h_uh_vh_w.
        \]
    \end{proof}
\end{proposition}
The volume of $V$ is $|V| = \int_{V}\,dV$.

\begin{example}
    Find the volume of the sphere $x ^ 2 + y ^ 2 + z ^ 2 = a ^ 2$.

    \begin{solution}
        Recall spherical coordinates:
        $\vec{x}(r, \theta, \phi) = r\sin(\theta)\cos(\phi)\vec{e}_1 + r\sin(\theta)\sin(\phi)\vec{e}_2 + r\cos(\theta)\vec{e}_3$.
        Using the previous proposition
        \[
        \left|\pd[\vec{x}]{u}\cdot\left(\pd[\vec{x}]{v} \times \pd[\vec{x}]{w}\right)\right| = h_rh_{\theta}h_{\phi} = (1)(r)(r\sin{\theta}) = r ^ 2\sin(\theta)
        \]
        so
        \begin{align*}
            |V| &= \int_{0}^{2\pi}\left(\int_{0}^{\pi}\left(\int_{0}^{a}(1)r ^ 2\sin(\theta)\,dr\right)\,d\theta\right)\,d\phi \\
            &= \int_{0}^{2\pi}\left(\int_{0}^{\pi}\left[\frac{r ^ 3}{3}\sin(\theta)\right]_{0}^{a}\,d\theta\right)\,d\phi \\
            &= \frac{a ^ 3}{3}\int_{0}^{2\pi}\left[-\cos(\theta)\right]_{0}^{\pi}\,d\phi \\
            &= \frac{2a ^ 3}{3}\int_{0}^{2\pi}\,d\phi = \frac{4\pi a ^ 3}{3}.
        \end{align*}
    \end{solution}
\end{example}

\begin{remark}
    Finding the correct limits can be tricky,
    always draw a diagram.
\end{remark}

\begin{example}
    Let $V$ be the region bounded by the planes $x = 0$,
    $y = 0$,
    $z = 2$ and the surface $z = x ^ 2 + y ^ 2$ for $x, y \geq 0$.
    Calculate $\int_{V}x\,dV$.

    \begin{solution}
        Note $z = x ^ 2 + y ^ 2$ is a paraboloid:
        The region $V$ is a quarter of the paraboloid.
        Due to rotational symmetry,
        the integral will be easiest to evaluate in cylindrical coordinates:
        \[
        \vec{x}(r, \theta, z) = r\cos(\theta)\vec{e}_1 + r\sin(\theta)\vec{e}_2 + z\vec{e}_3.
        \]
        Using the previous proposition
        \[
        \left|\pd[\vec{x}]{r} \cdot \left(\pd[\vec{x}]{\theta}\times\pd[\vec{x}]{z}\right)\right| = h_rh_{\theta}h_z = r.
        \]
        Limits:
        $y = 0 \iff \theta = 0$,
        $x = 0 \iff \theta = \pi / 2$,
        $r ^ 2 = x ^ 2 + y ^ 2 = z \iff r = \sqrt{z}$,
        so
        \begin{align*}
            \int_{V}x\,dV &= \int_{0}^{2}\left(\int_{0}^{\pi / 2}\left(\int_{0}^{\sqrt{z}}xr\,dr\right)\,d\theta\right)\,dz \\
            \int_{V}x\,dV &= \int_{0}^{2}\left(\int_{0}^{\pi / 2}\left(\int_{0}^{\sqrt{z}}r ^ 2\cos(\theta)\,dr\right)\,d\theta\right)\,dz \\
            &= \int_{0}^{2}\left(\int_{0}^{\pi / 2}\left[\frac{r ^ 3}{3}\right]_{0}^{\sqrt{z}}\cos(\theta)\,d\theta\right)\,dz \\
            &= \int_{0}^{2}\left(\int_{0}^{\pi / 2}\frac{z ^ {3 / 2}}{3}\cos(\theta)\,d\theta\right)\,dz \\
            &= \frac{1}{3}\int_{0}^{2}z ^ {3 / 2}\left[\sin(\theta)\right]_{0}^{\pi / 2}\,dz \\
            &= \frac{1}{3}\int_{0}^{2}z ^ {3 / 2}\,dz \\
            &= \frac{1}{3}\left[\frac{2}{5}z ^ {5 / 2}\right]_{0}^{2} \\
            &= \frac{8\sqrt{2}}{15}.
        \end{align*}
    \end{solution}
\end{example}

\begin{proposition}
    The volume integral of a scalar field is independent of the choice of parameterisation/coordinates.

    \begin{proof}
        Omitted.
    \end{proof}
\end{proposition}

\begin{remark}
    Can also have a vector-valued volume integral $\int_{V}\vec{f}\,dV$.
    Evaluate by changing to Cartesians:
    \[
    \int_{V}\vec{f}\,dV = \vec{e}_1\int_{V}f_1\,dV + \vec{e}_2\int_{V}f_2\,dV + \vec{e}_3\int_{V}f_3\,dV.
    \]
\end{remark}

\section{Differential Operators}

\subsection{Differentiating Scalar Fields}

The gradient of a scalar field $f : \R ^ 3 \to \R$ is the operator
\[
\bnabla f\,\biggr|_{\vec{x}} = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}f\hvec{n}\,dS
\]
where $S$ is a closed surface around $\vec{x}$ with outward normal $\hvec{n}$,
enclosed volume $|V|$ and diameter $d(S) = \sup_{\vec{x}, \vec{y} \in S}|\vec{x} - \vec{y}|$.

Note:
\begin{enumerate}[label = (\roman*)]
    \item $\bnabla f$ is a vector field $\R ^ 3 \to \R ^ 3$.

    \item $\bnabla f$ points in the direction of fastest increase of $f$,
    and $|\bnabla f|$ tells you the maximum rate of increase.

    \item To ensure the limit exists,
    usually $f$ is continuously differentiable
    ($C ^ {1}$)
    meaning $\pd[f]{x}$,
    $\pd[f]{y}$,
    $\pd[f]{z}$ all exist and are continuous.
\end{enumerate}

\begin{proposition}\label{mathmethII:prop:gradincart}
    In Cartesian coordinates,
    $\bnabla f = \pd[f]{x}\vec{e}_1 + \pd[f]{y}\vec{e}_2 + \pd[f]{z}\vec{e}_3$.
\end{proposition}

\begin{example}
    Calculate the gradient of $f(\vec{x}) = y + x ^ 2$.
    \[
    \bnabla f = \pd{x}(y + x ^ 2)\vec{e}_1 + \pd{y}(y + x ^ 2)\vec{e}_2 + \pd{z}(y + x ^ 2)\vec{e}_2 = 2x\vec{e}_1 + \vec{e}_2.
    \]
\end{example}

\begin{lemma}[Mean Value Theorem for Double Integrals]
    If $f : \R ^ 3 \to \R$ is a continuous scalar field on a closed,
    bounded and connected domain $A \subset \R ^ 2$ then there exists $\vec{x}_{*} \in A$ such that
    \[
    \int_{A}f\,dS = |A|f(\vec{x}_{*}).
    \]

    \begin{proof}
        Under the given conditions,
        we know that $f$ attains its maximum and minimum on $A$,
        say at $\vec{x}_{\text{max}}$ and $\vec{x}_{\text{min}}$.
        Also
        \[
        |A|f(\vec{x}_{\text{min}}) \leq \int_{A}f\,dS \leq |A|f(\vec{x}_{\text{max}}).
        \]
        Now let $\vec{x}(t)$ for $t \in [t_0, t_1]$ be a curve in $A$ such that
        \[
        \vec{x}(t_0) = \vec{x}_{\text{min}},\quad\vec{x}(t_1) = \vec{x}_{\text{max}}.
        \]
        Let $g(t) = |A|f(\vec{x}(t))$ so $g(t_0) = |A|f(\vec{x}_{\text{min}})$,
        $g(t_1) = |A|f(\vec{x}_{\text{max}})$.
        Since $g$ is continuous,
        the Intermediate Value Theorem shows that there exists $t_{*} \in [t_0, t_1]$ such that $g(t_{*}) = \int_{A}f\,dS$.

        Then at $\vec{x}_{*} = \vec{x}(t_{*})$ we have $|A|f(\vec{x}_{*}) = \int_{A}f\,dS$.
    \end{proof}
\end{lemma}

\begin{proposition}[continues = mathmethII:prop:gradincart]
    \begin{proof}
        We derive the $\vec{e}_1$ component.
        Let $S$ be a small cylinder radius $\epsilon$,
        length $h$,
        axis $\vec{e}_1$.

        \begin{align*}
            \vec{e}_1 \cdot \bnabla f &= \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}f\vec{e}_1\cdot\hvec{n}\,dS \\
            &= \lim_{\epsilon, h \to 0}\frac{1}{\pi \epsilon ^ 2h}\left[\int_{S_1}f\,dS + \int_{S_0}f(-1)\,dS\right]&&(\text{no contribution on curved surface $(\vec{e}_1 \cdot \hvec{n} = 0$)}) \\
            &= \lim_{\epsilon, h \to 0}\frac{1}{\pi \epsilon ^ 2 h}\left[\pi\epsilon ^ 2 f(\vec{x}_1) - \pi\epsilon ^ 2f(\vec{x}_0)\right] &&(\text{by the mean value theorem}) \\
            &= \lim_{\epsilon, h \to 0}\frac{1}{h}\left[f(\vec{x}_1) - f(\vec{x}_0)\right].
        \end{align*}
        Taking limit $\epsilon \to 0$,
        we find $\vec{x}_0 \to \vec{x}$ and $\vec{x}_1 \to \vec{x} + h\vec{e}_1$.
        So
        \begin{align*}
            \vec{e}_1 \cdot \bnabla f &= \lim{h \to 0}\frac{1}{h}\left[f(\vec{x} + h\vec{e}_1) - f(\vec{x})\right] = \pd[f]{x}\,\biggr|_{\vec{x}}.
        \end{align*}
    \end{proof}
\end{proposition}

\begin{proposition}
    In curvilinear coordinates $(u, v, w)$,
    \[
    \bnabla f = \frac{1}{h_u}\pd[f]{u}\vec{e}_u + \frac{1}{h_v}\pd[f]{v}\vec{e}_v + \frac{1}{h_w}\pd[f]{w}\vec{e}_w.
    \]

    \begin{proof}
        \begin{align*}
            \vec{e}_u \cdot \bnabla f &= \left(\frac{1}{h_u}\pd[\vec{x}]{u}\right)\cdot\bnabla f \\
            &= \frac{1}{h_u}\left(\pd[x]{u}\vec{e}_1 + \pd[y]{u}\vec{e}_2 + \pd[z]{u}\vec{e}_3\right)\cdot\left(\pd[f]{x}\vec{e}_1 + \pd[f]{y}\vec{e}_2 + \pd[f]{z}\vec{e}_3\right) \\
            &= \frac{1}{h_u}\left(\pd[x]{u}\pd[f]{x} + \pd[y]{u}\pd[f]{y} + \pd[z]{u}\pd[f]{z}\right) \\
            &= \frac{1}{h_u}\pd[f]{u}.
        \end{align*}
    \end{proof}
\end{proposition}

\begin{example}
    Compute $\bnabla z$ in both Cartesian and spherical coordinates.

    \begin{solution}
        In Cartesian coordinates:
        \[
        \bnabla z = \pd[z]{x}\vec{e}_1 + \pd[z]{y}\vec{e}_2 + \pd[z]{z}\vec{e}_3 = \vec{e}_3.
        \]

        In spherical coordinates:
        recall $z = r\cos{\theta}$,
        so
        \[
        \bnabla(r\cos{\theta}) = \frac{1}{h_r}\pd{r}(r\cos{\theta})\vec{e}_r + \frac{1}{h_{\theta}}\pd{\theta}(r\cos{\theta})\vec{e}_{\theta} + \frac{1}{h_{\phi}}\pd{\phi}(r\cos{\theta})\vec{e}_{\phi} = \cos(\theta)\vec{e}_r - \sin(\theta)\vec{e}_{\theta}\footnotemark.
        \]
        \footnotetext{This is the same as $\vec{e}_3$ in spherical coordinates.}
    \end{solution}
\end{example}

\begin{remark}
    $\bnabla f$ extends the scalar fields in any $\R ^ n$,
    always giving a vector in $\R ^ n$.
\end{remark}

\subsection{Differentiating Vector Fields}
There are two standard ways to differentiate a vector field $\vec{f} : \R ^ 3 \to \R ^ 3$
($\vec{f}$ assumed $C ^ {1}$):

\begin{enumerate}[label = (\roman*)]
    \item Divergence
    (scalar field)
    \[
    \bnabla \cdot \vec{f}\,\biggr|_{\vec{x}} = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n}\cdot \vec{f}\,dS.
    \]

    \item Curl
    (vector field)
    \[
    \bnabla \times \vec{f}\,\biggr|_{\vec{x}} = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n} \times \vec{f}\,dS
    \]
\end{enumerate}
$S$ is a closed surface around $\vec{x}$ with outward normal $\hvec{n}$,
enclosed volume $|V|$ and diameter $d(S)$.

Intuition for divergence -
measures how much $\vec{f}$ points outward/inward
(on average):
$\bnabla \vec{f} > 0$,
$\vec{f}$ is pointing away from $\vec{x}$.
$\bnabla \vec{f} < 0$,
$\vec{f}$ is pointing towards $\vec{x}$.

Intuition for curl -
measures how much $\vec{f}$ rotates around each axis:
$\vec{e}_3 \cdot (\bnabla \times \vec{f}) > 0$,
$\vec{f}$ is rotating anti-clockwise,
$\vec{e}_3 \cdot (\bnabla \times \vec{f}) < 0$,
$\vec{f}$ is rotating clockwise.

\begin{lemma}\label{mathmethII:lem:ctcrssscalareq}
    If $\vec{a} \in \R ^ 3$ is a constant vector and $f : \R ^ 3 \to \R$ is a $C ^ {1}$ scalar field,
    then
    \begin{enumerate}[label = (\roman*)]
        \item $\bnabla \cdot (f\vec{a}) = (\bnabla f)\cdot \vec{a}$.

        \item $\bnabla \times (f\vec{a}) = (\bnabla f)\times \vec{a}$.
    \end{enumerate}

    \begin{proof}
        \begin{enumerate}[label = (\roman*)]
            \item
            \begin{align*}
                \bnabla \cdot (f\vec{a}) &= \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n}\cdot(f\vec{a})\,dS \\
                &= \left(\lim_{d(S) \to 0}\frac{1}{|V|}\hvec{n}f\,dS\right)\cdot \vec{a} \\
                &= (\bnabla f) \cdot\vec{a}.
            \end{align*}

            \item
            \begin{align*}
                \bnabla \times (f\vec{a}) &= \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n}\times (f\vec{a})\,dS \\
                &= \left(\lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n}f\,dS\right) \times \vec{a} \\
                &= (\bnabla f)\times\vec{a}.
            \end{align*}
        \end{enumerate}
    \end{proof}
\end{lemma}

\begin{proposition}
    In Cartesian coordinates,
    \begin{enumerate}[label = (\roman*)]
        \item
        \[
        \bnabla \cdot \vec{f} = \pd[f_1]{x} + \pd[f_2]{y} + \pd[f_3]{z}.
        \]

        \item
        \[
        \bnabla \times \vec{f} = \begin{vmatrix}
            \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
            \pd{x} & \pd{y} & \pd{z} \\
            f_1 & f_2 & f_3
        \end{vmatrix} = \left(\pd[f_3]{y} - \pd[f_2]{z}\right)\vec{e}_1 + \left(\pd[f_1]{z} - \pd[f_3]{x}\right)\vec{e}_2 +\left(\pd[f_2]{x} - \pd[f_1]{y}\right)\vec{e}_3.
        \]
    \end{enumerate}

    \begin{proof}
        \begin{enumerate}[label = (\roman*)]
            \item
            \begin{align*}
                \bnabla \cdot \vec{f} &= \bnabla \cdot (f_1\vec{e}_1 + f_2\vec{e}_2 + f_3\vec{e}_3) \\
                &= \bnabla \cdot (f_1\vec{e}_1) + \bnabla \cdot (f_2\vec{e}_2) + \bnabla \cdot (f_3\vec{e}_3) \\
                &= (\bnabla f_1) \cdot \vec{e}_1 + (\bnabla f_2) \cdot \vec{e}_2 + (\bnabla f_3)\cdot \vec{e}_3 \\
                &= \pd[f_1]{x} + \pd[f_2]{y} + \pd[f_3]{z}.
            \end{align*}

            \item
            \begin{align*}
                \bnabla \times (f_1\vec{e}_1) &= (\bnabla f_1)\times \vec{e}_1 \\
                &= \left(\pd[f_1]{x}\vec{e}_1 + \pd[f_1]{y}\vec{e}_2 + \pd[f_1]{z}\vec{e}_3\right) \times \vec{e}_1 \\
                &= \pd[f_1]{x}\vec{e}_1 \times \vec{e}_1 + \pd[f_1]{y}\vec{e}_2 \times \vec{e}_1 + \pd[f_1]{z}\vec{e}_3 \times \vec{e}_1 \\
                &= -\pd[f_1]{y}\vec{e}_3 + \pd[f_1]{z}\vec{e}_2
            \end{align*}
            (similar for $f_2\vec{e}_2$ and $f_3\vec{e}_3$).
        \end{enumerate}
    \end{proof}
\end{proposition}

Note:
\begin{enumerate}[label = (\roman*)]
    \item The above shows that divergence and curl are differential operators.

    \item Justifies dot and cross notation,
    thinking of $\bnabla\  \text{"$=$"}\  \vec{e}_1\pd{x} + \vec{e}_2\pd{y} + \vec{e}_3\pd{z}$
\end{enumerate}

\begin{example}
    Compute the divergence and curl of the vector fields $\vec{f}(\vec{x}) = -\vec{x}$ and $\vec{g}(\vec{x}) = (y - x)\vec{e}_1 - (x + y)\vec{e}_2$.

    \begin{solution}
        Using a previous proposition
        \begin{align*}
            \bnabla \cdot \vec{f} &= \pd{x}(-x) + \pd{y}(-y) + \pd{z}(-z) = -3. \\
            \bnabla \cdot \vec{g} &= \pd{x}(y - x) + \pd{y}(-x - y) + \pd{z}(0) = -2.
            \intertext{They are both inward pointing.}
            \bnabla \times \vec{f} &= \begin{vmatrix}
                \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
                \pd{x} & \pd{y} & \pd{z} \\
                -x & -y & -x
            \end{vmatrix} = \left[\pd{y}(-z) - \pd{z}(-y)\right]\vec{e}_1 + \left[\pd{z}(-x) - \pd{x}(-z)\right]\vec{e}_2 + \left[\pd{x}(-y) - \pd{y}(-x)\right]\vec{e}_3 = \vec{0}. \\
            \bnabla \times \vec{g} &= \begin{vmatrix}
                \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
                \pd{x} & \pd{y} & \pd{z} \\
                y - x & -x - y & 0
            \end{vmatrix} = \left[\pd{y}(0) - \pd{z}(-x - y)\right]\vec{e}_1 + \left[\pd{z}(y - x) - \pd{x}(0)\right]\vec{e}_2 + \left[\pd{x}(-x - y) - \pd{y}(y - x)\right]\vec{e}_3 = -2\vec{e}_3.
        \end{align*}
        So $\vec{f}$ has no rotation and $\vec{g}$ has clockwise rotation around the $\vec{e}_3$ axis.
    \end{solution}
\end{example}

Note:
\begin{enumerate}[label = (\roman*)]
    \item If a vector field is $2$D
    (only $\vec{e}_2$ and $\vec{e}_2$ components and independent of $z$),
    then it never has $\vec{e}_1$ or $\vec{e}_2$ components of curl.

    \item In general,
    for non-linear vector fields,
    divergence and curl will not be constant in space.
\end{enumerate}

\begin{corollary}
    If $f : \R ^ 3 \to \R$ is a scalar field and $\vec{f} : \R ^ 3 \to \R ^ 3$ is a vector field,
    both with continuous second
    (partial)
    derivatives,
    then
    \begin{enumerate}[label = (\roman*)]
        \item $\bnabla \cdot (\bnabla \times \vec{f}) = 0$.

        \item $\bnabla \times (\bnabla f) = \vec{0}$.
    \end{enumerate}

    \begin{proof}
        \begin{enumerate}[label = (\roman*)]
            \item
            \begin{align*}
                \bnabla \cdot (\bnabla \times \vec{f}) &= \pd{x}\left(\pd[f_3]{y} - \pd[f_2]{z}\right) + \pd{y}\left(\pd[f_1]{z} - \pd[f_3]{x}\right) + \pd{z}\left(\pd[f_2]{x} - \pd[f_1]{y}\right) \\
                &= \frac{\partial ^ 2f_3}{\partial x\partial y} - \frac{\partial ^ 2f_2}{\partial x\partial z} + \frac{\partial ^ 2f_1}{\partial y\partial z} - \frac{\partial ^ 2f_3}{\partial y\partial x} + \frac{\partial ^ 2f_2}{\partial z\partial x} - \frac{\partial ^ 2f_1}{\partial z\partial y} \\
                &= 0.
            \end{align*}

            \item
            \begin{align*}
                \bnabla \times (\bnabla f) &= \left(\frac{\partial ^ 2f}{\partial y\partial z} - \frac{\partial ^ 2f}{\partial z\partial y}\right)\vec{e}_1 + \left(\frac{\partial ^ 2f}{\partial z\partial x} - \frac{\partial ^ 2f}{\partial x\partial z}\right)\vec{e}_2 + \left(\frac{\partial ^ 2f}{\partial x\partial y} - \frac{\partial ^ 2f}{\partial y\partial x}\right)\vec{e}_3 \\
                &= \vec{0}.
            \end{align*}
        \end{enumerate}
    \end{proof}
\end{corollary}

\subsection{Product Rules}
Scalar fields $g$,
$\bnabla g$ takes $g$ to a vector field.

There are six product rules:
\begin{proposition}[Product rules]
    If $f, g : \R ^ 3 \to \R$ are scalar fields and $\vec{f}, \vec{g} \in \R ^ 3 \to \R ^ 3$ are vector fields,
    all $C ^ {(1)}$,
    then
    \begin{enumerate}[label = (\arabic*)]
        \item
        \[
        \bnabla(fg) \equiv g\bnabla f + f\bnabla g.
        \]

        \item
        \[
        \bnabla \cdot (g\vec{f}) \equiv (\bnabla g) \cdot \vec{f} + g\bnabla \cdot \vec{f}.
        \]

        \item
        \[
        \bnabla \cdot (\vec{f} \times \vec{g}) \equiv \vec{g} \cdot (\bnabla \times \vec{f}) - \vec{f} \cdot (\bnabla \times \vec{g}).
        \]

        \item
        \[
        \bnabla \times (g\vec{f}) \equiv g(\bnabla \times \vec{f}) + (\bnabla g)\times \vec{f}.
        \]

        \item
        \[
        \bnabla (\vec{f} \times \vec{g}) \equiv (\vec{g} \cdot \bnabla)\vec{f} - (\vec{f}\cdot\bnabla)\vec{g} + \vec{f}(\bnabla \cdot\vec{g}) - \vec{g}(\bnabla \cdot \vec{f}).
        \]

        \item
        \[
        \bnabla(\vec{f} \cdot \vec{g}) \equiv (\vec{g}\cdot\bnabla)\vec{f} + (\vec{f}\cdot\bnabla)\vec{g} + \vec{f}\times(\bnabla \times \vec{g}) + \vec{g} \times (\bnabla \times \vec{f}).
        \]
    \end{enumerate}

    \begin{proof}
        Since both sides are coordinates independent,
        it suffices to prove them in Cartesian coordinates.
        
        \begin{enumerate}[label = (\arabic*)]
            \item
            \begin{align*}
                \bnabla(fg) &= \pd{x}(fg)\vec{e}_1 + \pd{y}(fg)\vec{e}_2 + \pd{z}(fg)\vec{e}_3 \\
                &= (f_xg + fg_x)\vec{e}_1 + (f_yg + fg_y)\vec{e}_2 + (f_zg + fg_z)\vec{e}_3 \\
                &= (f_x\vec{e}_1 + f_y\vec{e}_2 + f_z\vec{e}_3)g + f(g_x\vec{e}_1 + g_y\vec{e}_2 + g_z\vec{e}_3) \\
                &= g\bnabla f + f\bnabla g.
            \end{align*}

            \item 
            \begin{align*}
                \bnabla \cdot (g\vec{f}) &= \bnabla\cdot(gf_1\vec{e}_1 + gf_2\vec{e}_2 + gf_3\vec{e}_3) \\
                &= \bnabla (gf_1)\cdot\vec{e}_1 + \bnabla(gf_2)\cdot\vec{e}_2 + \bnabla(gf_3)\cdot \vec{e}_3 \\
                &= (g\bnabla f_1+ f_1\bnabla g) \cdot \vec{e}_1 + (g\bnabla f_2 + f_2\bnabla g) \cdot \vec{e}_2 + (g\bnabla f_3 + f_3\bnabla g)\cdot \vec{e}_3 \\
                &= g(f_1)_x + f_1g_x + g(f_2)_y + f_2g_y + g(f_3)_z + f_3g_z \\
                &= g((f_1)_x + (f_2)_y + (f_3)_z) + (f_1\vec{e}_1 f_2\vec{e}_2 + f_3\vec{e}_3) \cdot(g_x\vec{e}_1 + g_y\vec{e}_2 + g_z\vec{e}_3) \\
                &= g\bnabla\vec{f} + \vec{f}\cdot(\bnabla g).
            \end{align*}

            \item
            \begin{align*}
                \bnabla\cdot(\vec{f} \times \vec{g}) &= \bnabla \cdot \begin{vmatrix}
                    \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\ f_1 & f_2 & f_3 \\ g_1 & g_2 & g_3
                \end{vmatrix} \\
                &= \pd{x}(f_2g_3 - f_3g_2) + \pd{y}(f_3g_1 - f_1g_3) + \pd{z}(f_1g_2 - f_2g_1) \\
                &= g_1((f_3)_y - (f_2)_z) + g_2((f_1)_z - (f_3)_x) + g_3((f_2)_x - (f_1)_y) \\
                &+ f_1((g_2)_z - (g_3)_y) + f_2((g_3)_x - (g_1)_z) + f_3((g_1)_y - (g_2)_x) \\
                &= \vec{g} \cdot(\bnabla \times \vec{f}) - \vec{f}(\bnabla \times \vec{g}).
            \end{align*}

            \item
            \begin{align*}
                \bnabla \times (g\vec{f}) &= \bnabla \times (gf_1\vec{e}_1 + gf_2\vec{e}_2 + gf_3\vec{e}_3) \\
                &= \bnabla(gf_1)\times\vec{e}_1 + \bnabla(gf_2)\times\vec{e}_2 + \bnabla(gf_3)\times\vec{e}_3 \\
                &= \pd{y}(gf_1)\vec{e}_2 \times \vec{e}_1 + \pd{z}(gf_1)\vec{e}_3 \times \vec{e}_1 \\
                &+ \pd{x}(gf_2)\vec{e}_1 \times \vec{e}_2 + \pd{z}(gf_2)\vec{e}_3 \times \vec{e}_2 \\
                &+ \pd{x}(gf_3)\vec{e}_1 \times \vec{e}_3 + \pd{z}(gf_3)\vec{e}_2 \times \vec{e}_3 \\
                &= g\left[((f_3)_y - (f_2)_z)\vec{e}_1 + ((f_1)_z - (f_3)_{x})\vec{e}_2 + ((f_2)_x - (f_1)_y)\vec{e}_3\right] + (g_yf_3 - g_zf_2)\vec{e}_1 + (g_zf_1 - g_xf_3)\vec{e}_2 + (g_xf_2 - g_yf_1)\vec{e}_3 \\
                &= g(\bnabla\times\vec{f}) + (\bnabla g)\times \vec{f}.
            \end{align*}

            \item Omitted.
            
            \item Omitted.
        \end{enumerate}
    \end{proof}
\end{proposition}

\begin{example}
    If $u, v : \R ^ 3 \to \R$ are scalar fields,
    show that $\bnabla u \times \bnabla v = \bnabla \times (u \bnabla v)$.

    \begin{solution}
        \begin{align*}
            \bnabla \times (u \bnabla v) &= u(\bnabla \times \bnabla v) + (\bnabla u) \times (\bnabla v) \\
            &= 0 + (\bnabla u) \times (\bnabla v) \\
            &= (\bnabla u) \times (\bnabla v).
        \end{align*}
    \end{solution}
\end{example}

\subsection{Divergence and Curl in Curvilinear Coordinates}

\begin{proposition}
    In orthogonal curvilinear coordinates
    $(u, v, w)$,
    \begin{enumerate}[label = (\arabic*)]
        \item
        \[
        \bnabla \cdot \vec{f} = \frac{1}{h_uh_vh_w}\left[\pd{u}(h_vh_wf_u) + \pd{v}(h_uh_wf_v) + \pd{w}(h_uh_vf_w)\right].
        \]

        \item
        \[
        \bnabla \times \vec{f} = \frac{1}{h_uh_vh_w}\begin{vmatrix}
            h_u\vec{e}_u & h_v\vec{e}_v & h_w\vec{e}_w \\
            \pd{u} & \pd{v} & \pd{w} \\
            h_uf_u & h_vf_v & h_wf_w
        \end{vmatrix}.
        \]
    \end{enumerate}

    \begin{proof}
        Recall $\bnabla f = \frac{1}{h_u}f_u\vec{e}_u + \frac{1}{h_v}f_v\vec{e}_v + \frac{1}{h_w}f_w\vec{e}_w$,
        \[
        \bnabla u = \frac{1}{h_u}u_u\vec{e}_u + \frac{1}{h_v}u_v\vec{e}_v + \frac{1}{h_w}u_w\vec{e}_w \implies \vec{e}_u = h_u\bnabla u
        \]
        (similarly $\vec{e}_v = h_v\bnabla v$,
        $\vec{e}_w = h_w\bnabla w$).
        
        \begin{enumerate}[label = (\arabic*)]
            \item Consider the first component:
            \begin{align*}
                \bnabla \cdot (f_u\vec{e}_u) &= \bnabla \cdot (f_u\vec{e}_v \times \vec{e}_w). \\
                &= \bnabla \cdot (f_u(h_v\bnabla v) \times (h_w\bnabla w)) \\
                &= \bnabla \cdot (f_uh_vh_w (\bnabla v)\times(\bnabla w)) \\
                &= \bnabla(f_uh_vh_w) \cdot ((\bnabla v) \times (\bnabla w)) + f_uh_vh_w\bnabla \cdot ((\bnabla v) \times (\bnabla w)) \\
                &= \bnabla(f_uh_vh_w)\cdot\frac{\vec{e}_v \times \vec{e}_w}{h_vh_w} \\
                &= \frac{1}{h_vh_w}\vec{e}_u\cdot\bnabla(f_uh_vh_w) \\
                &= \frac{1}{h_uh_vh_w}\pd{u}(f_uh_vh_w)
            \end{align*}
            (the other components follow similarly).

            \item
            Again,
            consider the first component:
            \begin{align*}
                \bnabla \times (f_u\vec{e}_u) &= \bnabla \times (f_uh_u \bnabla u) \\
                &= f_uh_u\bnabla \times (\bnabla u) + \bnabla(f_uh_u) \times (\bnabla u) \\
                &= \bnabla(f_uh_u) \times \left(\frac{1}{h_u}\vec{e}_u\right) \\
                &= \frac{1}{h_u}\bnabla(f_uh_u) \times (\vec{e}_v \times \vec{e}_w)\footnotemark \\
                &= \frac{1}{h_u}\vec{e}_v(\vec{e}_w\cdot\bnabla(f_uh_u)) - \frac{1}{h_u}\vec{e}_w(\vec{e}_v\cdot\bnabla(f_uh_u)) \\
                &= \frac{1}{h_uh_w}\pd{w}(f_uh_u)\vec{e}_v - \frac{1}{h_uh_v}\pd{v}(f_uh_u)\vec{e}_w
            \end{align*}
            (the rest of the components follow similarly).
            \footnotetext{$\vec{a} \times (\vec{b} \times \vec{c}) = \vec{b}(\vec{a} \cdot \vec{c}) - \vec{c}(\vec{a} \cdot \vec{b})$.}
        \end{enumerate}
    \end{proof}
\end{proposition}

\begin{example}
    Calculate $\bnabla \times \vec{f}$ for $\vec{f}(\vec{x}) = r\sin(\theta)\vec{e}_{\phi}$ in spherical coordinates.

    \begin{solution}
        \begin{align*}
            \bnabla \times \vec{f} &= \frac{1}{h_rh_{\theta}h_{\phi}}\begin{vmatrix}
                h_r\vec{e}_r & h_{\theta}\vec{e}_{\theta} & h_{\phi}\vec{e}_{\phi} \\
                \pd{r} & \pd{\theta} & \pd{\phi} \\
                0 & 0 h_{\phi}f_{\phi}
            \end{vmatrix} \\
            &= \frac{1}{r ^ 2\sin{\theta}}\begin{vmatrix}
                \vec{e}_r & r\vec{e}_{\theta} & r\sin(\theta)\vec{e}_{\phi} \\
                \pd{r} & \pd{\theta} & \pd{\phi} \\
                0 & 0 & r ^ 2\sin ^ 2(\theta)
            \end{vmatrix} \\
            &= \frac{1}{r ^ 2\sin(\theta)}\pd{\theta}(r ^ 2\sin ^ 2\theta)\vec{e}_r - \frac{r}{r ^ 2\sin{\theta}}\pd{r}(r ^ 2\sin ^ 2\theta)\vec{e}_{\theta} \\
            &= 2\cos(\theta)\vec{e}_r - 2\sin(\theta)\vec{e}_{\theta}.
        \end{align*}

        Alternatively,
        could convert $\vec{f}$ to Cartesian components first:
        \[
        \vec{f}(\vec{x}) = r\sin(\theta)\vec{e}_{\phi} = r\sin(\theta)(-\sin(\phi)\vec{e}_1 + \cos(\phi)\vec{e}_2) = -y\vec{e}_1 + x\vec{e}_2.
        \]
        Then
        \[
        \bnabla \times \vec{f} = \begin{vmatrix}
            \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
            \pd{x} & \pd{y} & \pd{z} \\
            -y  & x & 0
        \end{vmatrix} = (x_x + y_y)\vec{e}_3 = 2\vec{e}_3 = 2(\cos(\theta)\vec{e}_r - \sin(\theta)\vec{e}_{\theta}
        \]
    \end{solution}
\end{example}

\textbf{Warning}:
\[
\bnabla \cdot \vec{f} \neq \frac{1}{h_u}(f_u)_u + \frac{1}{h_v}(f_v)_v + \frac{1}{h_w}(f_w)_w.
\]
Doesn't work because $\vec{e}_u, \vec{e}_v, \vec{e}_w$ are not constant vectors
(direction varies).
