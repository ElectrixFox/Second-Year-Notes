\section{Line Integrals}

\subsection{Curves}

A curve $C$ is a $1$D subset of $\R ^ n$,
for $n > 1$.
Usually describe a curve by a parameterisation $\vec{x}(t)$:
\[
\vec{x}(t) = x_1(t)\vec{e}_1 + x_2(t)\vec{e}_2 + \dotsc + x_n(t)\vec{e}_n
\]
which maps an interval $[t_0, t_1]$ into $\R ^ n$.
\textit{Think of $\vec{x}(t)$ as "trajectory" of a particle with $t$ being time.}

\begin{example}
    Find a parameterisation $\vec{x}(t)$ for the circle $x ^ 2 + y ^ 2 = a ^ 2$ in $\R ^ 2$.

    \begin{solution}
        Recall
        \[
        \begin{cases}
            x = r\cos{\theta} \\
            y = r\sin{\theta}
        \end{cases}
        \]
        so
        \[
        x ^ 2 + y ^ 2 = a ^ 2 \implies r ^ 2(\cos ^ 2{\theta} + \sin ^ 2{\theta}) = a ^ 2 \implies r = a.
        \]
        So we can set $t = \theta$ and obtain
        \[
        \vec{x}(t) = a\cos(t)\vec{e}_1 + a\sin(t)\vec{e}_2\quad\text{for } t \in [0, 2\pi].
        \]
    \end{solution}
\end{example}

A curve is closed if $\vec{x}(t_1) = \vec{x}(t_0)$.

A curve is simple if it doesn't cross/intersect itself.

\begin{example}
    Is the helix $\vec{x}(t) = \cos(t)\vec{e}_1 + \sin(t)\vec{e}_2 + t\vec{e}_3$ for $t \in [0, 6\pi]$ closed?
    Or simple?

    \begin{solution}
        $\vec{x}(0) = \vec{e}_1 \neq \vec{e}_1 + 6\pi\vec{e}_2 = \vec{x}(6\pi)$ so it is not closed,
        i.e. open.
        No self-intersections so it is simple.
    \end{solution}
\end{example}

\begin{example}
    Is the "clover" $\vec{x}(t) = \cos(3t)\cos(t)\vec{e}_1 + \cos(3t)\sin(t)\vec{e}_2$ for $t \in [0, \pi]$ closed? Or simple?

    \begin{solution}
        $\vec{x}(0) = \vec{e}_1 = \vec{e}_1 = \vec{x}(\pi)$ so it is closed.
        It is not simple because of the triple intersection at $\vec{x} = \vec{0}$,
        $\vec{x}\left(\frac{\pi}{6}\right) = \vec{x}\left(\frac{\pi}{2}\right) = \vec{x}\left(\frac{5\pi}{6}\right) = \vec{0}$.
    \end{solution}
\end{example}

A parameterisation $\vec{x}(t)$ is differentiable if it has a well-defined tangent vector
\[
\frac{d\vec{x}}{dt} = \lim_{dt \to 0}\frac{\vec{x}(t + dt) - \vec{x}(t)}{dt}
\]
points along the curve.

If $\vec{x}(t)$ is a trajectory then $\frac{d\vec{x}}{dt}$ is its velocity,
and $\left|\frac{d\vec{x}}{dt}\right|$ its speed.

\textit{Note:
the parameterisation of a curve is not unique
(could have opposite direction,
or different speed).}

\begin{example}
    Find the tangent vectors for
    \begin{enumerate}[label = (\roman*)]
        \item $\vec{x}(t) = \vec{a} + t(\vec{b} - \vec{a})$ for $t \in [0, 1]$.

        \item $\vec{x}(\tau) = \vec{a} + (\vec{b} - \vec{a})\sin{\tau}$ for $\tau \in \left[0, \frac{\pi}{2}\right]$.
    \end{enumerate}

    \begin{solution}
        \begin{enumerate}[label = (\roman*)]
            \item
            \[
            \frac{d\vec{x}}{dt} = \vec{b} - \vec{a}.
            \]

            \item
            \[
            \frac{d\vec{x}}{d\tau} = (\vec{b} - \vec{a})\cos{\tau}.
            \]
        \end{enumerate}

        They both describe the same straight line from $\vec{a}$ to $\vec{b}$
        (but with different speeds)
        because $\frac{d\vec{x}}{dt}$ and $\frac{d\vec{x}}{dt}$ have the same direction.

        If $\vec{x}(t)$ and $\vec{x}(\tau)$ are two parameterisation of the same curve,
        the Chain Rule gives
        \[
        \frac{d\vec{x}}{dt} = \frac{d\vec{x}}{d\tau}\frac{d\tau}{dt}.
        \]
    \end{solution}
\end{example}

\begin{example}
    Verify the chain rule formula for the two parametrisation in the previous example.

    At a given point on the curve,
    \[
    \vec{x}(t) = \vec{x}(\tau) \iff \vec{a} + t(\vec{b} - \vec{a}) = \vec{a} + (\vec{b} - \vec{a})\sin{\tau} \implies t = \sin{\tau} \implies \frac{dt}{d\tau}\cos{\tau}.
    \]
    Then
    \[
    \frac{d\vec{x}}{dt} = \frac{d\vec{x}}{d\tau}\frac{1}{\frac{dt}{d\tau}} = \frac{1}{\cos{\tau}}\frac{d\vec{x}}{dt}.
    \]
\end{example}

\begin{remark}
    A parameterisation $\vec{x}(t)$ is called regular if $\frac{d\vec{x}}{dt} \neq \vec{0}$ everywhere
    (otherwise it is singular).
    
    A curve is called regular if it has at least one regular parameterisation.
\end{remark}

\begin{example}
    Hypocycloid $\vec{x}(t) = \cos ^ 3(t)\vec{e}_1 + \sin ^ 3(t)\vec{e}_2$ for $t \in [0, 2\pi]$.

    Tangent vector:
    \[
    \frac{d\vec{x}}{dt} = -3\cos ^ 2(t)\sin(t)\vec{e}_1 + 3\sin ^ 2(t)\cos(t)\vec{e}_2.
    \]
    Vanishes when either $\sin(t) = 0$ or $\cos(t) = 0$ so $t = 0, \pi / 2, \pi, 3\pi / 2$ and $\vec{x}(0) = (1, 0)$,
    $\vec{x}\left(\frac{\pi}{2}\right) = (0, 1)$,
    $\vec{x}(\pi) = (-1, 0)$,
    $\vec{x}\left(\frac{3\pi}{2}\right) = (0, -1)$.
\end{example}

\subsection{Line integrals of scalar fields}

Recall:
a scalar field on $\R ^ n$ is a map $f : \R ^ n \to \R$
(assigns a real number to every point).

The line integral of a scalar field $f(\vec{x})$ along a curve $C$ with parameterisation $\vec{x}(t)$ for $t \in [t_0, t_1]$ is
\[
\int_{C}f\,d\ell = \int_{t_0}^{t_1}f(\vec{x}(t))\left|\frac{d\vec{x}}{dt}\right|\,dt.
\]

\begin{example}
    Integrate $f(\vec{x}) = y + x ^ 2$ around the unit circle.

    Parameterisation:
    \[
    \vec{x}(t) = \cos(t)\vec{e}_1 + \sin(t)\vec{e}_2 \text{ for } t \in [0, 2\pi].
    \]

    Tangent vector:
    \[
    \frac{d\vec{x}}{dt} = -\sin(t)\vec{e}_1 + \cos(t)\vec{e}_2 \implies \left|\frac{d\vec{x}}{dt}\right| = \sqrt{\sin ^ 2 t + \cos ^ 2 t} = 1
    \]
    so
    \begin{align*}
        \int_{C}f\,d\ell &= \int_{0}^{2\pi}(y + x ^ 2)(1)\,dt \\
        &= \int_{0}^{2\pi}(\sin(t) + \cos ^ 2(t))\,dt \\
        &= \int_{0}^{2\pi}\cos ^ 2t\,dt \\
        &= \frac{1}{2}\int_{0}^{2\pi}(1 + \cos(2t))\,dt \\
        &= \frac{1}{2}\left[t\right]_{0}^{2\pi} = \pi.
    \end{align*}
    \[
    \]
\end{example}

\begin{proposition}
    The line integral of a scalar field is independent of the choice of parameterisation.

    \begin{proof}
        Suppose we have two parameterisations of the same curve
        \[
        \begin{cases}
            \vec{x}(t) &\text{for } t \in [t_0, t_1] \\
            \vec{x}(\tau) &\text{for } \tau \in [\tau_0, \tau_1].
        \end{cases}
        \]
        By the chain rule
        \[
        \frac{d\vec{x}}{dt} = \frac{d\tau}{dt}\frac{d\vec{x}}{dt} \implies \left|\frac{d\vec{x}}{dt}\right| = \left|\frac{d\tau}{dt}\right|\left|\frac{d\vec{x}}{d\tau}\right|.
        \]

        There are now two cases to consider:

        Case $1$:
        \[
        \vec{x}(t_0) = \vec{x}(\tau_0)
        \]
        i.e. both have same direction.
        Then $\frac{dt}{d\tau} > 0$,
        so
        \[
        \int_{t_0}^{t_1}f(\vec{x}(t))\left|\frac{d\vec{x}}{dt}\right|\,dt = \int_{t_0}^{t_1}f(\vec{x}(t))\frac{d\tau}{dt}\left|\frac{d\vec{x}}{dt}\right|\,dt = \int_{\tau_0}^{\tau_1}f(\vec{x}(\tau))\left|\frac{d\vec{x}}{d\tau}\right|\,d\tau.
        \]
        
        Case $2$:
        $\vec{x}(t_0) = \vec{x}(\tau_1)$ i.e. opposite directions.
        Then $\frac{dt}{d\tau} < 0$,
        so
        \begin{align*}
            \int_{t_0}^{t_1}f(\vec{x}(t))\left|\frac{d\vec{x}}{dt}\right|\,dt &= \int_{t_0}^{t_1}f(\vec{x}(t))\left(-\frac{d\tau}{dt}\right)\left|\frac{d\vec{x}}{d\tau}\right|\,dt \\
            &= -\int_{\tau_1}^{\tau_0}f(\vec{x}(\tau))\left|\frac{d\vec{x}}{d\tau}\right|\,d\tau \\
            &= \int_{\tau_0}^{\tau_1}f(\vec{x}(\tau))\left|\frac{d\vec{x}}{d\tau}\right|\,d\tau.
        \end{align*}
    \end{proof}
\end{proposition}

For the special case $f(\vec{x}) \equiv 1$,
the line integral is called the arclength of the curve,
$L = \int_{C}d\ell = \int_{t_0}^{t_1}\left|\frac{d\vec{x}}{dt}\right|\,dt$.

\begin{remark}
    There is always an arclength parameterisation $\vec{x}(s)$ where $\left|\frac{d\vec{x}}{ds}\right| \equiv 1$ all along the curve.
    Then $s$ is the distance and $L = \int_{0}^{L}\,ds$.
\end{remark}

\begin{example}
    Calculate the arclength of the circle $x ^ 2 + y ^ 2 = a ^ 2$.

    \begin{solution}
        Parameterisation:
        $\vec{x}(t) = a\cos(t)\vec{e}_1 + a\sin(t)\vec{e}_2$ for $t \in [0, 2\pi]$.
        \[
        \frac{d\vec{x}}{dt} = -a\sin(t)\vec{e}_1 + a\cos(t)\vec{e}_2 \implies \left|\frac{d\vec{x}}{dt}\right| = a.
        \]
        So
        \[
        L = \int_{0}^{2\pi}\left|\frac{d\vec{x}}{dt}\,dt\right| = \int_{0}^{2\pi}a\,dt = 2\pi a.
        \]
    \end{solution}
\end{example}

\begin{example}
    Calculate the arclength of the hypocycloid $\vec{x}(t) = \cos ^ 3(t)\vec{e}_1 + \sin ^ 2(t)\vec{e}_2$ for $t \in [0, 2\pi]$.
    \begin{align*}
        \frac{d\vec{x}}{dt} &= -3\cos ^ 2(t)\sin(t)\vec{e}_1 + 3\sin ^ 2(t)\cos(t)\vec{e}_2 \\
        \implies \left|\frac{d\vec{x}}{dt}\right| &= \sqrt{9\cos ^ 4(t)\sin ^ 2(t) + 9\sin ^ 4(t)\cos ^ 2(t)} \\
        &= 3\sqrt{\cos ^ 2(t)\sin ^ 2(t)}\sqrt{\cos ^ 2t + \sin ^ 2t} \\
        &= 3|\cos(t)\sin(t)|.
    \end{align*}
    By symmetry,
    we only need to integrate over $t \in [0, \pi / 2]$,
    so
    \begin{align*}
        L &= \int_{0}^{2\pi}3|\cos(t)\sin(t)|\,dt \\
        &= 4\int_{0}^{\pi / 2}3\cos(t)\sin(t)\,dt \\
        &= 6\int_{0}^{\pi / 2}2\cos(t)\sin(t)\,dt \\
        &= 6\int_{0}^{\pi / 2}\sin(2t)\,dt \\
        &= 6\left[-\frac{1}{2}\cos(2t)\right]_{0}^{\pi / 2} = 6.
    \end{align*}
\end{example}

\subsection{Line integrals of vector fields}
A vector field is a function $\vec{f} : \R ^ n \to \R ^ n$ that assigns a vector to each point in $\R ^ n$.

For a curve with parameterisation $\vec{x}(t)$ for $t \in [t_0, t_1]$,
the unit tangent vector is:
\[
\hvec{t} = \frac{\frac{d\vec{x}}{dt}}{\left|\frac{d\vec{x}}{dt}\right|}
\]
\textit{"Only depends on direction,
not speed".}

An oriented curve is a curve together with a specified
(consistent)
choice of $\hvec{t}$.

The line integral of a vector field $\vec{f}(\vec{x})$ along an oriented curve $C$ with parameterisation $\vec{x}(t)$ for $t \in [t_0, t_1]$ is defined as
\[
\int_{C}\vec{f}\cdot\hvec{t}\,d\ell = \int_{t_0}^{t_1}\vec{f}(\vec{x}(t))\cdot\hvec{t}\left|\frac{d\vec{x}}{dt}\right|\,dt = \int_{t_0}^{t_1}\vec{f}(\vec{x}(t))\frac{d\vec{x}}{dt}\,dt
\]
the scalar line integral of the tangential component of $\vec{f}$.

\textbf{Warning:}
Changing the orientation $\hvec{t} \to -\hvec{t}$ will flip the sign of the integral.

\begin{example}
    Evaluate $\int_{C}\vec{f}\cdot\hvec{t}d\ell$ for $\vec{f}(\vec{x}) = -\vec{x}$ and $C$ the straight line from $(1, 0)$ to $(2, 0)$.

    Parameterisation:
    $\vec{x}(t) = (1 + t)\vec{e}_1$ for $t \in [0, 1]$.
    \[
    \frac{d\vec{x}}{dt} = \vec{e}_1.
    \]
    Then
    \[
    \int_{C}\vec{f}\cdot\hvec{t}\,d\ell = \int_{0}^{1}(-x\vec{e}_1 - y\vec{e}_2) \cdot \vec{e}_1\,dt = \int_{0}^{1}x\,dt = -\int_{0}^{1}(1 + t)\,dt = -\left[t + \frac{1}{2}t ^ 2\right]_{0}^{1} = -\frac{3}{2}.
    \]
\end{example}

If $C$ is closed we usually write
\[
\int_{C}\vec{f}\cdot\hvec{t}\,d\ell = \oint_{C}\vec{f}\cdot\hvec{t}\,d\ell
\]
called the circulation of $\vec{f}$ around $C$.

\begin{example}
    Evaluate $\oint_{C}\vec{f}\cdot\hvec{t}\,d\ell$ for $\vec{f}(\vec{x}) = -\vec{x}$ and $C$ the anti-clockwise unit circle.
    
    \begin{solution}
        Parameterisation:
        \[
        \vec{x}(t) = \cos(t)\vec{e}_1 + \sin(t)\vec{e}_2\qquad\text{for } t \in [0, 2\pi].
        \]
        \[
        \frac{d\vec{x}}{dt} = -\sin(t)\vec{e}_1 + \cos(t)\vec{e}_2.
        \]
        So
        \begin{align*}
            \oint_{C}\vec{f}\cdot\hvec{t}\,d\ell &= \int_{0}^{2\pi}(-x\vec{e}_1 - y\vec{e}_2)\cdot \frac{d\vec{x}}{dt}\,dt \\
            &= \int_{0}^{2\pi}(-\cos(t)\vec{e}_1 - \sin(t)\vec{e}_2)\cdot(-\sin(t)\vec{e}_1 + \cos(t)\vec{e}_2)\,dt \\
            &= \int_{0}^{2\pi}(\cos(t)\sin(t) - \sin(t)\cos(t))\,dt = 0.
        \end{align*}
    \end{solution}
\end{example}

\begin{example}
    Evaluate $\oint_{C}\vec{g}\cdot\hvec{t}\,d\ell$ for $\vec{g}(\vec{x}) = (y - x)\vec{e}_1 - (x + y)\vec{e}_2$ and $C$ the anti-clockwise unit circle.

    \begin{solution}
        \begin{align*}
            \oint_{C}\vec{g}\cdot\hvec{t}\,d\ell &= \int_{0}^{2\pi}\left[(\sin{t} - \cos{t})\vec{e}_1 - (\cos{t} + \sin{t})\vec{e}_2\right] \cdot \left[-\sin(t)\vec{e}_1 + \cos(t)\vec{e}_2\right] \\
            &= \int_{0}^{2\pi}(-\sin ^ 2{t} + \cos(t)\sin(t) - \cos ^ 2{t} - \sin(t)\cos(t))\,dt \\
            &= -\int_{0}^{2\pi}\,dt = -2\pi.
        \end{align*}
    \end{solution}
\end{example}

\begin{remark}
    \begin{enumerate}[label = (\roman*)]
        \item Alternative notation for vector line integrals:
        \[
        \int_{C}\vec{f}\cdot\hvec{t}\,d\ell = \int_{C}\vec{f}\cdot\vec{dx}.
        \]

        \item A physical example is the work done by a particle moving in a force-field $\vec{F}(\vec{x})$ under $m\ddot{\vec{x}} = \vec{F}(\vec{x})$.
        \[
        [\text{work done}] = K(T) - K(0) = \int_{0}^{T}\frac{dK}{dt}\,dt = \int_{0}^{T}m\dot{\vec{x}}\cdot\ddot{\vec{x}}\,dt = \int_{0}^{T}\vec{F}\cdot\dot{\vec{x}}\,dt = \int_{C}\vec{F}\cdot\hvec{t}\,d\ell
        \]
        change in kinetic energy
        \[
        K = \frac{1}{2}m|\dot{\vec{x}}| ^ 2 = \frac{1}{2}m\dot{\vec{x}}\cdot\dot{\vec{x}}.
        \]
    \end{enumerate}
\end{remark}

\subsection{Curvilinear Coordinates}
A system of curvilinear coordinates $(u,, v, w)$ for $\R ^ 3$ has coordinate lines $\vec{x}(u, v_0, w_0)$,
$\vec{x}(u_0, v, w_0)$,
$\vec{x}(u_0, v_0, w)$
which are curves with tangent vectors
\[
\pd[\vec{x}]{u}, \pd[\vec{x}]{v}, \pd[\vec{x}]{w}.
\]
The coordinate system is called orthogonal if the three tangent vectors are mutually orthogonal everywhere.
Then we get an orthonormal basis
\[
\vec{e}_u = \frac{1}{\left|\pd[\vec{x}]{u}\right|}\pd[\vec{x}]{u}\qquad\vec{e}_v = \frac{1}{\left|\pd[\vec{x}]{v}\right|}\pd[\vec{x}]{v}\qquad\vec{e}_w = \frac{1}{\left|\pd[\vec{x}]{w}\right|}\pd[\vec{x}]{w}.
\]
The scale factors $h_u, h_v, h_w$ are the bottom parts of the fractions.

Geometrically,
the scale factors tell us how much an infinitesimal region in $(u, v, w)$-space is stretched at different points in $\R ^ 3$.

\begin{example}
    Find the unit basis vectors for
    (plane)
    polar coordinates $\vec{x}(r, \theta) = r\cos(\theta)\vec{e}_1 + r\sin(\theta)\vec{e}_2$.

    \begin{solution}
        Differentiate to find coordinate tangent vectors:
        \[
        \pd[\vec{x}]{r} = \cos(\theta)\vec{e}_1 + \sin(\theta)\vec{e}_2\qquad\pd[\vec{x}]{\theta} = -r\sin(\theta)\vec{e}_1 + r\cos(\theta)\vec{e}_2.
        \]
        Scale factors:
        \[
        h_r = \left|\pd[\vec{x}]{r}\right| = \sqrt{\cos ^ 2\theta + \sin ^ 2\theta} = 1.
        \]
        \[
        h_{\theta} = \left|\pd[\vec{x}]{\theta}\right| = \sqrt{r ^ 2\cos ^ 2\theta + r ^ 2\sin ^ 2\theta} = r.
        \]
        So the basis vectors are
        \begin{align*}
            \vec{e}_r &= \frac{1}{h_r}\pd[\vec{x}]{r} = \cos(\theta)\vec{e}_1 + \sin(\theta)\vec{e}_2 \\
            \vec{e}_{\theta} &= \frac{1}{h_{\theta}}\pd[\vec{x}]{\theta} = -\sin(\theta)\vec{e}_1 + \cos(\theta)\vec{e}_2.
        \end{align*}
        (These are normal to $r = \text{constant}$ and $\theta = \text{constant}$ respectively)
    \end{solution}
\end{example}

Need to learn the two most common non-Cartesian systems in $\R ^ 3$
\begin{enumerate}[label = (\roman*)]
    \item Cylindrical coordinates:
    ($r > 0$,
    $\theta \in [0, 2\pi]$,
    $z \in \R$)
    \[
    \vec{x}(r, \theta, z) = r\cos(\theta)\vec{e}_1 + r\sin(\theta)\vec{e}_2 + z\vec{e}_3.
    \]
    Scale factors:
    $h_r = 1$,
    $h_{\theta} = r$,
    $h_z = 1$.
    Orthogonal.

    \item Spherical coordinates:
    ($r > 0$,
    $\theta \in [0, \pi)$,
    $\phi \in [0, 2\pi]$)
    \[
    \vec{x}(r, \theta, \phi) = r\sin(\theta)\cos(\phi)\vec{e}_1 + r\sin(\theta)\sin(\phi)\vec{e}_2 + r\cos(\theta)\vec{e}_3.
    \]
    Scale factors:
    $h_r = 1$,
    $h_{\theta} = r$,
    $h_{\theta} = r\sin{\theta}$.

    To derive the parameterisation:

    Side view of sphere and draw $z$ from top to bottom then derive using trigonometry and the radius line.

    Then do a top view with $y$ drawn as usual,
    use trigonometry to derive the others.
\end{enumerate}

\begin{example}
    Write the vector field $\vec{g}(\vec{x}) = (y - x)\vec{e}_1 - (x + y)\vec{e}_2$ in the cylindrical basis $\{\vec{e}_r, \vec{e}_{\theta}, \vec{e}_z\}$.

    \begin{solution}
        We found
        \begin{align*}
            \vec{e}_r &= \cos(\theta)\vec{e}_1 + \sin(\theta)\vec{e}_2 \\
            \vec{e}_{\theta} &= -\sin(\theta)\vec{e}_1 + \cos(\theta)\vec{e}_2 \\
            \vec{e}_z &= \vec{e}_3.
        \end{align*}
        We get
        \begin{align*}
            \vec{e}_1 &= \cos(\theta)\vec{e}_r - \sin(\theta)\vec{e}_{\theta} \\
            \vec{e}_2 &= -\sin(\theta)\vec{e}_r + \cos(\theta)\vec{e}_{\theta} \\
            \vec{e}_3 &= \vec{e}_z.
        \end{align*}
        So
        \begin{align*}
            \vec{g}(\vec{x}) &= (r\sin{\theta} - r\cos{\theta})(\cos(\theta)\vec{e}_r - \sin(\theta)\vec{e}_{\theta}) - (r\cos{\theta} + r\sin{\theta})(\sin(\theta)\vec{e}_r + \cos(\theta)\vec{e}_{\theta}) \\
            &= -r(\vec{e}_r + \vec{e}_{\theta}).
        \end{align*}
    \end{solution}
\end{example}

\section{Surface and Volume Integrals}

\subsection{Surfaces}

A $2$D surface embedded in $\R ^ 3$ can be defined parametrically as a map $\vec{x}(u, r)$ from $R ^ 2 \to \R ^ 3$.

For fixed $v = v_0$,
position $\vec{x}(u, v_0)$ is a curve with tangent vector $\pd[\vec{x}]{u}$
(and vice versa).

If $\vec{x}(u, v)$ is a well-defined parameterisation then $\pd[\vec{x}]{u}, \pd[\vec{x}]{v}$ are not collinear so we define the unit normal
\[
\hvec{n} = \frac{\pd[\vec{x}]{u}\times\pd[\vec{x}]{r}}{\left|\pd[\vec{x}]{u}\times\pd[\vec{x}]{r}\right|}.
\]
The sign of $\hvec{n}$ depends on the ordering of $\pd[\vec{x}]{u}$,
$\pd[\vec{x}]{v}$.
A surface together with a specified
(consistent)
choice of $\hvec{n}$ is called oriented.

\begin{example}
    Find the outward unit normal for the sphere $x ^ 2 + y ^ 2 + z ^ 2 = a ^ 2$.

    From spherical coordinates,
    $\hvec{n} = \vec{e}_r$
    (the basis vector).
    (Could derive this from $\pd[\vec{x}]{\theta} \times \pd[\vec{x}]{\phi} = (h_{\theta}\vec{e}_{\theta}) \times (h_{\phi}\vec{e}_{\phi}) = h_{\theta}h_{\phi}(\vec{e}_{\theta} \times \vec{e}_{\phi}) = h_{\theta}h_{\phi}\vec{e}_r$.)
\end{example}

\begin{example}
    Find the outward unit normal for the torus
    \[
    \vec{x}(u, v) = (R + \cos{v})\cos(u)\vec{e}_1 + (R + \cos{v})\sin(u)\vec{e}_2 + \sin(v)\vec{e}_3\qquad\text{for } u, v \in [0, 2\pi]
    \]
    with $R > 1$ constant.


    Tangent vectors:
    \begin{align*}
        \pd[\vec{x}]{u} &= -(R + \cos{v})\sin(u)\vec{e}_1 + (R +\cos{v})\cos(u)\vec{e}_2 \\
        \pd[\vec{x}]{v} &= -\sin(v)\cos(u)\vec{e}_1 - \sin(v)\sin(u)\vec{e}_2 + \cos(v)\vec{e}_3
    \end{align*}
    \begin{align*}
        \pd[\vec{x}]{u} \times \pd[\vec{x}]{v} &= \begin{vmatrix}
            \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
            -(R + \cos{v})\sin{u} & (R + \cos{v})\cos{u} & 0 \\
            -\sin(v)\cos(u) & -\sin(v)\sin(u) & \cos{v}
        \end{vmatrix} \\
        &=\vec{e}_1((R + \cos{v})\cos(u)\cos(V) - 0) + \vec{e}_2(0 + (R + \cos{v})\sin(u)\cos(v)) \\
        &+ \vec{e}_3((R + \cos{v})\sin(v)\sin ^ 2(u) + (R + \cos{v})\cos ^ 2(u)\sin(v) \\
        &= (R + \cos{v})(\cos(u)\cos(v)\vec{e}_1 + \sin(u)\cos(v)\vec{e}_2 + (\sin ^ 2u + \cos ^ 2u)\sin(v)\vec{e}_3.
    \end{align*}
    Magnitude:
    \begin{align*}
        \left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| &= |R + \cos{v}|\sqrt{\cos ^ 2(u)\cos ^ 2(v) + \sin ^ 2(u)\cos ^ 2(v) + \sin ^ 2(v)} \\
        &= |R + \cos{v}|\sqrt{\cos ^ 2{v} + \sin ^ 2{v}} = |R + \cos{v}| = R + \cos{v}\footnotemark
    \end{align*}
    \footnotetext{No absolute as $R > 1$ and $|\cos{x}| \leq 1$}.

    So unit normal is
    \[
    \hvec{n} = \frac{\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}}{\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|} = \cos(u)\cos(v)\vec{e}_1 + \sin(u)\cos(v)\vec{e}_2 + \sin(v)\vec{e}_3.
    \]

    Putting in $u = 0$,
    $v = 0$,
    then
    $\hvec{n} = \vec{e}_1$
    (so outward $v$).
\end{example}

A surface is smooth if $\hvec{n}$ varies continuously over the surface.
piecewise-smooth if it can be divided into finitely-many smooth pieces.

\begin{example}
    Sketch the surface $\vec{x}(r, \theta) = r\cos(\theta)\vec{e}_1 + r\sin(\theta)\vec{e}_2 + r\vec{e}_3$ for $r \in [0, \infty)$,
    $\theta \in [0, 2\pi]$,
    and determine whether or not it is smooth.

    \begin{solution}
        To sketch the surface,
        note that $x ^ 2 + y ^ 2 = z ^ 2$.
        This is a cone with its point at $\vec{x} = \vec{0}$.

        \ifplots
        \begin{figure}[h!]
             \begin{tikzpicture}
                 \begin{axis}
                 [
                 view={130}{30},
                 z buffer=sort,
                 ticks=none,
                 ]
                 \addplot3[surf,shader=interp,
                     domain=0:2*pi,
                     y domain = 0:3,
                     samples=40,
                     samples y=20,
                     colormap/cool,
                     shader=interp,
                 ]
                 ({y*cos(deg(x))},
                 {y*sin(deg(x))},
                 {y});
                 \end{axis}
             \end{tikzpicture}
        \end{figure}
        \fi

        To check smoothness,
        compute the normal:
        \[
        \pd[\vec{x}]{r} \times \pd[\vec{x}]{\theta} = \begin{vmatrix}
            \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
            \cos{\theta} & \sin{\theta} & 1 \\
            -r\sin{\theta} & r\cos{\theta} & 0
        \end{vmatrix} = -r\cos(\theta)\vec{e}_1 - r\sin(\theta)\vec{e}_2 + r\vec{e}_3
        \]
        which vanishes at $r = 0$ which means $\hvec{n}$ is not defined at $r = 0$ so the surface is not smooth.
    \end{solution}
\end{example}

A surface is open if it is bounded by a curve
(written $\partial S$),
or closed if it has no boundary.

\begin{example}
    Find the upward unit normal to the surface $z = h(x, y)$.

    \begin{solution}
        Parameterisation:
        \[
        \vec{x}(x, y) = x\vec{e}_1 + y\vec{e}_1 + h(x, y)\vec{e}_3
        \]
        so
        \[
        \pd[\vec{x}]{x} \times \pd[\vec{x}]{y} = \begin{vmatrix}
            \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
            1 & 0 & \pd[h]{x} \\
            0 & 1 & \pd[h]{y}
        \end{vmatrix} = -\pd[h]{x}\vec{e}_1 - \pd[h]{y}\vec{e}_2 + \vec{e}_3
        \]
        therefore the unit normal
        \[
        \hvec{n} = \frac{-\pd[h]{x}\vec{e}_1 - \pd[h]{y}\vec{e}_2 + \vec{e}_3}{\sqrt{1 + \left(\pd[h]{x}\right) ^ 2 + \left(\pd[h]{y}\right) ^ 2}}
        \]
    \end{solution}
\end{example}

\subsection{Surface Integrals of Scalar Fields}
If $f : \R ^ 3 \to \R$ is a scalar field and $\mathcal{S}$ is a surface with parameterisation $\vec{x}(u, v)$ then we define the surface integral of $f$ over $\mathcal{S}$ to be
\[
\int_{\mathcal{S}}f\,d\mathcal{S} = \int_{\mathcal{U}}f(\vec{x}(u, v))\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|\,dudv
\]
$\mathcal{U}$ is the preimage of $\mathcal{S}$ in the $(u, v)$-plane.

Idea:
the area element $\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|\,dudv$ is the area on $\mathcal{S}$ corresponding to an infinitesimal rectangle with sides $(du, dv)$ in the $(u, v)$-plane.

\begin{example}
    Find the area element for the plane $z = ax + b$ when parameterised by $(x, y)$.

    \begin{solution}
        Parameterisation:
        \[
        \vec{x}(x, y) = x\vec{e}_1 + y\vec{e}_2 + (ax + b)\vec{e}_3
        \]
        with
        \[
        \pd[\vec{x}]{x} \times \pd[\vec{x}]{y} = -\pd{x}(ax + b)\vec{e}_1 - \pd{y}(ax + b)\vec{e}_2 + \vec{e}_3 = -a\vec{e}_1 + \vec{e}_3.
        \]
        So the area element is 
        \[
        \left|\pd[\vec{x}]{x} \times \pd[\vec{x}]{y}\right|\,dxdy = \sqrt{1 + a ^ 2}\,dxdy
        \]
    \end{solution}
\end{example}

\begin{proposition}
    If $(u, v)$ are two of a set of orthogonal curvilinear coordinates,
    then
    \[
    \left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| = h_uh_v.
    \]

    \begin{proof}
        Recall $\pd[\vec{x}]{u} = h_u\vec{e}_u$ and $\pd[\vec{x}]{v} = h_V\vec{e}_v$ so
        \[
        \left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| = |h_uh_v(\vec{e}_u \times \vec{e}_v)| = h_uh_v|\vec{e}_u \times \vec{e}_v| = h_uh_v|\pm\vec{e}_w| = h_uh_v.
        \]
    \end{proof}
\end{proposition}

\textit{Note:}
\begin{enumerate}[label = (\textit{\arabic*})]
    \item
    \textit{The value of $\int_{\mathcal{S}}f\,d\mathcal{S}$ does not depend on the sign of $\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}$,
    i.e. orientation of the surface doesn't matter.}
    
    \item
    \textit{The surface area of $\mathcal{S}$ is $|\mathcal{S}| = \int_{\mathcal{S}}\,d\mathcal{S}$
    ($f(\vec{x}) \equiv 1$)
    (similarly to arclength).}
\end{enumerate}

\begin{example}
    Find the surface area of the sphere $x ^ 2 + y ^ 2 + z ^ 2 = a ^ 2$.

    \begin{solution}
        Parametrise with spherical coordinates,
        since $r = a$
        (constant).
        \[
        \vec{x}(\theta, \phi) = a\sin(\theta)\cos(\phi)\vec{e}_1 + a\sin(\theta)\sin(\phi)\vec{e}_2 + a\cos(\theta)\vec{e}_3
        \]
        for $\theta \in [0, \pi]$,
        $\phi \in [0, 2\pi]$.

        Recall $h_{\theta} = a$,
        $h_{\phi} = a\sin{\theta}$.

        So we have
        \[
        \left|\pd[\vec{x}]{\theta} \times \pd[\vec{x}]{\phi}\right| = h_{\theta}h_{\phi} = a ^ 2\sin{\theta}.
        \]
        Therefore
        \begin{align*}
            |S| &= \int_{\mathcal{S}}\,d\mathcal{S} \\
            &= \int_{0}^{2\pi}\left(\int_{0}^{\pi}a ^ 2\sin{\theta}\,d\theta\right)\,d\phi \\
            &= a ^ 2 \int_{0}^{2\pi}\left[-\cos{\theta}\right]_{0}^{\pi}\,d\phi \\
            &= 2a ^ 2\int_{0}^{2\pi}\,d\phi = 4\pi a ^ 2.
        \end{align*}
    \end{solution}
\end{example}

\begin{example}
    Find the surface area of the torus from a previous example.
    We had $\vec{x}(u, v)$ for $u, v \in [0, 2\pi]$ and $\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| = R + \cos{v}$.
    So
    \begin{align*}
        |\mathcal{S}| &= \int_{0}^{2\pi}\left(\int_{0}^{2\pi}(R + \cos{v})\,du\right)\,dv \\
        &= 2\pi\int_{0}^{2\pi}(R + \cos{v})\,dv = 4\pi ^ 2R.
    \end{align*}
\end{example}

\begin{proposition}
    The surface integral of a scalar field is independent of the choice of parameterisation.

    \begin{proof}
        Suppose $\vec{x}(u, v)$ and $\vec{x}(\mu, \nu)$ parametrise the same surface.
        \begin{align*}
            \pd[\vec{x}]{u} \times \pd[\vec{x}]{v} &= \left(\pd[\vec{x}]{\mu}\pd[\mu]{u} + \pd[\vec{x}]{\nu}\pd[\nu]{u}\right) \times \left(\pd[\vec{x}]{\mu}\pd[\mu]{v} + \pd[\vec{x}]{\nu}\pd[\nu]{v}\right) \\
            &= \pd[\vec{x}]{\mu}\pd[\mu]{u} \times \pd[\vec{x}]{\nu}\pd[\nu]{v} + \pd[\vec{x}]{\nu}\pd[\nu]{\mu} \times \pd[\vec{x}]{\mu}\pd[\mu]{v}. \\
            &= \underbrace{\left(\pd[\mu]{u}\pd[\nu]{v} - \pd[\mu]{v}\pd[\nu]{u}\right)}_{J - \text{Jacobian for change of variables $(\mu, \nu) \to (u, v)$}}\pd[\vec{x}]{\mu} \times \pd[\vec{x}]{\nu}
        \end{align*}
        which leads to
        \[
        \left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| = |J|\left|\pd[\vec{x}]{\mu} \times \pd[\vec{x}]{\nu}\right|
        \]
        so
        \[
        \int_{\mathcal{U}}f(\vec{x}(u, v))\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|\,dudv = \int_{\mathcal{U}'}f(\vec{x}(\mu, \nu)|J|\left|\pd[\vec{x}]{\mu} \times \pd[\vec{x}]{\nu}\right|\frac{1}{|J|}\,d\mu d\nu.
        \]
    \end{proof}
\end{proposition}

\begin{remark}
    If $\mathcal{S} \in \R ^ 2$ is a flat surface in the $(x, y)$-plane,
    then our parameterisation $\vec{x}(u, v)$ is just a change of variables and
    \[
    \left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| = \left|\pd[x]{u}\pd[y]{v} - \pd[x]{v}\pd[y]{u}\right| = |J|
    \]
    only has a $\vec{e}_3$ component.
    $|J|$ is the Jacobian for the change of coordinates in the plane.
\end{remark}

\subsection{Surface Integrals of Vector Fields}

If $\vec{f} : \R ^ 3 \to \R ^ 3$ is a vector field and $\mathcal{S}$ is an oriented surface with unit normal $\hvec{n}$,
we define the surface integral
(or flux)
of $\vec{f}$ through $\mathcal{S}$ as
\[
\int_{\mathcal{S}}\vec{f}\cdot\hvec{n}\,d\mathcal{S} = \int_{U}\vec{f}(\vec{x}(u, v))\cdot\hvec{n}\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|\,dudv = \int_{U}\vec{f}(\vec{x}(u, v))\cdot\left(\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right)\,dudv.
\]
\textit{Note:
sometimes see the notation $\hvec{n}\,d\mathcal{S} = d\vec{\mathcal{S}}$.
Unlike a scalar surface integral,
the flux also depends on the sign of $\hvec{n}$.
A non-zero flux requires a component of $\vec{f}$
"through"
the surface.}

\begin{example}
    Calculate the outward flux of $\vec{f}(\vec{x}) = \cos ^ 7(\theta)\vec{e}_r$
    [in spherical coordinates]
    through the hemisphere $r = a$,
    $z \geq 0$.

    \begin{solution}
        Parametrise in spherical coordinates:
        $\vec{x}(\theta, \phi)$.
        Recall:
        $\hvec{n} = \vec{e}_r$ and $\left|\pd[\vec{x}]{\theta} \times \pd[\vec{x}]{\phi}\right| = h_{\theta}h_{\phi} = a ^ 2\sin{\theta}$.

        So
        \begin{align*}
            \int_{\mathcal{S}}\vec{f}\cdot\hvec{n}\,d\mathcal{S} &= \int_{0}^{2\pi}\left(\int_{0}^{\pi / 2}\vec{f}\cdot\hvec{n}\left|\pd[\vec{x}]{\theta}\times\pd[\vec{x}]{\theta}\right|\,d\theta\right)\,d\phi \\
            &= \int_{0}^{2\pi}\left(\int_{0}^{\pi / 2}(\cos ^ 7(\theta)\vec{e}_r)\cdot\vec{e}_r a ^ 2\sin(\theta)\,d\theta\right)\,d\phi \\
            &= a ^ 2\int_{0}^{2\pi}\left(\int_{0}^{\pi / 2}(\cos ^ 7(\theta)\sin(\theta)\,d\theta\right)\,d\phi \\
            &= a ^ 2\int_{0}^{2\pi}\left[-\frac{1}{8}\cos ^ 8(\theta)\right]_{0}^{\pi / 2}\,d\phi \\
            &= \frac{a ^ 2}{8}\int_{0}^{2\pi}\,d\phi = \frac{\pi a ^ 2}{4}
        \end{align*}
    \end{solution}
\end{example}

If the surface is closed,
we write $\int_{\mathcal{S}}\vec{f}\cdot\hvec{n}\,d\mathcal{S} = \oint_{S}\vec{f}\cdot\hvec{n}\,d\mathcal{S}$.

\begin{example}
    Repeat the previous example for the full sphere.

    \begin{solution}
        Now $\theta \in [0, \pi]$.
        But $\vec{f} = \cos ^ 7(\theta)\vec{e}_r$ which is anti-symmetric about the equator $\theta = \pi / 2$ therefore $\oint_{\mathcal{S}}\vec{f}\cdot\hvec{n}\,d\mathcal{S} = 0$.
    \end{solution}
\end{example}

We will also meet vector-valued surface integrals of the form:
\[
\int_{\mathcal{S}}\vec{f}\,d\mathcal{S}
\]
for some vector field $\vec{f} : \R ^ 3 \to \R ^ 3$.

\begin{example}
    Evaluate $\int_{\mathcal{S}}\vec{e}_r\,d\mathcal{S}$ where $\mathcal{S}$ is the hemisphere $r = a$,
    $z \geq 0$ and $\vec{e}_r$ is the spherical basis vector.

    \begin{solution}
        Important -
        $\vec{e}_r$ is not constant over $\mathcal{S}$ because its direction varies which means that it cannot be taken out of the integral.

        Instead,
        write it in the Cartesian basis:
        \[
        \vec{e}_r = \sin(\theta)\cos(\phi)\vec{e}_1 + \sin(\theta)\sin(\phi)\vec{e}_2 + \cos(\theta)\vec{e}_3.
        \]
        So
        \begin{align*}
            \int_{\mathcal{S}}\vec{e}_r\,d\mathcal{S} &= \vec{e}_1\int_{\mathcal{S}}\sin(\theta)\cos(\phi)\,d\mathcal{S} + \vec{e}_2\int_{\mathcal{S}}\sin(\theta)\sin(\phi)\,d\mathcal{S} + \vec{e}_3\int_{\mathcal{S}}\cos(\theta)\,d\mathcal{S} \\
            &= 0 + 0 + \vec{e}_3\int_{\mathcal{S}}\cos(\theta)\,d\mathcal{S} \\
            &= \vec{e}_3\int_{0}^{2\pi}\left(\int_{0}^{\pi / 2}\cos(\theta)\left|\pd[\vec{x}]{\theta}\times\pd[\vec{x}]{\phi}\right|\,d\theta\right)\,d\phi \\
            &= a ^ 2\vec{e}_3\int_{0}^{2\pi}\left(\int_{0}^{\pi / 2}\cos(\theta)\sin(\theta)\,d\theta\right)\,d\phi \\
            &= a ^ 2\vec{e}_3\int_{0}^{2\pi}\left[\frac{1}{2}\sin ^ 2(\theta)\right]_{0}^{\pi / 2}\,d\phi \\
            &= \frac{a ^ 2}{2}\vec{e}_3\int_{0}^{2\pi}\,d\phi = \pi a ^ 2\vec{e}_3.
        \end{align*}
    \end{solution}
\end{example}

\textit{Note:
This is not the same as $\vec{e}_r\int_{\mathcal{S}}\,d\mathcal{S} = \vec{e}_r|S| = 2\pi a ^ 2\vec{e}_r$.
Can think of $\frac{1}{|\mathcal{S}|}\int_{\mathcal{S}}\vec{e}_r\,d\mathcal{S}$ as the
"average"
of $\vec{e}_r$ over $\mathcal{S}$.
}

\subsection{Volume Integrals}
The volume integral of a scalar field $f : \R ^ 3 \to \R$ over a $3$D subregion $\mathcal{V} \subset \R ^ 3$ is
\[
\int_{V}f\,dV = \int_{U}f(\vec{x}(u, v, w))\left|\pd[\vec{x}]{u}\cdot\left(\pd[\vec{x}]{v} \times \pd[\vec{x}]{w}\right)\right|\,dudvdw.
\]
The "volume element" $\left|\pd[\vec{x}]{u} \cdot\left(\pd[\vec{x}]{v} \times \pd[\vec{x}]{w}\right)\right|\,dudvdw$ is the volume of an infinitesimal parallelepiped.

Scalar tripple product:
\[
\begin{vmatrix}
    \pd[x]{u} & \pd[x]{v} & \pd[x]{w} \\
    \pd[y]{u} & \pd[y]{v} & \pd[y]{w} \\
    \pd[z]{u} & \pd[z]{v} & \pd[z]{w}
\end{vmatrix}.
\]

\begin{proposition}
    If $(u, v, w)$ are orthogonal curvilinear coordinates,
    then
    \[
    \left|\pd[\vec{x}]{u} \cdot \left(\pd[\vec{x}]{r} \times \pd[\vec{x}]{w}\right)\right| = h_uh_vh_w.
    \]

    \begin{proof}
        Recall $\left|\pd[\vec{x}]{u}\right| = h_u$ which implies $\pd[\vec{x}]{u} = h_u\vec{e}_u$.
        Similarly $\pd[\vec{x}]{v} = h_v\vec{e}_v$,
        $\pd[\vec{x}]{w} = h_w\vec{e}_w$.
        \[
        \left|\pd[\vec{x}]{u} \cdot \left(\pd[\vec{x}]{v} \times \pd[\vec{x}]{w}\right)\right| = |h_uh_vh_w\vec{e}_u \cdot(\vec{e}_r \times \vec{e}_w)| = |\pm h_uh_vh_w| = h_uh_vh_w.
        \]
    \end{proof}
\end{proposition}
The volume of $V$ is $|V| = \int_{V}\,dV$.

\begin{example}
    Find the volume of the sphere $x ^ 2 + y ^ 2 + z ^ 2 = a ^ 2$.

    \begin{solution}
        Recall spherical coordinates:
        $\vec{x}(r, \theta, \phi) = r\sin(\theta)\cos(\phi)\vec{e}_1 + r\sin(\theta)\sin(\phi)\vec{e}_2 + r\cos(\theta)\vec{e}_3$.
        Using the previous proposition
        \[
        \left|\pd[\vec{x}]{u}\cdot\left(\pd[\vec{x}]{v} \times \pd[\vec{x}]{w}\right)\right| = h_rh_{\theta}h_{\phi} = (1)(r)(r\sin{\theta}) = r ^ 2\sin(\theta)
        \]
        so
        \begin{align*}
            |V| &= \int_{0}^{2\pi}\left(\int_{0}^{\pi}\left(\int_{0}^{a}(1)r ^ 2\sin(\theta)\,dr\right)\,d\theta\right)\,d\phi \\
            &= \int_{0}^{2\pi}\left(\int_{0}^{\pi}\left[\frac{r ^ 3}{3}\sin(\theta)\right]_{0}^{a}\,d\theta\right)\,d\phi \\
            &= \frac{a ^ 3}{3}\int_{0}^{2\pi}\left[-\cos(\theta)\right]_{0}^{\pi}\,d\phi \\
            &= \frac{2a ^ 3}{3}\int_{0}^{2\pi}\,d\phi = \frac{4\pi a ^ 3}{3}.
        \end{align*}
    \end{solution}
\end{example}

\begin{remark}
    Finding the correct limits can be tricky,
    always draw a diagram.
\end{remark}

\begin{example}
    Let $V$ be the region bounded by the planes $x = 0$,
    $y = 0$,
    $z = 2$ and the surface $z = x ^ 2 + y ^ 2$ for $x, y \geq 0$.
    Calculate $\int_{V}x\,dV$.

    \begin{solution}
        Note $z = x ^ 2 + y ^ 2$ is a paraboloid:
        The region $V$ is a quarter of the paraboloid.
        Due to rotational symmetry,
        the integral will be easiest to evaluate in cylindrical coordinates:
        \[
        \vec{x}(r, \theta, z) = r\cos(\theta)\vec{e}_1 + r\sin(\theta)\vec{e}_2 + z\vec{e}_3.
        \]
        Using the previous proposition
        \[
        \left|\pd[\vec{x}]{r} \cdot \left(\pd[\vec{x}]{\theta}\times\pd[\vec{x}]{z}\right)\right| = h_rh_{\theta}h_z = r.
        \]
        Limits:
        $y = 0 \iff \theta = 0$,
        $x = 0 \iff \theta = \pi / 2$,
        $r ^ 2 = x ^ 2 + y ^ 2 = z \iff r = \sqrt{z}$,
        so
        \begin{align*}
            \int_{V}x\,dV &= \int_{0}^{2}\left(\int_{0}^{\pi / 2}\left(\int_{0}^{\sqrt{z}}xr\,dr\right)\,d\theta\right)\,dz \\
            \int_{V}x\,dV &= \int_{0}^{2}\left(\int_{0}^{\pi / 2}\left(\int_{0}^{\sqrt{z}}r ^ 2\cos(\theta)\,dr\right)\,d\theta\right)\,dz \\
            &= \int_{0}^{2}\left(\int_{0}^{\pi / 2}\left[\frac{r ^ 3}{3}\right]_{0}^{\sqrt{z}}\cos(\theta)\,d\theta\right)\,dz \\
            &= \int_{0}^{2}\left(\int_{0}^{\pi / 2}\frac{z ^ {3 / 2}}{3}\cos(\theta)\,d\theta\right)\,dz \\
            &= \frac{1}{3}\int_{0}^{2}z ^ {3 / 2}\left[\sin(\theta)\right]_{0}^{\pi / 2}\,dz \\
            &= \frac{1}{3}\int_{0}^{2}z ^ {3 / 2}\,dz \\
            &= \frac{1}{3}\left[\frac{2}{5}z ^ {5 / 2}\right]_{0}^{2} \\
            &= \frac{8\sqrt{2}}{15}.
        \end{align*}
    \end{solution}
\end{example}

\begin{proposition}
    The volume integral of a scalar field is independent of the choice of parameterisation/coordinates.

    \begin{proof}
        Omitted.
    \end{proof}
\end{proposition}

\begin{remark}
    Can also have a vector-valued volume integral $\int_{V}\vec{f}\,dV$.
    Evaluate by changing to Cartesians:
    \[
    \int_{V}\vec{f}\,dV = \vec{e}_1\int_{V}f_1\,dV + \vec{e}_2\int_{V}f_2\,dV + \vec{e}_3\int_{V}f_3\,dV.
    \]
\end{remark}

\section{Differential Operators}

\subsection{Differentiating Scalar Fields}

The gradient of a scalar field $f : \R ^ 3 \to \R$ is the operator
\[
\bnabla f\,\biggr|_{\vec{x}} = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}f\hvec{n}\,dS
\]
where $S$ is a closed surface around $\vec{x}$ with outward normal $\hvec{n}$,
enclosed volume $|V|$ and diameter $d(S) = \sup_{\vec{x}, \vec{y} \in S}|\vec{x} - \vec{y}|$.

Note:
\begin{enumerate}[label = (\roman*)]
    \item $\bnabla f$ is a vector field $\R ^ 3 \to \R ^ 3$.

    \item $\bnabla f$ points in the direction of fastest increase of $f$,
    and $|\bnabla f|$ tells you the maximum rate of increase.

    \item To ensure the limit exists,
    usually $f$ is continuously differentiable
    ($C ^ {1}$)
    meaning $\pd[f]{x}$,
    $\pd[f]{y}$,
    $\pd[f]{z}$ all exist and are continuous.
\end{enumerate}

\begin{proposition}\label{mathmethII:prop:gradincart}
    In Cartesian coordinates,
    $\bnabla f = \pd[f]{x}\vec{e}_1 + \pd[f]{y}\vec{e}_2 + \pd[f]{z}\vec{e}_3$.
\end{proposition}

\begin{example}
    Calculate the gradient of $f(\vec{x}) = y + x ^ 2$.
    \[
    \bnabla f = \pd{x}(y + x ^ 2)\vec{e}_1 + \pd{y}(y + x ^ 2)\vec{e}_2 + \pd{z}(y + x ^ 2)\vec{e}_2 = 2x\vec{e}_1 + \vec{e}_2.
    \]
\end{example}

\begin{lemma}[Mean Value Theorem for Double Integrals]
    If $f : \R ^ 3 \to \R$ is a continuous scalar field on a closed,
    bounded and connected domain $A \subset \R ^ 2$ then there exists $\vec{x}_{*} \in A$ such that
    \[
    \int_{A}f\,dS = |A|f(\vec{x}_{*}).
    \]

    \begin{proof}
        Under the given conditions,
        we know that $f$ attains its maximum and minimum on $A$,
        say at $\vec{x}_{\text{max}}$ and $\vec{x}_{\text{min}}$.
        Also
        \[
        |A|f(\vec{x}_{\text{min}}) \leq \int_{A}f\,dS \leq |A|f(\vec{x}_{\text{max}}).
        \]
        Now let $\vec{x}(t)$ for $t \in [t_0, t_1]$ be a curve in $A$ such that
        \[
        \vec{x}(t_0) = \vec{x}_{\text{min}},\quad\vec{x}(t_1) = \vec{x}_{\text{max}}.
        \]
        Let $g(t) = |A|f(\vec{x}(t))$ so $g(t_0) = |A|f(\vec{x}_{\text{min}})$,
        $g(t_1) = |A|f(\vec{x}_{\text{max}})$.
        Since $g$ is continuous,
        the Intermediate Value Theorem shows that there exists $t_{*} \in [t_0, t_1]$ such that $g(t_{*}) = \int_{A}f\,dS$.

        Then at $\vec{x}_{*} = \vec{x}(t_{*})$ we have $|A|f(\vec{x}_{*}) = \int_{A}f\,dS$.
    \end{proof}
\end{lemma}

\begin{proposition}[continues = mathmethII:prop:gradincart]
    \begin{proof}
        We derive the $\vec{e}_1$ component.
        Let $S$ be a small cylinder radius $\epsilon$,
        length $h$,
        axis $\vec{e}_1$.

        \begin{align*}
            \vec{e}_1 \cdot \bnabla f &= \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}f\vec{e}_1\cdot\hvec{n}\,dS \\
            &= \lim_{\epsilon, h \to 0}\frac{1}{\pi \epsilon ^ 2h}\left[\int_{S_1}f\,dS + \int_{S_0}f(-1)\,dS\right]&&(\text{no contribution on curved surface $(\vec{e}_1 \cdot \hvec{n} = 0$)}) \\
            &= \lim_{\epsilon, h \to 0}\frac{1}{\pi \epsilon ^ 2 h}\left[\pi\epsilon ^ 2 f(\vec{x}_1) - \pi\epsilon ^ 2f(\vec{x}_0)\right] &&(\text{by the mean value theorem}) \\
            &= \lim_{\epsilon, h \to 0}\frac{1}{h}\left[f(\vec{x}_1) - f(\vec{x}_0)\right].
        \end{align*}
        Taking limit $\epsilon \to 0$,
        we find $\vec{x}_0 \to \vec{x}$ and $\vec{x}_1 \to \vec{x} + h\vec{e}_1$.
        So
        \begin{align*}
            \vec{e}_1 \cdot \bnabla f &= \lim{h \to 0}\frac{1}{h}\left[f(\vec{x} + h\vec{e}_1) - f(\vec{x})\right] = \pd[f]{x}\,\biggr|_{\vec{x}}.
        \end{align*}
    \end{proof}
\end{proposition}

\begin{proposition}
    In curvilinear coordinates $(u, v, w)$,
    \[
    \bnabla f = \frac{1}{h_u}\pd[f]{u}\vec{e}_u + \frac{1}{h_v}\pd[f]{v}\vec{e}_v + \frac{1}{h_w}\pd[f]{w}\vec{e}_w.
    \]

    \begin{proof}
        \begin{align*}
            \vec{e}_u \cdot \bnabla f &= \left(\frac{1}{h_u}\pd[\vec{x}]{u}\right)\cdot\bnabla f \\
            &= \frac{1}{h_u}\left(\pd[x]{u}\vec{e}_1 + \pd[y]{u}\vec{e}_2 + \pd[z]{u}\vec{e}_3\right)\cdot\left(\pd[f]{x}\vec{e}_1 + \pd[f]{y}\vec{e}_2 + \pd[f]{z}\vec{e}_3\right) \\
            &= \frac{1}{h_u}\left(\pd[x]{u}\pd[f]{x} + \pd[y]{u}\pd[f]{y} + \pd[z]{u}\pd[f]{z}\right) \\
            &= \frac{1}{h_u}\pd[f]{u}.
        \end{align*}
    \end{proof}
\end{proposition}

\begin{example}
    Compute $\bnabla z$ in both Cartesian and spherical coordinates.

    \begin{solution}
        In Cartesian coordinates:
        \[
        \bnabla z = \pd[z]{x}\vec{e}_1 + \pd[z]{y}\vec{e}_2 + \pd[z]{z}\vec{e}_3 = \vec{e}_3.
        \]

        In spherical coordinates:
        recall $z = r\cos{\theta}$,
        so
        \[
        \bnabla(r\cos{\theta}) = \frac{1}{h_r}\pd{r}(r\cos{\theta})\vec{e}_r + \frac{1}{h_{\theta}}\pd{\theta}(r\cos{\theta})\vec{e}_{\theta} + \frac{1}{h_{\phi}}\pd{\phi}(r\cos{\theta})\vec{e}_{\phi} = \cos(\theta)\vec{e}_r - \sin(\theta)\vec{e}_{\theta}\footnotemark.
        \]
        \footnotetext{This is the same as $\vec{e}_3$ in spherical coordinates.}
    \end{solution}
\end{example}

\begin{remark}
    $\bnabla f$ extends the scalar fields in any $\R ^ n$,
    always giving a vector in $\R ^ n$.
\end{remark}

\subsection{Differentiating Vector Fields}
There are two standard ways to differentiate a vector field $\vec{f} : \R ^ 3 \to \R ^ 3$
($\vec{f}$ assumed $C ^ {1}$):

\begin{enumerate}[label = (\roman*)]
    \item Divergence
    (scalar field)
    \[
    \bnabla \cdot \vec{f}\,\biggr|_{\vec{x}} = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n}\cdot \vec{f}\,dS.
    \]

    \item Curl
    (vector field)
    \[
    \bnabla \times \vec{f}\,\biggr|_{\vec{x}} = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n} \times \vec{f}\,dS
    \]
\end{enumerate}
$S$ is a closed surface around $\vec{x}$ with outward normal $\hvec{n}$,
enclosed volume $|V|$ and diameter $d(S)$.

Intuition for divergence -
measures how much $\vec{f}$ points outward/inward
(on average):
$\bnabla \vec{f} > 0$,
$\vec{f}$ is pointing away from $\vec{x}$.
$\bnabla \vec{f} < 0$,
$\vec{f}$ is pointing towards $\vec{x}$.

Intuition for curl -
measures how much $\vec{f}$ rotates around each axis:
$\vec{e}_3 \cdot (\bnabla \times \vec{f}) > 0$,
$\vec{f}$ is rotating anti-clockwise,
$\vec{e}_3 \cdot (\bnabla \times \vec{f}) < 0$,
$\vec{f}$ is rotating clockwise.

\begin{lemma}\label{mathmethII:lem:ctcrssscalareq}
    If $\vec{a} \in \R ^ 3$ is a constant vector and $f : \R ^ 3 \to \R$ is a $C ^ {1}$ scalar field,
    then
    \begin{enumerate}[label = (\roman*)]
        \item $\bnabla \cdot (f\vec{a}) = (\bnabla f)\cdot \vec{a}$.

        \item $\bnabla \times (f\vec{a}) = (\bnabla f)\times \vec{a}$.
    \end{enumerate}

    \begin{proof}
        \begin{enumerate}[label = (\roman*)]
            \item
            \begin{align*}
                \bnabla \cdot (f\vec{a}) &= \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n}\cdot(f\vec{a})\,dS \\
                &= \left(\lim_{d(S) \to 0}\frac{1}{|V|}\hvec{n}f\,dS\right)\cdot \vec{a} \\
                &= (\bnabla f) \cdot\vec{a}.
            \end{align*}

            \item
            \begin{align*}
                \bnabla \times (f\vec{a}) &= \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n}\times (f\vec{a})\,dS \\
                &= \left(\lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n}f\,dS\right) \times \vec{a} \\
                &= (\bnabla f)\times\vec{a}.
            \end{align*}
        \end{enumerate}
    \end{proof}
\end{lemma}

\begin{proposition}
    In Cartesian coordinates,
    \begin{enumerate}[label = (\roman*)]
        \item
        \[
        \bnabla \cdot \vec{f} = \pd[f_1]{x} + \pd[f_2]{y} + \pd[f_3]{z}.
        \]

        \item
        \[
        \bnabla \times \vec{f} = \begin{vmatrix}
            \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
            \pd{x} & \pd{y} & \pd{z} \\
            f_1 & f_2 & f_3
        \end{vmatrix} = \left(\pd[f_3]{y} - \pd[f_2]{z}\right)\vec{e}_1 + \left(\pd[f_1]{z} - \pd[f_3]{x}\right)\vec{e}_2 +\left(\pd[f_2]{x} - \pd[f_1]{y}\right)\vec{e}_3.
        \]
    \end{enumerate}

    \begin{proof}
        \begin{enumerate}[label = (\roman*)]
            \item
            \begin{align*}
                \bnabla \cdot \vec{f} &= \bnabla \cdot (f_1\vec{e}_1 + f_2\vec{e}_2 + f_3\vec{e}_3) \\
                &= \bnabla \cdot (f_1\vec{e}_1) + \bnabla \cdot (f_2\vec{e}_2) + \bnabla \cdot (f_3\vec{e}_3) \\
                &= (\bnabla f_1) \cdot \vec{e}_1 + (\bnabla f_2) \cdot \vec{e}_2 + (\bnabla f_3)\cdot \vec{e}_3 \\
                &= \pd[f_1]{x} + \pd[f_2]{y} + \pd[f_3]{z}.
            \end{align*}

            \item
            \begin{align*}
                \bnabla \times (f_1\vec{e}_1) &= (\bnabla f_1)\times \vec{e}_1 \\
                &= \left(\pd[f_1]{x}\vec{e}_1 + \pd[f_1]{y}\vec{e}_2 + \pd[f_1]{z}\vec{e}_3\right) \times \vec{e}_1 \\
                &= \pd[f_1]{x}\vec{e}_1 \times \vec{e}_1 + \pd[f_1]{y}\vec{e}_2 \times \vec{e}_1 + \pd[f_1]{z}\vec{e}_3 \times \vec{e}_1 \\
                &= -\pd[f_1]{y}\vec{e}_3 + \pd[f_1]{z}\vec{e}_2
            \end{align*}
            (similar for $f_2\vec{e}_2$ and $f_3\vec{e}_3$).
        \end{enumerate}
    \end{proof}
\end{proposition}

Note:
\begin{enumerate}[label = (\roman*)]
    \item The above shows that divergence and curl are differential operators.

    \item Justifies dot and cross notation,
    thinking of $\bnabla\  \text{"$=$"}\  \vec{e}_1\pd{x} + \vec{e}_2\pd{y} + \vec{e}_3\pd{z}$
\end{enumerate}

\begin{example}
    Compute the divergence and curl of the vector fields $\vec{f}(\vec{x}) = -\vec{x}$ and $\vec{g}(\vec{x}) = (y - x)\vec{e}_1 - (x + y)\vec{e}_2$.

    \begin{solution}
        Using a previous proposition
        \begin{align*}
            \bnabla \cdot \vec{f} &= \pd{x}(-x) + \pd{y}(-y) + \pd{z}(-z) = -3. \\
            \bnabla \cdot \vec{g} &= \pd{x}(y - x) + \pd{y}(-x - y) + \pd{z}(0) = -2.
            \intertext{They are both inward pointing.}
            \bnabla \times \vec{f} &= \begin{vmatrix}
                \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
                \pd{x} & \pd{y} & \pd{z} \\
                -x & -y & -x
            \end{vmatrix} = \left[\pd{y}(-z) - \pd{z}(-y)\right]\vec{e}_1 + \left[\pd{z}(-x) - \pd{x}(-z)\right]\vec{e}_2 + \left[\pd{x}(-y) - \pd{y}(-x)\right]\vec{e}_3 = \vec{0}. \\
            \bnabla \times \vec{g} &= \begin{vmatrix}
                \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
                \pd{x} & \pd{y} & \pd{z} \\
                y - x & -x - y & 0
            \end{vmatrix} = \left[\pd{y}(0) - \pd{z}(-x - y)\right]\vec{e}_1 + \left[\pd{z}(y - x) - \pd{x}(0)\right]\vec{e}_2 + \left[\pd{x}(-x - y) - \pd{y}(y - x)\right]\vec{e}_3 = -2\vec{e}_3.
        \end{align*}
        So $\vec{f}$ has no rotation and $\vec{g}$ has clockwise rotation around the $\vec{e}_3$ axis.
    \end{solution}
\end{example}

Note:
\begin{enumerate}[label = (\roman*)]
    \item If a vector field is $2$D
    (only $\vec{e}_2$ and $\vec{e}_2$ components and independent of $z$),
    then it never has $\vec{e}_1$ or $\vec{e}_2$ components of curl.

    \item In general,
    for non-linear vector fields,
    divergence and curl will not be constant in space.
\end{enumerate}

\begin{corollary}
    If $f : \R ^ 3 \to \R$ is a scalar field and $\vec{f} : \R ^ 3 \to \R ^ 3$ is a vector field,
    both with continuous second
    (partial)
    derivatives,
    then
    \begin{enumerate}[label = (\roman*)]
        \item $\bnabla \cdot (\bnabla \times \vec{f}) = 0$.

        \item $\bnabla \times (\bnabla f) = \vec{0}$.
    \end{enumerate}

    \begin{proof}
        \begin{enumerate}[label = (\roman*)]
            \item
            \begin{align*}
                \bnabla \cdot (\bnabla \times \vec{f}) &= \pd{x}\left(\pd[f_3]{y} - \pd[f_2]{z}\right) + \pd{y}\left(\pd[f_1]{z} - \pd[f_3]{x}\right) + \pd{z}\left(\pd[f_2]{x} - \pd[f_1]{y}\right) \\
                &= \frac{\partial ^ 2f_3}{\partial x\partial y} - \frac{\partial ^ 2f_2}{\partial x\partial z} + \frac{\partial ^ 2f_1}{\partial y\partial z} - \frac{\partial ^ 2f_3}{\partial y\partial x} + \frac{\partial ^ 2f_2}{\partial z\partial x} - \frac{\partial ^ 2f_1}{\partial z\partial y} \\
                &= 0.
            \end{align*}

            \item
            \begin{align*}
                \bnabla \times (\bnabla f) &= \left(\frac{\partial ^ 2f}{\partial y\partial z} - \frac{\partial ^ 2f}{\partial z\partial y}\right)\vec{e}_1 + \left(\frac{\partial ^ 2f}{\partial z\partial x} - \frac{\partial ^ 2f}{\partial x\partial z}\right)\vec{e}_2 + \left(\frac{\partial ^ 2f}{\partial x\partial y} - \frac{\partial ^ 2f}{\partial y\partial x}\right)\vec{e}_3 \\
                &= \vec{0}.
            \end{align*}
        \end{enumerate}
    \end{proof}
\end{corollary}

\subsection{Product Rules}
Scalar fields $g$,
$\bnabla g$ takes $g$ to a vector field.

There are six product rules:
\begin{proposition}[Product rules]\label{mathmethII:prop:prodrules}
    If $f, g : \R ^ 3 \to \R$ are scalar fields and $\vec{f}, \vec{g} \in \R ^ 3 \to \R ^ 3$ are vector fields,
    all $C ^ {(1)}$,
    then
    \begin{enumerate}[label = (\arabic*)]
        \item
        \[
        \bnabla(fg) \equiv g\bnabla f + f\bnabla g.
        \]

        \item
        \[
        \bnabla \cdot (g\vec{f}) \equiv (\bnabla g) \cdot \vec{f} + g\bnabla \cdot \vec{f}.
        \]

        \item
        \[
        \bnabla \cdot (\vec{f} \times \vec{g}) \equiv \vec{g} \cdot (\bnabla \times \vec{f}) - \vec{f} \cdot (\bnabla \times \vec{g}).
        \]

        \item
        \[
        \bnabla \times (g\vec{f}) \equiv g(\bnabla \times \vec{f}) + (\bnabla g)\times \vec{f}.
        \]

        \item
        \[
        \bnabla (\vec{f} \times \vec{g}) \equiv (\vec{g} \cdot \bnabla)\vec{f} - (\vec{f}\cdot\bnabla)\vec{g} + \vec{f}(\bnabla \cdot\vec{g}) - \vec{g}(\bnabla \cdot \vec{f}).
        \]

        \item
        \[
        \bnabla(\vec{f} \cdot \vec{g}) \equiv (\vec{g}\cdot\bnabla)\vec{f} + (\vec{f}\cdot\bnabla)\vec{g} + \vec{f}\times(\bnabla \times \vec{g}) + \vec{g} \times (\bnabla \times \vec{f}).
        \]
    \end{enumerate}

    \begin{proof}
        Since both sides are coordinates independent,
        it suffices to prove them in Cartesian coordinates.
        
        \begin{enumerate}[label = (\arabic*)]
            \item
            \begin{align*}
                \bnabla(fg) &= \pd{x}(fg)\vec{e}_1 + \pd{y}(fg)\vec{e}_2 + \pd{z}(fg)\vec{e}_3 \\
                &= (f_xg + fg_x)\vec{e}_1 + (f_yg + fg_y)\vec{e}_2 + (f_zg + fg_z)\vec{e}_3 \\
                &= (f_x\vec{e}_1 + f_y\vec{e}_2 + f_z\vec{e}_3)g + f(g_x\vec{e}_1 + g_y\vec{e}_2 + g_z\vec{e}_3) \\
                &= g\bnabla f + f\bnabla g.
            \end{align*}

            \item 
            \begin{align*}
                \bnabla \cdot (g\vec{f}) &= \bnabla\cdot(gf_1\vec{e}_1 + gf_2\vec{e}_2 + gf_3\vec{e}_3) \\
                &= \bnabla (gf_1)\cdot\vec{e}_1 + \bnabla(gf_2)\cdot\vec{e}_2 + \bnabla(gf_3)\cdot \vec{e}_3 \\
                &= (g\bnabla f_1+ f_1\bnabla g) \cdot \vec{e}_1 + (g\bnabla f_2 + f_2\bnabla g) \cdot \vec{e}_2 + (g\bnabla f_3 + f_3\bnabla g)\cdot \vec{e}_3 \\
                &= g(f_1)_x + f_1g_x + g(f_2)_y + f_2g_y + g(f_3)_z + f_3g_z \\
                &= g((f_1)_x + (f_2)_y + (f_3)_z) + (f_1\vec{e}_1 f_2\vec{e}_2 + f_3\vec{e}_3) \cdot(g_x\vec{e}_1 + g_y\vec{e}_2 + g_z\vec{e}_3) \\
                &= g\bnabla\vec{f} + \vec{f}\cdot(\bnabla g).
            \end{align*}

            \item
            \begin{align*}
                \bnabla\cdot(\vec{f} \times \vec{g}) &= \bnabla \cdot \begin{vmatrix}
                    \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\ f_1 & f_2 & f_3 \\ g_1 & g_2 & g_3
                \end{vmatrix} \\
                &= \pd{x}(f_2g_3 - f_3g_2) + \pd{y}(f_3g_1 - f_1g_3) + \pd{z}(f_1g_2 - f_2g_1) \\
                &= g_1((f_3)_y - (f_2)_z) + g_2((f_1)_z - (f_3)_x) + g_3((f_2)_x - (f_1)_y) \\
                &+ f_1((g_2)_z - (g_3)_y) + f_2((g_3)_x - (g_1)_z) + f_3((g_1)_y - (g_2)_x) \\
                &= \vec{g} \cdot(\bnabla \times \vec{f}) - \vec{f}(\bnabla \times \vec{g}).
            \end{align*}

            \item
            \begin{align*}
                \bnabla \times (g\vec{f}) &= \bnabla \times (gf_1\vec{e}_1 + gf_2\vec{e}_2 + gf_3\vec{e}_3) \\
                &= \bnabla(gf_1)\times\vec{e}_1 + \bnabla(gf_2)\times\vec{e}_2 + \bnabla(gf_3)\times\vec{e}_3 \\
                &= \pd{y}(gf_1)\vec{e}_2 \times \vec{e}_1 + \pd{z}(gf_1)\vec{e}_3 \times \vec{e}_1 \\
                &+ \pd{x}(gf_2)\vec{e}_1 \times \vec{e}_2 + \pd{z}(gf_2)\vec{e}_3 \times \vec{e}_2 \\
                &+ \pd{x}(gf_3)\vec{e}_1 \times \vec{e}_3 + \pd{z}(gf_3)\vec{e}_2 \times \vec{e}_3 \\
                &= g\left[((f_3)_y - (f_2)_z)\vec{e}_1 + ((f_1)_z - (f_3)_{x})\vec{e}_2 + ((f_2)_x - (f_1)_y)\vec{e}_3\right] + (g_yf_3 - g_zf_2)\vec{e}_1 + (g_zf_1 - g_xf_3)\vec{e}_2 + (g_xf_2 - g_yf_1)\vec{e}_3 \\
                &= g(\bnabla\times\vec{f}) + (\bnabla g)\times \vec{f}.
            \end{align*}

            \item Omitted.
            
            \item Omitted.
        \end{enumerate}
    \end{proof}
\end{proposition}

\begin{example}
    If $u, v : \R ^ 3 \to \R$ are scalar fields,
    show that $\bnabla u \times \bnabla v = \bnabla \times (u \bnabla v)$.

    \begin{solution}
        \begin{align*}
            \bnabla \times (u \bnabla v) &= u(\bnabla \times \bnabla v) + (\bnabla u) \times (\bnabla v) \\
            &= 0 + (\bnabla u) \times (\bnabla v) \\
            &= (\bnabla u) \times (\bnabla v).
        \end{align*}
    \end{solution}
\end{example}

\subsection{Divergence and Curl in Curvilinear Coordinates}

\begin{proposition}
    In orthogonal curvilinear coordinates
    $(u, v, w)$,
    \begin{enumerate}[label = (\arabic*)]
        \item
        \[
        \bnabla \cdot \vec{f} = \frac{1}{h_uh_vh_w}\left[\pd{u}(h_vh_wf_u) + \pd{v}(h_uh_wf_v) + \pd{w}(h_uh_vf_w)\right].
        \]

        \item
        \[
        \bnabla \times \vec{f} = \frac{1}{h_uh_vh_w}\begin{vmatrix}
            h_u\vec{e}_u & h_v\vec{e}_v & h_w\vec{e}_w \\
            \pd{u} & \pd{v} & \pd{w} \\
            h_uf_u & h_vf_v & h_wf_w
        \end{vmatrix}.
        \]
    \end{enumerate}

    \begin{proof}
        Recall $\bnabla f = \frac{1}{h_u}f_u\vec{e}_u + \frac{1}{h_v}f_v\vec{e}_v + \frac{1}{h_w}f_w\vec{e}_w$,
        \[
        \bnabla u = \frac{1}{h_u}u_u\vec{e}_u + \frac{1}{h_v}u_v\vec{e}_v + \frac{1}{h_w}u_w\vec{e}_w \implies \vec{e}_u = h_u\bnabla u
        \]
        (similarly $\vec{e}_v = h_v\bnabla v$,
        $\vec{e}_w = h_w\bnabla w$).
        
        \begin{enumerate}[label = (\arabic*)]
            \item Consider the first component:
            \begin{align*}
                \bnabla \cdot (f_u\vec{e}_u) &= \bnabla \cdot (f_u\vec{e}_v \times \vec{e}_w). \\
                &= \bnabla \cdot (f_u(h_v\bnabla v) \times (h_w\bnabla w)) \\
                &= \bnabla \cdot (f_uh_vh_w (\bnabla v)\times(\bnabla w)) \\
                &= \bnabla(f_uh_vh_w) \cdot ((\bnabla v) \times (\bnabla w)) + f_uh_vh_w\bnabla \cdot ((\bnabla v) \times (\bnabla w)) \\
                &= \bnabla(f_uh_vh_w)\cdot\frac{\vec{e}_v \times \vec{e}_w}{h_vh_w} \\
                &= \frac{1}{h_vh_w}\vec{e}_u\cdot\bnabla(f_uh_vh_w) \\
                &= \frac{1}{h_uh_vh_w}\pd{u}(f_uh_vh_w)
            \end{align*}
            (the other components follow similarly).

            \item
            Again,
            consider the first component:
            \begin{align*}
                \bnabla \times (f_u\vec{e}_u) &= \bnabla \times (f_uh_u \bnabla u) \\
                &= f_uh_u\bnabla \times (\bnabla u) + \bnabla(f_uh_u) \times (\bnabla u) \\
                &= \bnabla(f_uh_u) \times \left(\frac{1}{h_u}\vec{e}_u\right) \\
                &= \frac{1}{h_u}\bnabla(f_uh_u) \times (\vec{e}_v \times \vec{e}_w)\footnotemark \\
                &= \frac{1}{h_u}\vec{e}_v(\vec{e}_w\cdot\bnabla(f_uh_u)) - \frac{1}{h_u}\vec{e}_w(\vec{e}_v\cdot\bnabla(f_uh_u)) \\
                &= \frac{1}{h_uh_w}\pd{w}(f_uh_u)\vec{e}_v - \frac{1}{h_uh_v}\pd{v}(f_uh_u)\vec{e}_w
            \end{align*}
            (the rest of the components follow similarly).
            \footnotetext{$\vec{a} \times (\vec{b} \times \vec{c}) = \vec{b}(\vec{a} \cdot \vec{c}) - \vec{c}(\vec{a} \cdot \vec{b})$.}
        \end{enumerate}
    \end{proof}
\end{proposition}

\begin{example}
    Calculate $\bnabla \times \vec{f}$ for $\vec{f}(\vec{x}) = r\sin(\theta)\vec{e}_{\phi}$ in spherical coordinates.

    \begin{solution}
        \begin{align*}
            \bnabla \times \vec{f} &= \frac{1}{h_rh_{\theta}h_{\phi}}\begin{vmatrix}
                h_r\vec{e}_r & h_{\theta}\vec{e}_{\theta} & h_{\phi}\vec{e}_{\phi} \\
                \pd{r} & \pd{\theta} & \pd{\phi} \\
                0 & 0 h_{\phi}f_{\phi}
            \end{vmatrix} \\
            &= \frac{1}{r ^ 2\sin{\theta}}\begin{vmatrix}
                \vec{e}_r & r\vec{e}_{\theta} & r\sin(\theta)\vec{e}_{\phi} \\
                \pd{r} & \pd{\theta} & \pd{\phi} \\
                0 & 0 & r ^ 2\sin ^ 2(\theta)
            \end{vmatrix} \\
            &= \frac{1}{r ^ 2\sin(\theta)}\pd{\theta}(r ^ 2\sin ^ 2\theta)\vec{e}_r - \frac{r}{r ^ 2\sin{\theta}}\pd{r}(r ^ 2\sin ^ 2\theta)\vec{e}_{\theta} \\
            &= 2\cos(\theta)\vec{e}_r - 2\sin(\theta)\vec{e}_{\theta}.
        \end{align*}

        Alternatively,
        could convert $\vec{f}$ to Cartesian components first:
        \[
        \vec{f}(\vec{x}) = r\sin(\theta)\vec{e}_{\phi} = r\sin(\theta)(-\sin(\phi)\vec{e}_1 + \cos(\phi)\vec{e}_2) = -y\vec{e}_1 + x\vec{e}_2.
        \]
        Then
        \[
        \bnabla \times \vec{f} = \begin{vmatrix}
            \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
            \pd{x} & \pd{y} & \pd{z} \\
            -y  & x & 0
        \end{vmatrix} = (x_x + y_y)\vec{e}_3 = 2\vec{e}_3 = 2(\cos(\theta)\vec{e}_r - \sin(\theta)\vec{e}_{\theta}
        \]
    \end{solution}
\end{example}

\textbf{Warning}:
\[
\bnabla \cdot \vec{f} \neq \frac{1}{h_u}(f_u)_u + \frac{1}{h_v}(f_v)_v + \frac{1}{h_w}(f_w)_w.
\]
Doesn't work because $\vec{e}_u, \vec{e}_v, \vec{e}_w$ are not constant vectors
(direction varies).

\section{Integral Theorems}

\subsection{Fundamental Theorem of Line Integrals}

\begin{theorem}[Fundamental Theorem of Calculus]
    If $f'(x)$ is continuous on $[a, b]$,
    then
    \[
    \int_{a}^{b}f'(t)\,dt = f(b) - f(a).
    \]
\end{theorem}
There are several generalisation to vector calculus.

\begin{theorem}[Fundamental Theorem of Line Integrals]
    If $f : \R ^ n \to \R$ is a $C ^ {1}$ scalar field and $C$ is an oriented curve with parameterisation $\vec{x}(t)$ for $t \in [t_0, t_1]$,
    then
    \[
    \int_{C}\bnabla f\cdot \hvec{t}\,d\ell = f(\vec{x}(t_1)) - f(\vec{x}(t_0)).
    \]

    \begin{proof}
        (For $n = 3$)
        \begin{align*}
            \int_{C}\bnabla f \cdot \hvec{t}\,d\ell &= \int_{t_0}^{t_1}\bnabla f \cdot \frac{d\vec{x}}{dt}\,dt \\
            &= \int_{t_0}^{t_1}\left(f_x\frac{d\vec{x}}{dt} + f_y\frac{d\vec{y}}{dt} + f_z\frac{d\vec{z}}{dt}\right)\,dt &&(\text{by chain rule}) \\
            &= \int_{t_0}^{t_1}\frac{df}{dt}\,dt = f(\vec{x}(t_1)) - f(\vec{x}(t_0)) &&(\text{by FTOC})
        \end{align*}
    \end{proof}
\end{theorem}

\textit{Note:
if $n = 1$,
then $C$ is just $[t_0, t_1]$ and $\bnabla f = f'\vec{e}_1$,
$\hvec{t} = \vec{e}_1$ reduces to the fundamental theorem of calculus.}

A vector field $\vec{f} : \R ^ n \to \R ^ n$ is conservative if it can be written as $\vec{f} = \bnabla g$ for some $C ^ {1}$ scalar field $g : \R ^ n \to \R$,
called a
(scalar)
potential.

\begin{example}
    Show that $\vec{f}(\vec{x}) = -\vec{x}$ in $\R ^ 2$ is conservative by finding a suitable potential.

    \begin{solution}
        Need 
        \begin{align}
            g_x &= -x \\
            g_y &= -y.
        \end{align}
        Integrating
        ($1$)
        with respect to $x$ gives $g(x, y) = -\frac{x ^ 2}{2} + h(y)$.
        Then
        ($2$)
        \[
        h'(y) = -y \implies h(y) = -\frac{y ^ 2}{2} + C.
        \]
        So
        \[
        g(x, y) = -\frac{x ^ 2 + y ^ 2}{2} + C.
        \]
        (\textit{Note:
        $\vec{f} = \bnabla g$ only defines $g$ up to a constant,
        $c$,
        because $\bnabla c = \vec{0}$.})
    \end{solution}
\end{example}

\begin{proposition}[Path Independence]
    A vector field is conservative if and only if it has path independent line integrals between each pair of points.

    \begin{proof}
        "$\implies$".
        Assume $\vec{f}$ conservative,
        i.e. $\vec{f} = \bnabla g$.

        Let $C_1, C_2$ be any two curves sharing both endpoints $\vec{x}_a, \vec{x}_b$.
        Then
        \[
        \int_{C_1}\vec{f}\cdot\hvec{t}\,d\ell = \int_{C_1}\bnabla g \cdot \hvec{t}\,d\ell = g(\vec{x}_b) - g(\vec{x}_a)
        \]
        but also
        \[
        \int_{C_2}\vec{f}\cdot\hvec{t}\,d\ell = \int_{C_2}\bnabla g \cdot \hvec{t}\,d\ell = g(\vec{x}_b) - g(\vec{x}_a)
        \]
        which implies
        \[
        \int_{C_1}\vec{f}\cdot\hvec{t}\,d\ell = \int_{C_2}\vec{f}\cdot\hvec{t}\,d\ell.
        \]

        "$\impliedby$".
        Assume $\vec{f}$ has path independent line integrals.
        Claim:
        \[
        g(\vec{x}) = \int_{C(\vec{x})}\vec{f}\cdot\hvec{t}\,d\ell\text{ satisfies } \bnabla g = \vec{f}.
        \]
        where $C(\vec{x})$ is any curve connecting $\vec{0}$ to $\vec{x}$.

        We have
        \begin{align*}
            g_x &= \lim_{\delta \to 0}\frac{1}{\delta}\left[\int_{C(\vec{x} + \delta\vec{e}_1)}\vec{f}\cdot\hvec{t}\,d\ell - \int_{C(\vec{x})}\vec{f}\cdot\hvec{t}\,d\ell\right] \\
            &= \lim_{\delta \to 0}\frac{1}{\delta}\left[\int_{C_1}\vec{f}\cdot\hvec{t}\,d\ell\right] &&(\text{by path independence})
            \intertext{where $C_1$ has $\vec{x}(t) = \vec{x} + t\vec{e}_1$ for $t \in [0, \delta]$}
            &= \lim_{\delta \to 0}\frac{1}{\delta}\int_{0}^{\delta}f_1(\vec{x} + t\vec{e}_1)\,dt \\
            &= \lim_{\delta \to 0}\frac{1}{\delta}\delta f_1(\vec{x} + t_{*}\vec{e}_1) &&(\text{by mean value theorem for integrals})
            \intertext{for some $t_{*} \in [0, \delta]$}
            &= f_1(\vec{x}) &&(\text{by squeezing})
        \end{align*}
        (Similarly for other components).
    \end{proof}
\end{proposition}

\begin{example}
    Compute $\int_{C}\vec{f}\cdot\hvec{t}\,d\ell$ where $\vec{f} = -\vec{x}$ and $C$ is the straight line from $(1, 0)$ to $(2, 0)$.

    \begin{solution}
        \[
        \vec{f} = -\bnabla\left(\frac{x ^ 2 + y ^ 2}{2}\right).
        \]
        So FTLI implies
        \[
        \int_{C}\vec{f}\cdot\hvec{t}\,d\ell = -\left(\frac{x ^ 2 + y ^ 2}{2}\right)\biggr|_{(2, 0)} + \left(\frac{x ^ 2 + y ^ 2}{2}\right)\biggr|_{(1, 0)} = -2 + \frac{1}{2} = -\frac{3}{2}.
        \]
    \end{solution}
\end{example}

\begin{remark}
    The name "conservative" comes from the fact that integrating along a closed path doesn't change $f$:
    $\vec{x}_a = \vec{x}_b$,
    \[
    \oint_{C}\bnabla f \cdot \hvec{t}\,d\ell = f(\vec{x}_a) - f(\vec{x}_a) = 0.
    \]
    E.g. gravitational force.
\end{remark}

\subsection{Divergence Theorem}
(Another generalisation of FTOC)

\begin{theorem}[Divergence/Gauss Theorem]
    If $\vec{f} : \R ^ 3 \to \R ^ 3$ is a $C ^ {1}$ vector field and $S$ is a closed surface with outward normal $\hvec{n}$,
    bounding a region $V$,
    then
    \[
    \oint_{S}\vec{f}\cdot\hvec{n}\,dS = \int_{V}\bnabla \cdot \vec{f}\,dV.
    \]

    \begin{proof}
        Idea:
        subdivide $V$ into a disjoint union of small domains $V_i$ for $i = 1, \dotsc, m$
        whose boundaries $S_i$ have unit normals $\hvec{n}_i$.

        Each interior segment of $S_i$ is shared by a neighbouring subdomain $V_j$ with equal and opposite normal $\hvec{n}_j = -\hvec{n}_i$.
        So
        \begin{align*}
            \oint_{S}\vec{f}\cdot\hvec{n}\,dS &= \sum_{i = 1}^{m}\oint_{S_i}\vec{f}\cdot\hvec{n}_i\,dS &&(\text{only contributions from global boundary survive}) \\
            &= \sum_{i = 1}^{m}\left(\frac{1}{|V_i|}\oint_{S_i}\vec{f}\cdot\hvec{n}_i\,dS\right)|V_i| \\
            &= \lim_{m \to \infty}\sum_{i = 1}^{m}\left(\frac{1}{|V_i|}\oint_{S_i}\vec{f}\cdot\hvec{n}_i\,dS\right)|V_i| \\
            &= \int_{V}\bnabla\cdot\vec{f}\,dV &&(\text{need $\vec{f} \in C ^ {1}$ for this to work})
        \end{align*}
        \[
        \]
    \end{proof}
\end{theorem}

\begin{remark}\phantom{}
    \begin{enumerate}[label = (\arabic*)]
        \item Generalises to $V \subset \R ^ n$
        ($n \neq 3$)
        with boundary $S$ and $(n - 1)$D hypersurface.

        \item For $n = 1$,
        we have $V = [a, b]$,
        $S = \{a, b\}$,
        $\hvec{n} = \begin{cases}
            -\vec{e}_1, &x = a \\
            \vec{e}_1, &x = b
        \end{cases}$,
        $\vec{f} = f(x)\vec{e}_1$,
        $\bnabla \cdot \vec{f} = f'(x)$.
        So the theorem reduces to $f(b) - f(a) = \int_{a}^{b}f'(x)\,dx$
        (FTOC).
    \end{enumerate}
\end{remark}

\begin{example}
    Verify the divergence theorem by computing both sides for $\vec{f} = r ^ 2\cos ^ 2(\theta)\vec{e}_r$ in spherical coordinates where $S$ is the sphere $x ^ 2 + y ^ 2 + z ^ 2 = a ^ 2$.

    \begin{enumerate}[label = (\arabic*)]
        \item Surface integral/flux
        \begin{align*}
            \oint_{S}\vec{f}\cdot\hvec{n}\,dS &= \int_{0}^{2\pi}\int_{0}^{\pi}\vec{f}\cdot\vec{e}_rh_{\theta}h_{\phi}\,d\theta\,d\phi  \\
            &= \int_{0}^{2\pi}\int_{0}^{\pi}a ^ 2\cos ^ 2(\theta)a ^ 2\sin ^ (\theta)\,d\theta\,d\phi \\
            &= a ^ 4\int_{0}^{2\pi}\int_{0}^{\pi}\cos ^ 2(\theta)\sin(\theta)\,d\theta\,d\phi \\
            &= a ^ 4\int_{0}^{2\pi}\left[-\frac{\cos ^ 3\theta}{3}\right]_{0}^{\pi}\,d\phi \\
            &= \frac{2a ^ 4}{3}\int_{0}^{2\pi}\,d\phi \\
            &= \frac{4}{3}\pi a ^ 4.
        \end{align*}

        \item Volume integral of $\bnabla \cdot \vec{f}$.
        First compute
        \begin{align*}
            \bnabla \cdot \vec{f} &= \frac{1}{h_rh_{\theta}h_{\phi}}\left[\pd{r}(h_{\theta}h_{\phi}f_r) + \pd{\theta}(h_rh_{\phi}f_{\theta}) + \pd{\phi}(h_rh_{\theta}f_{\phi})\right] \\
            &= \frac{1}{r ^ 2\sin(\theta)}\pd{r}(r ^ 2\sin(\theta)r ^ 2\cos ^ 2(\theta) \\
            &= \frac{\cos ^ 2(\theta)}{r ^ 2}\pd{r}(r ^ 4) \\
            &= 4r\cos ^ 2(\theta).
        \end{align*}
        So
        \begin{align*}
            \int_{V}\bnabla\cdot\vec{f}\,dV &= \int_{0}^{2\pi}\int_{0}^{\pi}\int_{0}^{a}4r\cos ^ 2(\theta)h_rh_{\theta}h_{\phi}\,dr\,d\theta\,d\phi \\
            &= \int_{0}^{2\pi}\int_{0}^{\pi}\int_{0}^{a}4r ^ 3\cos ^ 2(\theta)\sin(\theta)\,dr\,d\theta\,d\phi \\
            &= \int_{0}^{2\pi}\int_{0}^{\pi}\left[r ^ 4\cos ^ 2(\theta)\sin(\theta)\right]_{0}^{a}\,d\theta\,d\phi \\
            &= a ^ 4\int_{0}^{2\pi}\int_{0}^{\pi}\cos ^ 2(\theta)\sin(\theta)\,d\theta\,d\phi \\
            &= \frac{4}{3}\pi a ^ 4.
        \end{align*}
    \end{enumerate}
\end{example}

\begin{remark}
    \begin{enumerate}[label = (\arabic*)]
        \item The divergence theorem applies even when the boundary surface $S$ is not connected.
        E.g. two concentric spheres $S = S_0 \cup S_1$.
        \[
        \oint_{S_1}\vec{f}\cdot\hvec{n}_1\,dS + \oint_{S_1}\vec{f}\cdot\hvec{n}_2\,dS = \int_{V}\bnabla\cdot\vec{f}\,dV.
        \]

        \item The divergence theorem plays a fundamental role in connecting integral and differential forms of conservation laws.
    \end{enumerate}
\end{remark}

\subsection{Stokes' Theorem}
(Third generalisation of FTOC)

\begin{theorem}[Stokes' Theorem]\label{mathmethII:thm:stokesthm}
    If $\vec{f} : \R ^ 3 \to \R ^ 3$ is a $C ^ {1}$ vector field and $C$ is an oriented closed curve bounding a surface $S$,
    then
    \[
    \oint_{C}\vec{f}\cdot\hvec{t}\,d\ell = \int_{S}(\bnabla \times \vec{f})\cdot\hvec{n}\,dS
    \]
    where $\hvec{n}$ is chosen such that $\hvec{t} \times \hvec{n}$ is outward.
\end{theorem}

Notes:
\begin{enumerate}[label = (\arabic*)]
    \item $S$ is sometimes called a capping surface for $C$.

    \item Stokes' theorem says that the flux of $\bnabla \times \vec{f}$ is the same for every possible $S$ sharing the boundary $C$.

    \item Remember the correct orientation by the
    "right-hand rule".
\end{enumerate}

\begin{example}
    Verify Stokes' theorem by computing both sides for $\vec{f}(\vec{x}) = z\vec{e}_1 + x\vec{e}_2 - x\vec{e}_3$ with $S$ the unit disk $\{x ^ 2 + y ^ 2 \leq 1, z = 0\}$.

    LHS
    (circulation)
    Parametrise $C$ as $\vec{x}(t) = \cos(t)\vec{e}_1 + \sin(t)\vec{e}_2$ for $t \in [0, 2\pi]$.
    So
    \begin{align*}
        \oint_{C}\vec{f}\cdot\hvec{t}\,d\ell &= \int_{0}^{2\pi}\vec{f}\cdot\frac{d\vec{x}}{dt}\,dt \\
        &= \int_{0}^{2\pi}(\cos(t)\vec{e}_2 - \cos(t)\vec{e}_3)\cdot(-\sin(t)\vec{e}_1 + \cos(t)\vec{e}_2)\,dt \\
        &= \int_{0}^{2\pi}\cos ^ 2(t)\,dt \\
        &= \pi.
    \end{align*}

    RHS
    (flux of $\bnabla \times \vec{f}$)
    \[
    \bnabla \times \vec{f} = \begin{vmatrix}
        \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
        \pd{x} & \pd{y} & \pd{z} \\
        z & x & -x
    \end{vmatrix} = 2\vec{e}_2 + \vec{e}_3.
    \]
    Unit normal is $\hvec{n} = \vec{e}_3$,
    so
    \begin{align*}
        \int_{S}(\bnabla \times \vec{f})\cdot\hvec{n}\,dS &= \int_{S}(2\vec{e}_2 + \vec{e}_3)\cdot\vec{e}_3\,dS \\
        &= \int_{S}\,dS \\
        &= \pi(1) ^ 2 = \pi = \text{LHS}.
    \end{align*}
\end{example}

\begin{example}
    Repeat the surface integral in the previous example but with $S$ the hemisphere $\{x ^ 2 + y ^ 2 + z ^ 2 = 1, z \geq 0\}$.
    (Stokes' theorem says that the answer is still $\pi$).

    Now $\hvec{n} = \vec{e}_r$
    (spherical coordinates),
    so
    \begin{align*}
        \int_{S}(\bnabla \times \vec{f})\cdot\hvec{n}\,dS &= \int_{S}(2\vec{e}_2 + \vec{e}_3)\cdot\vec{e}_r\,dS \\
        &= \int_{S}(2\vec{e}_2 + \vec{e}_3)\cdot\left(\frac{x}{r}\vec{e}_1 + \frac{y}{r}\vec{e}_2 + \frac{z}{r}\vec{e}_3\right)\,dS \\
        &= \int_{S}\frac{2y}{r} + \frac{z}{r}\,dS \\
        &= \int_{S}2\sin(\theta)\sin(\phi) + \cos(\theta)\,dS \\
        &= \int_{0}^{2\pi}\int_{0}^{\pi / 2}(2\sin(\theta)\sin(\phi) + \cos(\theta))\sin(\theta)\,d\theta\,d\phi &&(\substack{\text{parametrise with} \\\text{spherical coordinates}}) \\
        &= \int_{0}^{2\pi}\int_{0}^{\pi / 2}2\sin ^ 2(\theta)\sin(\phi) + \cos(\theta)\sin(\theta)\,d\theta\,d\phi \\
        &= \int_{0}^{2\pi}\left[-\cos ^ 2(\theta) / 2\right]_{0}^{\pi / 2}\,d\phi \\
        &= \int_{0}^{2\pi}\frac{1}{2}\,d\phi = \pi.
    \end{align*}
    Same as before.
\end{example}

The special case where $S \subset \R ^ 2$ has a name:
\begin{corollary}[Green's Theorem]
    If $\vec{f} : \R ^ 2 \to \R ^ 2$ is a $C ^ {1}$ vector field and $C$ is a closed curve in $\R ^ 2$ oriented anti-clockwise,
    then
    \[
    \oint_{C}\vec{f}\cdot\hvec{t}\,d\ell = \int_{A}\left(\pd[f_2]{x} - \pd[f_1]{y}\right)\,dx\,dy.
    \]
\end{corollary}

\begin{corollary}
    If $\vec{f} : \R ^ 3 \to \R ^ 3$ is a $C ^ {1}$ vector field with $\bnabla \times \vec{f} = \vec{0}$ throughout some simply-connected region $V \subset \R ^ 3$,
    then $\vec{f}$ is conservative in $V$.

    \begin{proof}
        Recall that $\vec{f} = \bnabla g$ if and only if $\vec{f}$ has path independent line integrals.
        So consider two arbitrary curves $C_1, C_2$ between $\vec{x}_a, \vec{x}_b \in V$:
        Define $C = C_1 - C_2$.
        Since $V$ is simply-connected
        (by assumption),
        $\exists$ a capping surface $S \subset V$ for $C$.
        Stokes' theorem says
        \[
        \oint_{C}\vec{f}\cdot\hvec{t}\,d\ell = \int_{S}(\overbrace{\bnabla \times \vec{f}}^{= 0}) \cdot \hvec{n}\,dS = 0.
        \]
        Which implies
        \[
        \int_{C_1}\vec{f}\cdot\hvec{t}\,d\ell = \int_{C_2}\vec{f}\cdot\hvec{t}\,d\ell
        \]
        i.e. path independence holds so $\vec{f}$ is conservative.
    \end{proof}
\end{corollary}
A domain is simply-connected if any curve can be shrunk to a point within the domain.

\subsection{Proof of Stokes' Theorem}

\begin{lemma}
    Let $C$ be a planar curve in a plane with unit normal $\hvec{n}$,
    with diameter $d(C)$,
    enclosed area $|A|$ and containing the point $\vec{x}$.

    For any vector field $\vec{f} : \R ^ 3 \to \R ^ 3$,
    \[
    \hvec{n} \cdot (\bnabla \times \vec{f})\,\biggr|_{\vec{x}} = \lim_{d(C) \to 0}\frac{1}{|A|}\oint_{C}\vec{f}\cdot\hvec{t}\,d\ell
    \]
    where $C$ is oriented with $\hvec{t} = \hvec{n} \times \hvec{n}_{C}\footnotemark$.
    \footnotetext{Outward normal to the curve.}

    \begin{proof}
        "Extrude"
        $C$ by a small distance $h$ in the $\hvec{n}$-direction to form a thin
        "pill"
        surface $S$,
        with normal $\hvec{n}_{S}$.
        By the definition of curl,
        \begin{align*}
            \hvec{n}\cdot(\bnabla \times \vec{f})\,\biggr|_{\vec{x}} &= \hvec{n}\cdot\left[\lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n}_{S}\times \vec{f}\,dS\right] \\
            &= \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n}\cdot(\hvec{n}_S\times\vec{f})\,dS \\
            &= \lim_{d(S) \to 0}\frac{1}{|A|h}\oint_{S}\vec{f}\cdot(\hvec{n}\times\hvec{n}_{S})\,dS \\
            \intertext{$\hvec{n} \times \hvec{n}_{S}$ vanishes on top and bottom since $\hvec{n}_{S} = \pm \hvec{n}$.}
            &= \lim_{|A|, d(S) \to 0}\frac{1}{|A|h}\int_{\tilde{S}}\vec{f}\cdot(\hvec{n}\times\hvec{n}_{C})\,dS \\
            \intertext{$\tilde{S}$ is the curved surface}
        \end{align*}
        Parametrise $\tilde{S}$ by $\tilde{x}(t, v) = \vec{x}_{C}(t) + t\hvec{n}$ for $v \in [0, h]$,
        $t \in [t_0, t_1]$,
        $\vec{x}_{C}$ is a parametrisation of $C$.
        \[
        \left|\pd[\tilde{\vec{x}}]{t} \times \pd[\tilde{\vec{x}}]{v}\right| = \left|\frac{d\vec{x}_{C}}{dt}\times\hvec{n}\right| = \left|\frac{d\vec{x}_{C}}{dt}\right||\hvec{n}| = \left|\frac{d\vec{x}_{C}}{dt}\right|.
        \]
        So
        \begin{align*}
            \int_{\tilde{S}}\vec{f}\cdot(\hvec{n}\times\hvec{n}_{C})\,dS &= \int_{t_0}^{t_1}\int_{0}^{h}\vec{f}\cdot(\hvec{n}\times\hvec{n}_{C})\left|\frac{d\vec{x}_{C}}{dt}\right|\,dv\,dt \\
            &= \int_{t_0}^{t_1}(\hvec{n}\times\hvec{n}_{C})\cdot\left(\int_{0}^{h}\vec{f}\,dv\right)\left|\frac{d\vec{x}_{C}}{dt}\right|\,dt \\
            \intertext{$\int_{0}^{h}\vec{f}\,dv = h\vec{f}(t, v_{*}(t))$ for some $v_{*}(t) \in [0, h]$ by the mean value theorem for integrals}
            &= h\int_{t_0}^{t_1}(\hvec{n} \times \hvec{n}_{C})\cdot\vec{f}(t, v_{*}(t))\left|\frac{d\vec{x}_{C}}{dt}\right|\,dt.
        \end{align*}
        In the limit $h \to 0$,
        we have $v_{*}(t) \to 0$ for all $t$
        (squeezing theorem)
        so
        \begin{align*}
            \hvec{n}\cdot(\bnabla \times \vec{f})\,\biggr|_{\vec{x}} &= \lim_{|A| \to 0}\frac{1}{|A|}\int_{t_0}^{t_1}\underbrace{(\hvec{n}\times\hvec{n}_{C})}_{=\hvec{t}}\cdot\vec{f}(t, 0)\left|\frac{d\vec{x}_{C}}{dt}\right|\,dt \\
            &= \lim_{|A| \to 0}\frac{1}{|A|}\oint_{C}\vec{f}\cdot\hvec{t}\,d\ell.
        \end{align*}
        \[
        \]
        
    \end{proof}
\end{lemma}
I.e. projecting $\bnabla \times \vec{f}$ in some fixed direction $\hvec{n}$ gives
"circulation per unit area"
around that axis.

\begin{theorem}[continues = mathmethII:thm:stokesthm]
    \begin{proof}
        (Similar to divergence theorem)

        Subdivide $S$ into a disjoint union of smaller regions $S_i$ for $i = 1, \dotsc, m$ whose boundary curves have unit tangent vectors $\hvec{t}_i$.

        Observe that each boundary segment is shared by two neighbouring regions with equal and opposite tangents.

        So
        \begin{align*}
            \oint_{C}\vec{f}\cdot\hvec{t}\,d\ell &= \sum_{i = 1}^{m}\oint_{C_i}\vec{f}\cdot\hvec{t}_i\,d\ell \\
            &= \sum_{i = 1}^{m}\left(\frac{1}{|A_i|}\oint_{C_i}\vec{f}\cdot\hvec{t}_i\,d\ell\right)|A_i| \\
            &= \lim_{m \to \infty}\sum_{i = 1}^{m}\left(\frac{1}{|A_i|}\oint_{C_i}\vec{f}\cdot\hvec{t}_i\,d\ell\right)|A_i| \\
            &= \int_{S}(\bnabla \times \vec{f})\cdot\hvec{n}\,dS.
        \end{align*}
    \end{proof}
\end{theorem}

\begin{remark}
    There is a much more general form of
    "Stokes' theorem"
    in differential geometry,
    written elegantly as
    \[
    \int_{\Omega}d\omega = \int_{\partial\Omega}\omega.
    \]
\end{remark}

\section{Index Notation}

\subsection{Scalar Products}

Index/suffix notation -
a powerful method for simplifying vector equations.
E.g. $\vec{c} = \vec{a} + \vec{b} \to c_i = a_i + b_i\footnotemark$ where it is understood that it holds for $i = 1, \dotsc, n$
\footnotetext{$i$ would be $n$ for $\vec{a} \in \R ^ n$}.
(i.e.
any component in the Cartesian basis).

\begin{remark}
    In future courses may encounter "tensors" with more than one index.
\end{remark}

An index that appears once in each term is called a free index.
The choice of letter is arbitrary,
but all terms must have the same free index
(or indices).

In index notation,
we write a scalar product as:
\[
\vec{a} \cdot \vec{b} = \sum_{j = 1}^{n}a_jb_j
\]
repeated index implies a sum from $1$ to $n$
(called the Einstein summation convention).

An index that appears exactly twice in the same term is called a dummy index,
and always implies a summation.
Again the choice of letter is arbitrary.

\begin{example}
    Express $(\vec{a} \cdot \vec{b})(\vec{c} \cdot \vec{d})$ in index notation.

    Write $\vec{a} \cdot \vec{b} = a_jb_j$ and $\vec{c} \cdot \vec{d} = c_kd_k$
    (cannot use $j$ again).
    So
    \[
    (\vec{a}\cdot\vec{b})(\vec{c}\cdot\vec{d}) = a_jb_jc_kd_k.
    \]
\end{example}

\begin{example}
    Express $a_jb_ic_j$ in vector notation.

    Here:
    $i$ appears once so is a free index,
    $j$ appears twice so is a dummy index.
    Since the expression has one free index,
    it is the $i$th component of a vector:
    \[
    a_jb_ic_j = [\vec{b}(\vec{a} \cdot \vec{c})]_i.
    \]
\end{example}

\begin{example}
    Express the equation $\vec{u} + (\vec{a} \cdot \vec{b})\vec{v} = |\vec{a}| ^ 2(\vec{b} \cdot \vec{v})\vec{a}$ in index notation.

    Each side is a vector,
    so we need to write an equation for the $i$th component:
    \[
    u_i + (\vec{a} \cdot \vec{b})v_i = |\vec{a}| ^ 2(\vec{b} \cdot \vec{v})a_i.
    \]
    Introduce dummy indices for scalar products
    \[
    u_i + a_jb_jv_i = a_ja_jb_kv_ka_i.
    \]
\end{example}

Another way to think of $\vec{a} \cdot \vec{b}$ is as a quadratic form:
\[
\vec{a} \cdot \vec{b} = \begin{pmatrix}
    a_1 & a_2 & a_3
\end{pmatrix} \begin{pmatrix}
    1 & 0 & 0 \\ 0 & 1 & 0 \\ 0 & 0 & 1
\end{pmatrix}\begin{pmatrix}
    b_1 \\ b_2 \\ b_3
\end{pmatrix}.
\]
In index notation,
\[
\vec{a} \cdot \vec{b} = \delta_{ij}a_ib_j
\]
where $\delta_{ij} = \begin{cases}
    1 & i = j \\
    0 & i \neq j.
\end{cases}$
(Kronecker delta).

\begin{proposition}
    Multiplying an expression with free index $i$ by $\delta_{ij}$ is equivalent to replacing $i \to j$.

    \begin{proof}
        $a_i\delta_{ij}$,
        $i$ becomes a dummy.
        If $j = 1$,
        \[
        a_i\delta_{i1} = a_1\delta_{11} + a_2\delta_{21} + \dotsc + a_n\delta_{n1} = a_1.
        \]
        Similarly,
        for any $j$,
        we have $a_i\delta_{ij} = a_j$.
    \end{proof}
\end{proposition}

\begin{example}
    Simplify $\delta_{ij}\delta_{kj}$.

    \begin{solution}
        By the previous,
        multiplying by $\delta_{jk}$ is equivalent to replacing $j \to k$ so
        \[
        \delta_{ij}\delta_{kj} = \delta_{ik}.
        \]
    \end{solution}
\end{example}

\begin{example}
    Simplify $\delta_{ij}\delta_{ji}$.

    \begin{solution}
        By the previous proposition,
        multiplying by $\delta_{ij}$ is equivalent to replacing $j \to i$ so
        \[
        \delta_{ij}\delta_{ji} = \delta_{ii} = \sum_{r = 1}^{n}\delta_{rr} = \sum_{r = 1}^{n}1 = n.
        \]
    \end{solution}
\end{example}

\subsection{Vector Products}
In $\R ^ 3$,
we can write $\vec{a} \times \vec{b}$ in index notation by introducing the alternating tensor/Levi-Civita symbol
\[
\epsilon_{ijk} = \begin{cases}
    0 &\text{if any $i, j, k$ are equal} \\
    +1 &\text{if $(ijk) = (123), (231)$ or $(312)$} \\
    -1 &\text{if $(ijk) = (132), (213)$ or $(321)$}
\end{cases}
\]
with $+1$ the even permutations of $(123)$,
$-1$ the odd permutations of $(123)$.
It follows that
\begin{align*}
    \epsilon_{ijk} &= \epsilon_{jki} = \epsilon_{kij} \\
    \epsilon_{ijk} &= -\epsilon_{jik}.
\end{align*}
So
\[
[\vec{a} \times \vec{b}]_i = \epsilon_{ijk}a_jb_k.
\]

For example,
first component of right-hand side:
\begin{align*}
    \epsilon_{1jk}a_jb_k &= \epsilon_{11k}a_1b_k + \epsilon_{12k}a_1b_k + \epsilon_{13k}a_1b_k \\
    &= \epsilon_{123}a_2b_3 + \epsilon_{132}a_3b_2 \\
    &= a_2b_3 - a_3b_2 \\
    &= (\vec{a} \times \vec{b})\cdot\vec{e}_1.
\end{align*}

\begin{example}
    Use index notation to show that $\vec{a} \cdot(\vec{b} \times \vec{c}) = \vec{c} \cdot(\vec{a} \times \vec{b})$.

    \begin{solution}
        \begin{align*}
            \vec{a} \cdot (\vec{b}\times\vec{c}) &= a_i[\vec{b} \times \vec{c}]_i \\
            &= a_i\epsilon_{ijk}b_jc_k \\
            &= \epsilon_{ijk}a_ib_jc_k \\
            &= \epsilon_{ijk}c_ka_ib_j \\
            &= \epsilon_{kij}c_ka_ib_j \\
            &= c_k\epsilon_{kij}a_ib_j \\
            &= c_k[\vec{a} \times \vec{b}]_k \\
            &= \vec{c} \cdot (\vec{a} \times \vec{b}).
        \end{align*}
    \end{solution}
\end{example}

When simplifying more complicated expressions,
there is a useful identity for the product of two $\epsilon$'s that share an index.

\begin{proposition}
    \[
    \epsilon_{ijk}\epsilon_{klm} = \delta_{il}\delta_{jm} - \delta_{im}\delta_{jl}.
    \]

    \begin{proof}
        Left-hand side is
        \[
        \epsilon_{ijk}\epsilon_{klm} = \begin{cases}
            +1 &\text{if $(klm)$ even permutation of $(ijk)$}, \\
            -1 &\text{if $(klm)$ odd permutation of $(ijk)$}, \\
            0 &\text{if $(klm)$ is not a permutation of $(ijk)$}.
        \end{cases}
        \]
        So
        ($(klm) = (ijk), (ikj), (kij), (jik), (jki), (kji)$)
        \[
        \]
        \begin{align*}
            \epsilon_{ijk}\epsilon_{klm} &= \delta_{ik}\delta_{jl}\delta_{km} - \delta_{ik}\delta_{kl}\delta_{jm} + \delta_{kk}\delta_{il}\delta_{jm} - \delta_{jk}\delta_{il}\delta_{km} + \delta_{jk}\delta_{kl}\delta_{im} - \delta_{kk}\delta_{jl}\delta_{im} \\
            \intertext{for the first term $[]_k\delta_{km} = []_m$}
            &= \delta_{im}\delta_{jl} - \delta_{il}\delta_{jm} + 3\delta_{il}\delta_{jm} - \delta_{jm}\delta_{il} + \delta_{jl}\delta_{im} - 3\delta_{jl}\delta_{im} \\
            &= \delta_{il}\delta_{jm} - \delta_{im}\delta_{jl}.
        \end{align*}
    \end{proof}
\end{proposition}

\begin{example}
    Use index notation to prove that $\vec{a} \times (\vec{b} \times \vec{c}) = (\vec{a} \cdot \vec{c})\vec{b} - (\vec{a} - \vec{b})\vec{c}$.

    \begin{solution}
        The $i$th component of the left-hand side is
        \begin{align*}
            [\vec{a} \times (\vec{b} \times \vec{c})]_i &= \epsilon_{ijk}a_j[\vec{b} \times \vec{c}]_k \\
            &= \epsilon_{ijk}a_j(\epsilon_{klm}b_lc_m) \\
            &= \epsilon_{ijk}\epsilon_{klm}a_jb_lc_m \\
            &= (\delta_{il}\delta_{jm} - \delta_{im}\delta_{jl})a_jb_lc_m \\
            &= \delta_{il}\delta_{jm}a_jb_lc_m - \delta_{im}\delta_{jl}a_jb_lc_m \\
            &= a_j(\delta_{il}b_l)(\delta_{jm}c_m) - a_j(\delta_{im}c_m)(\delta_{jl}b_l) \\
            &= a_j(\delta_{il}b_l)(\delta_{jm}c_m) - a_j(\delta_{im}c_m)(\delta_{jl}b_l) \\
            &= a_jb_ic_j - a_jc_ib_j \\
            &= (a_jc_j)b_i - (a_jb_j)c_i \\
            &= (\vec{a} \cdot \vec{c})\vec{b} - (\vec{a} \cdot \vec{b})\vec{c}.
        \end{align*}
    \end{solution}
\end{example}

\subsection{Derivatives}

Recall the Cartesian expressions:
\[
\bnabla g = \pd[g]{x}\vec{e}_1 + \pd[g]{y}\vec{e}_2 + \pd[g]{z}\vec{e}_3 \to [\bnabla g]_{i} = \pd[g]{x_i}.
\]
\[
\bnabla \cdot \vec{f} = \pd[f_1]{x} + \pd[f_2]{y} + \pd[f_3]{z} \to \bnabla \cdot \vec{f} = \pd[f_j]{x_j}.
\]
\[
\bnabla \times \vec{f} = \begin{vmatrix}
    \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
    \pd{x} & \pd{y} & \pd{z} \\
    f_1 & f_2 & f_3
\end{vmatrix} \to [\bnabla \times \vec{f}]_{i} = \epsilon_{ijk}\pd{x_j}f_k.
\]

\begin{example}
    Use index notation to verify $\bnabla \cdot (\bnabla \times \vec{f}) = 0$.

    \begin{solution}
        \begin{align*}
            \bnabla \cdot (\bnabla \times \vec{f}) &= \pd{x_j}[\bnabla \times \vec{f}]_j \\
            &= \pd{x_j}\left(\epsilon_{jkl}\pd{x_k}f_l\right) \\
            &= \epsilon_{jkl}\pd{x_j}\left(\pd[f_l]{x_k}\right) \\
            &= \epsilon_{jkl}\pd{x_k}\left(\pd[f_l]{x_j}\right) \\
            &= \epsilon_{kjl}\pd{x_j}\left(\pd[f_l]{x_k}\right) &&(\epsilon_{kjl} = -\epsilon_{jkl}). \\
            &= -\epsilon_{jkl}\pd{x_j}\left(\pd[f_l]{x_k}\right) \\
            &= -\pd{x_j}\left(\epsilon_{jkl}\pd[f_l]{x_k}\right).
        \end{align*}
        Since this expression equals minus itself,
        it must be zero.
    \end{solution}
\end{example}

Index notation gives us a neater way to complete the following proofs:
\begin{proposition}[continues = mathmethII:prop:prodrules]\phantom{}
    \begin{enumerate}[label = (\arabic*)]
        \item[(5)]
        \[
        \bnabla (\vec{f} \times \vec{g}) \equiv (\vec{g} \cdot \bnabla)\vec{f} - (\vec{f}\cdot\bnabla)\vec{g} + \vec{f}(\bnabla \cdot\vec{g}) - \vec{g}(\bnabla \cdot \vec{f}).
        \]
    
        \item[(6)]
        \[
        \bnabla(\vec{f} \cdot \vec{g}) \equiv (\vec{g}\cdot\bnabla)\vec{f} + (\vec{f}\cdot\bnabla)\vec{g} + \vec{f}\times(\bnabla \times \vec{g}) + \vec{g} \times (\bnabla \times \vec{f}).
        \]
    \end{enumerate}

    \begin{proof}
        \begin{enumerate}[label = (\arabic*)]
            \item[(5)]
            \begin{align*}
                [\bnabla \times (\vec{f} \times \vec{g})]_i &= \epsilon_{ijk}\pd{x_j}[\vec{f} \times \vec{g}]_k \\
                &= \epsilon_{ijk}\pd{x_j}(\epsilon_{klm}f_lg_m) \\
                &= \epsilon_{ijk}\epsilon_{klm}\pd{x_j}(f_lg_m) \\
                &= \epsilon_{ijk}\epsilon_{klm}\left(\pd[f_l]{x_j}g_m + f_l\pd[g_m]{x_j}\right) \\
                &= (\delta_{il}\delta_{jm} - \delta_{im}\delta_{jl})\left(\pd[f_l]{x_j}g_m + f_l\pd[g_m]{x_j}\right) \\
                &= \delta_{il}\delta_{jm}\pd[f_l]{x_j}g_m + \delta_{il}\delta_{jm}f_l\pd[g_m]{x_j} - \delta_{im}\delta_{jl}\pd[g_m]{x_j}f_l - \delta_{im}\delta_{jl}g_m\pd[f_l]{x_j} \\
                &= \delta_{il}g_j\pd[f_l]{x_j} + \delta_{il}f_l\pd[g_j]{x_j} - \delta_{jl}\pd[g_i]{x_j}f_l - \delta_{jl}g_i\pd[f_l]{x_j} \\
                &= g_j\pd[f_i]{x_j} + f_i\pd[g_j]{x_j} - \pd[g_i]{x_j}f_j - g_i\pd[f_j]{x_j} \\
                &= [(\vec{g} \cdot \bnabla)\vec{f}]_i + [\vec{f}(\bnabla\cdot\vec{g})]_i - [(\vec{f}\cdot\bnabla)\vec{g}]_i - [(\bnabla \cdot \vec{f)\vec{g}}]_i.
            \end{align*}
            
            \item[(6)]
            Here the RHS has the triple products,
            so start there:
            \begin{align*}
                 [\vec{f}\times(\bnabla \times \vec{g})]_i &= \epsilon_{ijk}f_j[\bnabla \times \vec{g}]_k \\
                &= \epsilon_{ijk}f_j\epsilon_{klm}\pd{x_l}g_m \\
                &= \epsilon_{ijk}\epsilon_{klm}f_j\pd{x_l}g_m \\
                &= (\delta_{il}\delta_{jm} - \delta_{im}\delta_{jl})f_j\pd[g_m]{x_l} \\
                &= \delta_{il}\delta_{jm}f_j\pd[g_m]{x_l} - \delta_{im}\delta_{jl}f_j\pd[g_m]{x_l} \\
                &= \delta_{il}f_j\pd[g_j]{x_l} - \delta_{jl}f_j\pd[g_i]{x_l} \\
                &= f_j\pd[g_j]{x_i} - f_j\pd[g_i]{x_j}
            \end{align*}
            so RHS of the original identity is
            \begin{align*}
                []_i &= g_j\pd[f_i]{x_j} + f_j\pd[g_i]{x_j} + f_j\pd[g_j]{x_i} - f_j\pd[g_i]{x_j} + g_j\pd[f_j]{x_i} - g_j\pd[f_i]{x_j} \\
                &= f_j\pd[g_j]{x_i} + g_j\pd[f_j]{x_i} \\
                &= \pd{x_i}(f_jg_j) \\
                &= \pd{x_i}(\vec{f} \cdot \vec{g}) \\
                &= \left[\bnabla(\vec{f} \cdot \vec{g})\right]_i.
            \end{align*}
        \end{enumerate}
    \end{proof}
\end{proposition}

\begin{remark}
    The operator $(\vec{g} \cdot \bnabla)\vec{f}$ has $i$th
    (Cartesian)
    component $g_j\pd[f_i]{x_j}$,
    and can be defined as
    \[
    (\vec{g} \cdot \bnabla)\vec{f} = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\vec{f}(\vec{g}\cdot\hvec{n})\,dS.
    \]
\end{remark}

\subsection{Second Derivatives}
Derivative operators:
\begin{align*}
    &\bnabla \cdot (\bnabla g) \\
    &\bnabla \times (\bnabla g) = \vec{0} \\
    &\bnabla(\bnabla \cdot \vec{f}) \\
    &\bnabla \times (\bnabla \times \vec{f}) \\
    &\bnabla \cdot (\bnabla \times \vec{f}) = 0.
\end{align*}

So there are three non-trivial second derivative operators for a scalar field $g : \R ^ 3 \to \R$ and a vector field $\vec{f} : \R ^ 3 \to \R ^ 3$:
\begin{enumerate}[label = (\arabic*)]
    \item $\bnabla \cdot (\bnabla g) : \R \to \R$.
    
    \item $\bnabla (\bnabla \cdot \vec{f}) : \R ^ 3 \to \R ^ 3$.

    \item $\bnabla \times (\bnabla \times \vec{f}) : \R ^ 3 \to \R ^ 3$.
\end{enumerate}
Operator $1$ is called the Laplacian of a scalar field $\nabla ^ 2g$ or $\Delta g$.

\begin{example}
    Compute $\nabla ^ 2g$ for $g(\vec{x}) = x ^ 3 - 3xy ^ 2$.

    In Cartesians,
    \[
    \nabla ^ 2g = \bnabla \cdot (\bnabla g) = \pd{x_j}[\bnabla g]_j = \pd{x_j}\pd[g]{x_j} = \frac{\partial ^ 2g}{\partial x ^ 2} + \frac{\partial ^ 2g}{\partial y ^ 2} + \frac{\partial ^ 2g}{\partial z ^ 2}.
    \]
    So
    \[
    \nabla ^ 2g = \pd{x}(3x ^ 2 - 3y ^ 2) + \pd{y}(-6xy) = 6x - 6x = 0.
    \]
\end{example}

\begin{remark}
    Scalar fields satisfying $\nabla ^ 2g = 0$ are called harmonic functions.
\end{remark}

\begin{proposition}
    In orthogonal curvilinear coordinates $(u, v, w)$,
    the Laplacian is given by
    \[
    \nabla ^ 2g = \frac{1}{h_uh_vh_w}\left[\pd{u}\left(\frac{h_vh_w}{h_u}\pd[g]{u}\right) + \pd{v}\left(\frac{h_uh_w}{h_v}\pd[g]{v}\right) + \pd{w}\left(\frac{h_uh_v}{h_w}\pd[g]{w}\right)\right].
    \]

    \begin{proof}
        Recall:
        $\bnabla g = \frac{1}{h_u}\pd[g]{u}\vec{e}_u + \frac{1}{h_v}\pd[g]{v}\vec{e}_v + \frac{1}{h_w}\pd[g]{w}\vec{e}_w$ and $\bnabla\cdot\vec{f} = \frac{1}{h_uh_vh_w}\left[\pd{u}(h_vh_wf_u) + \pd{v}(h_uh_wf_v) + \pd{w}(h_uh_vf_w)\right]$.
        Simply let $\vec{f} = \bnabla g$.
    \end{proof}
\end{proposition}

The operator $2$ $\bnabla(\bnabla \cdot \vec{f})$ has $i$th component $[\bnabla(\bnabla \cdot \vec{f})]_i = \pd{x_i}\left(\pd[f_j]{x_j}\right)$.

The operator $3$ $\bnabla \times (\bnabla \times \vec{f})$ has $i$th component
\begin{align*}
    [\bnabla \times (\bnabla \times \vec{f})]_i &= \epsilon_{ijk}\pd{x_j}\left(\epsilon_{klm}\pd[f_m]{x_l}\right) \\
    &= \epsilon_{ijk}\epsilon_{klm}\pd{x_j}\left(\pd[f_m]{x_l}\right) \\
    &= (\delta_{il}\delta_{jm} - \delta_{im}\delta_{jl})\pd{x_j}\left(\pd[f_m]{x_l}\right) \\
    &= \delta_{il}\delta_{jm}\pd{x_j}\left(\pd[f_m]{x_l}\right) - \delta_{im}\delta_{jl}\pd{x_j}\left(\pd[f_m]{x_l}\right) \\
    &= \delta_{il}\pd{x_j}\left(\pd[f_j]{x_l}\right) - \delta_{jl}\pd{x_j}\left(\pd[f_i]{x_l}\right) \\
    &= \pd{x_j}\left(\pd[f_j]{x_i}\right) - \pd{x_j}\left(\pd[f_i]{x_j}\right) \\
    &= \pd{x_i}\left(\pd[f_j]{x_j}\right) - \frac{\partial ^ 2f_i}{\partial x_j\partial x_j} \\
    &= [\bnabla \cdot(\bnabla \cdot \vec{f})]_i - \nabla ^ 2f_i.
\end{align*}

We define the vector Laplacian as the vector field with Cartesian components
\[
[\bnabla ^ 2\vec{f}]_i = \nabla ^ 2f_i
\]
so
\[
\bnabla \times (\bnabla \times \vec{f}) = \bnabla(\bnabla \cdot \vec{f}) - \bnabla ^ 2\vec{f}.
\]

\begin{example}
    Verify this formula for the vector field $\vec{f}(\vec{x}) = x\vec{e}_1 + xy\vec{e}_2 + y ^ 3\vec{e}_3$.

    \begin{align*}
        \bnabla \cdot \vec{f} &= \pd{x}(x) + \pd{y}(xy) + \pd{z}(y ^ 3) = 1 + x. \\
        &\implies \bnabla (\bnabla \cdot \vec{f}) = \bnabla(1 + x) = \vec{e}_1. \\
        \bnabla\times\vec{f} &= \begin{vmatrix}
            \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
            \pd{x} & \pd{y} & \pd{z} \\
            x & xy & y ^ 3
        \end{vmatrix} = 3y ^ 2\vec{e}_1 + y\vec{e}_3 \\
        &\implies \bnabla \times (\bnabla \times \vec{f}) = \begin{vmatrix}
            \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\ \pd{x} & \pd{y} & \pd{z} \\ 3y ^ 2 & 0 & y
        \end{vmatrix} = \vec{e}_1 - 6y\vec{e}_3.
    \end{align*}
    In Cartesians,
    \begin{align*}
        \bnabla ^ 2\vec{f} &= (\nabla ^ 2f_1)\vec{e}_1 + (\nabla ^ 2f_2)\vec{e}_2 + (\nabla ^ 2f_3)\vec{e}_3 \\
        &= (0)\vec{e}_1 + (0)\vec{e}_2 + \frac{\partial ^ 2f_3}{\partial y ^ 2}\vec{e}_3 \\
        &= 6y\vec{e}_3.
    \end{align*}
    Therefore $\bnabla \times (\bnabla \times \vec{f}) = \bnabla(\bnabla \cdot \vec{f}) - \bnabla ^ 2\vec{f}$.
\end{example}

Warning:
Be careful when calculating $\bnabla ^ 2\vec{f}$ in non-Cartesian systems,
because
\[
[\bnabla ^ 2\vec{f}]\cdot\vec{e}_u \neq \nabla ^ 2f_u.
\]
To find the correct expression,
calculate
\[
\bnabla ^ 2\vec{f} = \bnabla(\bnabla\cdot\vec{f}) - \bnabla\times(\bnabla\times\vec{f}).
\]

\section{Partial Differential Equations}

\subsection{What is a PDE}

The heat equation
\[
\pd[u]{t} = D\left(\frac{\partial ^ 2u}{\partial x ^ 2} + \frac{\partial ^ 2u}{\partial y ^ 2}\right)
\]
dependent variable $u$,
independent variables $x, y, t$.

\begin{definition}
    A PDE is an equation containing an unknown function of two or more variables.
\end{definition}

\begin{definition}
    The order,
    the highest derivative
    ($2$ above).
\end{definition}

\begin{definition}
    Linear/non-linear.
    A PDE is linear/non-linear if it is linear/non-linear in the dependent variable
\end{definition}

\begin{example}
    \begin{enumerate}[label = \alph*)]
        \item
        \[
        \pd[u]{t} + \frac{\partial ^ 3u}{\partial x ^ 3} - 6u\pd[u]{t} = 0
        \]
        $u$ dependent variable,
        $x, t$ independent,
        $3$rd order.
        Non-linear due to $6u\pd[u]{x}$.

        \item
        \[
        \frac{\partial ^ 2u}{\partial t ^ 2} + \sin(u)\frac{\partial ^ 2u}{\partial x\partial y} = 0
        \]
        $2$nd order.
        Non-linear due to $\sin(u)$.

        \item
        \[
        \frac{\partial ^ 3u}{\partial t ^ 3} + \sin(x)\frac{\partial ^ 2u}{\partial x\partial y} = 0
        \]
        $3$rd order.
        Linear as $\sin(x)$ is in the independent variable.
    \end{enumerate}
\end{example}

\subsection{Solutions to PDEs}

\begin{definition}
    A solution to a PDE in some region $R$ of the space of independent variables is a function which possesses all the partial derivatives in the PDE in $R$,
    and satisfies the equation.
\end{definition}

\begin{example}
    \[
    \pd[u]{t} = D\left(\frac{\partial ^ 2u}{\partial x ^ 2} + \frac{\partial ^ 2u}{\partial y ^ 2}\right).
    \]
    $2$ spacial variables $x, y$,
    $1$ temporal variable $t$.
    Consider $R = [0, L_x] \times [0, L_y] \times (0, \infty)$
    $y \in [0, L_y]$,
    $x \in [0, L_x]$,
    $t \in (0, \infty)$.
    This is a $2$D PDE due to the $2$ spacial variables,
    even though $R$ is $3$D
    (standard nomenclature).

    Solutions
    \[
    u(x, y, t) = e ^ {-D\left(\frac{n ^ 2}{L_{x} ^ 2} + \frac{m ^ 2}{L_{y} ^ 2}\right)\pi ^ 2t}\cos\left(\frac{n\pi x}{L_{x}}\right)\cos\left(\frac{m\pi x}{L_{y}}\right)
    \]
    $n, m \in \Z$
    \[
    u(x, y, t) = e ^ {-D\left(\frac{n ^ 2}{L_{x} ^ 2} + \frac{m ^ 2}{L_{y} ^ 2}\right)\pi ^ 2t}\sin\left(\frac{n\pi x}{L_{x}}\right)\sin\left(\frac{m\pi x}{L_{y}}\right).
    \]
\end{example}

\subsubsection{Boundary conditions matter}
Two example boundary conditions matter:
\begin{enumerate}[label = (\alph*)]
    \item
    \[
    u(0, y, t) = u(L_x, y, t) = u(x, 0, t) = u(x, L_y, t) = 0.
    \]
    Zero all around the boundary.
    
    \item
    \[
    \pd[u]{x}(0, y, t) = \pd[u]{x}(L_x, y, t) = \pd[u]{y}(x, 0, t) = \pd[u]{y}(x, L_y, t) = 0.
    \]
    So $\bnabla u \cdot \hvec{n}$.
\end{enumerate}
Take (a)
$\sin() = \sin(n\pi) = 0$ implies $\sin\sin$ solution satisfy
(a)
but $\cos(0) = 0$ satisfies
(b)
but the $\sin\sin$ solution doesn't.

Key takeaway.
The boundary conditions
(a)
are an example of Dirichlet boundary conditions:
Conditions on the value of the function $u$ on the boundary.

Boundary conditions
(b)
are Neumann boundary conditions,
conditions on the derivative of the function on the boundary.

They are homogenous if the values are zero on the boundary and in homogeneous.

Key takeaway.
Solutions are not fully defined without boundary conditions.
In this case a big difference is that
(b)
allows $n = m = 0$ non-trivial.

\subsection{Principle of linear superposition in the heat equation}
Since $n, m$ were arbitrary integers we can sum over all their values,
i.e.
\[
u(x, y, t) = \sum_{n = 0}^{\infty}\sum_{m = 0}^{\infty}C_{nm}e ^ {-D\left(\frac{x ^ 2}{L_x ^ 2} + \frac{y ^ 2}{L_y ^ 2}\right)\pi ^ 2t}\cos\left(\frac{n\pi x}{L_x}\right)\cos\left(\frac{m\pi y}{L_y}\right)
\]
(assume Neumann homogeneous boundary conditions).

Some key properties of this summed solution
\begin{enumerate}[label = (\arabic*)]
    \item At $t = 0$ this is a Fourier series,
    we know this can represent any square integrable function.

    \item The terms of a Fourier series are linearly independent i.e. only zero if all $c_{nm} = 0$
\end{enumerate}
Point $1$ means we can expect a lot of complex behaviour.
Point $2$ means we can describe this complex behaviour in terms of simple functions.

\subsection{Fourier series recap}

\subsubsection{Fourier series on \texorpdfstring{$x \in [0, L]$}{}}

$1$D
\[
f(x) = \frac{a_0}{2} + \sum_{n = 1}^{\infty}\left[a_n\cos\left(\frac{n\pi x}{L}\right) + b_n\sin\left(\frac{n\pi x}{L}\right)\right]
\]
$a_0 = \frac{2}{L}\int_{0}^{L}f(x)\,dx$,
$a_n = \frac{2}{L}\int_{0}^{L}f(x)\cos\left(\frac{n\pi x}{L}\right)\,dx$,
$b_n = \frac{2}{L}\int_{0}^{L}f(x)\sin\left(\frac{n\pi x}{L}\right)\,dx$

\subsubsection{Fourier series recap \texorpdfstring{$[0, L_x] \times [0, L_y]$}{}}

\begin{align*}
    f(x, y) = \sum_{n = 0}^{\infty}\sum_{m = 0}^{\infty}\left[a_{nm}\cos\left(\frac{n\pi x}{L_x}\right)\cos\left(\frac{n\pi y}{L_y}\right)\right.& \\
    &+ b_{nm}\cos\left(\frac{n\pi x}{L_x}\right)\sin\left(\frac{n\pi y}{L_y}\right) \\
    &+ c_{nm}\sin\left(\frac{n\pi x}{L_x}\right)\cos\left(\frac{n\pi y}{L_y}\right) \\
    &\left.+ d_{nm}\sin\left(\frac{n\pi x}{L_x}\right)\sin\left(\frac{n\pi y}{L_y}\right)\right]
\end{align*}
\[
a_{nm} = \frac{4}{L_xL_y}\int_{0}^{L_x}\int_{0}^{L_y}f(x, y)\cos\left(\frac{n\pi x}{L_x}\right)\sin\left(\frac{m\pi y}{L_y}\right)\,dx\,dy
\]

Orthogonality of Fourier coefficients to the result on $x \in [a, b]$
\[
\int_{a}^{b}\cos\left(\frac{n\pi x(x - a)}{b - a}\right)\cos\left(\frac{m\pi(x - a)}{b - a}\right)\,dx = 0
\]
unless $m = n$,
same for $\sin\sin$.
\[
\int_{a}^{b}\cos\left(\frac{n\pi x(x - a)}{b - a}\right)\sin\left(\frac{m\pi(x - a)}{b - a}\right)\,dx = 0
\]
for any $n, m$.
Use to grab coefficients.
\[
\int_{a}^{b}f(x)\cos\left(\frac{n\pi x(x - a)}{b - a}\right)\,dx = \int_{a}^{b}\cos\left(\frac{n\pi(x - a)}{b - a}\right)\sum_{n = 0}^{\infty}\left(a_n\cos\left(\frac{m\pi(x - a)}{b - a}\right) + b_n\sin\left(\frac{m\pi(x - a)}{b - a}\right)\right)\,dx
\]
only the term
\[
a_n\int_{a}^{b}\cos\left(\frac{n\pi(x - a)}{b - a}\right)\cos\left(\frac{n\pi(x - a)}{b - a}\right)\,dx
\]
is non-zero.
So
\[
= a_m\int_{a}^{b}\cos ^ 2\left(\frac{m\pi(x - a)}{b - a}\right)\,dx
\]
\[
a_m = \frac{\int_{a}^{b}f(x)\cos\left(\frac{m\pi(x - a)}{b - a}\right)\,dx}{\int_{a}^{b}\cos ^ 2\left(\frac{m\pi(x - a)}{b - a}\right)\,dx}
\]
if $a = 0$,
$b = L$ so
\[
f(x) = \sum_{n = 0}^{\infty}(a_n\cos(n\pi(x - a) / (b - a)) + b_n\sin(n\pi(x - a) / (b - a))).
\]

\begin{example}
    \begin{enumerate}[label = (\alph*)]
        \item Find Fourier series of $f(x) = 1$ on $x \in [0, L]$
        here on $[0, L]$ $\cos$ is odd,
        as $1$ is even on this domain all $\cos$ coefficients $a_n$ vanish.
        So
        \[
        b_n = \frac{2}{L}\int_{0}^{L}\sin\left(\frac{n\pi x}{L}\right)\,dx = \frac{2}{n\pi}(1 - (-1) ^ n) = \begin{cases}
            \frac{4}{n\pi} &\text{if $n$ even} \\
            0 &\text{if $n$ odd}.
        \end{cases}
        \]

        \item $f(x, y) = \frac{dy}{L_xL_y}$ on $[-L_x, L_x] \times [-L_y, L_y]$.
        Function is odd on this domain and so is $\sin$.
        So only $\sin\sin$ coefficients are non-zero.
        \[
        d_{mn}\footnotemark = \frac{1}{L_xL_y}\int_{-L_x}^{L_x}\int_{-L_y}^{L_y}\frac{xy}{L_xL_y}\sin\left(\frac{m\pi x}{L_x}\right)\sin\left(\frac{n\pi y}{L_y}\right)\,dx\,dy
        \]
        \footnotetext{$d_{mn}$ not $4$ as this domain size is $2L_x \times 2L_y$.}
        Use
        \[
        \int_{-L_x}^{L_x}x\sin\left(\frac{n\pi x}{L_x}\right)\,dx = \frac{2Lx ^ 2}{2n\pi}(1 - (-1) ^ n)
        \]
        \[
        d_{mn} = \frac{4}{\pi ^ 2mn}(1 - (-1) ^ m)(1 - (-1) ^ n)
        \]
    \end{enumerate}
\end{example}

\subsection{The Heat equation as a solution to the Fourier series}
We can write
$[0, L_x] \times [0, L_y]$
\[
U(x, y, t) = \sum_{n = 0}^{\infty}\sum_{m = 0}^{\infty}T_{mn}(t)\cos\left(\frac{n\pi x}{L_x}\right)\cos\left(\frac{m\pi y}{L_y}\right)
\]
\begin{align*}
    \pd[U]{t} &= \sum_{n = 0}^{\infty}\sum_{m = 0}^{\infty}\frac{dT_{mn}}{dt}\cos\left(\frac{n\pi x}{L_x}\right)\cos\left(\frac{m\pi y}{L_y}\right) \\
    &= -D\sum_{n = 0}^{\infty}\sum_{m = 0}^{\infty}\left(\frac{n ^ 2\pi ^ 2}{L_x ^ 2} + \frac{m ^ 2\pi ^ 2}{L_y ^ 2}\right)T_{mn}(t)\cos\left(\frac{n\pi x}{L_x}\right)\cos\left(\frac{m\pi y}{L_y}\right).
\end{align*}
Put it together
\[
\sum_{n = 0}^{\infty}\sum_{m = 0}^{\infty}\left[\frac{dT_{mn}}{dt} + \left(\frac{n ^ 2\pi ^ 2}{L_x ^ 2} + \frac{m ^ 2\pi ^ 2}{L_y ^ 2}\right)DT_{mn}(t)\right]\cos()\cos() = 0.
\]
To make the whole sum zero we can demand.
\[
\frac{dT_{mn}}{dt} + D\left(\frac{n ^ 2\pi ^ 2}{L_x ^ 2} + \frac{m ^ 2\pi ^ 2}{L_y ^ 2}\right)T_{mn} = 0.
\]
This is assuming that the series can only be zero if each coefficient is identically zero.
We know this is true of the Fourier series and it relies on the linear independence of the Fourier series,
which in turn relies on orthogonality. 
\[
T_{mn}(t) = a_{nm}e ^ {-D\left(\frac{n ^ 2\pi ^ 2}{L_x ^ 2} + \frac{m ^ 2\pi ^ 2}{L_y ^ 2}\right)t}
\]

Initial conditions:
at $t = 0$
\[
U(x, y, 0) = \sum_{n = 0}^{\infty}\sum_{m = 0}^{\infty}a_{nm}\cos\left(\frac{n\pi x}{L_x}\right)\cos\left(\frac{m\pi y}{L_y}\right)
\]
(Neumann boundary conditions mean only $\cos\cos$)
\[
a_{00} = \frac{1}{L_xL_y}\int_{0}^{L_x}\int_{0}^{L_y}f(x, y)\,dx\,dy
\]
where $f(x, y) = U(x, y, 0)$ is the initial condition.
\[
a_{mn} = \frac{4}{L_xL_y}\int_{0}^{L_x}\int_{0}^{L_y}f(x, y)\cos\left(\frac{m\pi x}{L_x}\right)\cos\left(\frac{n\pi y}{L_y}\right)\,dx\,dy
\]
If we specify boundary conditions
\[
\pd[u]{x}(0, y, t) = \pd[u]{x}(L_x, y, t) = \pd[u]{y}(x, 0, t) = \pd[u]{y}(x, L_y, t) = 0
\]
and an initial condition $U(x, y, 0) = f(x, y)$ we fully specify the solution.

\subsubsection{Mode decay/growth}

as $t \to \infty$ all modes $T_{nm}(t) \to 0$ except $m = n = 0$ so $U(x, y, t \to \infty) \to a_{00}$.

\subsection{Separation of variables and varying mode behaviour}
Consider the wave equation:
\[
\frac{\partial ^ 2u}{\partial t ^ 2} = c ^ 2\left(\frac{\partial ^ 2u}{\partial x ^ 2} + \frac{\partial ^ 2u}{\partial y ^ 2}\right)
\]
with homogenous Neumann boundary conditions.
Use separation of variables to solve.
\[
u(x, y, t) = X(x)Y(y)T(t).
\]
Systematic method:
Perform the separation:
\[
\frac{T''}{c ^ 2}XY = X''YT + Y''XT \implies \frac{T''}{c ^ 2T} = \frac{X''}{X} + \frac{Y''}{Y}
\]
can only be satisfied for all $x, y, t$ if
\[
\frac{T''}{cT} = -\lambda_t,\qquad \frac{X''}{X} = -\lambda_x,\qquad \frac{Y''}{Y} = -\lambda_y
\]
\[
\lambda_t = \lambda_x + \lambda_y.
\]
The minus sign is a common convention.
\begin{align*}
    X'' + \lambda_xX &= 0 \\
    Y'' + \lambda_yY &= 0 \\
    T'' + c ^ 2\lambda_tT &= 0
\end{align*}
Neumann boundary conditions $X'(0) = X'(L_x) = 0$,
$Y'(0) = Y'(L_y) = 0$.
Reduction to ODEs.

We saw this solves the heat equation:
\[
u(x, y, t) = \sum_{n = 0}^{\infty}\sum_{m = 0}^{\infty}C_{mn}e ^ {-D\left(\frac{\pi ^ 2n ^ 2}{L_x ^ 2} + \frac{\pi ^ 2m ^ 2}{L_y ^ 2}\right)t}\cos\left(\frac{n\pi x}{L_x}\right)\cos\left(\frac{m\pi y}{L_y}\right).
\]
So we are missing something,
if there are non-separable solutions how do we find them are?

We can write $U ^ c$ in this sum format.
\[
U ^ c(x, y, t) = \sum_{n = 0}^{\infty}\sum_{m = 0}^{\infty}C_{nm}(t)\cos\left(\frac{n\pi x}{L_x}\right)\cos\left(\frac{m\pi y}{L_y}\right).
\]
\[
C_{mn}(t) = \frac{4}{L_xL_y}\int_{0}^{L_x}\int_{0}^{L_y}U ^ c(x, y, t)\cos\left(\frac{n\pi x}{L_x}\right)\cos\left(\frac{m\pi y}{L_y}\right)\,dx\,dy.
\]
So $U ^ c$ is actually amongst this class of summed functions.

\begin{proposition}
    Any solution to the heat equation on $[0, L_x] \times [0, L_y]$ with homogeneous Dirichlet/Neumann boundary conditions can be represented as a Fourier series.
\end{proposition}

Most linear PDE's are not separable.
E.g.
\[
u_{xx} + u_{yy} + xu_y = 0.
\]
Try separation:
\[
u(x, y) = X(x)Y(y)
\]
so
\[
\frac{X''}{X} + \frac{Y''}{Y} + x\frac{Y'}{Y} = 0
\]
They depend on $x$ and $y$ so cannot balance.

\subsubsection{Eigen Series
(mode decompositions)
are fundamental mathematics,
not separation of variables}

where $\phi_{nm}$ satisfied some eigen problem
(or system of eigen equations).

Crucial point are the properties of linear independence of $\phi_{nm}$ and the orthogonality of these functions still applies in the general case.

\subsection{A prototype problem and its solution via an eigen expansion}

The problem
\[
\pd[u]{t} = Lu + h(x, t)
\]
where $x \in [0, L]$
($1$D special problem),
$x \in [a, b]$.
$u(x, t)$ is a scalar function as is $h(x, t)$.

\subsubsection{The operator \texorpdfstring{$L$}{}}
Its linear
\[
L[au_1 + bu_2] = aLu_1 + bLu_2
\]
and it is a differential operator only dependent on $x$
(spacial operator)

\subsubsection{The
"source"
or
"forcing"
term \texorpdfstring{$h(x, t)$}{}}
Does not depend on $u$,
and provides some external affect on the system.
"source"
when it creates $u$ or
"forcing"
if it represents an external change
(or loss)
in the system.

If $h(x, t) = 0$ the problem is homogenous if not it is inhomogeneous.

Inhomogeneous - homogenous decomposition if $u_h$ solves $\pd[u]{t} = Lu$ and $u_{ih}$ solves $\pd[u]{t} = Lu + h(x, t)$,
$u_h + u_{ih}$ solves $\pd[u]{t} = Lu + h(x, t)$
(from linearity of $L$).

\subsubsection{Boundary conditions}
Need $n$ boundary conditions for an $n$th order linear operator
(almost exclusively second order in this course).
Denote
\[
BC_i[u, a, b] = 0\qquad i = 1, \dotsc, n
\]

\subsubsection{Initial conditions}
System is complete if we specify an initial condition:
\[
u(x, 0) = g(x).
\]

\subsubsection{A solution}

Assume we can expand $h(x, t)$ as this
\[
h(x, t) = \sum_{n = 0}^{\infty}y_n(x)h_n(t).
\]
We saw when $L = \frac{\partial ^ 2}{\partial x ^ 2}$ this will take the form of a Fourier or Bessel series.

Substitute into our model
\[
\sum_{n = 0}^{\infty}\frac{dC_n}{dt}y_n = \sum_{n = 0}^{\infty}C_n(t)Ly_n(x) + \sum_{n = 0}^{\infty}h_n(t)y_n(x).
\]

In our PDE solution we assumed we could decompose the forcing function $h(x, t)$,
\[
\sum_{n = 0}^{\infty}h_n(t)y_n(x) = h_n(t)
\]
where $y_n$ solved $Ly_n = -\lambda y_n$,
$Ly_n = -\lambda \Gamma(x)y_n$.
\[
\sum_{n = 0}^{\infty}h_n(t)\frac{Ly_n(x)}{\lambda_n} = h(x, t)
\]
Linearity
\[
L\sum_{n = 0}^{\infty}h_n(t)\frac{y_n}{\Gamma(x)\lambda_n} = h(x, t)
\]
if we fix a time $t = t_0$ we have an inhomogeneous problem
\[
u = \sum_{n = 0}^{\infty}h_n(t_0)\frac{y_n(x)}{\lambda_n}\cdot f(x) = h(x, t_0)
\]
we have an equation in the form $Lu = f$.
So seeking to solve $Lu = f$ is equivalent to asking if we can represent some function $h(x, t)$ in terms of the eigenfunctions of an operator $L$.
Crucial assumption in PDE solution.
There is a potential issue with the $\frac{1}{\lambda_n}$ factor.
If $\lambda_0 = 0$ then this does not work.

To solve
\[
Lu = f\qquad BC_1(a) = BC_2(b) = 0
\]
we take the following steps.
Solve the eigenvalue problem 
\[
Ly_n = \lambda_ny_n,\qquad BC_1(a, b) = BC_2(a, b) = 0
\]
to obtain the set of pairs $(\lambda_n, y_n)$.

Solve the adjoint eigenvalue problem
\[
L ^ {*}w_n = \lambda_nw_n,\qquad BC_1 ^ {*}(a, b) = BC_2 ^ {*}(a, b) = 0
\]
to get $(\lambda_n, w_n ^ {*})$.

Assume the eigenexpansion solution
\[
y = \sum_{i = 0}^{\infty}c_iy_i(x)
\]
start from $Ly = f$,
apply the inner product
\[
\langle Ly, w_k\rangle = \langle f, w_k\rangle
\]
apply inner product
\[
\langle y, L ^ {*}w_k\rangle = \langle f, w_k\rangle
\]
adjoint identity,
then
\[
\langle y, \Gamma(x)\lambda_kw_k\rangle = \langle f, w_k\rangle
\]
$L_{w_k}^{*} = \Gamma(x)\lambda_kw_k$,
sub in the eigenexpansion to get
\[
\lambda_k\left\langle\sum_{i = }^{\infty}c_iy_i, \Gamma(x)\lambda_kw_k\right\rangle = \langle f, w_k\rangle.
\]
Orthogonality means that $\lambda_kc_k\langle y_k, \Gamma(x)\lambda_kw_k\rangle = \langle f, w_k\rangle$.
$c_k = \frac{\langle f, w_k\rangle}{\lambda_k\langle y_k, \Gamma(x)\lambda_kw_k\rangle}$.

If the problem is fully self adjoint we get
\[
c_k = \frac{\langle f, y_k\rangle}{\lambda_k \langle y_k, \Gamma(x)y_k\rangle}
\]
e.g.
\[
\frac{d ^ 2y_n}{dx ^ 2} = -\lambda_ny_n\qquad \frac{dy_n}{dx}\biggr|_{x = a} = 0,\qquad \frac{dy_n}{dx}\biggr|_{x = b} = 0
\]
$L = \frac{d ^ 2}{dx ^ 2}$ know $y_n = \cos\left(\frac{n\pi(x - a)}{b - a}\right)$.
So
\[
\langle f, y_k \rangle = \int_{a}^{b}\cos\left(\frac{n\pi(x - a)}{b - a}\right)f(x)\,dx
\]
\[
\langle y_k, y_k\rangle = \int_{a}^{b}\cos ^ 2\left(\frac{kn(x - a)}{b - a}\right)\,dx = \frac{b - a}{2}.
\]

\subsubsection{Inhomogeneous boundary conditions}
What about
\[
Lu = f(x)\qquad B_iu = \delta_i,\quad i = 1, \dotsc, n
\]
$\delta_i$ are real constants.

We can exploit linearity to make the method almost identical.
Solve
\[
Lu_1 = f(x)\qquad B_iu = 0
\]
homogeneous boundary conditions.
Solve
\[
Lu_2 = 0\qquad B_iu = \delta_i.
\]
Linearity says $u_1 + u_2$ is a solution and it will match the required boundary conditions.

\begin{example}
    $y'' = f(x)$,
    $y(0) = \alpha$,
    $y(1) = \beta$,
    $0 < x < 1$.

    Solve $y_1$,
    \[
    Ly_1 = f(x)\qquad y_1(0) = 0, y_1(1) = 0.
    \]
    This has solution:
    \[
    c_k = \frac{\langle f, w_k\rangle}{\lambda_k\langle y_k, r(x)w_k\rangle} = 2\int_{0}^{1}\frac{f(x)\sin(k\pi x)}{k ^ 2\pi ^ 2}\,dx.
    \]
    \[
    Ly_2 = 0\qquad y_2(0) = \alpha,\quad y_2(1) = \beta
    \]
    \[
    \frac{d ^ 2y_2}{dk ^ 2} = 0\qquad y_2 = ax + b
    \]
    boundary conditions
    $y_2 = (\beta - \alpha)x + \alpha$.
    Full solution is
    \[
    y = y_1 + y_2.
    \]
\end{example}

\subsection{Strum-Liouville theory}

The weighted Strum-Liouville eigenvalue problem takes the form:
\[
Ly = \lambda r(x)y
\]
where $r(x) \geq 0$ is a weighting function,
and the operator $L$ is of the form
\[
Ly = \frac{d}{dx}\left(p(x)\frac{dy}{dx}\right) + q(x)y,\qquad a \leq x \leq b.
\]
Through the self-adjointness of the operator we can see that from
\[
Ly = f(x) \implies \langle Ly, y_k\rangle = \langle f, y_k\rangle
\]
we can derive
\[
c_k = \frac{\langle f, y_k\rangle}{\lambda_k\langle y_k, ry_k\rangle}
\]
giving us the full solution
\[
y = \sum_{k}c_ky_k.
\]

\textbf{Transforming an operator to Strum-Liouville form}

We can transform an operator of the form
\[
Ly \equiv a_2(x)y''(x) + a_1(x)y'(x) + a_0(x)y(x)
\]
with $a_2(x) \neq 0$ in the interval.

To \textbf{transform} we multiply by an integrating factor function $\mu(x)$:
\[
\mu a_2(x)y''(x) + \mu a_1y'(x) + \mu a_0y
\]
then choose our $\mu$ such that our first and second derivatives collapse,
i.e. so it can be expressed in the form
\[
\frac{d}{dx}(py') + qy = py'' + p'y' + qy = \mu a_2(x)y''(x) + \mu a_1y'(x) + \mu a_2y
\]
and we can solve for $\mu, p$ and $q$ by equating the coefficients of $y'', y'$ and $y$ then solving the ensuing ODE's.


\textbf{Orthogonality}

Due to the presence of the weighting function,
the orthogonality relation is
\[
\int_{a}^{b}y_k(x)y_j(x)r(x)\,dx = 0.
\]

\textbf{Real eigenvalues}

$p, q, r$ are real so $\bar{L} = L$.
Thus,
taking conjugate of both sides of $Ly_k = \lambda_ky_k$ gives
$\bar{\lambda}_k = \lambda_k$.
Thus all the eigenvalues are real.

\textbf{Eigenfunctions}

The $\{y_k\}$ are a complete set,
that is all $h(x)$ with $\int f  ^ 2r\,dx < \infty$ can be expanded as
\[
h(x) = \sum c_ky_k(x).
\]

\subsection{Existence of the eigen-expansion via the Fredholm alternative}
We have shown that if we can solve the eigen-equation
\[
Ly_n = \lambda_ny_n
\]
and indeed the problem
\[
Lu = f
\]
subject to some appropriate boundary conditions.
Then we can construct a solution using orthogonality and the coefficient formulae.

We need to know if the problem can be solved in the first place.
To do this we use the Fredholm alternative.

\begin{proposition}[Fredholm alternative]
    Either,
    the homogenous adjoint problem $L ^ {*}w = 0$ has a non-trivial solution.
    Or,
    the inhomogeneous boundary value problem $Ly = f(x)$,
    with relevant boundary conditions,
    has a unique solution.
\end{proposition}

If we are in case $1$ we have a further alternative:
\begin{proposition}
    Either,
    \[
    \langle w, f\rangle = 0
    \]
    and there is a non-unique solution to the problem $Ly = f$,
    with relevant boundary conditions.

    Or,
    this integral doe not vanish and there is no solution to the problem $Ly = f(x)$,
    with the relevant boundary conditions.
\end{proposition}

Here is an example:

\begin{example}
    Consider the following inhomogeneous BVP:
    \[
    \frac{d ^ 2y}{dx ^ 2} + g_0 ^ 2y = \sin(x)
    \]
    with $g_0$ constant,
    and with the boundary conditions:
    \[
    y(0) = 0, y(\pi) = 0.
    \]

    \begin{solution}
        Solve the adjoint problem.
        Consider the homogenous problem $L ^ {*}w = 0$:
        \[
        \frac{d ^ 2w}{dx ^ 2} + g_0 ^ 2w = 0,\qquad w(0) = 0,\qquad w(\pi) = 0
        \]
        $g_0 > 0$.
        The general solution is
        \[
        w(x) = A\sin(g_0x) + B\cos(g_0x).
        \]
        The boundary condition $w(0) = 0$ implies $B = 0$.
        The second condition:
        \[
        A\sin(g_0\pi x) = 0
        \]
        requires either $A = 0$ or $g_0 = 0, 1, 2, \dotsc, n$.
        So since we have a non-trivial solution to $L ^ {*}w = 0$ we need to check
        \[
        \langle w,\sin(x)\rangle = \int_{0}^{\pi}\sin(g_0x)\sin(x)\,dx = \frac{\sin(g_0\pi)}{g_0 ^ 2 - 1}
        \]
        so if $g_0 \neq \pm 1$ this is zero.
        If $g_0 = 1$ then the integral is not zero which implies we cannot solve the problem.
    \end{solution}
\end{example}

\section{Green's method}

\subsection{Motivation for Green's method}
Using our previous method we can get an eigenfunction expansion solution as such
\[
y(x) = \sum_{k = 1}^{\infty}\frac{\langle f, w_k\rangle}{\lambda_k\langle y_k, w_k\rangle}y_k(x)
\]
however this requires all $\lambda_k \neq 0$.
The case of zero eigenvalue has two subcases:
One where $\langle f, w_k\rangle \neq 0$ and one where the inner product is zero.
In the first case the BVP has no solution,
the latter is not unique as you can add any multiple of $y_k$ to the solution.

Assume this is not an issue for now.
Let $n_k = \langle y_k, w_k\rangle$,
then:
\begin{align*}
    y(x) &= \sum_{k = 1}^{\infty}\frac{1}{\lambda_kn_k}\left(\int_{a}^{b}f(t)w_k(t)\,dt\right)\,y_k(t) \\
    &= \int_{a}^{b}\left(\sum_{k = 1}^{\infty}\frac{1}{\lambda_kn_k}w_k(t)y_k(t)\right)\,f(t)\,dt \\
    &= \int_{a}^{b}g(x, t)f(t)\,dt
\end{align*}
where $g(x, t) = \sum_{k = 1}^{\infty}\frac{w_k(t)y_k(x)}{\lambda_kn_k}$.

Thus,
we have a solution to $Ly = f$ in the form
\[
y(x) = \int_{a}^{b}g(x, t)f(t)\,dt.
\]
$g(x, t)$ is called the Green's function
(GF).

If $L = L ^ {*}$,
then $w_k = y_k$ and:
\[
g(x, t) = \sum \frac{1}{\lambda_kn_k}y_k(t)y_k(x).
\]
In this case $g(x, t) = g(t, x)$,
and we have the important connection between a self-adjoint operator and a symmetric Green's function.

You can think of the Green's function as inverting the differential operator.
Look at the equation $A\vec{x} = \vec{b}$ the solution is given by $\vec{x} = A ^ {-1}\vec{b}$.

\begin{example}
    Consider the BVP:
    \begin{align*}
        Ly &\equiv -y'' = f(x),\qquad 0 < x < 1 \\
        y(0) &= y(1) = 0.
    \end{align*}
    Taking the Green's function
    (without working it out)
    as
    \[
    g(x, \xi) = \begin{cases}
        (1 - \xi)x & 0 < x < \xi \\
        (1 - x)\xi & \xi < x < 1.
    \end{cases}
    \]
    To check this we can use the Leibniz rule for differentiating integrals:
    \[
    \frac{d}{dx}\int_{a(x)}^{b(x)}f(x, \xi)\,d\xi = \int_{a(x)}^{b(x)}\frac{df(x, \xi)}{dx} + \frac{db(x)}{dx}f(b, \xi) - \pd[a(x)]{x}f(a, \xi).
    \]

    Showing that $y(x) = \int_{0}^{1}g(x, \xi)f(\xi)\,d\xi$ is a solution to this equation:
    \begin{align*}
        y(x) &= \int_{0}^{1}g(x, \xi)f(\xi)\,d\xi = \int_{0}^{x}(1 - x)\xi f(\xi)\,d\xi + \int_{x}^{1}(1 - \xi)xf(\xi)\,d\xi \\
        \frac{dy}{dx} &= -\int_{0}^{x}\xi f(\xi)\,d\xi + (1 - x)xf(x) + \int_{x}^{1}(1 - \xi)f(\xi)\,d\xi - (1 - x)xf(x) \\
        &= -\int_{0}^{x}\xi f(\xi)\,d\xi + \int_{x}^{1}(1 - \xi)f(\xi)\,d\xi, \\
        \frac{d ^ 2y}{dx ^ 2} &= -xf(x) - (1 - x)f(x) = -f(x).
    \end{align*}
    The Green's function vanishes at the boundaries it satisfies the boundary conditions.
\end{example}

The following properties of Green's function itself are easily checked:
\begin{enumerate}[label = (\roman*)]
    \item The GF satisfies $Lg = 0$ if $x \neq \xi$.

    \item $g(x, \xi)$ satisfies the boundary conditions as a function of $x$.

    \item $g$ is continuous on the whole interval $[0, 1]$.

    \item $g$ is differentiable everywhere except at $x = \xi$,
    where it suffers a jump in the derivative.
\end{enumerate}
The above are always true of the GF of a second order linear operator.

\subsection{Green's function via delta function}

To fix the context,
consider stationary heat conduction in a rod:
\[
-y''(x) = f(x)\quad 0 < x < 1,\quad y(0) = 0,\quad y(1) = 0
\]
where $y(x)$ is the temperature field and $f(x)$ is a given heat source density.

The delta function is described by
\[
\delta(x) = \begin{cases}
    0, & x \neq 0 \\
    \infty, & x = 0
\end{cases}
\]
and
\[
\int_{-\infty}^{\infty}\delta(x)\,dx = 1.
\]

We can approximate $\delta$ by $f_n(x)$ an increasingly narrower sequence of functions with normalised area where
\[
\forall n, \int_{-\infty}^{\infty}f_n(x)\,dx = 1,\quad \lim_{n \to \infty}f_n(x) = 0\quad \forall x \neq 0.
\]
Example:
"hat"
functions
\[
f_n(x) = \begin{cases}
    0 &\text{for } |x| > 1 / n \\
    n / 2 &\text{for } |x| \leq 1 / n
\end{cases}
\]
and $f_n(x) \to \delta(x)$ as $n \to \infty$.

There are some useful properties of the $\delta$ function:

Let $f(x)$ be continuous,
and $F(x) = \int^{x}f(s)\,ds$ its antiderivative.
Now consider the approximating sequences:
\begin{align*}
    \int_{-\infty}^{\infty}\delta(x - a)f(x)\,dx = \lim_{n \to \infty}\int_{-\infty}^{\infty}f_n(x - a)f(x)\,dx \\
    \intertext{and if $f_n$ are the hat functions}
    &= \lim_{n \to \infty}\int_{a - 1 / n}^{a + 1 / n}\frac{n}{2}f(x)\,dx \\
    &= \lim_{n \to \infty}\frac{F(a + (1 / n)) - F(a - (1 / n))}{2 / n} \\
    &= \lim_{s \to 0}\frac{F(a + s) - F(a - s)}{2s} = F'(a) = f(a).
\end{align*}
Thus,
we have
\[
\int_{-\infty}^{\infty}\delta(x - a)f(x)\,dx = f(a)\qquad\text{if $f$ is continuous at $a$}.
\]
In particular,
\[
\int_{-\infty}^{\infty}\delta(x)f(x)\,dx = f(x)\qquad\text{if $f$ is continuous at $x = 0$}.
\]
Thus,
the delta function can be sifted out the value of a function at a particular point

The antiderivative of $\delta(x)$.
The antiderivative of the delta function is the Heaviside function,
\[
\int_{-\infty}^{x}\delta(s)\,ds = H(x) \equiv \begin{cases}
    0 & x < 0 \\
    1 & x > 0.
\end{cases}
\]

Returning to the heat conduction BVP with a point heat source of unit strength at the centre of the rod:
\[
-y''(x) = \delta(x - 1 / 2),\qquad 0 < x < 1,\qquad y(0) = y(1) = 0.
\]
Since $\forall x \neq 1 / 2, \delta(x - 1 / 2) = 0$,
this implies
\[
-y''(x) = 0,\qquad 0 < x < 1 / 2,\quad 1 / 2 < x < 1.
\]
We can solve the second of the two above equations in each of the two separate domains $[0, 1 / 2)$ and $(1 / 2, 1]$ and then apply the boundary conditions of the first.

To derive the extra conditions we get at $x = 1 / 2$ we do the following:
\[
\int_{1 / 2-}^{1 / 2+}-y''(x)\,dx = \int_{1 / 2-}^{1 / 2+}\delta(x - 1 / 2)\,dx,
\]
where $1 / 2 - (1 / 2+)$ just to the left
(right)
of $1 / 2$.
We have
\[
-y'\biggr|_{1 / 2-}^{1 / 2+} = 1 \implies y'(1 / 2+) - y'(1 / 2-) = -1.
\]
The other extra conditions comes as a requirement that $y(x)$ is continuous across the point source,
that is
\[
y\biggr|_{1 / 2-}^{1 / 2+} = 0.
\]
Giving us the final solution
\[
u(x) = \begin{cases}
    \frac{x}{2} & 0 < x < 1 / 2 \\
    -\frac{x}{2} + \frac{1}{2} & 1 / 2 < x < 1.
\end{cases}
\]


To motivate the construction of Green's function,
consider the heat conduction problem with an arbitrary heat source:
\[
-y''(x) = f(x),\qquad 0 < x < 1,\qquad y(0) = y(1) = 0.
\]
Describing $f$ by a distribution of point heat sources with varying strength;
that is at point $x = \xi$ we imagine placing the point source $f(\xi)\delta(x - \xi)$.

The idea of the Green's function is to introduce such an extra parameter $\xi$,
and consider the system
\[
-g''(x, \xi) = \delta(x - \xi),\qquad 0 < x < 1,\qquad g(0, \xi) = g(1, \xi) = 0.
\]
We have replaced $f(x)$ by a delta function,
in order to solve for the Green's function $g(x, \xi)$.

We have already seen that the Green's function is
\[
g(x, \xi) = \begin{cases}
    (1 - \xi)x & 0 < x < \xi \\
    (1 - x)\xi & \xi < x < 1.
\end{cases}
\]

\subsection{General linear BVP}
Consider a general $n$th order linear BVP with arbitrary continuous forcing function,
\[
Ly(x) = a_ny ^ {(n)}(x) + a_{n - 1}y ^ {(n - 1)}(x) + \dotsi + a_1y'(x) + a_0y(x) = f(x)
\]
for $a < x < b$,
where each $a_i = a_i(x)$ is a continuous function,
and moreover $\forall x, a_n(x) \neq 0$.

Along with the above equation are $n$ boundary conditions,
each a linear combination of $y$ and derivatives up to $y ^ {(n - 1)}$,
evaluated at $x = a, b$.
For example:
in the case $n = 2$,
the general form is:
\begin{align*}
    B_1y &\equiv \alpha_{11}y(a) + \alpha_{12}y'(a) + \beta_{11}y(b) + \beta_{12}y'(b) = \gamma_1 \\
    B_2y &\equiv \alpha_{21}y(a) + \alpha_{22}y'(a) + \beta_{21}y(b) + \beta_{22}y'(b) = \gamma_2.
\end{align*}

To solve our general equation with homogeneous boundary conditions:
\[
B_iy = 0,\qquad i = 1,\dotsc, n - 1.
\]
We first determine the Green's function by solving
\begin{align*}
    Lg(x, \xi) &= \delta(x - \xi),\qquad a < x < b \\
    B_ig &= 0.
\end{align*}
As before,
\[
Lg(x, \xi) = \delta(x - \xi)
\]
implies
\[
Lg(x, \xi) = 0\qquad\text{on } a < x < \xi,\quad \xi < x < b,
\]
i.e.
we have a homogeneous problem to solve on two separate domains.
As before,
we need extra conditions,
which come by integrating $Lg(x, \xi) = \delta(x - \xi)$ across $x = \xi$:
\[
\int_{\xi-}^{\xi+}a_ng ^ {(n)}(x, \xi) + \dotsi + a_0g(x, \xi)\,d\xi = \int_{\xi-}^{\xi+}\delta(x - \xi)\,dx.
\]
The right hand side clearly integrates to one.
If we integrate by parts on the first term of the left hand side,
we would obtain
\[
a_n(x)g ^ {(n - 1)}(x, \xi)\biggr|_{\xi-}^{\xi+} + \int_{\xi-}^{\xi+}(a_{n - 1} - a_n')g ^ {(n - 1)} + \dotsi + a_0g(x, \xi)\,dx = 1.
\]
This equation is balanced by setting a jump condition on the $n - 1$st derivative:
\[
g ^ {(n - 1)}(x, \xi)\biggr|_{\xi-}^{\xi+} = \frac{1}{a_n(\xi)},
\]
taking all lower derivatives to be continuous across $x = \xi$:
\[
g ^ {(j)}(x, \xi)\biggr|_{\xi-}^{\xi+} = 0,\qquad j = 0, 1, \dotsc, n - 2.
\]
Once the Green's function is determined,
the solution to the BVP is given by
\[
y(x) = \int_{a}^{b}g(x, \xi)f(\xi)\,d\xi.
\]

\subsection{Three-dimensional Green's functions in
(pretty)
arbitrary domains}
We seek $G(x, \xi)$ such that
\[
-\nabla_{x}^{2}G(x, \xi) = \delta(x - \xi),\qquad G(x, \xi) \to 0\text{ as } |x| \to \infty.
\]
By translation and rotational symmetry,
$G$ depends only on the distance $r = |x - \xi|$.

\textbf{Step $1$}:
Solve away from the source.

For $r > 0$ we have $\nabla_{x}^{2}G = 0$,
so in spherical coordinates the radial Laplacian gives
\[
\Phi''(r) + \frac{2}{r}\Phi'(r) = 0.
\]
Integrating once:
$r ^ 2\Phi'(r) = A$ a constant,
hence $\Phi'(r) = \frac{A}{r ^ 2}$ and
\[
\Phi(r) = -\frac{A}{r} + B
\]
The decay condition $G \to 0$ as $r \to \infty$ forces $B = 0$,
so for $r > 0$
\[
G(x, \xi) = \Phi(r) = \frac{C}{r}\quad\text{with }C = -A.
\]

\textbf{Step $2$};
Fix the constant by normalisation.
Integrate the defining equation over a ball $B_{\varepsilon}(\xi)$ of radius $\varepsilon$ centred at $\xi$ and use the divergence theorem:
\[
\int_{B_{\varphi}(\xi)}-\nabla_{x}^{2}G\,dV = \int_{\partial B_{\varepsilon}(\xi)}-\pd[G]{n}\,dS = \int_{B_{\varepsilon}(\xi)}\delta(x - \xi)\,dV = 1.
\]
Since $G = C / r$,
its radial derivative is $\pd[G]{r} = -\frac{C}{r ^ 2}$.
On the sphere $|x - \xi| = \varepsilon$,
the outward normal derivative is $\pd[G]{n} = \pd[G]{r} = -\frac{C}{\varepsilon ^ 2}$,
hence
\[
\int_{\partial B_{\varepsilon}(\xi)}-\pd[G]{n}\,dS = \int_{\partial B_{\varphi}(\xi)}\frac{C}{\varepsilon ^ 2}\,dS = \frac{C}{\varepsilon ^ 2}\cdot 4\pi\varepsilon ^ 2 = 4\pi C.
\]
Therefore $4\pi C = 1$,
so $C = \frac{1}{4\pi}$ and
\[
G(x, \xi) = \frac{1}{4\pi|x - \xi|}.
\]

Let $\Omega \subset \R ^ 3$ be a bounded domain with smooth boundary $\partial\Omega$.
For
\[
-\nabla ^ 2u(x) = f(x) \in \Omega
\]
Green's representation formula is
\[
u(x) = \int_{\partial\Omega}\left(G(x, \xi)\pd[u]{n_{\xi}}(\xi) - u(\xi)\pd[G]{n_{\xi}}(x, \xi)\right)\,dS_{\xi} + \int_{\Omega}G(x, \xi)f(\xi)\,dV_{\xi},
\]
where $G$ is the Green's function appropriate to the boundary condition.

Dirichlet problem
($u = g$ on $\partial\Omega$):
choose the Dirichlet Green's function $G_{D}$ solving
\[
-\nabla_{x}^{2}G_{D}(x, \xi) = \delta(x - \xi),\qquad G_{D}(x, \xi) = 0\quad\text{for $x \in \partial\Omega$}.
\]
Then
\[
u(x) = -\int_{\partial\Omega}g(\xi)\pd[G_{D}]{n_{\xi}}(x, \xi)\,dS_{\xi} + \int_{\Omega}G_{D}(x, \xi)f(\xi)\,dV_{\xi},\qquad x \in \Omega.
\]
Hence $u$ is obtained from known boundary data $g$ and the volume source $f$.

Neumann problem
($\pd[u]{n} = h$ on $\partial\Omega$):
choose the Neumann function $G_{N}$ solving
\[
-\nabla_{x}^{2}G_{N}(x, \xi) = \delta(x - \xi),\qquad \pd[G_{N}]{n_x}(x, \xi) = 0\text{ for } x \in \partial\Omega,
\]
together with the usual compatibility $\int_{\Omega}f\,dV = \int_{\partial\Omega}h\,dS$.
Then
\[
u(x) = \int_{\partial\Omega}G_{N}(x, \xi)h(\xi)\,dS_{\xi} + \int_{\Omega}G_{N}(x, \xi)f(\xi)\,dV_{\xi},\qquad x \in \Omega
\]
which uses only the known Neumann data $h$ and $f$.
