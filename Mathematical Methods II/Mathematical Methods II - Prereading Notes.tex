\section{Line integrals}

\subsection{Curves}
It is convenient to describe a curve by a parameterisation,
$\mbf{x}(t)$,
which maps a real interval $[t_0, t_1]$ into $\R ^ n$.
This is a vector-valued function
\[
\mbf{x}(t) = x_1(t)\mbf{e}_1 + x_2(t)\mbf{e}_2 + \dotsc + x_n(t)\mbf{e}_n
\]
where $\{\mbf{e}_1, \mbf{e}_2, \dotsc, \mbf{e}_n\}$ denote the Cartesian unit vectors.

If $\mbf{x}(t_1) = \mbf{x}(t_2)$ then the curve is called closed.

A curve that does not cross itself is called simple.

A parameterisation $x(t)$ is differentiable if it has a well-defined tangent vector,
\[
\frac{d\mbf{x}}{dt} = \lim_{dt \to 0}\frac{\mbf{x}(t + dt) - \mbf{x}(t)}{dt}.
\]

\subsection{Line integrals of scalar fields}
A scalar field on $\R ^ n$ is a map $f : \R ^ n \to \R$.

The line integral of a scalar field $f(\mbf{x})$ along a curve $C$ with parameterisation $\mbf{x}(t)$,
for $t \in [t_0, t_1]$,
is defined as
\[
\int_{C}f\,d\ell = \int_{t_0}^{t_1}f(\mbf{x}(t))\left|\frac{d\mbf{x}}{dt}\right|\,dt.
\]

\begin{proposition}
    The line integral of a scalar field is independent of the choice of parameterisation.

    \begin{proof}
        Suppose we have two parameterisations of the same curve:
        $\mbf{x}(t)$ for $t \in [t_0, t_1]$,
        and $\mbf{x}(\tau)$ for $\tau \in [\tau_0, \tau_1]$.
        By the chain rule,
        the tangent vectors are related by
        \[
        \frac{d\mbf{x}}{dt} = \frac{d\tau}{dt}\frac{d\mbf{x}}{d\tau} \implies \left|\frac{d\mbf{x}}{dt}\right| = \left|\frac{d\mbf{x}}{dt}\right| = \left|\frac{d\tau}{dt}\right|\left|\frac{d\mbf{x}}{d\tau}\right|.
        \]
        There are now to cases:

        Case $1$:
        $\mbf{x}(\tau_0) = \mbf{x}(t_0)$,
        meaning both parameterisation trace the curve in the same direction.
        Then
        \[
        \frac{d\tau}{dt} > 0 \implies \int_{t_0}^{t_1}f(\mbf{x}(t))\left|\frac{d\mbf{x}}{dt}\right|\,dt = \int_{t_0}^{t_1}f(\mbf{x}(t))\left|\frac{d\mbf{x}}{d\tau}\right|\frac{d\tau}{dt}\,dt = \int_{\tau_0}^{\tau_1}f(\mbf{x}(\tau))\left|\frac{d\mbf{x}}{d\tau}\right|\,d\tau.
        \]
        Case $2$:
        $\mbf{x}(\tau_0) = \mbf{x}(t_1)$,
        meaning they trace the curve in opposite directions.
        Now
        \[
        \frac{d\tau}{dt} < 0 \implies \int_{t_0}^{t_1}f(\mbf{x}(t))\left|\frac{d\mbf{x}}{dt}\right|\,dt = \int_{t_0}^{t_1}f(\mbf{x}(t))\left|\frac{d\mbf{x}}{d\tau}\right|\left(-\frac{d\tau}{dt}\right)\,dt = -\int_{\tau_1}^{\tau_0}f(\mbf{x}(\tau))\left|\frac{d\mbf{x}}{d\tau}\right|\,d\tau.
        \]
        Both cases recover the original formula.
    \end{proof}
\end{proposition}

For the special case $f(\mbf{x}) \equiv 1$,
the line integral is called the arclength of the curve,
\[
L = \int_{C}\,d\ell = \int_{t_0}^{t_1}\left|\frac{d\mbf{x}}{dt}\right|\,dt.
\]

\subsection{Line integrals of vector fields}
A vector field is a function $\mbf{f} : \R ^ n \to \R ^ n$ that assigns a vector to each point in $\R ^ n$.
For $n = 2$,
we can visualise $\mbf{f}$ as an arrow at each point in the plane.

The unit target vector of a parameterisation is
\[
\hat{\mbf{t}} = \frac{\frac{d\mbf{x}}{dt}}{\left|\frac{d\mbf{x}}{dt}\right|}
\]

An oriented curve is a curve together with a specified
(consistent)
choice of $\hvec{t}$.
We indicate this by adding an arrow to the curve.

The line integral of a vector field $\mbf{f}(\mbf{x})$ along an oriented curve $C$ with parameterisation $\vec{x}(t)$,
for $t \in [t_0, t_1]$,
is defined as the scalar line integral of the tangential component:
\[
\int_{C}\vec{f}\cdot\hvec{t}\,d\ell = \int_{t_0}^{t_1}\vec{f}(\vec{x}(t))\cdot\hvec{t}\left|\frac{d\vec{x}}{dt}\right|\,dt = \int_{t_0}^{t_1}\vec{f}(\vec{x}(t))\cdot\frac{d\vec{x}}{dt}\,dt.
\]

If the curve $C$ is closed,
this is often emphasised by the notation
\[
\int_{C}\vec{f}\cdot\hvec{t}\,d\ell = \oint_{C}\vec{f}\cdot \hvec{t}\,d\ell,
\]
and the line integral is called the circulation of $\vec{f}$ around $C$.

\subsection{Curvilinear coordinates}
In a curvilinear coordinate system $(u, v, w)$,
the coordinate lines -
where two of the coordinates are constant and one varies -
may be curved.
These coordinates lines may be parameterised by the corresponding coordinate,
so are given by $\vec{x}(u, v, w)$ with two of the coordinates fixed.

The coordinate system is orthogonal if the three tangent vectors are mutually orthogonal at every point.
We can then define an orthonormal basis $\{\mbf{e}_u, \mbf{e}_v, \mbf{e}_w\}$ by normalising:
\[
\vec{e}_u = \frac{1}{h_u}\pd[\vec{x}]{u}\quad\vec{e}_v = \frac{1}{h_v}\pd[\vec{x}]{v},\quad\vec{e}_w = \frac{1}{h_w}\pd[\vec{x}]{w},
\]
where
\[
h_u = \left|\pd[\vec{x}]{u}\right|,\quad h_v = \left|\pd[\vec{x}]{v}\right|,\quad h_w = \left|\pd[\vec{x}]{w}\right|
\]
are called the scale factors.

In $\R ^ 3$,
there are two non-Cartesian systems in common use:

\textbf{Cylindrical coordinates} are useful for problems with axial symmetry:
\[
\vec{x}(r, \theta, z) = r\cos{\theta}\vec{e}_1 + r\sin{\theta}\vec{e}_2 + z\vec{e}_3,\quad r > 0,\quad \theta \in [0, 2\pi],\quad z \in \R,
\]
leading to scale factors $h_r = 1$,
$h_{\theta} = r$,
$h_z = 1$.

\textbf{Spherical coordinates} are useful for problems with spherical symmetry:
\[
\vec{x}(r, \theta, \phi) = r\sin{\theta}\cos{\phi}\vec{e}_1 + r\sin{\theta}\sin{\phi}\vec{e}_2 + r\cos{\theta}\vec{e}_3,\quad r > 0,\quad \theta \in [0, \pi],\quad \phi \in [0, 2\pi],
\]
leading to scale factors $h_r = 1$,
$h_{\theta} = r$,
$h_{\phi} = r\sin{\theta}$.

\section{Surface and volume integrals}

\subsection{Surfaces}

A $2$D surface embedded in $\R ^ 3$ can be defined parametrically as a map $\vec{x}(u, v)$ from $\R ^ 2 \to \R ^ 3$,
for $u \in [u_0, u_1]$ and $v \in [v_0, v_1]$.

For each value of $v$,
the position $\vec{x}(u, v)$ describes a curve on the surface with tangent vector $\pd[\vec{x}]{u}$.

Similarly,
for each value of $u$,
we get a curve with tangent vector $\pd[\vec{x}]{v}$.

If $(u, v)$ is a well-defined parameterisation,
then their tangent vectors cannot be collinear.
So the cross product of these tangent vectors allows us to define the unit normal to the surface:
\[
\hvec{n} = \frac{\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}}{\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|}.
\]
The sign of $\hvec{n}$ depends on the ordering of $(u, v)$.
A surface together with a specified
(consistent)
choice of $\hvec{n}$ is called an oriented surface.

A surface is called smooth if $\hvec{n}$ varies continuously over the surface.
It is called piecewise smooth if it can be divided into finitely many portions each of which is smooth.

A surface can be either open if it is bounded by a curve,
or closed if it has no boundary.
The boundary of an open surface $S$ is a curve,
often written $\partial S$.

A simple type of surface is an explicit surface given by the height of a two-dimensional function,
$z = h(x, y)$.

\subsection{Surface integrals of scalar fields}
If $f(\vec{x}) = f(x, y, z)$ is a scalar field and $S$ is a surface with parameterisation $\vec{x}(u, v)$,
then we define the surface integral of $f$ over $S$ to be
\[
\int_{S}f\,dS = \int_{U}f(\vec{x}(u, v))\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|\,dudv,
\]
where $U$ is the preimage of $S$ in the $(u, v)$ plane.

\begin{proposition}
    If $(u, v)$ are
    (two of a set of)
    orthogonal curvilinear coordinates,
    then
    \[
    \left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| = h_uh_v.
    \]

    \begin{proof}
        Recall that $\pd[\vec{x}]{u} = h_u\vec{e}_u$ and $\pd[\vec{x}]{v} = h_v\vec{e}_v$.
        Since $h_u, h_v \geq 0$,
        we have that
        \[
        \left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| = h_uh_v|\vec{e}_u \times \vec{e}_v| = h_uh_v|\pm\vec{e}_w| = h_uh_v.
        \]
    \end{proof}
\end{proposition}

The surface area of $S$ is computed by taking $f(\vec{x}) \equiv 1$,
giving $|S| = \int_{S}\,dS$.

\begin{proposition}\label{pre:mathmethII:prop:surfintscalindepparam}
    The surface integral of a scalar field is independent of the choice of parameterisation.

    \begin{proof}
        Suppose $\vec{x}(u, v)$ and $\vec{x}(\mu, \nu)$ are two regular parameterisations of the same surface.
        By the chain rule,
        \[
        \pd[\vec{x}]{u} = \pd[\vec{x}]{\mu}\pd[\mu]{u} + \pd[\vec{x}]{\nu}\pd[\nu]{u},\qquad \pd[\vec{x}]{v} = \pd[\vec{x}]{\mu}\pd[\mu]{v} + \pd[\vec{x}]{v}\pd[\nu]{v}.
        \]
        So
        \begin{align*}
            \pd[\vec{x}]{u} \times \pd[\vec{x}]{v} &= \left(\pd[\vec{x}]{\mu}\pd[\mu]{u} + \pd[\vec{x}]{\nu}\pd[\nu]{u}\right) \times \left(\pd[\vec{x}]{\mu}\pd[\mu]{v} + \pd[\vec{x}]{\nu}\pd[\nu]{v}\right) \\
            &= \left(\pd[\vec{x}]{\mu}\pd[\mu]{u} + \pd[\vec{x}]{\nu}\pd[\nu]{u}\right) + \left(\pd[\vec{x}]{\nu}\pd[\nu]{u} + \pd[\vec{x}]{\mu}\pd[\mu]{v}\right) \\
            &= \left(\pd[\mu]{u}\pd[\nu]{v} - \pd[\mu]{v}\pd[\nu]{u}\right)\pd[\vec{x}]{\mu} \times \pd[\vec{x}]{\nu}.
        \end{align*}
        This means
        \[
        \left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right| = |J|\left|\pd[\vec{x}]{\mu} \times \pd[\vec{x}]{\nu}\right|,
        \]
        where $J$ is the Jacobian.
        Since $|J|\,dudv = d\mu d\nu$,
        we obtain
        \[
        \int_{U}f(\vec{x}(u, v))\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|\,dudv = \int_{U'}f(\vec{x}(\mu, \nu))\left|\pd[\vec{x}]{\mu} \times \pd[\vec{x}]{\nu}\right|\,d\mu d\nu,
        \]
        where $U'$ is the corresponding region of the $(\mu, \nu)$ plane.
    \end{proof}
\end{proposition}

\subsection{Surface integrals of vector fields}
If $\vec{f} : \R ^ 3 \to \R ^ 3$ is a vector field and $S$ is an oriented surface with unit normal $\hvec{n}$,
we define the surface integral
(or flux)
of $\vec{f}$ through $S$ as
\[
\int_{S}\vec{f} \cdot\hvec{n}\,dS.
\]

We evaluate the integral using a specific parameterisation $\vec{x}(u, v)$,
so
\begin{align*}
    \int_{S}\vec{f}\cdot\hvec{n}\,dS &= \int_{U}\vec{f}(\vec{x}(u, v))\cdot\hvec{n}\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|\,dudv \\
    &= \int_{U}\vec{f}(\vec{x}(u, v)) \cdot \left(\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right)\,dudv,
\end{align*}
where $U$ is the preimage of $S$ in the $(u, v)$ plane.

If the surface $S$ is closed,
this is often emphasised by writing
\[
\int_{S}\vec{f}\cdot\hvec{n}\,dS = \oint_{S}\vec{f}\cdot\hvec{n}\,dS.
\]
We also have vector-valued surface integrals of the form
\[
\int_{S}\vec{f}\,dS
\]
for some vector field $\vec{f} : \R ^ 3 \to \R ^ 3$.

\subsection{Volume integrals}
The volume integral of a scalar field $f : \R ^ 3 \to \R$ over a $3$D subregion $V \subset \R ^ 3$ is
\[
\int_{V}f\,dV = \int_{U}f(\vec{x}(u, v, w))\left|\pd[\vec{x}]{u} \cdot \left(\pd[\vec{x}]{v} \times \pd[\vec{x}]{w}\right)\right|\,dudvdw,
\]
where $U$ is the preimage of $V$ in $(u, v, w)$ space,
for some well-defined coordinates $(u, v, w)$.

The factor $\left|\pd[\vec{x}]{u} \cdot \left(\pd[\vec{x}]{v} \times \pd[\vec{x}]{w}\right)\right|\,dudvdw$ is the volume of a parallelepiped corresponding to an infinitesimal cuboid $(du, dv, dw)$ in $(u, v, w)$ space.

\begin{proposition}
    If $(u, v, w)$ are orthogonal curvilinear coordinates,
    then
    \[
    \left|\pd[\vec{x}]{u} \cdot \left(\pd[\vec{x}]{v} \times \pd[\vec{x}]{w}\right)\right| = h_uh_vh_w.
    \]

    \begin{proof}
        Recall that $\pd[\vec{x}]{u} = h_u\vec{e}_u$,
        and similarly for the other two tangent vectors.
        Since $h_u, h_v, h_w \geq 0$,
        \[
        \left|\pd[\vec{x}]{u} \cdot \left(\pd[\vec{x}]{v} \times \pd[\vec{x}]{w}\right)\right| = h_uh_vh_w|\vec{e}_u \cdot (\vec{e}_v \times \vec{e}_w)| = h_uh_vh_w|\vec{e}_u \cdot (\pm\vec{e}_u)| = h_uh_vh_w.
        \]
    \end{proof}
\end{proposition}

\begin{proposition}
    The volume integral of a scalar field is independent of the choice of coordinates.

    \begin{proof}
        This is analogous to \autoref{pre:mathmethII:prop:surfintscalindepparam},
        using the fact that a $3$D change of variables obeys
        \[
        dx\,dy\,dz = \left|\pd[\vec{x}]{u} \cdot \left(\pd[\vec{x}]{v} \times \pd[\vec{x}]{w}\right)\right|\,du\,dv\,dw.
        \]
    \end{proof}
\end{proposition}

\section{Differential operators}

\subsection{Differentiating scalar fields}

The gradient of a scalar field $f : \R ^ 3 \to \R$ is the operator
\[
\bnabla f \biggr|_{x} = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}f\hvec{n}\,dS,
\]
where $S$ is a closed surface around $\vec{x}$ with an outward normal $\hvec{n}$,
enclosed volume $|V|$,
and diameter $d(S) = \sup_{x, y \in S}|\vec{x} - \vec{y}|$.

\begin{proposition}\label{pre:mathmethII:prop:gradincart}
    In Cartesian coordinates,
    \[
    \bnabla f = \pd[f]{x}\vec{e}_1 + \pd[f]{y}\vec{e}_2 + \pd[f]{z}\vec{e}_3.
    \]
\end{proposition}

\begin{lemma}[Mean value theorem for double integrals]
    If a scalar field $f : \R ^ 2 \to \R$ is continuous on a closed,
    bounded and connected domain $A \subset \R ^ 2$ then there exists $\vec{x}_{*} \in A$ such that
    \[
    \int_{A}f\,dS = |A|f(\vec{x}_{*}).
    \]

    \begin{proof}
        Under the given conditions,
        we know that $f$ attains both its maximum and minimum on $A$,
        say at some points $\vec{x}_{\text{max}}, \vec{x}_{\text{min}}$.
        We also know that
        \begin{equation}\label{pre:mathmethII:eq:eq1}\tag{$\dagger$}
            |A|f(\vec{x}_{\text{min}}) \leq \int_{A}f\,dS \leq |A|f(\vec{x}_{\text{max}}).
        \end{equation}
        Now let $\vec{x}(t)$ parametrise a curve in $A$ such that $\vec{x}(t_0) = \vec{x}_{\text{min}}$ and $\vec{x}(t_1) = \vec{x}_{\text{max}}$,
        and consider the function
        \[
        g(t) = |A|f(\vec{x}(t)).
        \]
        This is continuous with $g(t_0) = |A|f(\vec{x}_{\text{min}})$ and $g(t_1) = |A|f(\vec{x}_{\text{max}})$,
        so by the intermediate value theorem and \eqref{pre:mathmethII:eq:eq1} there exists $t_{*}$ with
        \[
        g(t_{*}) = \int_{A}f\,dS.
        \]
        Then at $\vec{x}_{*} = \vec{x}(t_{*})$ we have $|A|f(\vec{x}_{*}) = \int_{A}f\,dS$,
        as required.
    \end{proof}
\end{lemma}

\begin{proposition}[continues pre:mathmathII:prop:gradincart]
    \begin{proof}
        To find the $\vec{e}_1$ component of $\bnabla f$,
        let $V$ be a small cylinder of radius $\epsilon$ and length $h$,
        whose axis is the unit vector $\vec{e}_1$.
        The volume of the cylinder is $|V| = \pi\epsilon ^ 2h$.

        Then
        \begin{align*}
            \vec{e}_1 \cdot \bnabla f &= \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}f\vec{e}_1\cdot\hvec{n}\,dS \\
            &= \lim_{\epsilon, h \to 0}\frac{1}{\pi\epsilon ^ 2h}\left[\int_{S_1}f\,dS - \int_{S_0}f\,dS\right] &&(\text{since $\vec{e}_1 \cdot \hvec{n} = 0$ on the curved surface}) \\
            &= \lim_{\epsilon, h \to 0}\frac{1}{\pi\epsilon ^ 2h}\left[\pi \epsilon ^ 2f(\vec{x}_1) - \pi\epsilon ^ 2f(\vec{x}_0)\right]
        \end{align*}
        for some $\vec{x}_1 \in S_1$ and $\vec{x}_0 \in S_0$,
        by the mean value theorem for double integrals.
        Taking the $\varepsilon$ limit,
        these points tend to the centre of their faces by the squeezing theorem,
        so
        \[
        \vec{e}_1 \cdot \bnabla f = \lim_{h \to 0}\frac{f(\vec{x} + h\vec{e}_1) - f(\vec{x})}{h} = \pd[f]{x}.
        \]
        The other components are obtained analogously.
    \end{proof}
\end{proposition}

\begin{proposition}
    In curvilinear coordinates $(u, v, w)$,
    the gradient has the expression
    \[
    \bnabla f = \frac{1}{h_u}\pd[f]{u}\vec{e}_u + \frac{1}{h_v}\pd[f]{v}\vec{e}_v + \frac{1}{h_w}\pd[f]{w}\vec{e}_w.
    \]

    \begin{proof}
        This is a consequence of the previous proposition.
        Note that
        \[
        \vec{e}_u \cdot \bnabla f = \left(\frac{1}{h_u}\pd[x]{u}\right) \cdot\bnabla f = \frac{1}{h_u}\left(\pd[x]{u}\pd[f]{x} + \pd[y]{u}\pd[f]{y} + \pd[z]{u}\pd[f]{z}\right) = \frac{1}{h_u}\pd[f]{u},
        \]
        where the last step used the chain rule.
        The $\vec{e}_v$ and $\vec{e}_w$ components are analogous.
    \end{proof}
\end{proposition}

\subsection{Differentiating vector fields}
There are two standard ways to differentiate a vector field $\vec{f} : \R ^ 3 \to \R ^ 3$,
defined in a similar way to the gradient as follows.

$S$ is a closed surface enclosing $\vec{x}$ with outward normal $\hvec{n}$,
enclosed volume $|V|$,
and diameter $d(S)$.

\begin{enumerate}[label = \arabic*.]
    \item The \textbf{divergence} of $\vec{f}$ is the scalar field
    \[
    \bnabla \cdot \vec{f}\,\biggr|_{\vec{x}} = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n} \cdot \vec{f}\,dS.
    \]

    \item The \textbf{curl} of $\vec{f}$ is the vector field
    \[
    \bnabla \times \vec{f}\,\biggr|_{\vec{x}} = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n} \times \vec{f}\,dS.
    \]
\end{enumerate}

Divergence measures whether
(on average)
the vector field is pointing outward
($\bnabla \cdot \vec{f} > 0$)
or inward
($\bnabla \cdot \vec{f} < 0$)
at a particular point.

Curl measures how much
(on average)
the vector field at a particular point is rotating around any given axis.
$\vec{e}_3 \cdot \bnabla \times \vec{f} > 0$ for an anti-clockwise rotation and $\vec{e}_3 \cdot \bnabla \times \vec{f} < 0$ for a clockwise rotation.

\begin{lemma}\label{pre:mathmethII:lem:gradscfldprods}
    If $\vec{a} \in \R ^ 3$ is a constant vector and $f : \R ^ 3 \to \R$ is a $C ^ {1}$ scalar field,
    then
    \begin{enumerate}[label = (\roman*)]
        \item $\bnabla \cdot (f\vec{a}) = (\bnabla f)\cdot \vec{a}$,
        
        \item $\bnabla \times (f\vec{a}) = (\bnabla f)\times \vec{a}$.
    \end{enumerate}

    \begin{proof}
        These formulae follow from the integral definitions of $\bnabla \cdot \vec{f}$ and $\bnabla \times \vec{f}$.
        In particular,
        \[
        \bnabla \cdot (f\vec{a}) = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n}\cdot(f\vec{a})\,dS = \left(\lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}f\hvec{n}\,dS\right) \cdot \vec{a} = (\bnabla f)\cdot\vec{a},
        \]
        and
        \[
        \bnabla \times (f\vec{a}) = \lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}\hvec{n}\times(f\vec{a})\,dS = \left(\lim_{d(S) \to 0}\frac{1}{|V|}\oint_{S}f\hvec{n}\,dS\right) \times \vec{a} = (\bnabla f)\times\vec{a}.
        \]
    \end{proof}
\end{lemma}

\begin{proposition}
    In Cartesian coordinates,
    divergence and curl are given by
    \begin{enumerate}[label = (\roman*)]
        \item
        \[
        \bnabla \cdot \vec{f} = \pd[f_1]{x} + \pd[f_2]{y} + \pd[f_3]{z}.
        \]

        \item
        \[
        \bnabla \times \vec{f} = \begin{vmatrix}
            \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
            \pd{x} & \pd{y} & \pd{z} \\
            f_1 & f_2 & f_3
        \end{vmatrix} = \left(\pd[f_3]{y} - \pd[f_2]{z}\right)\vec{e}_1 + \left(\pd[f_1]{z} - \pd[f_3]{x}\right)\vec{e}_2 + \left(\pd[f_2]{x} - \pd[f_1]{y}\right)\vec{e}_3.
        \]
    \end{enumerate}

    \begin{proof}
        Using the previous lemma,
        we have
        \begin{align*}
            \bnabla \cdot \vec{f} &= \bnabla \cdot (f_1\vec{e}_1 + f_2\vec{e}_2 + f_3\vec{e}_3) \\
            &= \bnabla \cdot (f_1\vec{e}_1) + \bnabla \cdot (f_2\vec{e}_2) + \bnabla \cdot (f_3\vec{e}_3) &&(\text{since the divergence is clearly linear}) \\
            &= (\bnabla f_1)\vec{e}_1 + (\bnabla f_2)\vec{e}_2 + (\bnabla f_3)\vec{e}_3 \\
            &= \pd[f_1]{x} + \pd[f_2]{y} + \pd[f_3]{z} &&(\text{from definition of the gradient}).
        \end{align*}

        Using the previous lemma again,
        we have that
        \[
        \bnabla \times (f_1\vec{e}_1) = (\bnabla f_1) \times \vec{e}_1 = \pd[f_1]{y}\vec{e}_2 \times \vec{e}_1 + \pd[f_1]{z}\vec{e}_3 \times \vec{e}_1 = -\pd[f_1]{y}\vec{e}_3 + \pd[f_1]{z}\vec{e}_2.
        \]
        The other terms are obtained similarly.
    \end{proof}
\end{proposition}

\subsection{Product rules}
A consequence is that instead of the single product rule from single-variable calculus,
\begin{proposition}[Product rules]
    If $f, g : \R ^ 3 \to \R$ are scalar fields,
    and $\vec{f}, \vec{g} : \R ^ 3 \to \R ^ 3$ are vector fields,
    all $C ^ {1}$,
    then
    \begin{enumerate}[label = (\roman*)]
        \item $\bnabla (fg) \equiv g\bnabla f + f\bnabla g$,
        
        \item $\bnabla \cdot (g\vec{f}) \equiv (\bnabla g) \cdot f + g \bnabla \cdot \vec{f}$,
        
        \item $\bnabla \cdot (\vec{f} \times \vec{g}) \equiv \vec{g} \cdot (\bnabla \times \vec{f}) - \vec{f}\cdot(\bnabla \times \vec{g})$,
        
        \item $\bnabla \cdot (g\vec{f}) \equiv g(\bnabla \times \vec{f}) + (\bnabla g) \times \vec{f}$,

        \item $\bnabla \times (\vec{f} \times \vec{g}) \equiv (\vec{g} \cdot \bnabla)\vec{f} - (\vec{f} \cdot \bnabla)\vec{g} + \vec{f}(\bnabla \cdot \vec{g}) - \vec{g}(\bnabla \cdot \vec{f})$.

        \item $\bnabla (\vec{f} \cdot \vec{g}) \equiv (\vec{g} \cdot \bnabla)\vec{f} + (\vec{f} \cdot \bnabla)\vec{g} + \vec{f} \times (\bnabla \times \vec{g}) + \vec{g} \times (\bnabla \times \vec{f})$.
    \end{enumerate}

    \begin{proof}
        \begin{enumerate}[label = (\roman*)]
            \item We use the regular
            (single-variable)
            product rule to obtain
            \begin{align*}
                \bnabla (fg) &= \pd{x}(fg)\vec{e}_1 + \pd{y}(fg)\vec{e}_2 + \pd{z}(fg)\vec{e}_3 \\
                &= \left(\pd[f]{x}g + f\pd[g]{x}\right)\vec{e}_1 + \left(\pd[f]{y}g + f\pd[g]{y}\right)\vec{e}_2 + \left(\pd[f]{z}g + f\pd[g]{z}\right)\vec{e}_3 \\
                &= \left(\pd[f]{x}\vec{e}_1 + \pd[f]{y}\vec{e}_2 + \pd[f]{z}\vec{e}_3\right)g + f\left(\pd[g]{x}\vec{e}_1 + \pd[g]{y}\vec{e}_2 + \pd[g]{z}\vec{e}_3\right) \\
                &= g\bnabla f + f\bnabla g.
            \end{align*}

            \item By \autoref{pre:mathmethII:lem:gradscfldprods} we have
            \begin{align*}
                \bnabla \cdot (g\vec{f}) &= \bnabla \cdot (gf_1\vec{e}_1 + gf_2\vec{e}_2 + gf_3\vec{e}_3) \\
                &= \bnabla (gf_1)\cdot\vec{e}_1 + \bnabla (gf_2)\cdot\vec{e}_2 + \bnabla(gf_3)\cdot\vec{e}_3 \\
                &= f_1\bnabla g \cdot \vec{e}_1 + g\bnabla f_1 \cdot \vec{e}_1 + f_2\bnabla g \cdot \vec{e}_2 + g\bnabla f_2 \cdot \vec{e}_2 + f_3\bnabla g \cdot \vec{e}_3 + g\bnabla f_3 \cdot \vec{e}_3 \\
                &= (\bnabla g)\cdot(f_1\vec{e}_1 + f_2\vec{e}_2 + f_3\vec{e}_3) + g\left(\pd[f_1]{x} + \pd[f_2]{y} + \pd[f_3]{z}\right) \\
                &= (\bnabla g)\cdot\vec{f} + g\bnabla\cdot\vec{f}.
            \end{align*}

            \item Evaluating the cross product first gives
            \begin{align*}
                \bnabla \cdot (\vec{f} \times \vec{g}) &= \bnabla \cdot \begin{vmatrix}
                    \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
                    f_1 & f_2 & f_3 \\
                    g_1 & g_2 & g_3
                \end{vmatrix} \\
                &= \pd{x}(f_2g_3 - f_3g_2) + \pd{y}(f_3g_1 - f_1g_3) + \pd{z}(f_1g_2 - f_2g_1) \\
                &= g_1\left(\pd[f_3]{y} - \pd[f_2]{z}\right) + g_2\left(\pd[f_1]{z} - \pd[f_3]{z}\right) + g_3\left(\pd[f_2]{x} - \pd[f_1]{y}\right) \\
                &+ f_1\left(\pd[g_2]{z} - \pd[g_3]{y}\right) + f_2\left(\pd[g_3]{x} - \pd[g_1]{z}\right) + f_3\left(\pd[g_1]{y} - \pd[g_2]{x}\right) \\
                &= \vec{g}\cdot(\bnabla \times \vec{f}) - \vec{f}\cdot(\bnabla \times \vec{g}).
            \end{align*}

            \item By \autoref{pre:mathmethII:lem:gradscfldprods},
            we have
            \begin{align*}
                \bnabla \times (g\vec{f}) &= \bnabla \times (gf_1\vec{e}_1 + gf_2\vec{e}_2 + gf_3\vec{e}_3) \\
                &= \bnabla(gf_1) \times \vec{e}_1 + \bnabla (gf_2) \times \vec{e}_2 + \bnabla(gf_3)\times \vec{e}_3 \\
                &= \pd{y}(gf_1)\vec{e}_2 \times \vec{e}_1 + \pd{z}(gf_1)\vec{e}_3 \times \vec{e}_1 + \pd{x}(gf_2)\vec{e}_1 \times \vec{e}_2 + \pd{z}(gf_2)\vec{e}_3 \times \vec{e}_2 \\
                &+ \pd{x}(gf_3)\vec{e}_1 \times \vec{e}_3 + \pd{y}(gf_3)\vec{e}_2 \times \vec{e}_3 \\
                &= -\pd{y}(gf_1)\vec{e}_3 + \pd{z}(gf_1)\vec{e}_2 + \pd{x}(gf_2)\vec{e}_3 - \pd{z}(gf_2)\vec{e}_1 - \pd{x}(gf_3)\vec{e}_2 + \pd{y}(gf_3)\vec{e}_1 \\
                &= g\left[\left(\pd[f_3]{y} - \pd[f_2]{z}\right)\vec{e}_1 + \left(\pd[f_1]{z} - \pd[f_3]{x}\right)\vec{e}_2 + \left(\pd[f_2]{x} - \pd[f_1]{y}\right)\vec{e}_3\right] \\
                &\left(\pd[g]{y}f_3 - \pd[g]{z}f_2\right)\vec{e}_1 + \left(\pd[g]{z}f_1 - \pd[g]{x}f_3\right)\vec{e}_2 + \left(\pd[g]{x}f_2 - \pd[g]{y}f_1\right)\vec{e}_3 \\
                &= g(\bnabla \times \vec{f}) + (\bnabla g) \times \vec{f}.
            \end{align*}
        \end{enumerate}
    \end{proof}
\end{proposition}

\subsection{Divergence and curl in curvilinear coordinates}

\begin{proposition}
    In orthogonal curvilinear coordinates $(u, v, w)$,
    the divergence and curl are given by
    \begin{enumerate}[label = (\roman*)]
        \item
        \[
        \bnabla \cdot \vec{f} = \frac{1}{h_uh_vh_w}\left[\pd{u}(h_vh_wf_u) + \pd{v}(h_uh_wf_v) + \pd{w}(h_uh_vf_w)\right].
        \]
        
        \item
        \[
        \bnabla \times \vec{f} = \frac{1}{h_uh_vh_w}\begin{vmatrix}
            h_u\vec{e}_u & h_v\vec{e}_v & h_w\vec{e}_w \\
            \pd{u} & \pd{v} & \pd{w} \\
            h_uf_u & h_vf_v & h_wf_w
        \end{vmatrix}.
        \]
    \end{enumerate}

    \begin{proof}
        First,
        note from \autoref{pre:mathmethII:lem:gradscfldprods} that $\bnabla v = \frac{1}{h_v}\vec{e}_v$,
        so $\vec{e}_v = h_v\bnabla v$,
        and similarly $\vec{e}_w = h_w\bnabla w$.

        For the first part,
        note using orthogonality followed by the product rules,
        we get
        \begin{align*}
            \bnabla \cdot (f_u\vec{e}_u) &= \bnabla \cdot (f_u\vec{e}_v \times \vec{e}_w) \\
            &= \bnabla \cdot (f_uh_vh_w \bnabla v \times \bnabla w) \\
            &= \bnabla \cdot (f_uh_vh_w) \cdot \bnabla v \times \bnabla w + f_uh_vh_w\bnabla \cdot (\bnabla v \times \bnabla w) \\
            &= \bnabla \cdot (f_uh_vh_w) \cdot \bnabla v \times \bnabla w + f_uh_vh_w\bnabla \cdot (v \bnabla w]) \\
            &= \bnabla \cdot (f_uh_vh_w) \cdot \bnabla v \times \bnabla w \\
            &= \bnabla \cdot (f_uh_vh_w) \cdot \frac{\vec{e}_v \times \vec{e}_w}{h_vh_w} \\
            &= \frac{1}{h_vh_w}\vec{e}_u \cdot \bnabla(f_uh_vh_w) \\
            &= \frac{1}{h_uh_vh_w}\pd{u}(h_vh_wf_u).
        \end{align*}
        The other components are similar.

        For the second part,
        we use $\vec{e}_u = h_u\bnabla u$ and the product rules to find
        \begin{align*}
            \bnabla \times (f_u\vec{e}_u) &= \bnabla \times (f_uh_u \bnabla u) \\
            &= \bnabla(f_uh_u) \times \bnabla u + f_uh_u\bnabla \times \bnabla u \\
            &= \bnabla (f_uh_u) \times \bnabla u \\
            &= \frac{1}{h_u}\bnabla(f_uh_u) \times \vec{e}_u \\
            &= \frac{1}{h_u}\bnabla(f_uh_u) \times (\vec{e}_v \times \vec{e}_w) \\
            &= \frac{1}{h_u}\vec{e}_v[\vec{e}_w \cot \bnabla(f_uh_u)] - \vec{e}_w(\vec{e}_v \times \bnabla(f_uh_u) \\
            &= \frac{1}{h_uh_w}\pd{w}(h_uf_u)\vec{e}_v - \frac{1}{h_uh_v}\pd{v}(h_uf_u)\vec{e}_w.
        \end{align*}
        The penultimate step used the vector triple product identity.
        The other components are similar.
    \end{proof}
\end{proposition}
