\begin{problem}[$2012$ question $2$]
    Calculate $\int_{C}\vec{f}\cdot\hvec{t}\,d\ell$ where $\vec{f} = -x\vec{e}_1 + 4y\vec{e}_2$ and $C$ is the part of $x ^ 2 + 4y ^ 2 = 4$ from $(2, 0)$ to $(0, 1)$.

    \begin{solution}
        Parameterisation:
        \[
        \vec{x}(t) = 2\cos(t)\vec{e}_1 + \sin(t)\vec{e}_2\qquad\text{for } t \in [0, \frac{\pi}{2}].
        \]
        \begin{align*}
            \int_{C}\vec{f}\cdot\hvec{t}\,d\ell &= \int_{0}^{\frac{\pi}{2}}\vec{f}\cdot\hvec{t}\left|\frac{d\vec{x}}{dt}\right|\,dt \\
            &= \int_{0}^{\frac{\pi}{2}}\vec{f}\cdot\frac{d\vec{x}}{dt}\,dt \\
            &= \int_{0}^{\frac{\pi}{2}}\vec{f}\cdot(-2\sin(t)\vec{e}_1 + \cos(t)\vec{e}_2)\,dt \\
            &= \int_{0}^{\frac{\pi}{2}}(-x\vec{e}_1 + 4y\vec{e}_2)\cdot(-2\sin(t)\vec{e}_1 + \cos(t)\vec{e}_2)\,dt \\
            &= \int_{0}^{\frac{\pi}{2}}(-2\cos(t)\vec{e}_1 + 4\sin(t)\vec{e}_2)\cdot(-2\sin(t)\vec{e}_1 + \cos(t)\vec{e}_2)\,dt \\
            &= \int_{0}^{\frac{\pi}{2}}(4\cos(t)\sin(t) + 4\sin(t)\cos(t))\,dt \\
            &= 8\int_{0}^{\frac{\pi}{2}}\cos(t)\sin(t)\,dt \\
            &= 8\int_{1}^{0}(-u)\,du \\
            &= -8\left[\frac{u ^ 2}{2}\right]_{1}^{0} = 4.
        \end{align*}
    \end{solution}
\end{problem}


\begin{problem}[$2015$ question $8$]
    Calculate the line integral of $\vec{f}(\vec{x}) = z ^ 2\vec{e}_1 + x ^ 2\vec{e}_2 + y ^ 2\vec{e}_3$ along each of the following curves.

    \begin{enumerate}[label = (\alph*)]
        \item The straight line $C_1$ from $(0, 0, 0)$ to $(1, 1, 1)$.

        \item The circle $C_2$ with radius $a > 0$ in the $z = 0$ plane centred at $(1, 1, 0)$,
        traversed anti-clockwise.

        \item The triangle $C_3$ with vertices $(1, 1, 0) \to (1 + b, 1, 0) \to (1, 1 + b, 0)$ for $b > 0$.
    \end{enumerate}

    \begin{solution}
        \begin{enumerate}[label = (\alph*)]
            \item Need parameterisation:
            \[
            \vec{x}(t) = t\vec{e}_1 + t\vec{e}_2 + t\vec{e}_3\quad\text{for } t \in [0, 1]
            \]
            \[
            \frac{d\vec{x}}{dt} = \vec{e}_1 + \vec{e}_2 + \vec{e}_3.
            \]

            So
            \begin{align*}
                \int_{C_1}\vec{f}\cdot\hvec{t}\,d\ell &= \int_{0}^{1}(z ^ 2\vec{e}_1 + x ^ 2\vec{e}_2 + y ^ 2\vec{e}_3)\cdot(\vec{e}_1 + \vec{e}_2 + \vec{e}_3)\,dt \\
                &= \int_{0}^{1}(t ^ 2\vec{e}_1 + t ^ 2\vec{e}_2 + t ^ 2\vec{e}_3) \cdot (\vec{e}_1 + \vec{e}_2 + \vec{e}_3)\,dt \\
                &= \int_{0}^{1}3t ^ 2\,dt = \left[t ^ 3\right]_{0}^{1} = 1.
            \end{align*}

            \item Parameterisation for $C_2$:
            \[
            \vec{x}(t) = (1 + a\cos(t))\vec{e}_1 + (1 + a\sin(t))\vec{e}_2\quad\text{for } [0, 2\pi].
            \]
            \[
            \frac{d\vec{x}}{dt} = -a\sin(t)\vec{e}_1 + a\cos(t)\vec{e}_2.
            \]
            So
            \begin{align*}
                \int_{C_2}\vec{f}\hvec{t}\,d\ell &= \int_{0}^{2\pi}(z ^ 2\vec{e}_1 + x ^ 2\vec{e}_2 + y ^ 2\vec{e}_3) \cdot (-a\sin(t)\vec{e}_1 + a\cos(t)\vec{e}_2)\,dt \\
                &= \int_{0}^{2\pi}\left[(1 + a\cos(t)) ^ 2\vec{e}_2 + (1 + a\sin(t)) ^ 2\vec{e}_3\right]\cdot\left(-a\sin(t)\vec{e}_1 + a\cos(t)\vec{e}_2\right)\,dt \\
                &= \int_{0}^{2\pi}a\cos(t)(1 + a\cos(t)) ^ 2\,dt \\
                &= \int_{0}^{2\pi}a\cos(t)(1 + 2a\cos(t) + a ^ 2\cos ^ 2(t))\,dt \\
                &= a\int_{0}^{2\pi}\cos(t)\,dt + 2a ^ 2\int_{0}^{2\pi}\cos ^ 2(t)\,dt + a ^ 3\int_{0}^{2\pi}\cos ^ 3(t)\,dt \\
                &= 0 + 2\pi a ^ 2 + 0 = 2\pi a ^ 2.
            \end{align*}

            \item
            Split into three straight lines that are differentiable:
            \[
            \vec{x}(t) = \begin{cases}
                (1 + bt)\vec{e}_1 + \vec{e}_2 &\text{for } t \in [0, 1] \\
                (1 + b(1 -t))\vec{e}_1 + (1 + bt)\vec{e}_2 &\text{for } t \in [0, 1] \\
                \vec{e}_1 + (1 + b(1 -t))\vec{e}_2 &\text{for } t \in [0, 1].
            \end{cases}
            \]
            So
            \begin{align*}
                \oint_{C}\vec{f}\cdot\hvec{t}\,d\ell &= \int\vec{f}\cdot\hvec{t}\,d\ell + \int\vec{f}\cdot\hvec{t}\,d\ell + \int\vec{f}\cdot\hvec{t}\,d\ell \\
                &= \int_{0}^{1}(x ^ 2\vec{e}_2 + y ^ 2\vec{e}_3)\cdot(b\vec{e}_1)\,dt + \int_{0}^{1}(x ^ 2\vec{e}_2 + y ^ 2\vec{e}_3)\cdot(-b\vec{e}_1 + b\vec{e}_2)\,dt + \int_{0}^{1}(x ^ 2\vec{e}_2 + y ^ 2\vec{e}_3)\cdot(-b\vec{e}_2)\,dt \\
                &= \int_{0}^{1}0\,dt + \int_{0}^{1}bx ^ 2\,dt - \int_{0}^{1}bx ^ 2\,dt \\
                &= \int_{0}^{1}b(1 + b(1 - t)) ^ 2\,dt - \int_{0}^{1}b\,dt \\
                &= \int_{0}^{1}b(1 + b ^ 2 + b ^ 2t ^ 2 + 2b - 2bt)\,dt - \int_{0}^{1}b\,dt \\
                &= \left[b(1 + b ^ 2)t + \frac{b ^ 3}{3}t ^ 3 + 2b ^ 2t - b ^ 2t ^ 2 - b ^ 3t ^ 2\right]_{0}^{1} - \left[bt\right]_{0}^{1} \\
                &= b(1 + b ^ 2) + \frac{b ^ 3}{3} + 2b ^ 2 - b ^ 2 - b ^ 3 - b \\
                &= b ^ 2\left(\frac{b}{3} + 1\right).
            \end{align*}
        \end{enumerate}
    \end{solution}
\end{problem}

\begin{problem}[August $2024$ question $6$]
    The position vector in
    "ellipsoidal coordinates"
    is given by
    \[
    \vec{x}(r, \theta, \phi) = ar\sin(\theta)\cos(\phi)\vec{e}_1 + br\sin(\theta)\sin(\phi)\vec{e}_2 + cr\cos(\theta)\vec{e}_3
    \]
    where $r > 0$,
    $\theta \in [0, \pi]$,
    $\phi \in [0, 2\pi]$ and $a, b, c \in \R$ are given positive constants.

    Determine whether or not this $(r, \theta, \phi)$ coordinate system is orthogonal.

    \begin{solution}
        Calculate tangent vectors:
        \begin{align*}
            \pd[\vec{x}]{r} &= a\sin(\theta)\cos(\phi)\vec{e}_1 + b\sin(\theta)\sin(\phi)\vec{e}_2 + c\cos(\theta)\vec{e}_3 \\
            \pd[\vec{x}]{\theta} &= ar\cos(\theta)\cos(\phi)\vec{e}_1 + br\cos(\theta)\sin(\phi)\vec{e}_2 - cr\sin(\theta)\vec{e}_3
        \end{align*}
        then dotting them together
        \[
        \pd[\vec{x}]{r} \cdot \pd[\vec{x}]{\theta} \neq 0.
        \]
    \end{solution}
\end{problem}

\begin{problem}[$2013$ question $5$]
    Compute the surface integral $\int_{S}\vec{f}\cdot\hvec{n}\,dS$ of vector field $\vec{f} = (y, -x, x)$ over the surface $z = xy - \cos{x}$ for $x \in [0, 1]$,
    $y \in [0, 1]$,
    with upward normal.

    \begin{solution}
        \[
        \vec{f}(\vec{x}) = y\vec{e}_1 + x\vec{e}_2 + x\vec{e}_3.
        \]

        \begin{align*}
            \int_{S}\vec{f}\cdot\hvec{n}\,dS &= \int_{U}\vec{f}(\vec{x}(u, v)\cdot\hvec{n}\left|\pd[\vec{x}]{u}\times \pd[\vec{x}]{v}\right|\,dudv \\
            &= \int_{U}\vec{f}(\vec{x}(u, v)\cdot\left(\pd[\vec{x}]{u}\times \pd[\vec{x}]{v}\right)\,dudv.
        \end{align*}
        Parametrise the surface:
        \[
        z = xy - \cos{x}
        \]
        \[
        \vec{x}(x, y) = x\vec{e}_1 + y\vec{e}_2 + (xy - \cos{x})\vec{e}_3
        \]
        $x \in [0, 1]$,
        $y \in [0, 1]$.
        So
        \begin{align*}
            \pd[\vec{x}]{x} &= \vec{e}_1 + (y + \sin{x})\vec{e}_3 \\
            \pd[\vec{x}]{y} &= \vec{e}_2 + x\vec{e}_3
        \end{align*}
        \[
        \pd[\vec{x}]{x} \times \pd[\vec{x}]{y} = -(y + \sin{x})\vec{e}_1 + -x\vec{e}_2 + \vec{e}_3.
        \]
        Then
        \begin{align*}
            \int_{S}\vec{f}\cdot\hvec{n}\,dS &= \int_{0}^{1}\int_{0}^{1}(y\vec{e}_1 - x\vec{e}_2 + x\vec{e}_3)\cdot(-(y + \sin{x})\vec{e}_1 - x\vec{e}_2 + \vec{e}_3)\,dxdy \\
            &= \int_{0}^{1}\int_{0}^{1}(-y ^ 2 -y\sin{x} + x ^ 2 + x)\,dxdy \\
            &= \int_{0}^{1}\left[-y ^ 2x + y\cos{x} + \frac{1}{3}x ^ 3 + \frac{1}{2}x ^ 2\right]_{0}^{1}\,dy \\
            &= \int_{0}^{1}-y ^ 2 + y\cos(1) + \frac{5}{6} - y)\,dy \\
            &= \int_{0}^{1}-y ^ 2 + y(\cos(1) - 1) + \frac{5}{6})\,dy \\
            &= \left[-\frac{1}{3}y ^ 3 + \frac{1}{2}y ^ 2(\cos(1) - 1) + \frac{5}{6}y\right]_{0}^{1} \\
            &= -\frac{1}{3} + \frac{1}{2}\cos(1) + \frac{5}{6} - \frac{1}{2} = \frac{1}{2}\cos(1).
        \end{align*}
    \end{solution}
\end{problem}

\begin{problem}[$2020$ question $5$(a)]
    Calculate the surface integral $\oint_{S}\vec{f}\cdot\hvec{n}\,dS$ where $\vec{f}(\vec{x}) = \vec{x}$ and $S$ is the surface of the solid cylinder $\{x ^ 2 + y ^ 2 \leq 1, 0 \leq z \leq 4\}$,
    with outward normal.

    \begin{solution}
        \[
        \vec{f}(\vec{x}) = \vec{x} = x\vec{e}_1 + y\vec{e}_2 + z\vec{e}_3.
        \]

        $S$ is not smooth so divide into three pieces.
        $S_1$ -
        top,
        $S_2$ -
        bottom,
        $S_3$ -
        curved surface.

        Parametrise in cylindrical coordinates:
        \[
        \vec{x}(r, \theta, z) = r\cos(\theta)\vec{e}_1 + r\sin(\theta)\vec{e}_2 + z\vec{e}_3.
        \]
        $S_1$:
        \[
        \vec{x}(r, \theta) = r\cos(\theta)\vec{e}_1 + r\sin(\theta)\vec{e}_2 + 4\vec{e}_3\qquad\text{for } r \in [0, 1], \theta \in [0, 2\pi].
        \]
        \begin{align*}
            \int_{S_1}\vec{f}\cdot\hvec{n}\,dS &= \int_{S_1}\vec{f}\cdot\vec{e}_3\,dS \\
            &= \int_{0}^{2\pi}\int_{0}^{1}\vec{x}\cdot\vec{e}_3\left|\pd[\vec{x}]{r}\times\pd[\vec{x}]{\theta}\right|\,drd\theta \\
            &= \int_{0}^{2\pi}\int_{0}^{1}zr\,drd\theta \\
            &= 4\int_{0}^{2\pi}\int_{0}^{1}r\,drd\theta \\
            &= 4\pi.
        \end{align*}

        $S_2$:
        \[
        \vec{x}(r, \theta) = r\cos(\theta)\vec{e}_1 + r\sin(\theta)\vec{e}_2\qquad\text{for } r \in [0, 1], \theta \in [0, 2\pi].
        \]
        \begin{align*}
            \int_{S_2}\vec{f}\cdot\hvec{n}\,dS &= \int_{S_2}\vec{f}\cdot(-\vec{e}_3)\,dS \\
            &= \int_{0}^{2\pi}\int_{0}^{1}-zr\,drd\theta \\
            &= 0 &&(\text{z = 0}).
        \end{align*}


        $S_3$:
        \[
        \vec{x}(\theta, z) = \cos(\theta)\vec{e}_1 + \sin(\theta)\vec{e}_2 + z\vec{e}_3\quad\text{for } \theta \in [0, 2\pi], z \in [0, 4].
        \]
        \begin{align*}
            \int_{S_3}\vec{f}\cdot\hvec{n}\,dS &= \int_{S_3}\vec{f}\cdot\vec{e}_r\,dS
            \intertext{Either convert $\vec{e}_r$ to Cartesian or $\vec{f}$ to cylindrical}
            &= \int_{S_3}(r\vec{e}_r + z\vec{e}_z) \cdot \vec{e}_r\,dS \\
            &= \int_{0}^{4}\int_{0}^{2\pi}r\,d\theta dz \\
            &= 8\pi.
        \end{align*}
        So overall,
        \[
        \oint_{S}\vec{f}\cdot\hvec{n}\,dS = \int_{S_1}\vec{f}\cdot\hvec{n}\,dS + \int_{S_2}\vec{f}\cdot\hvec{n}\,dS + \int_{S_3}\vec{f}\cdot\hvec{n}\,dS = 12\pi
        \]
    \end{solution}
\end{problem}

\begin{problem}[August $2025$ question $2$(b)]
    Compute the volume of a solid chocolate egg given by
    \[
    \{r < a\cos ^ 3{\theta},\quad\theta \in [0, \pi / 2]\}
    \]
    in spherical coordinates,
    where $a > 0$ is constant.

    \begin{solution}
        Parametrise in spherical coordinates
        \[
        \vec{x}(r, \theta, \phi) = r\sin(\theta)\cos(\phi)\vec{e}_1 + r\sin(\theta)\sin(\phi)\vec{e}_2 + r\cos(\theta)\vec{e}_2
        \]

        \begin{align*}
            |V| &= \int_{V}\,dV \\
            &= \int_{0}^{2\pi}\int_{0}^{\pi / 2}\int_{0}^{a\cos ^ 3\theta}h_rh_{\theta}h_{\phi}\,drd\theta d\phi \\
            &= \int_{0}^{2\pi}\int_{0}^{\pi / 2}\int_{0}^{a\cos ^ 3\theta}r ^ 2\sin(\theta)\,drd\theta d\phi \\
            &= \int_{0}^{2\pi}\int_{0}^{\pi / 2}\left[\frac{1}{3}r ^ 3\sin(\theta)\right]_{0}^{a\cos ^ 3\theta}\,d\theta d\phi \\
            &= \frac{a ^ 3}{3}\int_{0}^{2\pi}\int_{0}^{\pi / 2}\cos ^ 9(\theta)\sin(\theta)\,d\theta d\phi \\
            &= \frac{a ^ 3}{3}\int_{0}^{2\pi}\left[-\frac{1}{10}\cos ^ {10}(\theta)\right]_{0}^{\pi / 2}\,d\phi \\
            &= \frac{a ^ 3}{30}\int_{0}^{2\pi}\,d\phi = \frac{\pi a ^ 3}{15}.
        \end{align*}
    \end{solution}
\end{problem}

\begin{problem}[$2007$ Question $1$]
    Let $\vec{x}$ be the position vector in $\R ^ 3$.
    Calculate
    \begin{enumerate}[label = (\roman*)]
        \item $\bnabla \cdot (3\vec{x})$.

        \item $\bnabla (3\vec{x} \cdot \vec{x})$.
        
        \item $\bnabla \times (\vec{e}_3 \times \vec{x})$.
    \end{enumerate}

    \begin{solution}
        \begin{enumerate}[label = (\roman*)]
            \item
            \begin{align*}
                \bnabla \cdot (3\vec{x}) &= \bnabla \cdot (3x\vec{e}_1 + 3y\vec{e}_2 + 3z\vec{e}_3) \\
                &= \pd{x}(3x) + \pd{y}(3y) + \pd{z}(3z) \\
                &= 3 + 3 + 3 \\
                &= 9.
            \end{align*}

            \item
            \begin{align*}
                \bnabla (3\vec{x} \cdot \vec{x}) &= \bnabla(3(x ^ 2 + y ^ 2 + z ^ 2)) \\
                &= \pd{x}(3(x ^ 2 + y ^ 2 + z ^ 2)\vec{e}_1 + \pd{y}(3(x ^ 2 + y ^ 2 + z ^ 2)\vec{e}_2 + \pd{z}(3(x ^ 2 + y ^ 2 + z ^ 2)\vec{e}_3 \\
                &= 6x\vec{e}_1 + 6y\vec{e}_2 + 6z\vec{e}_3 \\
                &= 6\vec{x}.
            \end{align*}

            \item
            \begin{align*}
                \vec{e}_3 \times \vec{x} &= \begin{vmatrix}
                    \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
                    0 & 0 & 1 \\
                    x & y & z
                \end{vmatrix} \\
                &= ((0)z - (1)(y))\vec{e}_1 + ((1)(x) - (0)(z))\vec{e}_2 + ((0)(y) - (0)(x))\vec{e}_3 \\
                &= -y\vec{e}_1 + x\vec{e}_2 \\
                \bnabla \times (\vec{e}_3 \times \vec{x}) &= \begin{vmatrix}
                    \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
                    \pd{x} & \pd{y} & \pd{z} \\
                    -y & x & 0
                \end{vmatrix} \\
                &= \left(\pd{y}(0) - \pd{z}(x)\right)\vec{e}_1 + \left(\pd{z}(-y) - \pd{x}(0)\right)\vec{e}_2 + \left(\pd{x}(x) - \pd{y}(-y)\right)\vec{e}_3 \\
                &= 2\vec{e}_3.
            \end{align*}
        \end{enumerate}
    \end{solution}
\end{problem}

\begin{problem}[$2004$ Question $1$]
    Calculate the divergence of the vector field
    \[
    \vec{f}(r, \theta, z) = r\vec{e}_r + \cos(\theta)\vec{e}_{\theta} + \vec{e}_z.
    \]

    \textit{Recall}:
    $\vec{e}_r = \cos(\theta)\vec{e}_1 + \sin(\theta)\vec{e}_2$,
    $\vec{e}_{\theta} = -\sin(\theta)\vec{e}_1 + \cos(\theta)\vec{e}_2$,
    $\vec{e}_z = \vec{e}_3$.

    \begin{solution}
        $\vec{f}$ is in cylindrical coordinates,
        use formula for divergence in orthogonal curvilinear coordinates:
        \[
        \bnabla \cdot \vec{f} = \frac{1}{h_uh_vh_w}\left(\pd{u}(h_vh_wf_u) + \pd{v}(h_uh_wf_v) + \pd{w}(h_uh_vf_w)\right).
        \]
        Here $(u, v, w) = (r, \theta, z)$,
        $h_r = 1$,
        $h_{\theta} = r$,
        $h_z = 1$.

        So
        \begin{align*}
            \bnabla \cdot \vec{f} &= \frac{1}{r}\left[\pd{r}(rf_r) + \pd{\theta}(f_{\theta}) + \pd{z}(rf_z)\right] \\
            &= \frac{1}{r}\left[\pd{r}(r ^ 2) + \pd{\theta}(\cos{\theta}) + \pd{z}(r)\right] \\
            &= \frac{1}{r}(2r - \sin{\theta}) = 2 - \frac{\sin{\theta}}{r}
        \end{align*}
    \end{solution}
\end{problem}

\begin{problem}[$2024$ August Question $5$]
    Let $\vec{f}(\vec{x})$ be the vector field
    \[
    \vec{f}(\vec{x}) = x ^ 2e ^ {5z}\vec{e}_1 + x\cos(y)\vec{e}_2 + 3y\vec{e}_3
    \]
    and let $S$ be the disk bounded by the circle $C$ with parameterisation
    \[
    \vec{x}(t) = (2 + 2\cos(t))\vec{e}_2 + (2 + 2\sin(t))\vec{e}_3,\qquad t \in [0, 2\pi].
    \]

    \begin{enumerate}[label = (\alph*)]
        \item Compute $\bnabla \times \vec{f}$.

        \item By evaluating an appropriate surface integral,
        calculate the flux of $\bnabla \times \vec{f}$ through $S$,
        taking the surface normal to be the positive $x$-direction.

        \item Evaluate the circulation of $\vec{f}$ around $C$,
        in the direction of the parameterisation.
    \end{enumerate}

    \begin{solution}
        \begin{enumerate}[label = (\alph*)]
            \item
            \begin{align*}
                \bnabla \times \vec{f} &= \begin{vmatrix}
                    \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
                    \pd{x} & \pd{y} & \pd{z} \\
                    x ^ 2e ^ {5z} & x\cos{y} & 3y
                \end{vmatrix} \\
                &= \left[\pd{y}(3y) - \pd{z}(x\cos{y})\right]\vec{e}_1 + \left[\pd{z}(x ^ 2e ^ {5z}) - \pd{x}(3y)\right]\vec{e}_2 + \left[\pd{x}(x\cos{y}) - \pd{y}(x ^ 2e ^ {5z})\right]\vec{e}_3 \\
                &= 3\vec{e}_1 + 5x ^ 2e ^ {5z}\vec{e}_2 + \cos(y)\vec{e}_3.
            \end{align*}

            \item
            $C$ has centre at $(2, 2)$,
            radius $2$,
            in $(y, z)$-plane
            (from the parameterisation)

            Parametrise $S$ with polar coordinates in $(y, z)$ plane with centre at $(2, 2)$:
            \[
            \vec{x}(u, v) = (2 + u\cos(v))\vec{e}_2 + (2 + u\sin(v))\vec{e}_3\qquad\text{for } u \in [0, 2], v \in [0, 2\pi].
            \]
            Want to calculate
            \begin{align*}
                \int_{S}(\bnabla \times \vec{f})\cdot\hvec{n}\,dS &= \int_{0}^{2\pi}\int_{0}^{2}(\bnabla \times \vec{f}) \cdot \hvec{n}\left|\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right|\,dudv \\
                &= \int_{0}^{2\pi}\int_{0}^{2}(\bnabla \times \vec{f}) \cdot \left(\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right)\,dudv \\
                \pd[\vec{x}]{u} \times \pd[\vec{x}]{v} &= \begin{vmatrix}
                    \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
                    0 & \cos{v} & \sin{v} \\
                    0 & -u\sin{v} & u\cos{v}
                \end{vmatrix} \\
                &= \vec{e}_1(u\cos ^ 2v + u\sin ^ 2v) = u\vec{e}_1 \\
                \int_{0}^{2\pi}\int_{0}^{2}(\bnabla \times \vec{f}) \cdot \left(\pd[\vec{x}]{u} \times \pd[\vec{x}]{v}\right)\,dudv &= \int_{0}^{2\pi}\int_{0}^{2}(3\vec{e}_1 + 5x ^ 2e ^ {5z}\vec{e}_2 + \cos(y)\vec{e}_3) \cdot (u\vec{e}_1)\,dudv \\
                &= \int_{0}^{2\pi}\int_{0}^{2}3u\,dudv \\
                &= \int_{0}^{2\pi}\left[\frac{3u ^ 2}{2}\right]_{0}^{2}\,dv \\
                &= 6\int_{0}^{2\pi}\,dv = 12\pi.
            \end{align*}

            \item
            Circulation:
            \begin{align*}
                \oint_{C}\vec{f}\cdot\hvec{t}\,d\ell &= \int_{0}^{2\pi}\left(x ^ 2e ^ {5z}\vec{e}_1 + x\cos(y)\vec{e}_2 + 3y\vec{e}_3\right)\cdot\frac{d\vec{x}}{dt}\,dt \\
                &= \int_{0}^{2\pi}-2\sin(t)(2 + 2\cos(t))\cos(2 + 2\sin(t)) + 6\cos(t)(2 + 2\sin(t))\,dt
            \end{align*}
            will see how to do this a different way next week.
        \end{enumerate}
    \end{solution}
\end{problem}

\begin{problem}[$2019$ Question $3$]\phantom{}
    \begin{enumerate}[label = (\alph*)]
        \item Let $\vec{f}(\vec{x})$ be the vector field in $\R ^ 3$ defined by
        \[
        f(x, y, z) = 2xy\sin(z)\vec{e}_1 + (x ^ 2 + 2y)\sin(z)\vec{e}_2 + (x ^ 2y + y ^ 2)\cos(z)\vec{e}_3.
        \]
        Find a scalar field $g(x, y, z)$ such that $\vec{f} = \bnabla g$.

        \item Using your answer to part
        (a)
        or otherwise,
        compute the line integral
        \[
        \int_{C}\vec{f}\cdot\hvec{t}\,d\ell
        \]
        where $C$ is the straight line running from the origin to the point $(2, 3, 3\pi / 2)$.

        \item Now let $\vec{v}(\vec{x})$ be the vector field
        \[
        \vec{v}(x, y, z) = 2xy\sin(z)\vec{e}_1 + 2xy\sin(z)\vec{e}_2 + x ^ 2y\cos(z)\vec{e}_3.
        \]
        Is it possible to write $\vec{v}$ as $\vec{v} = \bnabla k$ for some scalar field $k(x, y, z)$?
        Justify your answer.
    \end{enumerate}

    \begin{solution}\phantom{}
        \begin{enumerate}[label = (\alph*)]
            \item
            We need
            \[
            \pd[g]{x} = \underbrace{2xy\sin(z)}_{f_1}\qquad\pd[g]{y} = \underbrace{(x ^ 2 + 2y)\sin(z)}_{f_2}\qquad\pd[g]{z} = \underbrace{(x ^ 2y + y ^ 2)\cos(z)}_{f_3}.
            \]
            Integrate $\pd[g]{x}$ with respect to $x$:
            \[
            g = x ^ 2y\sin{z} + h(y, z).
            \]
            Then $\pd[g]{y}$ implies
            \begin{align*}
                x ^ 2\sin{z} + \pd[h]{y} = (x ^ 2 + 2y)\sin{z} &\implies \pd[h]{y} = 2y\sin{z} \\
                &\implies h = y ^ 2\sin{z} + m(z).
            \end{align*}
            Then $\pd[g]{z}$ implies
            \begin{align*}
                x ^ 2y\cos{z} + y ^ 2\cos{z} + m'(z) = (x ^ 2y + y ^ 2)\cos{z} &\implies m'(z) = 0 \\
                &\implies m = \text{constant}.
            \end{align*}
            So
            \[
            g(x, y, z) = x ^ 2y\sin{z} + y ^ 2\sin{z} + \text{constant}.
            \]

            \item
            \[
            \int_{C}\vec{f}\cdot\hvec{t}\,d\ell.
            \]
            Need to parametrise $C$:
            \[
            \vec{x}(t) = 2t\vec{e}_1 + 3t\vec{e}_2 + \frac{3\pi}{2}t\vec{e}_3\qquad\text{for } t \in [0, 1].
            \]
            So
            \begin{align*}
                \int_{C}\vec{f}\cdot\hvec{t}\,d\ell &= \int_{0}^{1}\vec{f}\cdot\frac{d\vec{x}}{dt}\,d\ell \\
                &= \int_{0}^{1}\vec{f}\cdot\left(2\vec{e}_1 + 3\vec{e}_2 + \frac{3\pi}{2}\vec{e}_3\right)\,dt \\
                &= \int_{0}^{1}\left[4(2t)(3t)\sin\left(\frac{3\pi}{2}t\right) + 3((2t) ^ 2 + 2(3t))\sin\left(\frac{3\pi}{2}t\right) + \frac{3\pi}{2}\left[(2t) ^ 2(3t) + (3t) ^ 2\right]\cos\left(\frac{3\pi}{2}t\right)\right]\,dt.
            \end{align*}
            Simpler method:
            Use the fact that $\vec{f} = \bnabla g$ so that the fundamental theorem of line integrals gives
            \begin{align*}
                \int_{C}\vec{f}\cdot\hvec{t}\,d\ell &= \int_{C}\bnabla g\cdot\hvec{t}\,d\ell \\
                &= g(\vec{x}(t_1)) - g(\vec{x}(t_0)) \\
                &= g(2, 3, 3\pi / 2) - g(0, 0, 0) \\
                &= 12\sin{\frac{3\pi}{2}} + 9\sin{\frac{3\pi}{2}} + \text{constant} - 0 - 0 - \text{constant} \\
                &= 12(-1) + 9(-1) = -21.
            \end{align*}

            \item
            Recall
            \[
            \vec{v} = \bnabla k \implies \bnabla \times \vec{v} = \vec{0}.
            \]
            Here
            \[
            \bnabla \times \vec{v} = \begin{vmatrix}
                \vec{e}_1 & \vec{e}_2 & \vec{e}_3 \\
                \pd{x} & \pd{y} & \pd{z} \\
                2xy\sin{z} & 2xy\sin{z} & x ^ 2y\cos{z}
            \end{vmatrix} = \vec{e}_1\underbrace{(x ^ 2\cos{z} - 2xy\cos{z})}_{\neq 0} + \dotsi
            \]
            so $\bnabla \times \vec{v} \neq \vec{0}$ so $\vec{v}$ is not conservative.
        \end{enumerate}
    \end{solution}
\end{problem}

\begin{problem}[$2016$ Question $5$]\phantom{}
    \begin{enumerate}[label = (\alph*)]
        \item State the Divergence Theorem.

        \item Use it to calculate $\oint_{S}\vec{f}\cdot\hvec{n}\,dS$ where $\vec{f}(x, y, z) = xy\vec{e}_1 - \frac{1}{2}y ^ 2\vec{e}_2 + z\vec{e}_3$ and $S$ is the boundary of the volume enclosed by $z = 4 - 3x ^ 2 - 3y ^ 2$ on the top,
        and $z = 0$ on the bottom.
    \end{enumerate}

    \begin{solution}\phantom{}
        \begin{enumerate}[label = (\alph*)]
            \item If $\vec{f} : \R ^ 3 \to \R ^ 3$ is a $C ^ {1}$ vector field and $S$ is a closed surface with outward normal $\hvec{n}$ bounding a region $V$,
            then
            \[
            \oint_{S}\vec{f}\cdot\hvec{n}\,dS = \int_{V}(\bnabla \cdot \vec{f})\,dV.
            \]

            \item Need to calculate $\int_{V}(\bnabla \cdot \vec{f})\,dV$.
            \begin{align*}
                \bnabla \cdot \vec{f} &= \pd[f_1]{x} + \pd[f_2]{y} + \pd[f_3]{z} \\
                &= \pd{x}(xy) + \pd{y}\left(-\frac{y ^ 2}{2}\right) + \pd{z}(z)  \\
                &= y - y + 1 = 1.
            \end{align*}
            Top:
            \[
            z = 4 - 3x ^ 2 - 3y ^ 2 = 4 - 3(x ^ 2 + y ^ 2) \underset{\mathclap{\text{Cylindrical coordinates}}}{=} 4 -3r ^ 2
            \]
            Bottom:
            \[
            z = 0.
            \]
            So
            \begin{align*}
                \int_{V}(\bnabla \cdot \vec{f})\,dV &= \int_{V}\,dV \\
                &= \int_{0}^{4}\int_{0}^{2\pi}\int_{0}^{\sqrt{\frac{4 - z}{3}}}r\,dr\,d\theta\,dz \\
                &= \int_{0}^{4}\int_{0}^{2\pi}\left[\frac{r ^ 2}{2}\right]_{0}^{\sqrt{\frac{4 - z}{3}}}\,d\theta\,dz \\
                &= \int_{0}^{4}\int_{0}^{2\pi}\frac{4 - z}{6}\,d\theta\,dz \\
                &= \frac{\pi}{3}\int_{0}^{4}4 - z\,dz \\
                &= \frac{\pi}{3}\left[4z - \frac{z ^ 2}{2}\right]_{0}^{4} \\
                &= \frac{\pi}{3}(16 - 16 / 2) = \frac{16}{6}\pi = \frac{8}{3}\pi.
            \end{align*}
        \end{enumerate}
    \end{solution}
\end{problem}

\begin{problem}[$2014$ Question $9$]\phantom{}
    \begin{enumerate}[label = (\alph*)]
        \item For any closed surface $S$,
        show that
        \[
        \oint_{S}(\bnabla \times \vec{f})\cdot\hvec{n}\,dS = 0
        \]
        by (i) the Divergence Theorem,
        and (ii) Stokes' Theorem.
        Show all your working.

        \item Calculate
        \[
        \int_{S}(\bnabla \times \vec{f})\cdot\hvec{n}\,dS
        \]
        for $f(x, y, z) = xz\vec{e}_1 + x ^ 2y ^ 2z\vec{e}_2 + xyz\vec{e}_3$,
        where $S$ is the open surface given by the cylinder
        (with no ends)
        of radius $a$,
        centred on the $x$-axis with equation $y ^ 2 + z ^ 2 = a ^ 2$ and $0 \leq x \leq 5$.
    \end{enumerate}

    \begin{solution}\phantom{}
        \begin{enumerate}[label = (\alph*)]
            \item \phantom{}
            \begin{enumerate}[label = (\roman*)]
                \item By Divergence Theorem
                \[
                \oint_{S}(\bnabla \times \vec{f})\cdot\hvec{n}\,dS = \int_{V}\bnabla\cdot(\bnabla \times \vec{f})\,dV = 0.
                \]
    
                \item
                Recall:
                Stokes' Theorem
                \[
                \oint_{C}\vec{f}\cdot\hvec{t}\,d\ell = \int_{S}(\bnabla \times \vec{f})\cdot\hvec{n}\,dS.
                \]
                Split $S$ into $S_1 \cup S_2$;
                with boundary curve $C$ between them.
                Then
                \begin{align*}
                    \oint_{S}(\bnabla \times \vec{f})\cdot\hvec{n}\,dS &= \int_{S_1}(\bnabla \times \vec{f})\cdot\hvec{n}\,dS + \int_{S_2}(\bnabla \times \vec{f})\cdot\hvec{n}\,dS \\
                    &= \oint_{C}\vec{f}\cdot\hvec{t}\,d\ell - \oint_{C}\vec{f}\cdot\hvec{t}\,d\ell = 0. &&(\text{Stokes' Theorem})
                \end{align*}
            \end{enumerate}

            \item
            Either:
            evaluate directly.
            Alternatively:
            \[
            \int_{S}(\bnabla \times \vec{f})\cdot\hvec{n}\,dS = \oint_{C_1}\vec{f}\cdot\hvec{t}\,d\ell - \oint_{C_2}\vec{f}\cdot\hvec{t}\,d\ell
            \]
            with $C_1, C_2$ the boundary curves.
            
        \end{enumerate}
    \end{solution}
\end{problem}

\begin{problem}[$2010$ Question $2$]
    Given that $S_{ij} = x_ix_j - c\delta_{ij}|\vec{x}|$ for $\vec{x} \in \R ^ 3$:

    \begin{enumerate}[label = (\alph*)]
        \item Find the value of $c$ such that $S_{ii} = 0$.

        \item Simplify as much as possible $S_{ij}\epsilon_{ijk}$.
    \end{enumerate}

    \begin{solution}
        \begin{enumerate}[label = (\alph*)]
            \item
            \begin{align*}
                S_{ii} &\left[= \sum_{i = 1}^{3}S_{ii}\right] \\
                &= x_ix_i - c\delta_{ij}|\vec{x}| ^ 2
            \end{align*}
            so
            \begin{align*}
                0 &= x_ix_i - c\delta_{ii}|\vec{x}| ^ 2 \\
                &= |\vec{x}| ^ 2 - 3c|\vec{x}| ^ 2 \\
                &= |\vec{x}| ^ 2(1 - 3c) \implies c = \frac{1}{3}.
            \end{align*}

            \item
            \begin{align*}
                S_{ij}\epsilon_{ijk} &= x_ix_j\epsilon_{ijk} - c\delta_{ij}|\vec{x}| ^ 2\epsilon_{ijk} \\
                \intertext{relabel $i \to j$}
                &= x_jx_i\epsilon_{jik} - c\delta_{ji}|\vec{x}| ^ 2\epsilon_{jik} \\
                &= x_ix_j\epsilon_{jik} - c\delta_{ij}|\vec{x}| ^ 2\epsilon_{jik} \\
                &= -x_ix_j\epsilon_{ijk} + c\delta_{ij}|\vec{x}| ^ 2\epsilon_{ijk} \\
                &= -S_{ij}\epsilon_{ijk} \implies S_{ij}\epsilon_{ijk} = 0.
            \end{align*}
        \end{enumerate}
    \end{solution}
\end{problem}

\begin{problem}[$2009$ Question $5$]\phantom{}
    \begin{enumerate}[label = (\alph*)]
        \item Show that if $g$ is a scalar field and $\vec{f}$ is a vector field,
        then
        \[
        \bnabla \times (g\vec{f}) = (\bnabla g)\times\vec{f} + g\bnabla \times \vec{f}.
        \]

        \item Hence,
        or otherwise,
        show that for some constant vector $\vec{c}$,
        \[
        \bnabla \times (|\vec{x}| ^ 2\vec{c}) = 2\vec{x} \times \vec{c}.
        \]

        \item Use Stokes' Theorem to show that
        \[
        \oint_{C}|\vec{x}| ^ 2\hvec{t}\,d\ell = -\int_{S}2\vec{x} \times \hvec{n}\,dS,
        \]
        where $C$ is the boundary of the surface $S$.
    \end{enumerate}

    \begin{solution}\phantom{}
        \begin{enumerate}[label = (\alph*)]
            \item
            Prove with index notation:
            the $i$th component of LHS is
            \begin{align*}
                [\bnabla \times (g\vec{f})]_i &= \epsilon_{ijk}\pd{x_j}[g\vec{f}]_k \\
                &= \epsilon_{ijk}\pd{x_j}(gf_k) \\
                &= \epsilon_{ijk}\left(\pd[g]{x_j}f_k + g\pd[f_k]{x_j}\right) \\
                &= [(\bnabla g) \times \vec{f}]_i + [g\bnabla \times \vec{f}]_i.
            \end{align*}

            \item
            In formula from $(a)$,
            take $\vec{f} = \vec{c}$ and $g = |\vec{x}| ^ 2$ so
            \[
            \bnabla \times (|\vec{x}| ^ 2\vec{c}) = (\bnabla |\vec{x}| ^ 2) \times \vec{c} + |\vec{x}| ^ 2\bnabla \times \vec{c}.
            \]
            Now calculate:
            \begin{align*}
                [\bnabla|\vec{x}| ^ 2]_i &= \pd{x_i}|\vec{x}| ^ 2 \\
                &= \pd{x_i}(\vec{x} \cdot \vec{x}) \\
                &= \pd{x_i}(x_jx_j) \\
                &= \pd[x_j]{x_i}x_j + x_j\pd[x_j]{x_i} \\
                &= 2x_j\pd[x_j]{x_i} \\
                &= 2x_j\delta_{ij} \\
                &= 2x_i
            \end{align*}
            so
            \[
            \bnabla|\vec{x}| ^ 2 = 2\vec{x}.
            \]
            So
            \[
            \bnabla \times (|\vec{x}| ^ 2\vec{c}) = 2\vec{x} \times \vec{c}.
            \]
            $\left[\text{Alternatively, }
            \bnabla|\vec{x}| ^ 2 = \bnabla(r ^ 2) = \frac{1}{h_r}\pd{r}(r ^ 2)\vec{e}_r = 2r\vec{r}_r = 2r\frac{\vec{x}}{r} = 2\vec{x}\right]$.

            \item 
            Try integrating the flux of each side in $(b)$ over $S$:
            \begin{align*}
                \int_{S}\bnabla\times(|\vec{x}| ^ 2\vec{c})\cdot\hvec{n}\,dS = \int_{S}(2\vec{x} \times \vec{c})\cdot\hvec{n}\,dS &\iff \oint_{C}(|\vec{x}| ^ 2\vec{c})\cdot\hvec{t}\,d\ell = \int_{S}(2\vec{x} \times \vec{c})\cdot\hvec{n}\,dS \\
                &\iff \vec{c}\cdot\oint_{C}|\vec{x}| ^ 2\hvec{t}\,d\ell = \vec{c}\cdot\int_{S}(\hvec{n} \times 2\vec{x}) \\
                &\iff \vec{c}\cdot\left[\oint_{C}|\vec{x}| ^ 2\hvec{t}\,d\ell - \int_{S}\hvec{n}\times 2\vec{x}\,dS\right] = 0
                \intertext{If the above holds for all $\vec{c} \in \R ^ 3$ then}
                &\implies \oint_{C}|\vec{x}| ^ 2\hvec{t}\,d\ell = \int_{S}\hvec{n}\times 2\vec{x} = -\int_{S}2\vec{x} \times \hvec{n}\,dS.
            \end{align*}
        \end{enumerate}
    \end{solution}
\end{problem}

\begin{problem}[$2009$ Question $8$]\phantom{}
    \begin{enumerate}[label = (\alph*)]
        \item Calculate $\int_{V}\rho\,dV$ where
        \[
        \rho(\vec{x}) = \frac{1}{|\vec{x}| ^ 4}\exp\left(-\frac{1}{|\vec{x}|}\right)
        \]
        and $V$ is a sphere of radius $R$ centred on the origin in $\R ^ 3$.

        \item Let $\vec{B}(\vec{x})$ be a vector field in $\R ^ 3$ of the form
        \[
        \vec{B}(\vec{x}) = f(|\vec{x}|)\frac{\vec{x}}{|\vec{x}|}.
        \]
        Given that $\vec{B}(\vec{x})$ is related to $\rho$ by $\bnabla \cdot \vec{B} = \rho$,
        use the Divergence Theorem to find $f(|\vec{x}|)$.

        \item Use index notation to show that for any suitably differentiable function of $|\vec{x}|$,
        $g(|\vec{x}|)$,
        we have
        \[
        \bnabla g = g'(|\vec{x}|)\frac{\vec{x}}{|\vec{x}|}.
        \]

        \item Using the result of part $(c)$,
        or otherwise,
        show that the line integral $\int_{C}\vec{B}\cdot\hvec{t}\,d\ell$ along the curve $C$ from $\vec{x} = \vec{a}$ to $\vec{x} = \vec{b}$ is independent of the path taken.
        Hence calculate this integral when $C$ is the path
        \[
        \vec{x}(t) = \cos(t)\vec{e}_1 + \sin(t)\vec{e}_2 + t\vec{e}_3\qquad t \in [0, \infty].
        \]
    \end{enumerate}

    \begin{solution}\phantom{}
        \begin{enumerate}[label = (\alph*)]
            \item
            \begin{align*}
                \int_{V}\rho\,dV &= \int_{0}^{2\pi}\int_{0}^{\pi}\int_{0}^{R}\rho r ^ 2\sin(\theta)\,dr\,d\theta\,d\phi \\
                &= \int_{0}^{2\pi}\int_{0}^{\pi}\int_{0}^{R}\frac{1}{r ^ 2}e ^ {-1 / r}\,dr\,d\theta\,d\phi \\
                \intertext{$u = \frac{1}{r}$}
                &= \int_{0}^{2\pi}\int_{0}^{\pi}\int_{\infty}^{1 / R}e ^ {-u}\,du\sin(\theta)\,d\theta\,d\phi \\
                &= \int_{0}^{2\pi}\int_{0}^{\pi}\left[e ^ {-u}\right]_{\infty}^{1 / R}\sin(\theta)\,d\theta\,d\phi \\
                &= e ^ {-1 / R}\int_{0}^{2\pi}\int_{0}^{\pi}\sin(\theta)\,d\theta\,d\phi \\
                &= e ^ {-1 / R}\int_{0}^{2\pi}\left[-\cos(\theta)\right]_{0}^{\pi}\,d\phi \\
                &= 2e ^ {-1 / R}\int_{0}^{2\pi}\,d\phi = 4\pi e ^ {-1 / R}.
            \end{align*}

            \item 
            $\bnabla \cdot \vec{B} = \rho$.
            Divergence theorem implies
            \begin{align*}
                &\int_{V}\bnabla\cdot\vec{B}\,dV = \oint_{S}\vec{B}\cdot\hvec{n}\,dS \\
                &\implies \int_{V}\rho\,dV = \oint_{S}\vec{B}\cdot\vec{e}_r\,dS \\
                &\implies 4\pi e ^ {-1 / R} = \oint_{S}f(r)\,dS = f(r)\oint_{S}\,dS = 4\pi R ^ 2f(r) \\
                &\implies f(R) = \frac{1}{R ^ 2}e ^ {-1 / R} \\
                &\implies f(r) = \frac{1}{r ^ 2}e ^ {-1 / r}.
            \end{align*}

            \item
            \begin{align*}
                [\bnabla g]_i &= \pd{x_i}(g(r)) \\
                &= \pd[r]{x_i}\frac{dg}{dr} \\
                \pd[r]{x_i} &= \pd{x_i}\left(|\vec{x}|\right) = \pd{x_i}\left((x_jx_j) ^ {1 / 2}\right) = \frac{1}{2}(x_jx_j) ^ {1 / 2}\pd{x_i}(x_jx_j) \\
                &= \frac{1}{2r}2x_j\pd[x_j]{x_i} \\
                &= \frac{x_j}{r}\delta_{ij} = \frac{x_i}{r}.
                \intertext{So}
                [\bnabla g]_i &= \frac{x_i}{r}g' = [g'\vec{x} / |\vec{x}|]_i
            \end{align*}

            \item
            Path independent iff $\vec{B} = \bnabla g$.
            But $\vec{B} = f(r)\frac{\vec{x}}{r}$ so by $(c)$ if $g'(r) = f(r)$ then $\vec{B} = \bnabla g$,
            we have
            \[
            g' = \frac{1}{r ^ 2}e ^ {-1 / r} \implies g = \int_{0}^{r}\frac{1}{r ^ 2}e ^ {-1 / r}\,dr = e ^ {-1 / r} + C.
            \]
            So $\vec{B} = \bnabla(e ^ {-1 / r})$.
            So FTLI implies
            \begin{align*}
                \int_{C}\vec{B} \cdot\hvec{t}\,d\ell &= \int_{C}\bnabla g\cdot\hvec{t}\,d\ell \\
                &= g(\vec{b}) - g(\vec{a}) \\
                r &= |\vec{x}| = \sqrt{1 + t ^ 2} \\
                &= g(\infty) - g(1) \\
                &= 1 - e ^ {-1} = 1 - e ^ {-1}.
            \end{align*}
        \end{enumerate}
    \end{solution}
\end{problem}
