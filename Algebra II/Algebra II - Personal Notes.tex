\section{Rings, subrings and fields}

\textit{Informal definition}:
A ring is set of mathematical objects where we can add,
subtract and multiply two elements while remaining in the set.

\subsection{Definition of a ring}
\textbf{Binary operations},
these are operations which take two elements as arguments.

A binary operation on a set $R$ is a mapping of the Cartesian product of two elements of $R$,
$R \times R$ to $R$.
For addition this would be
\[
+(x, y) : R \times R \to R\qquad(\text{with $x, y \in R$}).
\]

A ring $R$ has a binary operation $+$ which makes it into an abelian group.

$R$ is an abelian group with respect to $+$ means that:
\begin{enumerate}[label = \roman*)]
    \item $R$ contains an additive identity element.

    \item Associativity.

    \item For every $x \in R$ there exists a $y \in R$ such that $x + y = y + x = 0$.
    Such that $y$ is unique and generally denoted $-x$.

    \item To make the group abelian we need for $x + y = y + x$ for all $x, y \in R$.
\end{enumerate}

\begin{definition}[Ring]
    A ring $R$ is a set together with two binary operations
    (usually written $+, \cdot$)
    called "addition" and "multiplication",
    such that $+$ makes $R$ into an abelian group,
    and such that the following conditions hold:
    
    \begin{enumerate}[label = \roman*)]
        \item \phantom{}[(multiplicative) Identity]
        There exists an element $1 \in R$ such that $1 \cdot x = x \cdot 1 = x$,
        for all $x \in R$.

        \item \phantom{}[Associativity]
        For any $x, y, z \in R$ we have $x(yz) = (xy)z$.

        \item \phantom{}[Distributivity]
        For any $x, y, z \in R$ we have $x(y + z) = xy + xz$ and $(y + z)x = yx + zx$.
    \end{enumerate}
\end{definition}

If the ring is commutative distributivity is equivalent in both cases.

\subsection{The integers modulo \texorpdfstring{$n$}{}}
A
(binary)
relation $\sim$,
on a set $S$ is said to be an equivalence relation,
if and only if it is reflexive,
symmetric and transitive.
That is,
for all $a, b$ and $c$ in the set $S$ we have:
\begin{enumerate}[label = \roman*)]
    \item 
    $a \sim a$
    (reflexivity).

    \item
    $a \sim b \iff b \sim a$
    (symmetry).

    \item
    If $a \sim b$ and $b \sim c$ then $a \sim c$
    (transitivity).
\end{enumerate}

The equivalence class of $a$ under the given relation $\sim$,
denoted by $[a]$
(or $\bar{a}$)
is defined as
\[
[a] = \{a \in S\,:\, x \sim a\}.
\]
For two $a, b \in S$ with $a \sim b$ we have $[a] = [b]$.
When we write $[a]$ we call $a$ the representative of the class $[a]$.

Given a set $S$ with an equivalence relation $\sim$ as above we may construct another set
\[
S / \sim \coloneqq \{[a]\, :\, a \in S\}
\]
That is,
the set of equivalence classes on the set $S$ with respect to the given equivalence relation $\sim$.


\textbf{The integers modulo} $n$.

Let $n \in \N$ be a natural number.
Consider the set of integer numbers $\Z$ and define the following relation:
two integers $a, b$ are equivalent $a \sim b$ if $n \mid (a - b)$.
The most common notation for this is $a \equiv b \mod n$.

For an integer $a$ we denote $\bar{a}$ the equivalence class of $a$,
that is
\[
\bar{a} = \{x \in \Z\,:\, n\text{ divides } x - a\} = \{x \in \Z\,:\, x \equiv a\mod n\}
\]
since when we divide an integer by $n$ we will always get a remainder among the numbers
\[
0, 1, \dotsc, n - 1.
\]
That is,
every integer belongs to one of the following equivalence classes:
\[
\{\bar{0}, \bar{1}, \dotsc, \overline{n - 1}\},
\]
we will denote this set
(of equivalence classes)
by $\Z / n$.
The elements of $\Z / n$ are called residue classes.

\begin{definition}[Subring]
    A subring $S$ of a ring $R$ is a set $S \subseteq R$ such that
    \begin{enumerate}[label = \roman*)]
        \item $0 \in S, 1 \in S$,

        \item $a, b \in S \implies a + b \in S$,
        
        \item $a, b \in S \implies ab \in S$,
        
        \item $a \in S \implies -a \in S$,
    \end{enumerate}
\end{definition}
This can be summed up as "A subring contains $0$ and $1$ and is closed under addition,
subtraction and multiplication".

\begin{definition}[Field]
    A ring $R$ is called a field if
    
    \begin{enumerate}[label = \roman*)]
        \item $R$ is commutative,

        \item $1 \neq 0$ in $R$
        (i.e. $R$ has at least two elements),


        \item For any $a \in R$ such that $a \neq 0$,
        there exists a $b \in R$ such that $ab = 1$
        ($b$ is called the inverse of $a$).
    \end{enumerate}
\end{definition}

\section{Integral domains,
units,
polynomial rings}

\begin{definition}
    A ring $R$ which is commutative,
    has at least two elements
    ($0 \neq 1$),
    and does not have any zero divisors apart from $0$,
    is called an integral domain.
\end{definition}
\textit{Note:
A non zero divisor is a number $b$ such that $ab = 0$ where $b \neq 0$.}

\subsection{Examples and non-examples of integral domains}
\begin{proposition}
    If $R$ is an integral domain,
    then the ring of polynomials $R[x] = \{a_0 + a_1x + \dotsi + a_nx ^ n\mid a_i \in R\}$ is also an integral domain.

    \begin{proof}
        We can clearly see that $R[x]$ is commutative and that $0 \neq 1$
        (as this holds in $R$).

        We need to now show that there are no non-zero divisors in $R[x]$ apart from $0$.
        Assume not,
        then for non-zero polynomials $f(x), g(x) \in R[x]$,
        $f(x)g(x) = 0$.
        Then we get
        \begin{align*}
            f(x) &= a_0 + \dotsc + a_nx ^ n,\quad a_n \neq 0, \\
            g(x) &= b_0 + \dotsc + b_nx ^ m,\quad b_m \neq 0
        \end{align*}
        if we expand $f(x)g(x)$ then the term of the highest degree will be $a_nb_mx ^ {n + m}$.
        Since $f(x)g(x) = 0$ $a_nb_mx ^ {n + m} = 0$ which implies $a_nb_m = 0$ and since $a_n, b_m \in R$ which is an integral domain, $a_n = 0$ or $b_m = 0$.
        Contradiction!
    \end{proof}
\end{proposition}

\subsection{The group of units in a ring}
Let $R$ be a ring.
$a \in R$ is called a unit if there exists a $b \in R$ such that $ab = ba = 1$
($b$ is the inverse of $a$ and $b = a ^ {-1}$).

The inverse is unique as,
if $b_1, b_2 \in R$ such that $ab_1 = b_1a = 1 = ab_2 = b_2a$ then
\[
b_1 = b_11 = b_1(ab_2) = (b_1a)b_2 = 1b_2 = b_2.
\]

The set of all units of $R$ is denoted $R ^ {\times}$
(or $R ^ *$).

$R ^ {\times}$ is always a group,
under the multiplication from the ring $R$:
\[
a, b \in R ^ {\times} \implies ab \in R ^ {\times}
\]
if $c, d \in R$ are elements such that $ac = 1$ and $bd = 1$,
then $(ab)(dc) = abcd = ad = 1 = (dc)(ab)$,
so $ab$ is a unit.
We have an identity element $1$ in the group,
multiplication is associative
(as it is in the ring $R$)
and every element in $R ^ {\times}$ has an inverse by definition.
Thus $R ^ {\times}$ is a group.


\begin{proposition}\label{prep:algII:prop:eleunitiffgcd}
    An element $\bar{a} \in \Z / n$  is a unit if and only if $\gcd(a, n) = 1$.

    \begin{proof}
        Let $d = \gcd(a, n)$.
        Then $d \mid a$ and $d \mid n$.

        Suppose $\bar{a}$ is a unit and let $\bar{a} = \bar{b} ^ {-1}$,
        then $\bar{a}\bar{b} = \bar{1}$ which is equivalent to $ab \equiv 1\mod{n}$.
        Thus $ab = 1 + kn$,
        for some $k \in \Z$.
        Since $d \mid (ab)$ and $d \mid kn$ we must have $d \mid 1$.
        Hence $d = 1$.
        
        Conversely,
        assume that $d = 1$.
        The Euclidean algorithm shows that there exist $x, y \in \Z$ such that 
        \[
        xa + ny = d = 1.
        \]
        Therefore $xa \equiv \mod{n}$,
        which means that $\bar{x}\bar{a} = \bar{1}$ in $\Z / n$.
        Hence $\bar{a}$ is a unit.
    \end{proof}
\end{proposition}

\begin{corollary}\label{prep:algII:cor:zpfieldifprime}
    $\Z / p$ is a field if and only if $p$ is a prime.

    \begin{proof}
        By \autoref{prep:algII:prop:eleunitiffgcd},
        if $p$ is a prime,
        then every element $\bar{1}, \dotsc, \overline{p - 1}$ is a unit,
        so $\Z / p$ is a field.

        Going forwards,
        if $\Z / p$ is a field,
        every non-zero $\bar{a} \in \Z / p$ is a unit,
        so by \autoref{prep:algII:prop:eleunitiffgcd},
        $\gcd(a, o) = 1$ for all $a$ such that $1 \leq a \leq p - 1$.
        This means that $p$ has no divisors between $1$ and $p - 1$ other than $1$,
        so $p$ is prime.
    \end{proof}
\end{corollary}

\begin{proposition}
    The ring $\Z / n$ is an integral domain if and only if $n$ is a prime.

    \begin{proof}
        If $n$ is a prime,
        which \autoref{prep:algII:cor:zpfieldifprime} shows that this means that $\Z / n$ is a field and any field is an integral domain.

        Assume $n$ is not prime,
        then $n = ab$,
        $a, b \in \Z$,
        where $2 \leq a, b \leq n - 1$.
        Then
        \[
        \bar{a}\bar{b} = \bar{n} = \bar{0} \in \Z / n,
        \]
        so $\bar{a}$ and $\bar{b}$ are zero divisors in $\Z / n$,
        and so the ring is not an integral domain.
    \end{proof}
\end{proposition}

\subsection{Polynomials over a field and the division algorithm}
From now on,
$F$ will be a field
(e.g. $\Q, \R$ or $\Z / p$ for $p$ prime).

For $f(x) = a_0 + a_1x + \dotsc + a_nx ^ n \in F[x]$,
we define its degree as
\[
\deg{f} \coloneqq \begin{cases}
    \max\{i\mid a_i \neq 0\} &\text{if } f(x) \neq 0 \\
    -\infty &\text{if } f(x) = 0.
\end{cases}
\]
Then we have the following properties for all $f(x), g(x) \in F[x]$:
\begin{itemize}
    \item $\deg(fg) = \deg{f} + \deg{g}$,
    \item $\deg(f + g) \leq \max\{\deg{f}, \deg{g}\}$.
    Note that if $\deg{f} \neq \deg{g}$ then equality.
\end{itemize}


\textbf{The division algorithm in} $\Z$

\begin{proposition}[Division algorithm]\label{prep:algII:prop:divalgpoly}
    Let $f(x), g(x) \in F[x]$ with $g(x) \neq 0$.
    Then there exist unique polynomials $q(x), r(x) \in F[x]$ with $\deg{r} < \deg{g}$ such that
    \[
    f(x) = q(x)g(x) + r(x).
    \]

    \begin{proof}
        (Existence)
        If $\deg{g} > \deg{f}$,
        take $q(x) = 0$ and $r(x) = f(x)$.
        If $\deg{g} \leq \deg{f}$,
        let
        \begin{align*}
            f(x) &= a_0 + \dotsc + a_nx ^ n,\quad a_n \neq 0, \\
            g(x) &= b_0 + \dotsc + b_mx ^ m,\quad b_m \neq 0.
        \end{align*}
        Let $d = m - n \geq 0$.
        Using induction on $d$:
        $m = n$ as $0 = m - n \geq 0$.
        Set $q(x) = a_n / b_m$ and
        \[
        r(x) \coloneqq f(x) - q(x)g(x).
        \]
        Note that $\deg{r} < n = \deg{g}$,
        then we have the required $q(x), r(x)$.

        Assume the existence of $q(x), r(x)$ for all $d < k$,
        for some $k \geq 0$.
        Now consider $d = k$,
        so that $n = m + k$.
        Let
        \[
        f_1(x) \coloneqq f(x) - \frac{a_n}{b_m}x ^ {n - m}g(x).
        \]
        Note $\deg{f_1} < \deg{f}$,
        so by induction,
        there exists polynomials $q_1(x), r(x)$ such that $\deg{r} < \deg{g}$ and
        \[
        f_1(x) = q_1(x)g(x) + r(x).
        \]
        Thus
        \[
        f(x) = f_1(x) + \frac{a_n}{b_m}x ^ {n - m}g(x) = (q_1(x) + \frac{a_n}{b_m}x ^ {n - m})g(x) + r(x)
        \]
        so the result is true for $d = k$,
        and hence for all $d \geq 0$ by induction.

        (Uniqueness)
        Suppose $f(x) = q(x)g(x) + r(x) = Q(x)g(x) + R(x)$,
        with $\deg{r} < \deg{g}$ and $\deg{R} < \deg{g}$.
        This implies $\deg(R - r) < \deg{g}$.
        Rearranging,
        we have
        \begin{equation}\label{prep:algII:eq:eq1}\tag{*}
            R(x) - r(x) = (q(x) - Q(x))g(x),
        \end{equation}
        so by the first basic property,
        we get
        \[
        \deg(R - r) = \deg(q - Q) + \deg{g}.
        \]
        Since $\deg(R - r) < \deg{g}$,
        we must have $\deg(q - Q) < 0$.
        This can only happen if $q(x) = Q(x)$.
        Then by \eqref{prep:algII:eq:eq1} we finally get
        \[
        R(x) = r(x).
        \]
    \end{proof}
\end{proposition}

\section{Divisibility and greatest common divisor in a ring}
For this section let $R$ be a commutative ring.

\subsection{The notions of divisibility and of greatest common divisor}
\begin{definition}
    Let $a, b \in R$.
    We say that $a$ divides $b$ in $R$
    ($a \mid b$)
    if there exists an $r \in R$ such that $b = ra$.
\end{definition}

\begin{definition}\label{prep:algII:def:gcd}
    Let $R$ be as above and $a, b \in R$.
    An element $d \in R$ is called a greatest common divisor,
    $\gcd(a, b)$,
    of $a$ and $b$ if
    
    \begin{enumerate}[label = (\roman*)]
        \item $d \mid a$ and $d \mid b$.
        \item If $e \in R$ such that $e \mid a$ and $e \mid b$,
        then $e \mid d$.
    \end{enumerate}
\end{definition}


\subsection{The Euclidean algorithm}

\begin{lemma}
    Let $R$ be an integral domain.
    Let $a, b \in R$ and assume that $d \coloneqq \gcd(a, b)$ exists.
    The $ud$ is also a $\gcd(a, b)$,
    for any unit $u \in R ^ {\times}$.
    Moreover,
    for any two $\gcd$s $d$,
    $d''$ of $a$ and $b$,
    there exists a unit $u \in R ^ {\times}$ such that $d = ud'$
    (i.e.,
    the $\gcd$ is unique up to units).

    \begin{proof}
        First,
        we show that $ud$ is a $\gcd$ of $a$ and $b$.
        Since $d \mid a$,
        there is an $m \in R$ such that $dm = a$.
        Thus
        \[
        du(u ^ {-1}m) = a,
        \]
        so $du$ divides $a$.
        Similarly $ud \mid b$.
        Now,
        if $e \in R$ is such that $e \mid a$ and $e \mid b$,
        then we know $e \mid d$,
        that is,
        there is a $k \in R$ such that $ek = d$.
        But then $eku = du$,
        so $e \mid du$.
        Thus $du$ is a $\gcd$,
        by \autoref{prep:algII:def:gcd}.
        Next,
        assume that $d$ and $d'$ are two $\gcd$s.
        Then,
        by definition both divide $a$ and $b$ and both must divide each other.
        This means that
        \[
        d = d'u,\quad d' = dv,
        \]
        for some $u, v \in R$.
        Thus $d = duv$.
        If $d = 0$ then also $d' = 0$,
        so we may take $u = 1$.
        If $d \neq 0$,
        then
        (since $R$ is an integral domain),
        we can cancel $d$ and obtain $uv = 1$.
        Hence $u$ and $v$ are units,
        so we are done.
    \end{proof}
\end{lemma}

A polynomial in $F[x]$ is called monic if its leading coefficient is $1$.

\begin{theorem}
    Let $R$ be either $\Z$ or $F[x]$,
    and let $a, b \in R$.
    Then
    \begin{enumerate}[label = \roman*)]
        \item A $\gcd(a, b)$ exists.

        \item If $a$,
        and $b$ are not both zero,
        we can compute a $\gcd(a, b)$ by the Euclidean algorithm.

        \item If $d$ is a $\gcd(a, b)$,
        then there exist $x, y \in R$ such that $ax + by = d$.
    \end{enumerate}

    \begin{proof}
        For $R = F[x]$.

        Set $r_{-1}(x) \coloneqq a \in F[x]$ and $r_0(x) \coloneqq b \in F[x]$.
        We start with the pair $(r_{-1}(x), r_0(x))$ for which we obtain a decomposition $r_{-1}(x) = q_1r_0(x) + r_1(x)$ with $q_1(x)$ and $r_1(x) \in F[x]$ where we have that $\deg(r_1(x)) < \deg(r_0(x))$.
        Then we are done if $r_1(x)$ is the zero polynomial,
        as then $r_0(x)$ already divides $r_{-1}(x)$,
        and otherwise we indicate this process for the pair $(r_0(x), r_1(x))$.
        In equation form we get
        \begin{align*}
            r_{-1}(x) &= q_1(x)r_0(x) + r_1(x),\quad 0 \leq \deg(r_1(x)) < \deg(r_0(x)), \\
            r_{0}(x) &= q_2(x)r_1(x) + r_2(x),\quad 0 \leq \deg(r_2(x)) < \deg(r_1(x)), \\
            &\vdots \\
            r_{i - 2}(x) &= q_i(x)r_{i - 1}(x) + r_i(x),\quad 0 \leq \deg(r_i(x)) < \deg(r_{i - 1}(x)), \\
            &\vdots \\
            r_{n - 2}(x) &= q_n(x)r_{n - 1}(x) + r_n(x),\quad 0 \leq \deg(r_n(x)) < \deg(r_{n - 1}(x)), \\
            r_{n - 1}(x) &= q_{n + 1}(x)r_{n}(x) + 0.
        \end{align*}
        Note the process must stop after finitely many steps,
        as the degree is a non-negative integer,
        so if $\deg(r_0(x)) = n \geq 0$,
        there are at most $n + 1$ steps before the remainder becomes $0$.

        Now the 'last non-zero' remainder $r_n(x)$ is a common divisor of both $r_{-1}(x)$ and $r_0(x)$:
        by noticing that $r_n(x) \mid r_{n - 1}(x)$ and hence it divides $r_{n - 1}(x)$,
        the LHS divides the RHS and going up we get $d(x)$ the common divisor of both $r_{-1}(x)$ and $r_0(x)$ must also divide $r_(x)$:
        this becomes apparent by going down the ladder,
        we get that $d(x)$ also divides $r_1(x), r_2(x), \dotsc, r_n(x)$.
        For the last statement we work as in the examples.
        Namely we start from the equation $r_n(x) = r_{n - 2}(x) - q_n(x)r_{n - 1}(x)$.
        The equation before gives $r_{n - 1}(x) = r_{n - 3}(x) - q_{n - 1}(x)r_{n - 2}(x)$.
        Using this,
        we substitute $r_{n - 1}$ into the equation.
        Collect terms and get
        \[
        r_n(x) = h(x)r_{n - 2}(x) + g(x)r_{n - 3}(x)
        \]
        for some $h(x), g(x) \in F[x]$.
        Now replace $r_{n - 2}(x)$ if we are going up the ladder we end up with $r_{-1}(x)$ and $r_0(x)$.
        We can write $r_n(x) = A(x)r_{-1}(x) + B(x)r_0(x)$ for some $A(x), B(x) \in F[x]$.
    \end{proof}
\end{theorem}

\section{Factorisation in rings}

\subsection{Irreducible polynomials in \texorpdfstring{$F[x]$}{}}

Some polynomials can be factored into a product of other polynomials,
e.g.
\[
x ^ 3 - 1 = (x ^ 2 + x + 1)(x - 1) \in \Q[x].
\]
Since constants are also polynomials we can factorise them too,
e.g. $13x + 13 = 13(x + 1)$.

\begin{definition}
    Let $R$ be a commutative ring.
    An element $0 \neq r \in R$ is called irreducible if the following conditions hold:
    \begin{enumerate}[label = \roman*)]
        \item $r$ is not a unit.
        \item If $r = ab$ for $a, b \in R$,
        then $a$ is a unit,
        or $b$ is a unit.
    \end{enumerate}
\end{definition}

Let $F$ be a field and $f(x) \in F[x]$.
An element $a \in F$ is called a root of $f$ in $F$ if $f(a) = 0$.

If $\deg{f} = 2, 3$,
then it is irreducible if and only if it has no roots in $F$.
\begin{proof}
    Let $\alpha \in F$ be a root of $f(x)$.
    Write $f(x) = q(x)(x - \alpha) + r(x)$ with $\deg{r} \leq 0$,
    using \autoref{prep:algII:prop:divalgpoly}.
    Thus $r(x)$ is a constant.
    Then $0 = f(\alpha) = r(\alpha)$,
    but $r(x)$ is constant,
    so $f(x) = q(x)(x - \alpha)$,
    and so is not irreducible since $\deg(q) \geq 1$.
    Conversely,
    if $f(x)$ is not irreducible,
    then $f(x) = g(x)h(x)$ with $\deg{g}, h \geq 1$.
    But $\deg{f} = \deg{g} + \deg{h}$ so if $\deg{f} = 2$ or $3$,
    we must have either $\deg{g} = 1$ or $\deg{h} = 1$,
    say $g$.
    Then $g(x) = ax + b$,
    for some $a, b \in F[x]$,
    $a \neq 0$.
    Thus
    \[
    0 = g(-b / a) = f(-b / a),
    \]
    so $f(x)$ has a root.
\end{proof}

\begin{proposition}
    Let $f(x) = a_0 + a_1x + \dotsc + a_nx ^ n \in \Z[x]$ with $\deg{f} \geq 1$.
    If $f(p / q) = 0$,
    where $p, q \in \Z$ and $\gcd(p, q) = 1$,
    then
    \[
    p \mid a_0 \qquad\text{and}\qquad q \mid a_n.
    \]

    \begin{proof}
        We have
        \[
        f(p / q) = a_0 + a_1(p / q) + \dotsc + a_n(p / q) ^ n = 0.
        \]
        Multiply both sides by $q ^ n$ and take out a factor of $p$:
        \[
        p(a_1q ^ {n - 1} + a_2pq ^ {n - 2} + \dotsc + a_np ^ {n - 1}) = -a_0q ^ n.
        \]
        Thus $p \mid (a_0q ^ n)$,
        but $\gcd(p, q) = 1$,
        so $p \mid a_0$.
        Now shift the leading term of the first equation to the right hand side and multiply by $q ^ n$,
        we get
        \[
        q(a_0q ^ {n - 1} + \dotsc + a_{n - 1}p ^ {n - 1}) = -a_np ^ n.
        \]
        Thus $q \mid (a_np ^ n)$,
        and so $q \mid a_n$.
    \end{proof}
\end{proposition}

\begin{example}
    Show that $f(x) = x ^ 3 - 2x + 2$ is irreducible in $\Q[x]$.

    \begin{solution}
        From the above we would need $p / q \in \Q$ with $\gcd(p, q) = 1$ and $p \mid 2$ and $q \mid 1$ leaving us with only two choices $\pm 1$ and $\pm 2$,
        easily we can see that neither $f(1 / 2)$ nor $f(2 / 1)$ is a root and hence $f$ is irreducible in $\Q[x]$.
    \end{solution}
\end{example}

\begin{theorem}[Gauss's lemma]\label{prep:algII:thm:guasslem}
    Let $f(x) = a_0 + a_1x + \dotsc + a_nx ^ n \in \Z[x]$ be a non-constant polynomial.
    Then $f(x)$ is irreducible in $\Z[x]$ if and only if it is irreducible in $\Q[x]$ and $\gcd(a_0, a_1, \dotsc, a_n) = 1$.

    \begin{proof}
        "$\impliedby$"
        Assume that $f(x)$ is irreducible in $\Q[x]$ and $\gcd(a_0, a_1, \dotsc, a_n) = 1$.
        Suppose that $f(x) = g(x)h(x)$ for some $g(x), h(x) \in \Z[x]$.
        If $\deg{g} \geq 1$ and $\deg{h} \geq 1$,
        then we would also have a proper factorisation in $\Q[x]$,
        violating irreducibility in $\Q[x]$.
        Thus either $\deg{g} = 0$ or $\deg{h} = 0$,
        say it's $g$.
        Thus $g(x) \in \Z$.
        If $g(x) \neq \pm 1$,
        then there will exist a prime number $p \in \Z$ dividing $g(x)$,
        and thus $f(x)$,
        violating $\gcd(a_0, a_1, \dotsc, a_n) = 1$.
        Thus $g(x) = \pm 1$
        (a unit),
        so that $f(x)$ is irreducible in $\Z[x]$.

        "$\implies$"
        Omitted.
    \end{proof}
\end{theorem}

\begin{corollary}\label{prep:algII:cor:gauslemcorol}
    Let $f(x) = a_0 + a_1x + \dotsc + a_nx ^ n \in \Z[x]$ with $\gcd(a_0, \dotsc, a_n) = 1$.
    Then $f(x)$ is irreducible in $\Z[x]$ if and only if it is irreducible in $\Q[x]$.
\end{corollary}

\begin{lemma}
    If a monic polynomial in $\Z[x]$ factors in $\Q[x]$,
    then it factors into integer monic polynomials.

    \begin{proof}
        Let $f(x) = a_0 + \dotsc + a_nx ^ n$,
        $a_n = 1$.

        Thus,
        \autoref{prep:algII:cor:gauslemcorol} implies that if it factors in $\Q[x]$ then it factors in $\Z[x]$.
        Thus,
        \[
        f(x) = g(x)h(x) = (b_0 + \dotsc + b_mx ^ m)(c_0 + \dotsc + c_lx ^ l),
        \]
        where $m + l = n$ and $b_i, c_j \in \Z$.
        Now equating coefficients of $x ^ n$ on both sides,
        $b_mc_l = 1$,
        thus either $b_m = c_l = 1$ or $b_m = c_l = -1$.
        In the first case $g(x), h(x)$ are monic and in the second case $-g(x), -h(x)$ are monic and $f(x) = (-g(x)(-h(x))$.
    \end{proof}
\end{lemma}

\begin{lemma}
    Let $R$ be a commutative ring,
    $x \in R$ and irreducible element and $u \in R ^ {\times}$.
    Then $ux$ is also irreducible.

    \begin{proof}
        Since $x \neq 0$ we also have $ux \neq 0$.
        Now we need to show that $ux$ is not a unit.
        If $ux$ is a unit then there exists a $b \in R$ such that $v(ux) = 1$ or equivalently $(vu)x = 1$.
        That is $x$ is a unit.
        Contradiction.

        Assume now that $ux = ab$ with $a, b \in R$.
        We need to show that either $a$ or $b$ is a unit.
        Then we may write $x = abu ^ {-1} = a(bu ^ {-1})$.
        Since $x$ is irreducible we have that either $a \in R ^ {\times}$ or $bu ^ {-1} \in R ^ {\times}$.
        But the latter condition also gives that $b \in R ^ {\times}$ as the units form a group under multiplication.
        That is,
        $ux$ is irreducible.
    \end{proof}
\end{lemma}

\begin{proposition}[Eisenstein's criterion]
    Let $f(x) = a_0 + a_1x + \dotsc + a_nx ^ n \in \Z[x]$,
    and let $p \in \Z$ be a prime such that $p \mid a_0, p \mid a_1, \dotsc, p \mid a_{n - 1}$ and $p \nmid a_n$ and $p ^ 2 \nmid a_0$.
    Then $f(x)$ is irreducible in $\Q[x]$.

    \begin{proof}
        Set $d \coloneqq \gcd(a_0, a_1, \dotsc, a_n)$.
        Set $b_i \coloneqq \frac{a_i}{d}$.
        Then
        \[
        F(x) \coloneqq \frac{1}{d}f(x) = b_0 + b_1x + \dotsc + b_nx ^ n \in \Z[x]
        \]
        and $\gcd(b_0, b_1, \dotsc, b_n) = 1$.
        By the above if we show $F(x)$ is irreducible in $\Q[x]$ then so is $f(x)$.
        We note that since $p \nmid a_n$ we have $p \nmid d$ and so $p \mid b_0, p \mid b_1, \dotsc, p \mid b_{n - 1}, p \nmid b_n$ and $p ^ 2 \nmid b_0$.

        We assume now $F(x)$ is not irreducible in $\Q[x]$ and we will derive a contradiction.
        By Gauss's lemma we have $F(x) = g(x)h(x)$ with $g(x), h(x) \in \Z[x]$ and $\deg(g), \deg(h) \geq 1$.
        We now take reduction modulo $p$ and obtain
        \[
        \bar{g}(x)\bar{h}(x) = \bar{F}(x) = \bar{b}_0 + \bar{b}_1x + \dotsc + \bar{b}_nx ^ n = \bar{b}_nx ^ n.
        \]
        We now write
        \[
        \bar{g}(x) = \bar{\alpha}_0 + \dotsc + \bar{\alpha}_mx ^ m,\quad m = \deg(\bar{g}) = \deg(g), \bar{\alpha}_m \neq \bar{0},
        \]
        and
        \[
        \bar{h}(x) = \bar{\beta}_0 + \dotsc + \bar{\beta}_kx ^ k,\quad k = \deg(\bar{h}) = \deg(h), \bar{\beta}_k \neq \bar{0}.
        \]
        From $\bar{g}(x)\bar{h}(x)$ we obtain $\alpha_0\beta_0$ is divisible by $p$.
        Since $p$ is prime it has to divide at least one of them,
        we aim to show that it divides both since then we can conclude $p ^ 2$ divides $b_0$.
        Assume $\alpha_0$ is divisible by $p$.
        As we just mentioned if $\beta_0$ is also divisible by $p$ we derive our contradiction.
        Assume $\beta_0$ is not divisible by $p$.
        First show all other coefficients of $g$ except $\alpha_m$,
        that is $\alpha_i$ with $i = 1, \dotsc, m - 1$ are divisible by $p$.

        Since $F(x) = g(x)h(x)$,
        we have $b_1 = \alpha_0\beta_1 + \beta_0\alpha_1$.
        Since $p \mid b_1$ and $p \mid \alpha_0$ and since $p \nmid \beta_0$ we must have $p \mid \alpha_1$.
        Continuing inductively we show that all the terms from $\alpha_1$ to $\alpha_{m - 1}$ are divisible by $p$.
        In particular from $\bar{g}(x)\bar{h}(x)$ we obtain
        \[
        \bar{\alpha}_0\bar{\beta}_0 + \bar{\alpha}_0\bar{\beta}_kx ^ k + \bar{\beta}_0\bar{\alpha}_mx ^ m + \dotsc + \bar{\alpha}_m\bar{\beta}_kx ^ {n + k} = \bar{b}_nx ^ n.
        \]
        From the above we have that $\bar{\beta}_0\bar{\alpha}_m = \bar{0}$ and since $\bar{\alpha}_m \neq \bar{0}$ we have $\bar{\beta}_0 = \bar{0}$.
        That is,
        both $\alpha_0$ and $\beta_0$ are divisible by $p$ and hence we have that $\alpha_0\beta_0$ is divisible by $p ^ 2$.
        But $b_0 = \alpha_0\beta_0$ and so $p ^ 2$ divides $b_0$.
        Contradiction.
    \end{proof}
\end{proposition}
